# 文件操作

## C++ 流方式

```cpp
#include <fstream>

// 写入文件
std::ofstream out("output.txt");
out << "Hello, file!" << std::endl;
out.close();

// 读取文件
std::ifstream in("input.txt");
std::string line;
while (std::getline(in, line)) {
    std::cout << line << std::endl;
}
in.close();

// 同时读写
std::fstream file("data.bin", std::ios::in | std::ios::out | std::ios::binary);
```

## C++17 filesystem

```cpp
#include <filesystem>
namespace fs = std::filesystem;

fs::path p = "dir/subdir/file.txt";

// 路径操作
std::cout << p.parent_path() << std::endl;   // dir/subdir
std::cout << p.filename() << std::endl;       // file.txt
std::cout << p.stem() << std::endl;          // file
std::cout << p.extension() << std::endl;      // .txt

// 遍历目录
for (const auto& entry : fs::directory_iterator(".")) {
    std::cout << entry.path() << std::endl;
}

// 创建目录
fs::create_directories("new_dir/sub_dir");

// 检查存在性
if (fs::exists("file.txt")) {
    // 文件存在
}
```

## 二进制 I/O

### 读写二进制数据

```cpp
#include <fstream>
#include <vector>
#include <cstdint>

// 写入二进制
std::ofstream out("data.bin", std::ios::binary);
int numbers[] = {1, 2, 3, 4, 5};
out.write(reinterpret_cast<const char*>(numbers), sizeof(numbers));
out.close();

// 读取二进制
std::ifstream in("data.bin", std::ios::binary);
int read_numbers[5];
in.read(reinterpret_cast<char*>(read_numbers), sizeof(read_numbers));
in.close();

// 使用 vector 读写
std::vector<double> values = {1.1, 2.2, 3.3};
std::ofstream out2("doubles.bin", std::ios::binary);
out2.write(reinterpret_cast<const char*>(values.data()),
           static_cast<std::streamsize>(values.size() * sizeof(double)));
out2.close();
```

### 结构体二进制序列化

```cpp
#pragma pack(push, 1)  // 取消结构体对齐
struct Person {
    char name[32];
    int age;
    double height;
};
#pragma pack(pop)

// 写入
Person p = {"Alice", 30, 165.5};
std::ofstream out("person.bin", std::ios::binary);
out.write(reinterpret_cast<const char*>(&p), sizeof(Person));
out.close();

// 读取
Person read_p;
std::ifstream in2("person.bin", std::ios::binary);
in2.read(reinterpret_cast<char*>(&read_p), sizeof(Person));
in2.close();
```

## 随机访问文件

### seekg / seekp / seek

```cpp
#include <fstream>

// 打开文件并定位
std::fstream file("data.bin", std::ios::binary | std::ios::in | std::ios::out);

// 获取文件大小
file.seekg(0, std::ios::end);
std::streamsize size = file.tellg();

// 定位到开头
file.seekg(0, std::ios::beg);

// 定位到指定位置（从头算起）
file.seekg(10);  // 移动到第10字节

// 当前读写位置
std::streamoff pos = file.tellg();  // 读位置
std::streamoff pos2 = file.tellp(); // 写位置

// 相对定位
file.seekg(5, std::ios::cur);  // 从当前位置向后移动5字节
file.seekg(-5, std::ios::cur); // 从当前位置向前移动5字节

// 读取第 i 个 int（每个 int 4 字节）
int getIntAt(std::fstream& f, int index) {
    int val;
    f.seekg(index * sizeof(int));
    f.read(reinterpret_cast<char*>(&val), sizeof(int));
    return val;
}
```

### 同时读写示例

```cpp
std::fstream file("records.bin", std::ios::binary | std::ios::in | std::ios::out);
const int RECORD_SIZE = 64;
const int RECORD_COUNT = 100;

// 更新第 record_idx 条记录的第 offset 字节
void updateRecord(int record_idx, int offset, char value) {
    file.seekp(record_idx * RECORD_SIZE + offset);
    file.write(&value, 1);
}
```

## 文件锁（flock）

### POSIX 文件锁

```cpp
#include <sys/file.h>
#include <fcntl.h>
#include <unistd.h>

int fd = open(".lockfile", O_CREAT | O_RDWR, 0666);

// 阻塞式加锁
flock(fd, LOCK_EX);  // 独占锁（写锁）

// 非阻塞加锁
int result = flock(fd, LOCK_EX | LOCK_NB);
if (result == -1) {
    // 锁已被持有
}

// 临界区操作
// ...

// 解锁
flock(fd, LOCK_UN);

close(fd);
```

### 跨平台文件锁（C++17）

```cpp
#include <filesystem>
#include <system_error>

bool tryAcquireLock(const std::filesystem::path& lock_path) {
    // 创建锁文件
    std::ofstream(lock_path) << std::flush;
    // 尝试原子重命名（原子性保证单实例）
    std::error_code ec;
    std::filesystem::rename(lock_path, lock_path, ec);
    return !ec;
}
```

## 内存映射（mmap）

### POSIX mmap

```cpp
#include <sys/mman.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <cstring>

// 打开文件
int fd = open("data.bin", O_RDWR);
struct stat sb;
fstat(fd, &sb);
size_t file_size = sb.st_size;

// 内存映射
void* addr = mmap(nullptr, file_size, PROT_READ | PROT_WRITE,
                  MAP_SHARED, fd, 0);

if (addr == MAP_FAILED) {
    // 映射失败
}

// 使用映射内存
char* data = static_cast<char*>(addr);
data[0] = 'X';  // 直接修改，操作系统自动写回文件

// 解除映射
munmap(addr, file_size);
close(fd);
```

### 私有映射（写时复制）

```cpp
// MAP_PRIVATE：修改不影响原文件
void* addr = mmap(nullptr, file_size, PROT_READ | PROT_WRITE,
                  MAP_PRIVATE, fd, 0);
```

### 无文件映射（匿名 mmap）

```cpp
// 分配 4096 字节内存页
void* buf = mmap(nullptr, 4096, PROT_READ | PROT_WRITE,
                 MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);

// 使用完后释放
munmap(buf, 4096);
```

### Windows 内存映射

```cpp
#include <windows.h>

// 创建文件映射内核对象
HANDLE hFile = CreateFileA("data.bin", GENERIC_READ | GENERIC_WRITE,
                           FILE_SHARE_READ | FILE_SHARE_WRITE,
                           nullptr, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, nullptr);

HANDLE hMap = CreateFileMappingA(hFile, nullptr, PAGE_READWRITE,
                                  0, 0, nullptr);

void* addr = MapViewOfFile(hMap, FILE_MAP_ALL_ACCESS, 0, 0, 0);

// 使用 addr ...

// 清理
UnmapViewOfFile(addr);
CloseHandle(hMap);
CloseHandle(hFile);
```

## 文件操作工具函数

### 读取整个文件到 string

```cpp
#include <fstream>
#include <sstream>

std::string readFile(const std::string& path) {
    std::ifstream in(path);
    if (!in) return "";
    std::ostringstream ss;
    ss << in.rdbuf();
    return ss.str();
}

// 读取到 vector<string>（每行一个）
std::vector<std::string> readLines(const std::string& path) {
    std::vector<std::string> lines;
    std::ifstream in(path);
    std::string line;
    while (std::getline(in, line)) {
        lines.push_back(line);
    }
    return lines;
}

// C++17 更简洁写法
std::string readFile(const std::filesystem::path& path) {
    return std::filesystem::read_string(path);
}
```

### 原子写入

```cpp
#include <filesystem>
#include <fstream>

void atomicWrite(const std::filesystem::path& path,
                 const std::string& content) {
    auto tmp = path.string() + ".tmp";
    std::ofstream out(tmp);
    out << content;
    out.close();
    std::filesystem::rename(tmp, path);  // 原子替换
}
```

## 文件属性查询（C++17 filesystem）

```cpp
namespace fs = std::filesystem;

fs::path p = "example.txt";

fs::file_size(p);              // 文件大小（字节）
fs::last_write_time(p);         // 最后修改时间
fs::status(p).type();           // 文件类型（regular_file, directory, ...）
fs::status(p).permissions();   // 权限位

// 快捷判断
fs::is_regular_file(p);         // 普通文件
fs::is_directory(p);           // 目录
fs::is_symlink(p);             // 符号链接
fs::is_empty(p);               // 空文件或空目录

// 递归遍历
for (const auto& entry : fs::recursive_directory_iterator(".")) {
    if (entry.is_regular_file()) {
        std::cout << entry.path() << " size=" << entry.file_size() << "\n";
    }
}
```
