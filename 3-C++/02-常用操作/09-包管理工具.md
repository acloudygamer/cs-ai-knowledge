# C++ 包管理工具

C++ 生态中有多个包管理工具，本章介绍最流行的 vcpkg 和 Conan。

## vcpkg

vcpkg 是微软开发的 C++ 包管理器，支持 Windows、Linux 和 macOS。

### 安装

```bash
# Windows
git clone https://github.com/Microsoft/vcpkg.git
cd vcpkg
.\bootstrap-vcpkg.bat

# Linux/macOS
git clone https://github.com/Microsoft/vcpkg.git
cd vcpkg
./bootstrap-vcpkg.sh

# 安装后可选
vcpkg integrate install  # 集成到全局 CMake
```

### 基础使用

```bash
# 搜索包
vcpkg search json

# 安装包
vcpkg install nlohmann-json        # 安装到默认 triplet
vcpkg install nlohmann-json:x64-linux  # 指定平台

# 列出已安装
vcpkg list

# 移除包
vcpkg remove nlohmann-json
```

### CMake 集成

```cmake
# CMakeLists.txt
cmake_minimum_required(VERSION 3.16)
project(MyProject)

# vcpkg 提供集成，无需 find_package
find_package(nlohmann-json REQUIRED)

add_executable(myapp main.cpp)
target_link_libraries(myapp PRIVATE nlohmann-json::nlohmann-json)
```

### 使用 vcpkg 安装的包

```cpp
// main.cpp
#include <nlohmann/json.hpp>
#include <iostream>

int main() {
    nlohmann::json j = {
        {"name", "Alice"},
        {"age", 30}
    };
    std::cout << j.dump(2) << std::endl;
}
```

### vcpkg triplet 配置

```bash
# 查看可用 triplets
vcpkg help triplet

# 常见 triplets
vcpkg install boost:x64-windows
vcpkg install boost:x64-linux
vcpkg install boost:arm64-osx

# 自定义 triplet（vcpkg.json 目录）
{
    "dependencies": ["nlohmann-json"],
    "builtin-baseline": "..."
}
```

### manifest 模式（推荐）

```bash
# 在项目根目录创建 vcpkg.json
{
    "name": "my-project",
    "version": "1.0.0",
    "dependencies": [
        "nlohmann-json",
        "spdlog",
        "boost-filesystem"
    ]
}

# 安装
vcpkg install

# CMake 使用
# CMakeLists.txt 中无需 find_package，由 vcpkg 提供
```

### CMake Toolchain 文件

```bash
# 使用 vcpkg 提供 toolchain
cmake -B build -S . \
    -DCMAKE_TOOLCHAIN_FILE=$VCPKG_ROOT/scripts/buildsystems/vcpkg.cmake

# 指定 triplet
cmake -B build -S . \
    -DCMAKE_TOOLCHAIN_FILE=$VCPKG_ROOT/scripts/buildsystems/vcpkg.cmake \
    -DVCPKG_TARGET_TRIPLET=x64-linux
```

## Conan

Conan 是去中心化的 C++ 包管理器，灵活且功能强大。

### 安装

```bash
# pip 安装
pip install conan

# 或下载安装器
# https://conan.io/downloads.html

# 初始化配置
conan profile detect  # 自动检测编译器
```

### 基础命令

```bash
# 搜索包
conan search nlohmann-json

# 安装包
conan install . -pr:b=default  # 从 conanfile.txt
conan install . -pr:h=default  # 指定主机配置

# 创建包
conan create . pkg/1.0.0@user/channel

# 列出本地缓存
conan list
```

### conanfile.txt

```ini
[requires]
nlohmann-json/3.10.5
spdlog/1.12.0

[generators]
CMakeDeps
CMakeToolchain
```

### conanfile.py

```python
from conan import ConanFile

class MyPackage(ConanFile):
    name = "mylib"
    version = "1.0.0"
    settings = "os", "compiler", "build_type", "arch"

    def requirements(self):
        self.requires("nlohmann-json/[>=3.10.0]")

    def layout(self):
        self.folders.generators = "build/generators"

    def generate(self):
        tc = CMakeToolchain(self)
        tc.generate()
        deps = CMakeDeps(self)
        deps.generate()
```

### CMake 集成

```cmake
# CMakeLists.txt
cmake_minimum_required(VERSION 3.16)
project(MyProject)

# 包含 Conan 生成的文件
include(${CMAKE_BINARY_DIR}/generators/conanbuildinfo.cmake)
conan_basic_setup()

find_package(nlohmann-json REQUIRED)

add_executable(myapp main.cpp)
target_link_libraries(myapp CONAN_PKG::nlohmann-json)
```

### Conan 配置文件

```bash
# ~/.conan2/profiles/default
include(default)

# 自定义编译器
ccache:range = native

# 构建选项
default_optimized_profile: linux-release
```

### Conan Center 与自定义仓库

```bash
# Conan Center（中央仓库）
conan search nlohmann-json -r conancenter

# 添加自定义远程仓库
conan remote add myserver https://myserver.com/artifactory/api/conan/myrepo

# 使用私有仓库
conan install . -r myserver
```

### 跨平台构建

```bash
# Windows MSVC
conan install . -pr:h=default -s:h build_type=Release

# Linux GCC
conan install . -pr:h=linux-gcc -s:h build_type=Release

# macOS Clang
conan install . -pr:h=apple-clang -s:h build_type=Release
```

## vcpkg vs Conan 对比

| 特性 | vcpkg | Conan |
|------|-------|-------|
| 架构 | 集中式（git 子模块） | 去中心化 |
| 安装速度 | 快（编译后缓存） | 依赖网络 |
| CMake 集成 | 原生支持 | 需生成器 |
| 二进制缓存 | 支持 | 支持 |
| 跨平台 | 优秀 | 优秀 |
| 企业支持 | 微软 | JFrog |

## 推荐工作流

### 小型项目（推荐 vcpkg）

```bash
# 1. 创建项目
mkdir myproject && cd myproject

# 2. 创建 vcpkg.json
echo '{"dependencies": ["nlohmann-json", "spdlog"]}' > vcpkg.json

# 3. CMake 使用
cmake -B build -S . -DCMAKE_TOOLCHAIN_FILE=$VCPKG_ROOT/scripts/buildsystems/vcpkg.cmake
cmake --build build
```

### 大型项目（推荐 Conan）

```bash
# 1. 使用 Conanfile.py 描述复杂依赖
# 2. 支持多个构建配置
# 3. 二进制制品管理
conan create . mylib/1.0.0@user/channel
conan upload mylib/1.0.0@user/channel --all
```

## 常见问题

### 头文件库 vs 预编译库

```bash
# 头文件库（header-only）
vcpkg install nlohmann-json  # 只需头文件

# 预编译库
vcpkg install boost --download
vcpkg install boost --head
```

### 版本约束

```bash
# vcpkg
vcpkg install "nlohmann-json>=3.10"

# Conan
self.requires("nlohmann-json/[>=3.10.0]")
```

### 依赖冲突

```bash
# Conan 处理依赖冲突
conan install . --update  # 尝试更新依赖
conan graph info --requires=conflict  # 查看冲突

# vcpkg
vcpkg remove problematic-package
vcpkg explicit
```
