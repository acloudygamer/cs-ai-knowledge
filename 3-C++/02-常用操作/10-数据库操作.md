# C++ 数据库操作

C++ 中操作数据库主要有三种方式：使用原生驱动、使用第三方库、或通过 ORM 框架。

## SQLite3

SQLite3 是嵌入式数据库，C++ 标准库不直接支持，需要使用 sqlite3 API。

### 安装

```bash
# Ubuntu/Debian
apt install libsqlite3-dev

# macOS
brew install sqlite3

# vcpkg
vcpkg install sqlite3

# Conan
conan install sqlite3/3.45.0
```

### 基础操作

```cpp
#include <sqlite3.h>
#include <iostream>
#include <string>

int main() {
    sqlite3* db;
    char* err_msg = nullptr;

    // 打开/创建数据库
    int rc = sqlite3_open("test.db", &db);
    if (rc != SQLITE_OK) {
        std::cerr << "Cannot open database: " << sqlite3_errmsg(db) << std::endl;
        return 1;
    }

    // 创建表
    const char* sql = R"(
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            email TEXT UNIQUE,
            age INTEGER
        )
    )";

    rc = sqlite3_exec(db, sql, nullptr, nullptr, &err_msg);
    if (rc != SQLITE_OK) {
        std::cerr << "SQL error: " << err_msg << std::endl;
        sqlite3_free(err_msg);
    }

    // 插入数据
    const char* insert = "INSERT INTO users (name, email, age) VALUES (?, ?, ?)";
    sqlite3_stmt* stmt;

    rc = sqlite3_prepare_v2(db, insert, -1, &stmt, nullptr);
    if (rc == SQLITE_OK) {
        sqlite3_bind_text(stmt, 1, "Alice", -1, SQLITE_STATIC);
        sqlite3_bind_text(stmt, 2, "alice@example.com", -1, SQLITE_STATIC);
        sqlite3_bind_int(stmt, 3, 30);

        if (sqlite3_step(stmt) == SQLITE_DONE) {
            std::cout << "Inserted: " << sqlite3_last_insert_rowid(db) << std::endl;
        }
        sqlite3_finalize(stmt);
    }

    // 查询数据
    const char* query = "SELECT id, name, email, age FROM users WHERE age > ?";
    rc = sqlite3_prepare_v2(db, query, -1, &stmt, nullptr);
    if (rc == SQLITE_OK) {
        sqlite3_bind_int(stmt, 1, 25);

        while (sqlite3_step(stmt) == SQLITE_ROW) {
            int id = sqlite3_column_int(stmt, 0);
            const char* name = (const char*)sqlite3_column_text(stmt, 1);
            const char* email = (const char*)sqlite3_column_text(stmt, 2);
            int age = sqlite3_column_int(stmt, 3);

            std::cout << id << ": " << name << " (" << email << ") age " << age << std::endl;
        }
        sqlite3_finalize(stmt);
    }

    sqlite3_close(db);
    return 0;
}
```

### 事务处理

```cpp
#include <sqlite3.h>

void transaction_example(sqlite3* db) {
    char* err_msg = nullptr;

    // 开始事务
    sqlite3_exec(db, "BEGIN TRANSACTION", nullptr, nullptr, &err_msg);

    try {
        // 执行多条 SQL
        sqlite3_exec(db, "INSERT INTO users (name) VALUES ('Bob')", nullptr, nullptr, &err_msg);
        sqlite3_exec(db, "INSERT INTO users (name) VALUES ('Carol')", nullptr, nullptr, &err_msg);
        sqlite3_exec(db, "INSERT INTO invalid SQL", nullptr, nullptr, &err_msg);  // 会失败

        // 提交
        sqlite3_exec(db, "COMMIT", nullptr, nullptr, &err_msg);
    } catch (...) {
        // 回滚
        sqlite3_exec(db, "ROLLBACK", nullptr, nullptr, &err_msg);
    }
}
```

### C++ 封装（现代化写法）

```cpp
#include <sqlite3.h>
#include <string>
#include <optional>
#include <variant>
#include <stdexcept>

class Database {
    sqlite3* db_ = nullptr;

public:
    Database(const std::string& path) {
        if (sqlite3_open(path.c_str(), &db_) != SQLITE_OK) {
            throw std::runtime_error(sqlite3_errmsg(db_));
        }
    }

    ~Database() {
        if (db_) sqlite3_close(db_);
    }

    // 禁止复制
    Database(const Database&) = delete;
    Database& operator=(const Database&) = delete;

    // 允许移动
    Database(Database&& other) noexcept : db_(other.db_) {
        other.db_ = nullptr;
    }

    // 执行无返回值的 SQL
    void execute(const std::string& sql) {
        char* err = nullptr;
        if (sqlite3_exec(db_, sql.c_str(), nullptr, nullptr, &err) != SQLITE_OK) {
            std::string msg = err ? err : "unknown error";
            sqlite3_free(err);
            throw std::runtime_error(msg);
        }
    }

    // 查询单行
    template<typename... Cols>
    std::optional<std::tuple<Cols...>> query_one(
        const std::string& sql,
        auto bind_fn,  // lambda: [](sqlite3_stmt* stmt) { sqlite3_bind_xxx(...); }
        auto read_fn   // lambda: [](sqlite3_stmt* stmt) { return sqlite3_column_xxx(...); }
    ) {
        sqlite3_stmt* stmt = nullptr;
        if (sqlite3_prepare_v2(db_, sql.c_str(), -1, &stmt, nullptr) != SQLITE_OK) {
            throw std::runtime_error(sqlite3_errmsg(db_));
        }

        std::optional<std::tuple<Cols...>> result;

        bind_fn(stmt);
        if (sqlite3_step(stmt) == SQLITE_ROW) {
            result = read_fn(stmt);
        }

        sqlite3_finalize(stmt);
        return result;
    }
};
```

## SOCI

SOCI 是一个连接导向的数据库抽象库，支持多种后端。

### 安装

```bash
# vcpkg
vcpkg install soci

# 依赖后端库
vcpkg install soci[sqlite3]:x64-windows
vcpkg install soci[postgresql]:x64-windows
```

### 基础使用

```cpp
#include <soci/soci.h>
#include <soci/sqlite3/soci-sqlite3.h>
#include <iostream>
#include <string>

int main() {
    soci::session sql(soci::sqlite3, "test.db");

    // 创建表
    sql << "CREATE TABLE IF NOT EXISTS users ("
           "id INTEGER PRIMARY KEY, "
           "name TEXT NOT NULL, "
           "age INTEGER)";

    // 插入（使用参数绑定）
    int id = 1;
    std::string name = "Alice";
    int age = 30;
    sql << "INSERT INTO users (id, name, age) VALUES (:id, :name, :age)",
        soci::use(id), soci::use(name), soci::use(age);

    // 查询（into 提取结果）
    int retrieved_id;
    std::string retrieved_name;
    int retrieved_age;

    sql << "SELECT id, name, age FROM users WHERE id = :id",
        soci::into(retrieved_id),
        soci::into(retrieved_name),
        soci::into(retrieved_age),
        soci::use(id);

    std::cout << retrieved_id << ": " << retrieved_name << ", age " << retrieved_age << std::endl;

    // 使用 rowset（类似 STL 迭代器）
    soci::rowset<soci::row> rs = (sql.prepare <<
        "SELECT id, name, age FROM users");

    for (auto& row : rs) {
        std::cout << row.get<int>(0) << ": " << row.get<std::string>(1) << std::endl;
    }
}
```

### 事务处理

```cpp
#include <soci/soci.h>

void transaction_example(soci::session& sql) {
    try {
        soci::transaction tr(sql);

        sql << "INSERT INTO users (name) VALUES ('Bob')";
        sql << "INSERT INTO users (name) VALUES ('Carol')";

        tr.commit();  // 提交事务
    } catch (const soci::soci_error& e) {
        // 异常时自动回滚
        std::cerr << "Transaction failed: " << e.what() << std::endl;
    }
}
```

### 数据类型映射

```cpp
// SOCI 支持的类型
int i;              // INTEGER
double d;           // REAL
std::string s;      // TEXT
std::tm t;          // DATETIME
std::vector<char> blob;  // BLOB
```

## libpqxx

libpqxx 是 PostgreSQL 的 C++ 客户端库。

### 安装

```bash
# Ubuntu/Debian
apt install libpqxx-dev

# vcpkg
vcpkg install libpqxx

# Conan
conan install libpqxx/7.7.5
```

### 基础使用

```cpp
#include <pqxx/pqxx>
#include <iostream>
#include <string>

int main() {
    try {
        // 连接数据库
        pqxx::connection conn(
            "host=localhost "
            "port=5432 "
            "dbname=testdb "
            "user=myuser "
            "password=mypass"
        );

        // 执行 SQL
        pqxx::work w(conn);

        // 创建表
        w.exec("CREATE TABLE IF NOT EXISTS users ("
               "id SERIAL PRIMARY KEY, "
               "name TEXT NOT NULL, "
               "age INTEGER)");

        // 插入并返回 ID
        pqxx::row r = w.exec1(
            "INSERT INTO users (name, age) VALUES ('Alice', 30) RETURNING id"
        );
        int new_id = r[0].as<int>();
        std::cout << "Inserted with id: " << new_id << std::endl;

        // 参数化查询
        std::string name = "Bob";
        int age = 25;
        w.exec("INSERT INTO users (name, age) VALUES ($1, $2)",
               pqxx::params(name, age));

        w.commit();  // 提交事务
    } catch (const pqxx::sql_error& e) {
        std::cerr << "SQL error: " << e.what() << std::endl;
        std::cerr << "Query: " << e.query() << std::endl;
    } catch (const std::exception& e) {
        std::cerr << "Error: " << e.what() << std::endl;
    }
}
```

### 查询结果处理

```cpp
#include <pqxx/pqxx>

void query_example(pqxx::connection& conn) {
    // 使用参数化查询
    int min_age = 20;
    pqxx::result res = conn.exec(
        "SELECT id, name, age FROM users WHERE age >= $1 ORDER BY age",
        pqxx::params(min_age)
    );

    // 遍历结果
    for (auto row : res) {
        int id = row["id"].as<int>();
        std::string name = row["name"].as<std::string>();
        int age = row["age"].as<int>();

        std::cout << id << ": " << name << ", " << age << std::endl;
    }

    // 使用索引访问
    for (auto row : res) {
        int id = row[0].as<int>();
        std::string name = row[1].as<std::string>();
    }

    // 检查 NULL
    for (auto row : res) {
        if (!row["name"].is_null()) {
            std::string name = row["name"].as<std::string>();
        }
    }
}
```

### 事务处理

```cpp
#include <pqxx/pqxx>

void transaction_example(pqxx::connection& conn) {
    try {
        // 启动事务
        pqxx::work w(conn);

        // 在事务中执行
        w.exec("INSERT INTO users (name) VALUES ('Carol')");
        w.exec("INSERT INTO users (name) VALUES ('Dave')");

        // 准备语句（预编译）
        w.prepare(
            "insert_user",
            "INSERT INTO users (name, age) VALUES ($1, $2)"
        );
        w.exec_prepared("insert_user", "Eve", 35);

        w.commit();  // 提交
    } catch (const pqxx::sql_error& e) {
        std::cerr << "Transaction failed: " << e.what() << std::endl;
        // 析构函数会自动回滚未提交的事务
    }
}
```

## ORM 框架：Drogon-ORM

Drogon 是一个现代 C++ Web 框架，内置 ORM。

### 模型定义

```cpp
#include <drogon/orm/SqlBinder.h>

// User.h
#pragma once
#include <drogon/orm/Model.h>

class User : public drogon::orm::Model<User> {
public:
    int id;
    std::string name;
    std::string email;
    int age;

    // 表名
    static constexpr const char* TableName = "users";

    // 列定义
    enum Field {
        F_id,
        F_name,
        F_email,
        F_age
    };
};
```

### CRUD 操作

```cpp
#include <drogon/orm/Mapper.h>
#include <drogon/orm/Criteria.h>

using namespace drogon;

void orm_example() {
    orm::Mapper<User> mapper(app().getDbClient());

    // 插入
    User new_user;
    new_user.name = "Alice";
    new_user.email = "alice@example.com";
    new_user.age = 30;
    mapper.insert(new_user);

    // 查询单个
    auto user = mapper.findByPrimaryKey(1);

    // 条件查询
    auto users = mapper.findBy(
        Criteria(User::Field::age, CompareOperator::GT, 25)
    );

    // 分页查询
    auto paged = mapper.paginate(1, 10);  // 第1页，每页10条

    // 更新
    user.age = 31;
    mapper.update(user);

    // 删除
    mapper.deleteByPrimaryKey(1);
}
```

## 数据库连接池

### SOCI 连接池

```cpp
#include <soci/soci.h>
#include <soci/sqlite3/soci-sqlite3.h>

int main() {
    // 创建连接池
    soci::connection_pool pool(4);  // 4 个连接

    for (int i = 0; i < 4; ++i) {
        pool.at(i).open(soci::sqlite3, "test.db");
    }

    // 使用连接
    {
        soci::session sql(pool);
        sql << "SELECT COUNT(*) FROM users";
    }  // 连接归还池中
}
```

### libpqxx 连接池（自定义实现）

```cpp
#include <pqxx/pqxx>
#include <queue>
#include <mutex>

class PgPool {
    std::queue<std::unique_ptr<pqxx::connection>> pool_;
    std::mutex mutex_;
    size_t max_size_;

public:
    PgPool(const std::string& conn_info, size_t size = 10)
        : max_size_(size) {
        for (size_t i = 0; i < size; ++i) {
            pool_.push(std::make_unique<pqxx::connection>(conn_info));
        }
    }

    class Guard {
        PgPool& pool_;
        std::unique_ptr<pqxx::connection> conn_;
    public:
        Guard(PgPool& p, std::unique_ptr<pqxx::connection> conn)
            : pool_(p), conn_(std::move(conn)) {}

        pqxx::connection& get() { return *conn_; }

        ~Guard() {
            std::lock_guard<std::mutex> lock(pool_.mutex_);
            pool_.pool_.push(std::move(conn_));
        }
    };

    Guard get() {
        std::lock_guard<std::mutex> lock(mutex_);
        auto conn = std::move(pool_.front());
        pool_.pop();
        return Guard(*this, std::move(conn));
    }
};
```

## 数据库选择指南

| 数据库 | 适用场景 | 库 |
|--------|----------|-----|
| SQLite | 嵌入式、轻量级、本地存储 | sqlite3 |
| PostgreSQL | 企业级应用、复杂查询 | libpqxx |
| MySQL/MariaDB | Web 应用、高并发 | mysqlclient, SOCI |
| Oracle | 企业核心系统 | OCI |
| Redis | 缓存、实时数据 | hiredis |

## 最佳实践

### 防止 SQL 注入

```cpp
// 总是使用参数化查询
sql << "SELECT * FROM users WHERE name = $1",
    pqxx::params(user_input);  // 安全

// 避免字符串拼接
// 错误
// sql << "SELECT * FROM users WHERE name = '" + user_input + "'";  // 不安全
```

### 异常处理

```cpp
#include <drogon/orm/DbClient.h>

try {
    // 数据库操作
} catch (const drogon::orm::DbException& e) {
    // 连接错误、超时等
    std::cerr << "Database error: " << e.what() <<  std::endl;
}
```

### 性能优化

```cpp
// 使用预编译语句
sql.prepare("select_user", "SELECT * FROM users WHERE id = $1");

// 批量插入
pqxx::copy_writer copy = w.copy_init("COPY users FROM STDIN");

copy.complete();
```
