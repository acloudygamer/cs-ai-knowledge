# 面向对象

## 类与对象

### 基本定义

```cpp
class Rectangle {
    // 默认 private
    double width_;
    double height_;

public:
    // 构造函数
    Rectangle() : width_(0), height_(0) {}
    Rectangle(double w, double h) : width_(w), height_(h) {}

    // 成员函数
    double area() const { return width_ * height_; }
    double perimeter() const { return 2 * (width_ + height_); }

    // setter
    void set_width(double w) { width_ = w; }
    void set_height(double h) { height_ = h; }

    // getter（const 成员函数）
    double width() const { return width_; }
    double height() const { return height_; }
};

// 使用
Rectangle r(10, 5);
std::cout << r.area() << std::endl;  // 50
```

### 构造函数的分类与初始化

```cpp
class Person {
    std::string name_;
    int age_;

public:
    // 默认构造函数
    Person() : name_(""), age_(0) {}

    // 参数化构造函数
    explicit Person(const std::string& name, int age = 0)
        : name_(name), age_(age) {}

    // 拷贝构造函数
    Person(const Person& other)
        : name_(other.name_), age_(other.age_) {}

    // 移动构造函数（C++11）
    Person(Person&& other) noexcept
        : name_(std::move(other.name_)), age_(other.age_) {
        // other 被置为有效但未定义的状态
    }

    // 拷贝赋值
    Person& operator=(const Person& other) {
        if (this != &other) {
            name_ = other.name_;
            age_ = other.age_;
        }
        return *this;
    }

    // 移动赋值（C++11）
    Person& operator=(Person&& other) noexcept {
        if (this != &other) {
            name_ = std::move(other.name_);
            age_ = other.age_;
        }
        return *this;
    }

    ~Person() {
        // 析构函数，释放资源
    }
};

// 统一初始化（C++11）
Person p1;                       // 默认构造
Person p2{"Alice", 30};         // 参数化构造
Person p3 = Person{"Bob", 25};  // 显式构造
Person p4{p3};                   // 拷贝构造
Person p5 = std::move(p2);      // 移动构造
```

### 类的静态成员

```cpp
class Counter {
    int count_;
    static int total_count_;  // 所有对象共享

public:
    Counter() : count_(0) { ++total_count_; }
    ~Counter() { --total_count_; }

    static int get_total() { return total_count_; }  // 静态成员函数
    int get_count() const { return count_; }
};

// 静态成员必须在类外定义和初始化
int Counter::total_count_ = 0;

Counter c1, c2;
std::cout << Counter::get_total() << std::endl;  // 2
```

## 继承

### 基本语法

```cpp
// 基类
class Animal {
protected:  // 派生类可访问，外部不可访问
    std::string name_;

public:
    Animal(const std::string& name) : name_(name) {}
    virtual ~Animal() {}  // 基类析构函数应为 virtual

    virtual void speak() const {
        std::cout << name_ << " makes a sound" << std::endl;
    }

    std::string get_name() const { return name_; }
};

// 派生类
class Dog : public Animal {
    std::string breed_;

public:
    Dog(const std::string& name, const std::string& breed)
        : Animal(name), breed_(breed) {}

    void speak() const override {
        std::cout << name_ << " barks: Woof!" << std::endl;
    }

    void fetch() const {
        std::cout << name_ << " fetches the ball" << std::endl;
    }
};

Animal* animal = new Dog("Buddy", "Golden Retriever");
animal->speak();  // 调用 Dog::speak()，输出 "Buddy barks: Woof!"
delete animal;
```

### 继承方式

| 继承方式 | 基类 public | 基类 protected | 基类 private |
|---------|-------------|-----------------|--------------|
| `public` | public | protected | 不可访问 |
| `protected` | protected | protected | 不可访问 |
| `private` | private | private | 不可访问 |

```cpp
// public 继承：保持基类成员的访问级别
class Dog : public Animal {
    // name_ 保持 protected，get_name() 保持 public
};

// protected 继承：基类 public 成员变为 protected
class GuardDog : protected Animal {
    // name_ 和 get_name() 都变为 protected
};

// private 继承：基类所有成员变为 private
class SecretDog : private Animal {
    // name_ 和 get_name() 都变为 private
};
```

### 多继承

```cpp
class Flyable {
public:
    virtual void fly() const = 0;
    virtual ~Flyable() {}
};

class Bird : public Animal {
public:
    Bird(const std::string& name) : Animal(name) {}
};

class Sparrow : public Bird, public Flyable {
public:
    Sparrow(const std::string& name) : Bird(name) {}

    void fly() const override {
        std::cout << get_name() << " flies" << std::endl;
    }
};

// 菱形继承问题需要虚继承解决
//     Base
//    /    \
//   A      B
//    \    /
//     C
// 若 A、B 非虚继承 Base，则 C 会包含两份 Base 子对象，导致歧义
class Base {
public:
    int value_;
};

class A : virtual public Base {};  // 虚继承：Base 在 A 中为共享子对象
class B : virtual public Base {};  // 虚继承：Base 在 B 中为共享子对象
class C : public A, public B {
    // 虚继承后，C 只有一个 Base 子对象，消除歧义
};
```

## 多态

### 虚函数与动态绑定

```cpp
class Shape {
public:
    // 虚函数：允许派生类重写
    virtual double area() const = 0;  // 纯虚函数
    virtual ~Shape() {}
};

class Circle : public Shape {
    double radius_;

public:
    Circle(double r) : radius_(r) {}

    double area() const override {
        return 3.14159 * radius_ * radius_;
    }
};

class Rectangle : public Shape {
    double width_, height_;

public:
    Rectangle(double w, double h) : width_(w), height_(h) {}

    double area() const override {
        return width_ * height_;
    }
};

// 使用基类指针调用派生类实现
std::vector<Shape*> shapes;
shapes.push_back(new Circle(5));
shapes.push_back(new Rectangle(4, 6));

for (const auto* s : shapes) {
    std::cout << s->area() << std::endl;  // 动态分派
}

// 使用 unique_ptr 更安全
std::vector<std::unique_ptr<Shape>> shapes2;
shapes2.push_back(std::make_unique<Circle>(5));
shapes2.push_back(std::make_unique<Rectangle>(4, 6));
```

### override 与 final（C++11）

```cpp
class Base {
public:
    virtual void foo() const {}
    virtual void bar() {}
};

// override：明确表示要重写基类虚函数，编译器检查
class Derived : public Base {
public:
    void foo() const override {}  // 编译器确保基类有可重写的 foo()
    void bar() override {}         // 如果基类没有 virtual void bar()，编译错误
};

// final：禁止进一步重写或继承
class NoOverride final {
    virtual void process() final {}  // 禁止派生类重写
};

class NoInherit : public Base {};  // 禁止继承
class Blocked : public NoInherit {};  // 编译错误
```

### 抽象类与纯虚函数

```cpp
// 含有纯虚函数的类是抽象类，不能直接实例化
class Drawable {
public:
    virtual void draw() const = 0;  // 纯虚函数
    virtual ~Drawable() {}
};

// 派生类必须实现所有纯虚函数才能实例化
class Line : public Drawable {
    Point p1_, p2_;

public:
    void draw() const override {
        std::cout << "Drawing line" << std::endl;
    }
};

Drawable d;           // 错误！抽象类不能实例化
Drawable* ptr = &line; // 可以，指向派生类对象
```

### 虚析构函数

**基类析构函数必须为 virtual**，否则通过基类指针删除派生类对象时，派生类析构函数不会被调用：

```cpp
class Base {
public:
    Base() { std::cout << "Base constructor\n"; }
    // 非 virtual 析构函数
    ~Base() { std::cout << "Base destructor\n"; }
};

class Derived : public Base {
    int* data_;

public:
    Derived() : data_(new int[100]) {
        std::cout << "Derived constructor\n";
    }
    ~Derived() {
        delete[] data_;
        std::cout << "Derived destructor\n";
    }
};

Base* ptr = new Derived();
delete ptr;  // 只调用 Base::~Base()，内存泄漏！Derived 析构函数未调用

// 正确做法：
class Base2 {
public:
    virtual ~Base2() {}
};

class Derived2 : public Base2 {
    ~Derived2() {}
};

Base2* ptr2 = new Derived2();
delete ptr2;  // 正确调用 Derived2::~Derived2() 然后 Base2::~Base2()
```

## 访问控制

### 访问级别

| 关键字 | 自身类 | 友元 | 派生类 | 其他 |
|--------|--------|------|--------|------|
| `private` | Yes | Yes | No | No |
| `protected` | Yes | Yes | Yes | No |
| `public` | Yes | Yes | Yes | Yes |

```cpp
class MyClass {
    int private_member_;      // 仅自身类可访问
protected:
    int protected_member_;    // 自身 + 派生类可访问
public:
    int public_member_;       // 所有人可访问
};
```

### 友元（friend）

```cpp
class Person {
    std::string name_;
    int age_;

public:
    Person(const std::string& name, int age)
        : name_(name), age_(age) {}

    // 声明友元函数
    friend void print_person(const Person& p);
    friend class PersonPrinter;  // 友元类
};

// 友元函数可以访问 private 成员
void print_person(const Person& p) {
    // 可以访问 p.name_ 和 p.age_
    std::cout << p.name_ << ", " << p.age_ << std::endl;
}

// 友元类
class PersonPrinter {
public:
    void print(const Person& p) {
        // 可以访问 Person 的所有 private 成员
        std::cout << p.name_ << " is " << p.age_ << " years old" << std::endl;
    }
};

// 注意：友元破坏封装，慎用
```

### = default 与 = delete

```cpp
class NonCopyable {
    std::vector<int> data_;

public:
    NonCopyable() = default;  // 使用编译器生成的默认实现
    NonCopyable(const NonCopyable&) = delete;  // 禁止拷贝
    NonCopyable& operator=(const NonCopyable&) = delete;  // 禁止拷贝赋值

    NonCopyable(NonCopyable&&) = default;  // 允许移动
    NonCopyable& operator=(NonCopyable&&) = default;
};

NonCopyable obj1;
NonCopyable obj2 = obj1;  // 编译错误！拷贝被禁用
```

### 隐式生成的特殊成员函数行为

| 情况 | 拷贝构造 | 移动构造 |
|------|---------|---------|
| 没有定义任何特殊成员函数 | 逐成员拷贝 | 逐成员移动 |
| 定义了拷贝构造 | 不生成 | 不生成 |
| 定义了移动构造 | 不生成 | 不生成 |
| 定义了拷贝构造或移动构造 | - | 需要手动定义另一个 |

```cpp
class Example {
    std::vector<int> vec_;

public:
    Example() = default;  // 默认构造
    // 编译器不再生成拷贝构造和移动构造，需要手动定义
};
```

## 嵌套类

```cpp
class Outer {
    int outer_value_;

public:
    class Inner {
        int inner_value_;

    public:
        explicit Inner(int v) : inner_value_(v) {}
        void process(const Outer& o) {
            // 可以访问 o.outer_value_ 如果是 public
        }
    };

    Inner create_inner(int v) { return Inner(v); }
};

Outer::Inner i(10);  // 嵌套类可以这样构造
```

## 结构体与类的区别

```cpp
// 结构体：默认 public 继承和成员
struct PointStruct {
    int x;  // 默认 public
    int y;
    PointStruct(int x, int y) : x(x), y(y) {}
};

// 类：默认 private 继承和成员
class PointClass {
    int x;  // 默认 private
    int y;
public:
    PointClass(int x, int y) : x(x), y(y) {}
};

// 通常：结构体用于 POD 数据，类用于具有行为的对象
```
