# C++ 应用场景

C++ 是应用最广泛的系统级编程语言之一，在对性能要求苛刻的领域几乎是不二之选。本章详细介绍 C++ 的主要应用领域及其特点。

## 游戏开发

游戏行业是 C++ 最大的应用领域之一。主流游戏引擎如 **Unreal Engine** 和 **Unity**（核心部分）都使用 C++ 开发。

### 为什么游戏偏爱 C++

```cpp
// 游戏开发中的典型场景
```

```cpp
    std::chrono::steady_clock::time_point last_time_;

    void run() {
        last_time_ = std::chrono::steady_clock::now();
```

```cpp
    volatile uint32_t* control_reg_;
    volatile uint32_t* data_reg_;

public:
    bool initialize(uintptr_t mmio_base) {
        control_reg_ = (volatile uint32_t*)mmio_base;
        data_reg_ = (volatile uint32_t*)(mmio_base + 4);

        // 重置设备
        *control_reg_ = 0x01;
        return wait_for_ready();
    }

    size_t read(void* buffer, size_t size) {
        char* dest = static_cast<char*>(buffer);
        for (size_t i = 0; i < size; ++i) {
            // 轮询等待数据就绪
            while (!(*control_reg_ & 0x02)) {}
            dest[i] = (char)*data_reg_;
        }
        return size;
    }
};
```

```cpp
// 典型的 IoT 网关结构
class IoTGateway {
    std::array<uint8_t, 64> rx_buffer_;
    std::array<uint8_t, 64> tx_buffer_;
    size_t buffer_index_ = 0;

public:
    enum class State {
        IDLE,
        HEADER,
        PAYLOAD,
        CHECKSUM
    };

    State current_state_ = State::IDLE;

    bool validate_checksum() {
        return true;  // 示例实现
    }

    void handle_message() {
        // 处理消息
    }

    void process_byte(uint8_t byte) {
        switch (current_state_) {
            case State::IDLE:
                if (byte == 0xAA) {  // 帧头
                    buffer_index_ = 0;
                    rx_buffer_[buffer_index_++] = byte;
                    current_state_ = State::HEADER;
                }
                break;

            case State::HEADER:
                rx_buffer_[buffer_index_++] = byte;
                if (buffer_index_ >= 2) {
                    current_state_ = State::PAYLOAD;
                }
                break;

            case State::PAYLOAD:
                rx_buffer_[buffer_index_++] = byte;
                if (buffer_index_ >= 60) {
                    current_state_ = State::CHECKSUM;
                }
                break;

            case State::CHECKSUM:
                if (buffer_index_ < rx_buffer_.size()) {
                    rx_buffer_[buffer_index_++] = byte;
                }
                if (validate_checksum()) {
                    handle_message();
                }
                current_state_ = State::IDLE;
                break;
        }
    }
};
```

```cpp
template<typename T>
class LockFreeQueue {
    struct Node {
        std::atomic<Node*> next;
        std::atomic<T> data;
    };

    std::atomic<Node*> head_;
    std::atomic<Node*> tail_;

public:
    void enqueue(T value) {
        Node* node = new Node{{nullptr}, {value}};

        while (true) {
            Node* tail = tail_.load();
            Node* next = tail->next.load();

            if (tail == tail_.load()) {
                if (next == nullptr) {
                    if (tail->next.compare_exchange_weak(next, node)) {
                        tail_.compare_exchange_weak(tail, node);
                        return;
                    }
                } else {
                    tail_.compare_exchange_weak(tail, next);
                }
            }
        }
    }
};
```

## 图形图像处理

### 图像处理基础

```cpp
#include <cstdint>
#include <vector>
#include <algorithm>

// 像素结构
struct Pixel {
    uint8_t r, g, b, a;
};

class Image {
    uint32_t width_;
    uint32_t height_;
    std::vector<Pixel> pixels_;

public:
    // 卷积操作
    void apply_convolution(const std::vector<float>& kernel, size_t ksize) {
        std::vector<Pixel> result(pixels_.size());

        for (uint32_t y = 0; y < height_; ++y) {
            for (uint32_t x = 0; x < width_; ++x) {
                float sum_r = 0, sum_g = 0, sum_b = 0;

                for (size_t ky = 0; ky < ksize; ++ky) {
                    for (size_t kx = 0; kx < ksize; ++kx) {
                        int px = std::clamp<int>(x + kx - ksize/2, 0, width_-1);
                        int py = std::clamp<int>(y + ky - ksize/2, 0, height_-1);
                        const Pixel& p = pixels_[py * width_ + px];

                        float k = kernel[ky * ksize + kx];
                        sum_r += p.r * k;
                        sum_g += p.g * k;
                        sum_b += p.b * k;
                    }
                }

                Pixel& out = result[y * width_ + x];
                out.r = std::clamp<uint8_t>(sum_r, 0, 255);
                out.g = std::clamp<uint8_t>(sum_g, 0, 255);
                out.b = std::clamp<uint8_t>(sum_b, 0, 255);
                out.a = 255;
            }
        }

        pixels_ = std::move(result);
    }
};
```

```cpp
        Matrix result(rows_, other.cols_);

        for (size_t i = 0; i < rows_; ++i) {
```
