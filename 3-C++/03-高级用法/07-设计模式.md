# 设计模式

设计模式是针对特定问题的通用解决方案。C++ 支持多种编程范式，使实现设计模式更加灵活。

## 创建型模式

### 单例模式（Singleton）

确保一个类只有一个实例，并提供全局访问点：

```cpp
// 线程安全的单例（懒汉式，C++11）
class Singleton {
    static std::unique_ptr<Singleton> instance_;
    static std::mutex mutex_;

    Singleton() = default;
    Singleton(const Singleton&) = delete;
    Singleton& operator=(const Singleton&) = delete;

public:
    static Singleton& get_instance() {
        std::lock_guard<std::mutex> lock(mutex_);
        if (!instance_) {
            instance_ = std::unique_ptr<Singleton>(new Singleton());
        }
        return *instance_;
    }

    void do_something() {}
};

// 双重检查锁定（优化）
class SingletonDCL {
    static std::atomic<SingletonDCL*> instance_;
    static std::mutex mutex_;

public:
    static SingletonDCL* get_instance() {
        SingletonDCL* tmp = instance_.load(std::memory_order_acquire);
        if (tmp == nullptr) {
            std::lock_guard<std::mutex> lock(mutex_);
            tmp = instance_.load(std::memory_order_relaxed);
            if (tmp == nullptr) {
                tmp = new SingletonDCL();
                instance_.store(tmp, std::memory_order_release);
            }
        }
        return tmp;
    }
};

// 使用 C++11 static 局部变量（最简单，线程安全）
class SimpleSingleton {
public:
    static SimpleSingleton& get_instance() {
        static SimpleSingleton instance;
        return instance;
    }

private:
    SimpleSingleton() = default;
    SimpleSingleton(const SimpleSingleton&) = delete;
    SimpleSingleton& operator=(const SimpleSingleton&) = delete;
};
```

### 工厂方法模式（Factory Method）

定义创建对象的接口，让子类决定实例化哪个类：

```cpp
// 产品接口
class Shape {
public:
    virtual ~Shape() = default;
    virtual void draw() const = 0;
};

// 具体产品
class Circle : public Shape {
public:
    void draw() const override { std::cout << "Drawing Circle\n"; }
};

class Square : public Shape {
public:
    void draw() const override { std::cout << "Drawing Square\n"; }
};

// 工厂接口
class ShapeFactory {
public:
    virtual ~ShapeFactory() = default;
    virtual std::unique_ptr<Shape> create_shape() const = 0;
};

// 具体工厂
class CircleFactory : public ShapeFactory {
public:
    std::unique_ptr<Shape> create_shape() const override {
        return std::make_unique<Circle>();
    }
};

class SquareFactory : public ShapeFactory {
public:
    std::unique_ptr<Shape> create_shape() const override {
        return std::make_unique<Square>();
    }
};

// 使用
void draw_shape(const ShapeFactory& factory) {
    auto shape = factory.create_shape();
    shape->draw();
}

int main() {
    CircleFactory cf;
    SquareFactory sf;
    draw_shape(cf);  // Drawing Circle
    draw_shape(sf);  // Drawing Square
}
```

### 建造者模式（Builder）

分步创建复杂对象：

```cpp
class Person {
    std::string name_;
    int age_ = 0;
    std::string address_;
    std::string phone_;

public:
    friend class PersonBuilder;  // 允许 Builder 访问私有成员

    void print() const {
        std::cout << "Name: " << name_
                  << ", Age: " << age_;
        if (!address_.empty()) std::cout << ", Address: " << address_;
        if (!phone_.empty()) std::cout << ", Phone: " << phone_;
        std::cout << "\n";
    }
};

class PersonBuilder {
protected:
    Person person_;  // 构建中的对象

public:
    PersonBuilder() = default;
    virtual ~PersonBuilder() = default;

    PersonBuilder& name(const std::string& n) {
        person_.name_ = n;
        return *this;
    }

    PersonBuilder& age(int a) {
        person_.age_ = a;
        return *this;
    }

    PersonBuilder& address(const std::string& addr) {
        person_.address_ = addr;
        return *this;
    }

    PersonBuilder& phone(const std::string& p) {
        person_.phone_ = p;
        return *this;
    }

    Person build() { return std::move(person_); }
};

int main() {
    PersonBuilder builder;

    Person p1 = builder.name("Alice").age(30).build();
    Person p2 = builder.name("Bob").age(25).address("123 Main St").phone("555-1234").build();

    p1.print();  // Name: Alice, Age: 30
    p2.print();  // Name: Bob, Age: 25, Address: 123 Main St, Phone: 555-1234
}
```

## 结构型模式

### 适配器模式（Adapter）

将一个类的接口转换成客户期望的另一个接口：

```cpp
// 目标接口
class MediaPlayer {
public:
    virtual ~MediaPlayer() = default;
    virtual void play(const std::string& filename) const = 0;
};

// 需要适配的类（Adaptee）
class AdvancedMediaPlayer {
public:
    void play_mp4(const std::string& filename) {
        std::cout << "Playing MP4: " << filename << "\n";
    }
    void play_vlc(const std::string& filename) {
        std::cout << "Playing VLC: " << filename << "\n";
    }
};

// 适配器
class MediaAdapter : public MediaPlayer {
    AdvancedMediaPlayer advanced_player_;
    std::string format_;

public:
    MediaAdapter(const std::string& format) : format_(format) {}

    void play(const std::string& filename) const override {
        if (format_ == "mp4") {
            advanced_player_.play_mp4(filename);
        } else if (format_ == "vlc") {
            advanced_player_.play_vlc(filename);
        }
    }
};

// 使用适配器
class Player : public MediaPlayer {
public:
    void play(const std::string& filename) const override {
        if (filename.find(".mp4") != std::string::npos) {
            MediaAdapter adapter("mp4");
            adapter.play(filename);
        } else if (filename.find(".vlc") != std::string::npos) {
            MediaAdapter adapter("vlc");
            adapter.play(filename);
        } else {
            std::cout << "Unsupported format: " << filename << "\n";
        }
    }
};
```

### 装饰器模式（Decorator）

动态给对象添加额外职责：

```cpp
#include <iostream>
#include <string>

// 组件接口
class Coffee {
public:
    virtual ~Coffee() = default;
    virtual int get_cost() const = 0;
    virtual std::string get_description() const = 0;
};

// 具体组件
class SimpleCoffee : public Coffee {
public:
    int get_cost() const override { return 10; }
    std::string get_description() const override { return "Simple Coffee"; }
};

// 装饰器基类
class CoffeeDecorator : public Coffee {
protected:
    std::unique_ptr<Coffee> coffee_;

public:
    explicit CoffeeDecorator(std::unique_ptr<Coffee> coffee)
        : coffee_(std::move(coffee)) {}
};

// 具体装饰器
class MilkDecorator : public CoffeeDecorator {
public:
    using CoffeeDecorator::CoffeeDecorator;

    int get_cost() const override {
        return coffee_->get_cost() + 5;
    }

    std::string get_description() const override {
        return coffee_->get_description() + ", Milk";
    }
};

class SugarDecorator : public CoffeeDecorator {
public:
    using CoffeeDecorator::CoffeeDecorator;

    int get_cost() const override {
        return coffee_->get_cost() + 2;
    }

    std::string get_description() const override {
        return coffee_->get_description() + ", Sugar";
    }
};

int main() {
    std::unique_ptr<Coffee> coffee = std::make_unique<SimpleCoffee>();
    std::cout << coffee->get_description() << ": " << coffee->get_cost() << "\n";
    // Simple Coffee: 10

    coffee = std::make_unique<MilkDecorator>(std::move(coffee));
    std::cout << coffee->get_description() << ": " << coffee->get_cost() << "\n";
    // Simple Coffee, Milk: 15

    coffee = std::make_unique<SugarDecorator>(std::move(coffee));
    std::cout << coffee->get_description() << ": " << coffee->get_cost() << "\n";
    // Simple Coffee, Milk, Sugar: 17
}
```

### 组合模式（Composite）

将对象组合成树形结构表示"部分-整体"层次：

```cpp
class Graphic {
public:
    virtual ~Graphic() = default;
    virtual void draw() const = 0;
    virtual double price() const = 0;
};

class Circle : public Graphic {
    double radius_;

public:
    explicit Circle(double r) : radius_(r) {}
    void draw() const override { std::cout << "Circle(r=" << radius_ << ")\n"; }
    double price() const override { return 3.14 * radius_ * radius_; }
};

class Rectangle : public Graphic {
    double width_, height_;

public:
    Rectangle(double w, double h) : width_(w), height_(h) {}
    void draw() const override { std::cout << "Rectangle(" << width_ << "x" << height_ << ")\n"; }
    double price() const override { return width_ * height_; }
};

// 组合图形
class CompositeGraphic : public Graphic {
    std::vector<std::unique_ptr<Graphic>> children_;

public:
    void add(std::unique_ptr<Graphic> g) {
        children_.push_back(std::move(g));
    }

    void draw() const override {
        for (const auto& child : children_) {
            child->draw();
        }
    }

    double price() const override {
        double total = 0;
        for (const auto& child : children_) {
            total += child->price();
        }
        return total;
    }
};

int main() {
    auto circle1 = std::make_unique<Circle>(1.0);
    auto circle2 = std::make_unique<Circle>(2.0);
    auto rect = std::make_unique<Rectangle>(3.0, 4.0);

    CompositeGraphic group;
    group.add(std::move(circle1));
    group.add(std::move(circle2));

    CompositeGraphic scene;
    scene.add(std::move(&group));
    scene.add(std::move(rect));

    scene.draw();
    std::cout << "Total price: " << scene.price() << "\n";
}
```

### 代理模式（Proxy）

为其他对象提供一种代理以控制对这个对象的访问：

```cpp
// 主题接口
class Image {
public:
    virtual ~Image() = default;
    virtual void display() const = 0;
};

// 真实对象（可能开销很大）
class RealImage : public Image {
    std::string filename_;

public:
    explicit RealImage(const std::string& filename) : filename_(filename) {
        load_from_disk();  // 模拟昂贵的加载操作
    }

    void display() const override {
        std::cout << "Displaying: " << filename_ << "\n";
    }

private:
    void load_from_disk() {
        std::cout << "Loading: " << filename_ << " from disk...\n";
    }
};

// 代理对象（延迟加载）
class ImageProxy : public Image {
    std::string filename_;
    mutable std::unique_ptr<RealImage> real_image_;

public:
    explicit ImageProxy(const std::string& filename)
        : filename_(filename), real_image_(nullptr) {}

    void display() const override {
        if (!real_image_) {
            real_image_ = std::make_unique<RealImage>(filename_);
        }
        real_image_->display();
    }
};

int main() {
    std::unique_ptr<Image> image = std::make_unique<ImageProxy>("photo.jpg");

    std::cout << "Image created (RealImage not loaded yet)\n";
    // 在这个点，图片还没有从磁盘加载

    image->display();  // 第一次调用会触发加载
    image->display();  // 第二次调用直接使用已加载的图片
}
```

## 行为型模式

### 观察者模式（Observer）

定义对象间的一对多依赖关系：

```cpp
#include <vector>
#include <memory>

class Observer {
public:
    virtual ~Observer() = default;
    virtual void update(int value) = 0;
};

class Subject {
    std::vector<std::weak_ptr<Observer>> observers_;
    int state_ = 0;

public:
    void attach(std::shared_ptr<Observer> observer) {
        observers_.push_back(observer);
    }

    void detach(std::weak_ptr<Observer> observer) {
        // 需要清理已释放的 weak_ptr
        for (auto it = observers_.begin(); it != observers_.end(); ) {
            if (it->expired()) {
                it = observers_.erase(it);
            } else {
                ++it;
            }
        }
    }

    void set_state(int state) {
        state_ = state;
        notify();
    }

    void notify() {
        for (auto& observer : observers_) {
            if (auto obs = observer.lock()) {
                obs->update(state_);
            }
        }
    }
};

class ConcreteObserver : public Observer {
    std::string name_;

public:
    explicit ConcreteObserver(const std::string& name) : name_(name) {}

    void update(int value) override {
        std::cout << name_ << " received update: " << value << "\n";
    }
};

int main() {
    Subject subject;

    auto obs1 = std::make_shared<ConcreteObserver>("Observer1");
    auto obs2 = std::make_shared<ConcreteObserver>("Observer2");

    subject.attach(obs1);
    subject.attach(obs2);

    subject.set_state(42);
    // Observer1 received update: 42
    // Observer2 received update: 42
}
```

### 策略模式（Strategy）

定义一系列算法，使它们可以相互替换：

```cpp
class SortStrategy {
public:
    virtual ~SortStrategy() = default;
    virtual void sort(std::vector<int>& vec) const = 0;
};

class BubbleSort : public SortStrategy {
public:
    void sort(std::vector<int>& vec) const override {
        std::cout << "BubbleSort\n";
        for (size_t i = 0; i < vec.size(); ++i) {
            for (size_t j = 0; j < vec.size() - i - 1; ++j) {
                if (vec[j] > vec[j + 1]) {
                    std::swap(vec[j], vec[j + 1]);
                }
            }
        }
    }
};

class QuickSort : public SortStrategy {
public:
    void sort(std::vector<int>& vec) const override {
        std::cout << "QuickSort\n";
        std::sort(vec.begin(), vec.end());  // 实际用 STL
    }
};

class Context {
    std::unique_ptr<SortStrategy> strategy_;

public:
    void set_strategy(std::unique_ptr<SortStrategy> strategy) {
        strategy_ = std::move(strategy);
    }

    void sort(std::vector<int>& vec) {
        if (strategy_) {
            strategy_->sort(vec);
        }
    }
};

int main() {
    Context ctx;
    std::vector<int> data = {3, 1, 4, 1, 5, 9, 2, 6};

    ctx.set_strategy(std::make_unique<BubbleSort>());
    ctx.sort(data);

    ctx.set_strategy(std::make_unique<QuickSort>());
    ctx.sort(data);
}
```

### 命令模式（Command）

将请求封装成对象：

```cpp
class Document {
    std::string content_;

public:
    void append(const std::string& text) { content_ += text; }
    void remove(size_t pos, size_t len) { content_.erase(pos, len); }
    std::string get() const { return content_; }
};

class Command {
public:
    virtual ~Command() = default;
    virtual void execute() = 0;
    virtual void undo() = 0;
};

class AppendCommand : public Command {
    Document& doc_;
    std::string text_;

public:
    AppendCommand(Document& doc, const std::string& text)
        : doc_(doc), text_(text) {}

    void execute() override { doc_.append(text_); }
    void undo() override { doc_.remove(doc_.get().size() - text_.size(), text_.size()); }
};

class CommandManager {
    std::vector<std::unique_ptr<Command>> commands_;
    size_t current_ = 0;

public:
    void execute(std::unique_ptr<Command> cmd) {
        cmd->execute();
        // 删除当前位置之后的历史（用于不支持重做的简单实现）
        commands_.erase(commands_.begin() + current_, commands_.end());
        commands_.push_back(std::move(cmd));
        current_ = commands_.size();
    }

    void undo() {
        if (current_ > 0) {
            --current_;
            commands_[current_]->undo();
        }
    }

    void redo() {
        if (current_ < commands_.size()) {
            commands_[current_]->execute();
            ++current_;
        }
    }
};

int main() {
    Document doc;
    CommandManager cm;

    cm.execute(std::make_unique<AppendCommand>(doc, "Hello"));
    cm.execute(std::make_unique<AppendCommand>(doc, " World"));
    std::cout << doc.get() << "\n";  // Hello World

    cm.undo();
    std::cout << doc.get() << "\n";  // Hello

    cm.redo();
    std::cout << doc.get() << "\n";  // Hello World
}
```

### 迭代器模式（Iterator）

提供一种顺序访问集合元素的方法：

```cpp
template <typename T>
class Iterator {
public:
    virtual ~Iterator() = default;
    virtual bool has_next() const = 0;
    virtual T next() = 0;
};

template <typename T>
class List {
    std::vector<T> data_;

public:
    void add(const T& item) { data_.push_back(item); }

    class ListIterator : public Iterator<T> {
        const List& list_;
        size_t pos_ = 0;

    public:
        explicit ListIterator(const List& list) : list_(list) {}

        bool has_next() const override { return pos_ < list_.data_.size(); }
        T next() override { return list_.data_[pos_++]; }
    };

    std::unique_ptr<Iterator<T>> create_iterator() const {
        return std::make_unique<ListIterator>(*this);
    }
};

int main() {
    List<int> list;
    list.add(1);
    list.add(2);
    list.add(3);

    auto it = list.create_iterator();
    while (it->has_next()) {
        std::cout << it->next() << " ";  // 1 2 3
    }
}
```
