# 性能优化

## 移动语义（C++11）

```cpp
class Buffer {
    char* data_;
    size_t size_;
public:
    Buffer(size_t size) : size_(size), data_(new char[size]) {}

    // 移动构造函数
    Buffer(Buffer&& other) noexcept : data_(other.data_), size_(other.size_) {
        other.data_ = nullptr;
        other.size_ = 0;
    }

    // 移动赋值运算符
    Buffer& operator=(Buffer&& other) noexcept {
        if (this != &other) {
            delete[] data_;
            data_ = other.data_;
            size_ = other.size_;
            other.data_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }

    ~Buffer() { delete[] data_; }
};

Buffer create_buffer() {
    Buffer buf(1000);
    return buf;  // 移动构造，无需复制
}
```

## 内存池

```cpp
// 简单对象池
class ObjectPool {
    std::queue<std::unique_ptr<Object>> pool_;
    std::mutex mutex_;
public:
    std::unique_ptr<Object> acquire() {
        std::lock_guard lock(mutex_);
        if (!pool_.empty()) {
            auto obj = std::move(pool_.front());
            pool_.pop();
            return obj;
        }
        return std::make_unique<Object>();
    }
    void release(std::unique_ptr<Object> obj) {
        std::lock_guard lock(mutex_);
        pool_.push(std::move(obj));
    }
};
```

## 缓存优化

CPU 缓存行大小通常 64 字节，缓存友好代码性能差异可达 10-100 倍。

### 缓存行与数据布局

```cpp
// 缓存不友好：数组结构
struct BadStruct {
    float position[3];
    float velocity[3];
    float acceleration[3];
    float mass;
    // ...
};

std::vector<BadStruct> particles(1000000);

// 遍历时，mass 和其他字段被一起加载但只用一次
for (size_t i = 0; i < particles.size(); ++i) {
    auto& p = particles[i];
    // 访问 position[0], velocity[0], acceleration[0] 等，缓存效率低
    p.position[0] += p.velocity[0] * dt;
    p.position[1] += p.velocity[1] * dt;
    p.position[2] += p.velocity[2] * dt;
}

// 缓存友好：结构数组（SoA）
struct ParticleSystem {
    std::vector<float> position_x, position_y, position_z;
    std::vector<float> velocity_x, velocity_y, velocity_z;
    // ...
};

ParticleSystem particles;
particles.position_x.resize(1000000);

// 遍历时只加载需要的字段，缓存命中率高
for (size_t i = 0; i < count; ++i) {
    particles.position_x[i] += particles.velocity_x[i] * dt;
}
```

### 循环分块（Cache Blocking）

```cpp
// 未优化：矩阵乘法
void matrix_multiply(std::vector<float>& C, const std::vector<float>& A,
                     const std::vector<float>& B, int n) {
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            for (int k = 0; k < n; ++k) {
                C[i*n + j] += A[i*n + k] * B[k*n + j];
            }
        }
    }
}

// 分块优化：将数据分成 64x64 的块，适应 L1 缓存
void matrix_multiply_blocked(std::vector<float>& C, const std::vector<float>& A,
                              const std::vector<float>& B, int n, int block_size = 64) {
    for (int i = 0; i < n; i += block_size) {
        for (int j = 0; j < n; j += block_size) {
            for (int k = 0; k < n; k += block_size) {
                // 处理每个小块
                for (int ii = i; ii < std::min(i + block_size, n); ++ii) {
                    for (int jj = j; jj < std::min(j + block_size, n); ++jj) {
                        float sum = 0;
                        for (int kk = k; kk < std::min(k + block_size, n); ++kk) {
                            sum += A[ii*n + kk] * B[kk*n + jj];
                        }
                        C[ii*n + jj] += sum;
                    }
                }
            }
        }
    }
}
```

### 预取（Prefetching）

```cpp
// 手动预取提示
void process_with_prefetch(float* data, int n) {
    constexpr int prefetch_distance = 16;  // 提前多少个元素

    for (int i = 0; i < n; ++i) {
        // 提前预取后续数据
        if (i + prefetch_distance < n) {
            __builtin_prefetch(&data[i + prefetch_distance], 0, 3);
        }

        // 处理当前数据
```

## LTO（Link-Time Optimization）

链接时优化允许编译器在链接阶段进行跨翻译单元的优化。

### 编译时启用

```bash
# GCC/Clang
g++ -flto -O2 a.cpp b.cpp -o program
# 或分步编译
g++ -flto -c a.cpp -o a.o
g++ -flto -c b.cpp -o b.o
g++ -flto a.o b.o -o program

# MSVC
cl /GL /O2 a.cpp b.cpp  # /GL 启用 LTO
link /LTCG a.obj b.obj  # /LTCG 链接时优化
```

### 优化效果

```cpp
// a.cpp
inline int helper(int x) { return x * 2; }

// b.cpp
extern int process();  // 编译器可能不知道 helper 实现

// 有 LTO：编译器可以看到 helper 的实现，选择内联
// 无 LTO：必须按外部函数调用处理
```

### 完整示例

```bash
# 生产构建建议
g++ -flto -O2 -fno-exceptions -fno-rtti \
    -ffunction-sections -fdata-sections \
    a.cpp b.cpp -o program

# 链接时丢弃未使用代码
g++ -flto -Wl,--gc-sections a.o b.o -o program
```

## PGO（Profile-Guided Optimization）

使用实际运行数据指导编译器优化。

### 工作流程

```bash
# 1. 插桩编译
g++ -fprofile-generate -O2 a.cpp b.cpp -o program

# 2. 运行测试
./program < test_input.txt

# 3. 基于 profiling 数据优化编译
g++ -fprofile-use -O2 a.cpp b.cpp -o program_optimized
```

### 优化类型

```cpp
// PGO 帮助编译器做出更好的决策：

// 1. 分支预测优化
void process(bool flag) {
    if (flag) {  // PGO 发现 90% 为 true
        fast_path();  // 编译器优化为更可能的分支
    } else {
        slow_path();
    }
}

// 2. 内联决策
// 热点函数被更激进地内联

// 3. 缓存友好
// 调整代码布局，提高指令缓存命中率
```

### 实际应用场景

### 手动 SIMD（Intel SSE/AVX）

```cpp
#include <immintrin.h>

// AVX 256位：一次处理 8 个 float
void add_vectors_avx(float* a, float* b, float* c, int n) {
    int i = 0;
    // 处理对齐的 8 的倍数
    for (; i + 7 < n; i += 8) {
        __m256 va = _mm256_loadu_ps(&a[i]);  // 加载 8 个 float
        __m256 vb = _mm256_loadu_ps(&b[i]);
        __m256 vc = _mm256_add_ps(va, vb);   // 加法
        _mm256_storeu_ps(&c[i], vc);         // 存储
    }
    // 处理剩余元素
    for (; i < n; ++i) {
        c[i] = a[i] + b[i];
    }
}

// AVX 矩阵点积
float dot_product_avx(const float* a, const float* b, int n) {
    __m256 sum = _mm256_setzero_ps();

    int i = 0;
    for (; i + 7 < n; i += 8) {
        __m256 va = _mm256_loadu_ps(&a[i]);
        __m256 vb = _mm256_loadu_ps(&b[i]);
        sum = _mm256_add_ps(sum, _mm256_mul_ps(va, vb));
    }

    // 水平相加
    float result[8];
    _mm256_storeu_ps(result, sum);
    float total = result[0] + result[1] + result[2] + result[3] +
                  result[4] + result[5] + result[6] + result[7];

    for (; i < n; ++i) {
        total += a[i] * b[i];
    }
    return total;
}
```

### 条件操作

```cpp
// 掩码操作
void clamp_avx(float* data, int n, float min_val, float max_val) {
    __m256 vmin = _mm256_set1_ps(min_val);
    __m256 vmax = _mm256_set1_ps(max_val);

    for (int i = 0; i < n; i += 8) {
        __m256 v = _mm256_loadu_ps(&data[i]);
        v = _mm256_max_ps(vmin, _mm256_min_ps(v, vmax));  // 截断
        _mm256_storeu_ps(&data[i], v);
    }
}
```

### 性能考虑

```cpp
// 1. 确保数据对齐
alignas(32) float data[1024];

// 2. 确保循环足够长（抵消 SIMD 设置开销）
void process_aligned(float* data, int n) {
    for (int i = 0; i < n; ++i) {  // 小循环可能更慢
        data[i] *= 2.0f;
    }
}

// 3. 分离读写（避免依赖）
void process_separate(float* a, float* b, float* c, int n) {
    // 先读完再写
    for (int i = 0; i < n; ++i) {
        a[i] = b[i] + c[i];  // b、c 读取完成后再写 a
    }
}

// 4. 使用restrict提示无别名
```
