# C++ 单元测试

C++ 主流单元测试框架包括 GoogleTest（gtest）和 Catch2，两者都支持现代 C++ 特性。

## GoogleTest

GoogleTest 是 Google 开发的跨平台测试框架，支持丰富的断言和测试组织。

### 基础用法

```cpp
#include <gtest/gtest.h>

// 测试套件
TEST(MathTest, Addition) {
    EXPECT_EQ(2 + 2, 4);
    EXPECT_NE(2 + 2, 5);
}

TEST(MathTest, Division) {
    EXPECT_FLOAT_EQ(10.0 / 3.0, 3.333333f);
}

// 主函数（可选，gtest gtest_main 可自动生成）
int main(int argc, char** argv) {
    testing::InitGoogleTest(&argc, argv);
    return RUN_ALL_TESTS();
}
```

### 断言类型

```cpp
// 致命断言（ASSERT_*）- 失败时终止当前测试
ASSERT_EQ(a, b);
ASSERT_NE(a, b);
ASSERT_TRUE(condition);
ASSERT_FALSE(condition);
ASSERT_FLOAT_EQ(a, b);
ASSERT_DOUBLE_EQ(a, b);
ASSERT_STREQ(a, b);

// 非致命断言（EXPECT_*）- 失败时继续执行
EXPECT_EQ(a, b);
EXPECT_NE(a, b);
EXPECT_TRUE(condition);
EXPECT_LT(a, b);      // less than
EXPECT_GT(a, b);      // greater than
EXPECT_LE(a, b);      // less or equal
EXPECT_GE(a, b);      // greater or equal
EXPECT_THROW(statement, exception_type);
EXPECT_NO_THROW(statement);
EXPECT_DEATH(statement, regex);  // 期望崩溃
```

### 测试夹具（Fixtures）

```cpp
class DatabaseTest : public testing::Test {
protected:
    void SetUp() override {
        db_ = new Database("test.db");
        db_->connect();
    }

    void TearDown() override {
        db_->disconnect();
        delete db_;
    }

    Database* db_;
};

TEST_F(DatabaseTest, Insert) {
    EXPECT_TRUE(db_->insert("key", "value"));
}

TEST_F(DatabaseTest, Query) {
    db_->insert("key", "value");
    EXPECT_EQ(db_->query("key"), "value");
}
```

### 参数化测试

```cpp
// 测试参数组合
class AddTest : public testing::TestWithParam<std::tuple<int, int, int>> {
};

TEST_P(AddTest, Values) {
    auto [a, b, expected] = GetParam();
    EXPECT_EQ(add(a, b), expected);
}

INSTANTIATE_TEST_CASE_P(
    AddTests,
    AddTest,
    testing::Values(
        std::make_tuple(1, 2, 3),
        std::make_tuple(0, 0, 0),
        std::make_tuple(-1, 1, 0)
    )
);
```

### 死亡测试（Death Tests）

```cpp
TEST(DeathTest, DivisionByZero) {
    EXPECT_DEATH({
        int x = 1;
        int y = 0;
        int z = x / y;  // 崩溃
    }, "division");
}
```

## Catch2

Catch2 以单头文件著称，使用更自然的测试语法。

### 基础用法

```cpp
#define CATCH_CONFIG_MAIN
#include <catch2/catch_test_macros.hpp>

TEST_CASE("Math operations", "[math]") {
    REQUIRE(2 + 2 == 4);
    CHECK(3 * 3 == 9);
}

TEST_CASE("String operations", "[string]") {
    std::string s = "hello";
    CHECK(s.length() == 5);
    CHECK(s.substr(0, 2) == "he");
}
```

### SECTION（多次执行）

```cpp
TEST_CASE("Vector operations", "[vector]") {
    std::vector<int> v;

    SECTION("push_back increases size") {
        v.push_back(1);
        REQUIRE(v.size() == 1);
    }

    SECTION("empty vector has size 0") {
        REQUIRE(v.empty());
    }

    // 每个 SECTION 独立执行，v 每次都是新的
}
```

### 模板测试

```cpp
template <typename T>
T max_of(T a, T b) {
    return a > b ? a : b;
}

TEST_CASE("max_of works with various types") {
    STATIC_REQUIRE(max_of(1, 2) == 2);
    STATIC_REQUIRE(max_of(1.5, 2.5) == 2.5);
    STATIC_REQUIRE(max_of('a', 'b') == 'b');
}
```

### 参数化（生成器）

```cpp
TEST_CASE("Addition", "[math]") {
    auto [a, b, expected] = GENERATE(
        table<int, int, int>({
            {1, 2, 3},
            {0, 0, 0},
            {-1, 1, 0}
        })
    );

    REQUIRE(max_of(a, b) == expected);
}
```

## CMake 集成

### GoogleTest + CMake

```cmake
cmake_minimum_required(VERSION 3.16)
project(MyProject)

enable_testing()

# FetchContent 方式（推荐 CMake 3.24+）
include(FetchContent)
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG v1.14.0
)
FetchContent_MakeAvailable(googletest)

# 防止每次编译都运行 gtest的入口函数（自动生成 main）
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)

add_executable(myapp src/main.cpp)
add_executable(tests test_main.cpp utils_test.cpp)

target_link_libraries(tests PRIVATE
    GTest::gtest
    GTest::gtest_main
)

include(GoogleTest)
gtest_discover_tests(tests)

# 或者直接运行
add_test(NAME MyTests COMMAND tests)
```

## 测试最佳实践

### AAA 模式（Arrange-Act-Assert）

```cpp
TEST_CASE("Process pending orders", "[order]") {
    // Arrange
    OrderProcessor processor;
    Order order{"id-123", 100.0, OrderStatus::Pending};

    // Act
    processor.add_order(order);
    processor.process_pending();

    // Assert
    auto result = processor.get_order("id-123");
    REQUIRE(result.status == OrderStatus::Processed);
}
```

### 独立测试原则

```cpp
TEST_CASE("Vector capacity after clear") {
    std::vector<int> v{1, 2, 3};
    v.clear();
    // 不要依赖之前测试的状态
    REQUIRE(v.capacity() >= 0);  // 明确验证
}
```

### Mock 对象

```cpp
// 接口
class IUserRepository {
public:
    virtual ~IUserRepository() = default;
    virtual User find_by_id(int id) = 0;
};

// Mock 实现
class MockUserRepository : public IUserRepository {
public:
    MOCK_METHOD(User, find_by_id, (int id), (override));
};

TEST_CASE("UserService with mock") {
    MockUserRepository mock_repo;
    EXPECT_CALL(mock_repo, find_by_id(1))
        .WillOnce(Return(User{"Alice", 25}));

    UserService service(&mock_repo);
    auto user = service.get_user(1);

    CHECK(user.name == "Alice");
}
```

### 基准测试

```cpp
#include <benchmark/benchmark.h>

static void BM_StringCreation(benchmark::State& state) {
    for (auto _ : state) {
        std::string s = "hello";
        benchmark::DoNotOptimize(s);
    }
}
BENCHMARK(BM_StringCreation);

BENCHMARK_MAIN();
```
