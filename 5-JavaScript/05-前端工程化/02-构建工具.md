# 构建工具

## 构建工具概述

构建工具用于处理代码转换、依赖管理、代码优化等任务，提高开发效率和应用性能。

### 常见构建工具

| 工具 | 特点 |
|------|------|
| Vite | 基于 ESM，开发体验快 |
| Webpack | 功能全面，生态丰富 |
| Rollup | 适合库打包，Tree Shaking 优秀 |
| esbuild | 极快，用 Go 编写 |
| Parcel | 零配置 |
| SWC | Rust 编写，替代 Babel |

---

## Vite

### 特点

- 基于 ES Module 的开发服务器
- 极快的冷启动
- HMR（热模块替换）快速
- 生产环境使用 Rollup 打包

### 项目创建

```bash
# 创建项目
npm create vite@latest my-app
npm create vite@latest my-app -- --template vanilla
npm create vite@latest my-app -- --template vue
npm create vite@latest my-app -- --template react
npm create vite@latest my-app -- --template svelte

# 安装依赖
cd my-app
npm install

# 开发
npm run dev

# 构建
npm run build

# 预览
npm run preview
```

### 配置

```javascript
// vite.config.js
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';

export default defineConfig({
  plugins: [vue()],

  // 基础路径
  base: '/',

  // 构建输出
  build: {
    outDir: 'dist',
    assetsDir: 'assets',
    sourcemap: false,
    minify: 'esbuild',
    rollupOptions: {
      output: {
        manualChunks: {
          vendor: ['vue', 'vue-router']
        }
      }
    }
  },

  // 开发服务器
  server: {
    port: 3000,
    host: true,
    proxy: {
      '/api': {
        target: 'http://localhost:8080',
        changeOrigin: true
      }
    }
  },

  // 路径别名
  resolve: {
    alias: {
      '@': '/src',
      '~': '/src'
    }
  }
});
```

### 常用功能

```javascript
// 环境变量
// .env
VITE_API_BASE_URL=http://localhost:8080

// 使用
console.log(import.meta.env.VITE_API_BASE_URL);

// 动态导入
const module = await import('./module.js');

// CSS 预处理器
// npm install -D sass
import './styles.scss';
```

## Webpack

### 基础配置

```javascript
// webpack.config.js
const path = require('path');

module.exports = {
  mode: 'production',  // production | development | none

  entry: './src/index.js',

  output: {
    path: path.resolve(__dirname, 'dist'),
    filename: '[name].[contenthash].js',
    clean: true
  },

  module: {
    rules: [
      {
        test: /\.js$/,
        exclude: /node_modules/,
        use: {
          loader: 'babel-loader',
          options: {
            presets: ['@babel/preset-env']
          }
        }
      },
      {
        test: /\.css$/,
        use: ['style-loader', 'css-loader']
      },
      {
        test: /\.(png|svg|jpg|jpeg|gif)$/i,
        type: 'asset/resource'
      }
    ]
  },

  plugins: [
    new HtmlWebpackPlugin({
      template: './src/index.html',
      minify: true
    })
  ],

  devServer: {
    port: 3000,
    hot: true,
    proxy: {
      '/api': 'http://localhost:8080'
    }
  },

  optimization: {
    splitChunks: {
      chunks: 'all'
    }
  }
};
```

### 常用 Loader

```javascript
module.exports = {
  module: {
    rules: [
      // Babel
      {
        test: /\.js$/,
        exclude: /node_modules/,
        use: 'babel-loader'
      },

      // TypeScript
      {
        test: /\.ts$/,
        use: 'ts-loader'
      },

      // CSS
      {
        test: /\.css$/,
        use: ['style-loader', {
          loader: 'css-loader',
          options: { modules: true }
        }]
      },

      // Sass/SCSS
      {
        test: /\.scss$/,
        use: ['style-loader', 'css-loader', 'sass-loader']
      },

      // 图片
      {
        test: /\.(png|svg|jpg|jpeg|gif)$/i,
        type: 'asset/resource'
      },

      // 字体
      {
        test: /\.(woff|woff2|eot|ttf|otf)$/i,
        type: 'asset/resource'
      },

      // CSV/JSON
      {
        test: /\.(csv|xml)$/i,
        type: 'asset/resource'
      },

      // JSON
      {
        test: /\.json$/,
        type: 'json'
      }
    ]
  }
};
```

### 常用 Plugin

```javascript
const HtmlWebpackPlugin = require('html-webpack-plugin');
const MiniCssExtractPlugin = require('mini-css-extract-plugin');
const TerserPlugin = require('terser-webpack-plugin');

    new MiniCssExtractPlugin({
      filename: '[name].[contenthash].css'

```

## Rollup

### 基础配置

```javascript
// rollup.config.js
import resolve from '@rollup/plugin-node-resolve';
import commonjs from '@rollup/plugin-commonjs';
import terser from '@rollup/plugin-terser';

export default {
  input: 'src/index.js',

  output: {
    file: 'dist/bundle.js',
    format: 'esm',
    sourcemap: true
  },

  plugins: [
    resolve(),
    commonjs(),
    terser()
  ]
};
```

### 输出格式

```javascript
// ES Module
export default { /* */ };

// CommonJS
module.exports = { /* */ };

// UMD
(function(root, factory) {
  // ...
}(this, function() { /* */ }));

// IIFE
(function() { /* */ })();
```

### Tree Shaking

```javascript
// rollup 自动进行 Tree Shaking
// 只打包使用的导出

// src/math.js
export function add(a, b) { return a + b; }
export function multiply(a, b) { return a * b; }  // 未使用，不会被打包

// src/index.js
import { add } from './math.js';
console.log(add(1, 2));
```

## esbuild

### 快速使用

```bash
# 安装
npm install -g esbuild

# 打包
esbuild app.js --bundle --outfile=out.js

# 带 loader
esbuild app.jsx --bundle --loader:.jsx=jsx --outfile=out.js

# 监听
esbuild app.js --bundle --watch

# 生产模式
esbuild app.js --bundle --minify --outfile=out.js
```

### 配置文件

```javascript
// esbuild.config.js
const { build } = require('esbuild');

build({
  entryPoints: ['src/index.js'],
  bundle: true,
  outfile: 'dist/bundle.js',
  minify: process.env.NODE_ENV === 'production',
  sourcemap: true,
  target: ['es2020'],
  platform: 'browser',
  loader: {
    '.js': 'jsx',
    '.ts': 'ts'
  }
}).catch(() => process.exit(1));
```

## 持续集成

### CI 配置

```yaml
# .github/workflows/build.yml
name: Build

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3

      - uses: actions/setup-node@v3
        with:
          node-version: '18'
          cache: 'npm'

      - run: npm ci

      - run: npm run lint

      - run: npm run test -- --coverage

      - run: npm run build
        env:
          NODE_ENV: production

      - uses: actions/upload-artifact@v3
        with:
          name: dist
          path: dist
```
