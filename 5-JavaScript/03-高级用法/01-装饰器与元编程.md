# 装饰器与元编程

## Symbol（符号）

### 基本用法

```javascript
// Symbol 创建唯一标识符
const sym1 = Symbol('description');
const sym2 = Symbol('description');

console.log(sym1 === sym2);  // false，每个 Symbol 都独一无二

// 用于对象属性键
const obj = {
  [sym1]: 'value1',
  [sym2]: 'value2',
  regularKey: 'value3'
};

console.log(obj[sym1]);  // 'value1'
console.log(Object.keys(obj));  // ['regularKey'] - Symbol 键不枚举
```

### 常用内置符号

```javascript
// Symbol.iterator - 使对象可迭代
const iterable = {
  *[Symbol.iterator]() {
    yield 1;
    yield 2;
    yield 3;
  }
};

for (const val of iterable) {
  console.log(val);  // 1, 2, 3
}

// Symbol.toStringTag - 自定义 toString() 标签
const User = {
  [Symbol.toStringTag]: 'UserClass'
};
console.log(Object.prototype.toString.call(User));  // '[object UserClass]'

// Symbol.hasInstance - 自定义 instanceof 行为
class EvenNumbers {
  static [Symbol.hasInstance](instance) {
    return typeof instance === 'number' && instance % 2 === 0;
  }
}
console.log(2 instanceof EvenNumbers);  // true
console.log(3 instanceof EvenNumbers);  // false

// Symbol.toPrimitive - 自定义类型转换
const counter = {
  [Symbol.toPrimitive](hint) {
    if (hint === 'number') return 42;
    if (hint === 'string') return 'forty-two';
    return 42;
  }
};
console.log(+counter);  // 42 (hint: number)
console.log(`${counter}`);  // 'forty-two' (hint: string)
```

## Proxy（代理）

### Proxy 陷阱完整列表

```javascript
const target = { name: 'Alice', age: 25 };
const handler = {
  // 属性读取
  get(target, prop, receiver) {
    if (prop === 'age') {
      return target[prop] + ' years old';  // 转换
    }
    return Reflect.get(target, prop, receiver);
  },

  // 属性写入
  set(target, prop, value, receiver) {
    if (prop === 'age' && (typeof value !== 'number' || value < 0 || value > 150)) {
      throw new TypeError('Invalid age');
    }
    return Reflect.set(target, prop, value, receiver);
  },

  // 属性删除
  deleteProperty(target, prop) {
    if (prop === 'name') {
      throw new Error('Cannot delete name');
    }
    return Reflect.deleteProperty(target, prop);
  },

  // 属性检查
  has(target, prop) {
    if (prop === 'secret') {
      return false;  // 隐藏属性
    }
    return Reflect.has(target, prop);
  },

  // Object.getOwnPropertyDescriptor
  getOwnPropertyDescriptor(target, prop) {
    const desc = Reflect.getOwnPropertyDescriptor(target, prop);
    if (prop === 'age') {
      return { ...desc, enumerable: false };  // 不可枚举
    }
    return desc;
  },

  // Object.defineProperty
  defineProperty(target, prop, descriptor) {
    if (prop === 'age' && descriptor.value < 0) {
      return false;  // 拒绝定义
    }
    return Reflect.defineProperty(target, prop, descriptor);
  },

  // Object.preventExtensions
  preventExtensions(target) {
    if (target.age < 18) {
      return false;  // 不允许锁定
    }
    return Reflect.preventExtensions(target);
  },

  // Object.isExtensible
  isExtensible(target) {
    return Reflect.isExtensible(target);
  },

  // Object.getPrototypeOf
  getPrototypeOf(target) {
    return Reflect.getPrototypeOf(target);
  },

  // Object.setPrototypeOf
  setPrototypeOf(target, proto) {
    return Reflect.setPrototypeOf(target, proto);
  },

  // 函数调用
  apply(target, thisArg, args) {
    console.log(`Calling ${target.name} with`, args);
    return Reflect.apply(target, thisArg, args);
  },

  // new 操作符
  construct(target, args) {
    console.log('Constructing with', args);
    return Reflect.construct(target, args);
  }
};

const proxy = new Proxy(target, handler);

// 使用代理
console.log(proxy.age);        // '25 years old'
proxy.age = 30;                // OK
// proxy.name = 'Bob';          // 可正常修改
delete proxy.name;             // Error: Cannot delete name
console.log('name' in proxy);  // false - 隐藏的属性
```

### 实际应用：响应式系统

```javascript
function reactive(obj) {
  const handlers = new Map();

  function track(target, key) {
    const handler = handlers.get(target);
    if (handler && handler.get) {
      handler.get(key);
    }
  }

  function trigger(target, key) {
    const handler = handlers.get(target);
    if (handler && handler.set) {
      handler.set(key);
    }
  }

  return new Proxy(obj, {
    get(target, key, receiver) {
      track(target, key);
      const value = Reflect.get(target, key, receiver);

      // 嵌套对象也需要响应式
      if (value !== null && typeof value === 'object') {
        return reactive(value);
      }

      return value;
    },

    set(target, key, value, receiver) {
      const result = Reflect.set(target, key, value, receiver);
      trigger(target, key);
      return result;
    }
  });
}

// 使用
const state = reactive({ user: { name: 'Alice', age: 25 } });
state.user.name = 'Bob';  // 触发更新
```

### 实际应用：数据验证

```javascript
function validated(schema) {
  return new Proxy({}, {
    set(target, prop, value) {
      const validator = schema[prop];
      if (validator && !validator(value)) {
        throw new Error(`Invalid value for ${prop}`);
      }
      target[prop] = value;
      return true;
    },
    get(target, prop) {
      return target[prop];
    }
  });
}

const user = validated({
  name: (v) => typeof v === 'string' && v.length >= 2,
  age: (v) => typeof v === 'number' && v >= 0 && v <= 150,
  email: (v) => typeof v === 'string' && v.includes('@')
});

user.name = 'Alice';  // OK
// user.name = 'A';    // Error
```

## Reflect（反射）

### Reflect API 完整方法

```javascript
const obj = { name: 'Alice', age: 25, greet() { return 'Hi'; } };

// 属性操作
Reflect.get(obj, 'name');                    // 获取属性
Reflect.set(obj, 'name', 'Bob');              // 设置属性
Reflect.has(obj, 'name');                     // 检查属性
Reflect.deleteProperty(obj, 'age');           // 删除属性
Reflect.ownKeys(obj);                         // 所有自身属性键

// 对象操作
Reflect.preventExtensions(obj);               // 阻止扩展
Reflect.isExtensible(obj);                    // 是否可扩展
Reflect.getPrototypeOf(obj);                  // 获取原型
Reflect.setPrototypeOf(obj, Array.prototype);  // 设置原型

// 属性描述符
Reflect.getOwnPropertyDescriptor(obj, 'name');
Reflect.defineProperty(obj, 'name', { value: 'Charlie' });

// 函数操作
Reflect.apply(obj.greet, obj, []);           // 函数调用
Reflect.construct(Array, [1, 2, 3]);         // 构造函数调用
```

### 与 Proxy 配合：代理所有操作

```javascript
const withLogging = (target) => new Proxy(target, {
  get(target, prop, receiver) {
    console.log(`Getting ${String(prop)}`);
    return Reflect.get(target, prop, receiver);
  },
  set(target, prop, value, receiver) {
    console.log(`Setting ${String(prop)} = ${JSON.stringify(value)}`);
    return Reflect.set(target, prop, value, receiver);
  },
  apply(target, thisArg, args) {
    console.log(`Calling ${target.name || 'function'} with`, args);
    return Reflect.apply(target, thisArg, args);
  }
});

const user = withLogging({ name: 'Alice', age: 25 });
user.name;              // Logging: Getting name
user.age = 30;          // Logging: Setting age = 30
```

### 类装饰器

```javascript
//  sealed - 阻止类被扩展
function sealed(target) {
  Object.seal(target);
  Object.seal(target.prototype);
  return target;
}

//  自定义名称
function named(name) {
  return function(target) {
    target.displayName = name;
    return target;
  };
}

@sealed
@named('PersonClass')
class Person {
  name = 'Alice';
}

console.log(Person.displayName);  // 'PersonClass'
```

### 方法装饰器

```javascript
// 日志装饰器
function log(target, name, descriptor) {
  const original = descriptor.value;
  descriptor.value = function(...args) {
    console.log(`Calling ${name}(${args.join(', ')})`);
    const start = performance.now();
    const result = original.apply(this, args);
    const elapsed = performance.now() - start;
    console.log(`${name} returned ${JSON.stringify(result)} in ${elapsed.toFixed(2)}ms`);
    return result;
  };
  return descriptor;
}

// 防抖装饰器
function debounce(ms) {
  return function(target, name, descriptor) {
    const original = descriptor.value;
    let timer = null;
    descriptor.value = function(...args) {
      clearTimeout(timer);
      timer = setTimeout(() => original.apply(this, args), ms);
    };
    return descriptor;
  };
}

// 重试装饰器
function retry(maxAttempts) {
  return function(target, name, descriptor) {
    const original = descriptor.value;
    descriptor.value = async function(...args) {
      for (let attempt = 1; attempt <= maxAttempts; attempt++) {
        try {
          return await original.apply(this, args);
        } catch (error) {
          if (attempt === maxAttempts) throw error;
          console.log(`Retry ${attempt}/${maxAttempts} for ${name}`);
        }
      }
    };
    return descriptor;
  };
}

class UserService {
  @log
  @debounce(300)
  fetchUser(id) {
    return { id, name: 'Alice' };
  }

  @retry(3)
  async fetchData(url) {
    const response = await fetch(url);
    if (!response.ok) throw new Error('Failed');
    return response.json();
  }
}
```

### 属性装饰器

```javascript
// 只读装饰器
function readonly(target, name, descriptor) {
  descriptor.writable = false;
  return descriptor;
}

// 最小值验证
function minValue(min) {
  return function(target, name, descriptor) {
    const setter = descriptor.set;
    descriptor.set = function(value) {
      if (value < min) {
        throw new Error(`${name} must be at least ${min}`);
      }
      setter.call(this, value);
    };
    return descriptor;
  };
}

// 最大值验证
function maxValue(max) {
  return function(target, name, descriptor) {
    const setter = descriptor.set;
    descriptor.set = function(value) {
      if (value > max) {
        throw new Error(`${name} must be at most ${max}`);
      }
      setter.call(this, value);
    };
    return descriptor;
  };
}

class Product {
  name = 'Unknown';

  @readonly
  price = 0;

  @minValue(0)
  @maxValue(100)
  discount = 0;
}
```

### 参数装饰器

```javascript
// 必需参数标记
function required(target, name, index) {
  // 标记逻辑
}

// 参数验证
function validate(target, name, descriptor) {
  const original = descriptor.value;
  descriptor.value = function(...args) {
    const paramTypes = Reflect.getMetadata('design:paramtypes', target, name);
    for (let i = 0; i < paramTypes.length; i++) {
      if (args[i] === undefined && paramTypes[i] !== undefined) {
        throw new Error(`Argument ${i} is required`);
      }
    }
    return original.apply(this, args);
  };
  return descriptor;
}

class Calculator {
  @validate
  add(@required a, @required b) {
    return a + b;
  }
}
```

### 装饰器组合

```javascript
// 多个装饰器组合执行顺序
// 1. 参数装饰器从右到左
// 2. 方法装饰器从右到左
// 3. 类装饰器从下到上

@sealed
@named('MyClass')
class MyClass {
  @readonly
  @log
  method(@required arg) {
    return arg;
  }
}

// 执行顺序：required -> log -> readonly -> named -> sealed
```

### 装饰器工厂

```javascript
// 带参数的装饰器工厂
function typeCheck(expectedType) {
  return function(target, name, descriptor) {
    const original = descriptor.value;
    descriptor.value = function(...args) {
      args.forEach((arg, i) => {
        if (typeof arg !== expectedType) {
          throw new Error(`Argument ${i} must be ${expectedType}`);
        }
      });
      return original.apply(this, args);
    };
    return descriptor;
  };
}

// 使用
class MathUtils {
  @typeCheck('number')
  add(a, b) {
    return a + b;
  }
}

new MathUtils().add(1, 2);      // OK
// new MathUtils().add('a', 2);  // Error
```

## 元编程实用模式

### 自动属性代理

```javascript
class AutoProperties {
  constructor(initialValues = {}) {
    const handler = {
      get(target, prop) {
        if (prop in target) {
          return target[prop];
        }
        // 自动创建新属性
        target[prop] = new AutoProperties();
        return target[prop];
      },
      set(target, prop, value) {
        target[prop] = value;
        return true;
      }
    };

    return new Proxy(this, handler);
  }
}

const obj = new AutoProperties();
obj.user.name = 'Alice';  // 自动创建嵌套结构
console.log(obj.user.name);  // 'Alice'
```

class SecureUser {

### 唯读代理

```javascript
function deepFreeze(obj) {
  const freeze = (o) => {
    Object.freeze(o);
    Object.keys(o).forEach(key => {
      if (typeof o[key] === 'object' && o[key] !== null) {
        freeze(o[key]);
      }
    });
    return o;
  };
  return new Proxy(freeze(obj), {
    set() { throw new Error('Cannot modify frozen object'); },
    deleteProperty() { throw new Error('Cannot delete from frozen object'); }
  });
}

const config = deepFreeze({ db: { host: 'localhost', port: 5432 } });
// config.db.host = 'remote';  // Error
```

### 异步属性访问

```javascript
function lazy(target, name, descriptor) {
  const original = descriptor.get;
  let initialized = false;
  let value;

  descriptor.get = function() {
    if (!initialized) {
      value = original.call(this);
      initialized = true;
    }
    return value;
  };

  return descriptor;
}

class Database {
  @lazy
  get connection() {
    console.log('Connecting to database...');
    return { query: () => {} };
  }
}

const db = new Database();
console.log('Before');  // 不触发连接
db.connection;          // 触发连接
db.connection;          // 使用缓存的连接
```
