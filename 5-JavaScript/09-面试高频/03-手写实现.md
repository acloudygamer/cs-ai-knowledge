## 数组方法实现

### Q: 如何实现 Array.prototype.map？

```javascript
Array.prototype.myMap = function(fn, thisArg) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (i in this) {
      result[i] = fn.call(thisArg, this[i], i, this);
    }
  }
  return result;
};

// 测试
[1, 2, 3].myMap(x => x * 2);  // [2, 4, 6]
```

### Q: 如何实现 Array.prototype.filter？

```javascript
Array.prototype.myFilter = function(fn, thisArg) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (i in this && fn.call(thisArg, this[i], i, this)) {
      result.push(this[i]);
    }
  }
  return result;
};

// 测试
[1, 2, 3, 4].myFilter(x => x % 2 === 0);  // [2, 4]
```

### Q: 如何实现 Array.prototype.reduce？

```javascript
Array.prototype.myReduce = function(fn, initialValue) {
  let accumulator = initialValue;
  let startIndex = 0;

  if (accumulator === undefined) {
    accumulator = this[0];
    startIndex = 1;
  }

  for (let i = startIndex; i < this.length; i++) {
    accumulator = fn(accumulator, this[i], i, this);
  }

  return accumulator;
};

// 测试
[1, 2, 3].myReduce((acc, cur) => acc + cur, 0);  // 6
[1, 2, 3].myReduce((acc, cur) => acc + cur);     // 6
```

### Q: 如何实现 Array.prototype.flat？

```javascript
// 指定深度展平
Array.prototype.myFlat = function(depth = 1) {
  if (depth <= 0) return this.slice();

  const result = [];
  for (const item of this) {
    if (Array.isArray(item)) {
      result.push(...item.myFlat(depth - 1));
    } else {
      result.push(item);
    }
  }
  return result;
};

// 无限深度展平
Array.prototype.myFlatDeep = function() {
  const result = [];
  for (const item of this) {
    if (Array.isArray(item)) {
      result.push(...item.myFlatDeep());
    } else {
      result.push(item);
    }
  }
  return result;
};

// 测试
[1, [2, [3, [4]]]].myFlat(1);     // [1, 2, [3, [4]]]
[1, [2, [3, [4]]]].myFlatDeep();  // [1, 2, 3, 4]
```

### Q: 如何实现 call/apply/bind？

```javascript
// call
Function.prototype.myCall = function(context, ...args) {
  context = context ?? globalThis;
  const fn = Symbol('fn');
  context[fn] = this;
  const result = context[fn](...args);
  delete context[fn];
  return result;
};

// apply
Function.prototype.myApply = function(context, args = []) {
  context = context ?? globalThis;
  const fn = Symbol('fn');
  context[fn] = this;
  const result = context[fn](...args);
  delete context[fn];
  return result;
};

// bind
Function.prototype.myBind = function(context, ...initialArgs) {
  const fn = this;
  return function(...laterArgs) {
    return fn.apply(context, [...initialArgs, ...laterArgs]);
  };
};

// 测试
function greet(greeting, punct) {
  return `${greeting}, ${this.name}${punct}`;
}
greet.myCall({ name: 'Alice' }, 'Hello', '!');  // 'Hello, Alice!'
greet.myApply({ name: 'Bob' }, ['Hi', '.']);   // 'Hi, Bob.'
greet.myBind({ name: 'Carol' })('Hey', '?');     // 'Hey, Carol?'
```

### Q: 如何实现 new？

```javascript
function myNew(constructor, ...args) {
  const obj = Object.create(constructor.prototype);
  const result = constructor.apply(obj, args);

  // 如果构造函数返回对象，则返回该对象
  // 否则返回新创建的对象
  return typeof result === 'object' && result !== null
    ? result
    : obj;
}

// 测试
function Person(name, age) {
  this.name = name;
  this.age = age;
}
const p = myNew(Person, 'Alice', 30);
console.log(p instanceof Person);  // true
console.log(p.name);               // 'Alice'
```

### Q: 如何实现 instanceof？

```javascript
function myInstanceOf(left, right) {
  if (left === null || typeof left !== 'object') return false;

  let proto = left.__proto__;
  const prototype = right.prototype;

  while (proto !== null) {
    if (proto === prototype) return true;
    proto = proto.__proto__;
  }

  return false;
}

// 测试
myInstanceOf([], Array);        // true
myInstanceOf({}, Object);       // true
myInstanceOf('hello', String);  // false
```

## 数据结构实现

### Q: 如何实现防重复请求（请求缓存）？

```javascript
function createRequestCache() {
  const cache = new Map();
  const pending = new Map();

  return async function request(url, options) {
    const key = `${options?.method || 'GET'}:${url}`;

    // 缓存命中
    if (cache.has(key)) {
      return cache.get(key);
    }

    // 请求中，等待已有请求完成
    if (pending.has(key)) {
      return pending.get(key);
    }

    // 发起新请求
    const promise = fetch(url, options)
      .then(r => r.json())
      .finally(() => {
        pending.delete(key);
        // 可选：设置缓存过期
      });

    pending.set(key, promise);
    cache.set(key, promise);

    return promise;
  };
}
```

### Q: 如何实现简易版 EventEmitter？

```javascript
class EventEmitter {
  constructor() {
    this.events = new Map();
  }

  on(event, listener) {
    if (!this.events.has(event)) {
      this.events.set(event, []);
    }
    this.events.get(event).push(listener);
    return this;
  }

  once(event, listener) {
    const wrapper = (...args) => {
      listener(...args);
      this.off(event, wrapper);
    };
    return this.on(event, wrapper);
  }

  off(event, listener) {
    if (!this.events.has(event)) return this;
    const listeners = this.events.get(event);
    const index = listeners.indexOf(listener);
    if (index > -1) listeners.splice(index, 1);
    return this;
  }

  emit(event, ...args) {
    if (!this.events.has(event)) return false;
    this.events.get(event).forEach(listener => listener(...args));
    return true;
  }
}

// 测试
const emitter = new EventEmitter();
emitter.on('data', d => console.log('Received:', d));
emitter.emit('data', { msg: 'hello' });  // 'Received: { msg: "hello" }'
```
