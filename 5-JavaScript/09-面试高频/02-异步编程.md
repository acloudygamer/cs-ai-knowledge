## 手写 Promise

### Q: 如何实现一个 Promise？

```javascript
class MyPromise {
  constructor(executor) {
    this.state = 'pending';
    this.value = undefined;
    this.handlers = [];

    const resolve = (value) => {
      if (this.state !== 'pending') return;
      this.state = 'fulfilled';
      this.value = value;
      this.handlers.forEach(this.#handle);
    };

    const reject = (reason) => {
      if (this.state !== 'pending') return;
      this.state = 'rejected';
      this.value = reason;
      this.handlers.forEach(this.#handle);
    };

    try {
      executor(resolve, reject);
    } catch (e) {
      reject(e);
    }
  }

  #handle = ({ onFulfilled, onRejected, resolve, reject }) => {
    if (this.state === 'fulfilled') {
      queueMicrotask(() => onFulfilled?.(this.value));
    } else if (this.state === 'rejected') {
      queueMicrotask(() => onRejected?.(this.value));
    } else {
      this.handlers.push({ onFulfilled, onRejected, resolve, reject });
    }
  };

  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      this.#handle({
        onFulfilled: (v) => {
          try {
            resolve(onFulfilled?.(v));
          } catch (e) { reject(e); }
        },
        onRejected: (e) => {
          try {
            resolve(onRejected?.(e));
          } catch (err) { reject(err); }
        },
        resolve, reject
      });
    });
  }

  catch(onRejected) {
    return this.then(null, onRejected);
  }

  finally(onFinally) {
    return this.then(
      v => { onFinally(); return v; },
      e => { onFinally(); throw e; }
    );
  }
}
```

## async/await 深度

### Q: async/await 的错误处理方式？

```javascript
// 方式 1：try/catch
async function fetchData() {
  try {
    const res = await fetch('/api/data');
    const data = await res.json();
    return data;
  } catch (error) {
    console.error('Error:', error.message);
  }
}

// 方式 2：.catch()
const data = await fetch('/api/data')
  .then(r => r.json())
  .catch(error => console.error(error));

// 方式 3：封装为 Result 模式
async function to(promise) {
  try {
    const data = await promise;
    return [data, null];
  } catch (error) {
    return [null, error];
  }
}

const [data, error] = await to(fetch('/api/data').then(r => r.json()));
```

### Q: 并发请求如何实现？

```javascript
// 串行（一个个来）
async function serial() {
  const users = await fetchUsers();
  const posts = await fetchPosts();
  // 等待一个完成再执行下一个
}

// 并发（同时请求）
async function parallel() {
  const [users, posts] = await Promise.all([
    fetchUsers(),
    fetchPosts()
  ]);
}

// 竞态处理：取消旧请求
let controller;
async function search(query) {
  controller?.abort();  // 取消上一个
  controller = new AbortController();
  const res = await fetch(`/api/search?q=${query}`, {
    signal: controller.signal
  });
}
```

### Q: async/await 相对于 Promise 的优势？

```javascript
// 1. 代码更直观，像同步代码
// Promise
function fetchData() {
  return fetch('/api/data')
    .then(r => r.json())
    .then(data => {
      return fetch(`/api/user/${data.userId}`)
        .then(r => r.json())
        .then(user => ({ data, user }));
    });
}

// async/await
async function fetchData() {
  const dataRes = await fetch('/api/data');
  const data = await dataRes.json();
  const userRes = await fetch(`/api/user/${data.userId}`);
  const user = await userRes.json();
  return { data, user };
}

// 2. 调试更容易（能在 await 处设置断点）
// 3. 错误堆栈更清晰
```
