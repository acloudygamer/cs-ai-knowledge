# 变量与类型

## let / const / var

- `var`: 函数作用域，可重复声明，存在变量提升（hoisting）
- `let`: 块级作用域，不可重复声明，暂时性死区（TDZ）
- `const`: 块级作用域，声明时必须初始化，引用不可变（但属性可变）

```javascript
var name = 'Alice';          // 函数作用域
let age = 25;                // 块级作用域
const PI = 3.14159;          // 常量，不可重新赋值

// var 变量提升
console.log(hoisted);       // undefined（不是错误）
var hoisted = 'I am hoisted';

// let 暂时性死区
// console.log(tdz);         // ReferenceError
let tdz = 'temporal dead zone';
```

## 基本类型（7种）

- `number`: 整数和浮点数，`1`, `3.14`, `Infinity`, `NaN`
- `string`: 字符串，`'hello'`, `"world"`, `` `template ${var}` ``
- `boolean`: `true` / `false`
- `null`: 空值（主动赋值），`typeof null === 'object'`（历史bug）
- `undefined`: 未定义（未赋值），表示缺少值
- `symbol`: 唯一标识符，用于对象属性键
- `bigint`: 大整数，`10n`，超过 Number 安全整数范围时使用

## 对象类型

- `Object`: 键值对集合 `{ key: value }`
- `Array`: 有序列表，`[1, 2, 3]`
- `Function`: 函数对象，`function() {}` 或 `() => {}`
- `Date`, `RegExp`, `Map`, `Set`, `WeakMap`, `WeakSet`

## 类型判断

- `typeof`: 返回类型字符串，但 `typeof null === 'object'`
- `instanceof`: 检查原型链，`obj instanceof Array`
- `Object.prototype.toString.call()`: 精确类型检测

```javascript
typeof 123           // 'number'
typeof 'str'        // 'string'
typeof true         // 'boolean'
typeof undefined    // 'undefined'
typeof null         // 'object'（历史bug）
typeof {}            // 'object'
typeof []            // 'object'
typeof function(){}  // 'function'

// 精确判断
Object.prototype.toString.call(123)     // '[object Number]'
Object.prototype.toString.call([])       // '[object Array]'
Object.prototype.toString.call(null)    // '[object Null]'

// Array 检查
Array.isArray([])                       // true
[] instanceof Array                      // true
```

## 类型转换

### 强制类型转换

```javascript
// 字符串转换
String(123)              // '123'
(123).toString()         // '123'
123 + ''                  // '123'

// 数字转换
Number('123')            // 123
parseInt('123px', 10)    // 123（解析整数）
parseFloat('3.14')       // 3.14
+'123'                    // 123（一元加号）

// 浮点精度问题
0.1 + 0.2                // 0.30000000000000004
// 解决方案：使用整数运算后除法，或使用 decimal.js

// 布尔转换
Boolean(1)               // true
Boolean('')              // false
Boolean('0')             // true
!!1                       // true（双重取反）

// BigInt 转换
BigInt(123)              // 123n
BigInt('123')            // 123n
```

### 隐式类型转换

```javascript
// 字符串拼接
'5' + 3                  // '53'（数字被转为字符串）

// 数学运算
'6' - 1                  // 5（字符串转为数字）
'6' * '2'                // 12

// 逻辑运算
!!'hello'               // true（非空字符串为真）
!!0                      // false
!!''                    // false
!!null                  // false

// 相等性比较
1 == '1'                 // true（宽松相等，类型转换）
1 === '1'               // false（严格相等，不转换）
```

## 类型守卫

类型守卫是在运行时检查类型的方式，用于在条件分支中缩小变量类型范围。

### typeof 类型守卫

```javascript
function process(value) {
  if (typeof value === 'string') {
    // value 在这里被推断为 string
    return value.toUpperCase();
  }
  if (typeof value === 'number') {
    // value 在这里被推断为 number
    return value.toFixed(2);
  }
  // value 在这里是 unknown
}
```

### instanceof 类型守卫

```javascript
class Dog {
  bark() { console.log('Woof!'); }
}

class Cat {
  meow() { console.log('Meow!'); }
}

function speak(animal) {
  if (animal instanceof Dog) {
    animal.bark();   // TypeScript 推断为 Dog
  }
  if (animal instanceof Cat) {
    animal.meow();  // TypeScript 推断为 Cat
  }
}
```

### in 类型守卫

```javascript
function play(instrument) {
  if ('play' in instrument) {
    instrument.play();  // instrument 有 play 方法
  }
}
```

### 自定义类型守卫

JavaScript 中可以使用函数进行类型检查：

```javascript
// JavaScript 类型守卫函数
function isString(value) {
  return typeof value === 'string';
}

function isNumber(value) {
  return typeof value === 'number';
}

function isArray(value) {
  return Array.isArray(value);
}

// 使用类型守卫缩小类型范围
function process(value) {
  if (isString(value)) {
    console.log(value.toUpperCase());
  } else if (isNumber(value)) {
    console.log(value.toFixed(2));
  }
}
```

## Symbol 妙用

Symbol 是 ES6 引入的原始数据类型，用于创建唯一标识符。

### 基本用法

```javascript
// 创建 Symbol
const sym1 = Symbol();
const sym2 = Symbol('description');  // 描述仅用于调试
const sym3 = Symbol('description');

sym1 === sym2        // false（每个 Symbol 都唯一）
sym2 === sym3        // false
Symbol('desc') === Symbol('desc')  // false

// 作为对象属性键
const obj = {
  [Symbol('id')]: 1,
  [Symbol('id')]: 2,
  name: 'Alice'
};
Object.getOwnPropertySymbols(obj)  // [Symbol(id), Symbol(id)]
```

### 常见使用场景

#### 1. 避免属性名冲突

```javascript
const firstName = Symbol('firstName');
const lastName = Symbol('lastName');

const user = {
  [firstName]: 'John',
  [lastName]: 'Doe',
  name: 'Anonymous'  // 普通属性，与 Symbol 不冲突
};

// 访问 Symbol 属性需要用 Object.getOwnPropertySymbols
Object.getOwnPropertySymbols(user).forEach(sym => {
  console.log(user[sym]);
});
```

#### 2. 定义类的私有成员

```javascript
const _password = Symbol('password');
const _validate = Symbol('validate');

class User {
  constructor(name, password) {
    this.name = name;
    this[_password] = password;  // 私有属性
  }

  [_validate]() {
    return this[_password].length >= 8;
  }

  checkPassword(pwd) {
    // Symbol 属性外部无法直接访问
    return pwd === this[_password];
  }
}

const user = new User('Alice', 'secret123');
user._password          // undefined（不是真正的私有）
user[Symbol.for('password')]  // undefined
```

#### 3. Symbol.for 创建全局 Symbol

```javascript
// Symbol.for 创建全局注册表中的 Symbol
const globalSym1 = Symbol.for('app.version');
const globalSym2 = Symbol.for('app.version');

globalSym1 === globalSym2  // true（同一引用）

// Symbol.keyFor 获取全局 Symbol 的键
Symbol.keyFor(globalSym1)   // 'app.version'
```

#### 4. 内置 Symbol 扩展点

```javascript
// Symbol.iterator - 自定义迭代行为
class Counter {
  constructor(start = 0, end = 5) {
    this.start = start;
    this.end = end;
    this.current = start;
  }

  [Symbol.iterator]() {
    return {
      next: () => {
        if (this.current <= this.end) {
          return { value: this.current++, done: false };
        }
        return { value: undefined, done: true };
      }
    };
  }
}

[...new Counter(1, 3)]  // [1, 2, 3]

// Symbol.toPrimitive - 自定义类型转换
constTemperature = (value) => ({
  [Symbol.toPrimitive](hint) {
    if (hint === 'number') return value;
    if (hint === 'string') return `${value}°C`;
    return value;
  }
});

const temp = Temperature(25);
+temp       // 25（hint: 'number'）
`${temp}`   // '25°C'（hint: 'string'）
temp + 0    // 25（hint: 'default'）

// Symbol.toStringTag - 自定义 toString 结果
class Person {
  get [Symbol.toStringTag]() {
    return 'Person';
  }
}
new Person().toString()  // '[object Person]'
```

#### 5. 禁用扩展的对象行为

```javascript
// Symbol.unscopables - 指定排除的属性
const obj = { a: 1, b: 2, c: 3 };
obj[Symbol.unscopables] = { b: true };

with (obj) {
  console.log(a);  // 1
  console.log(b);  // ReferenceError: b is not defined（被排除）
  console.log(c);  // 3
}
```

### Symbol 工具函数

```javascript
// 获取对象所有 Symbol 属性
Object.getOwnPropertySymbols(obj)

// 获取所有符号属性（包括不可枚举）
Reflect.ownKeys(obj)

// 检查是否为 Symbol
Symbol.prototype.toString.call(sym)  // 'Symbol(description)'
typeof sym                           // 'symbol'
```
