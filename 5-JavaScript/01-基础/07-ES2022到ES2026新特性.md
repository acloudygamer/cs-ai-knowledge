# ES2022 到 ES2026 新特性

## ES2022 新特性

### Top-level await

```javascript
// ES2022 之前：await 只能在 async 函数内使用
// ES2022：在模块顶层直接使用 await

const data = await fetch('/api/data').then(r => r.json());
export { data };

// 条件式初始化
const translations = language === 'zh'
  ? await import('./zh.json')
  : await import('./en.json');

// 回退机制
const config = await fetch('/api/config').catch(() => defaultConfig);
```

### 私有类字段

```javascript
class BankAccount {
  #balance = 0;
  #id;

  constructor(id) {
    this.#id = id;
  }

  deposit(amount) {
    if (amount <= 0) throw new Error('Invalid amount');
    this.#balance += amount;
  }

  withdraw(amount) {
    if (amount > this.#balance) throw new Error('Insufficient funds');
    this.#balance -= amount;
  }

  getBalance() {
    return this.#balance;
  }

  #validate(amount) {
    return amount > 0 && Number.isFinite(amount);
  }
}

// account.#balance  // SyntaxError: Private field

// 私有字段检测
class MyClass {
  #privateField = 42;
  hasPrivateField() {
    return #privateField in this;
  }
}
```

### 静态类字段与静态代码块

```javascript
class Config {
  static DEFAULT_TIMEOUT = 5000;
  static MAX_RETRIES = 3;
  static #instanceCount = 0;

  static {
    console.log('Config class loaded');
    Config.#instanceCount++;
  }

  static getInstanceCount() {
    return Config.#instanceCount;
  }
}
```

### at() 方法

```javascript
const arr = [1, 2, 3, 4, 5];
arr.at(0);   // 1
arr.at(-1);  // 5（倒数第一个）
arr.at(10);  // undefined

const str = 'hello';
str.at(-1);  // 'o'
```

### Object.hasOwn()

```javascript
const obj = { name: 'Alice' };

// 旧方式
Object.prototype.hasOwnProperty.call(obj, 'name');  // true

// 新方式（更安全，不受 hasOwnProperty 重写影响）
Object.hasOwn(obj, 'name');     // true
Object.hasOwn(obj, 'toString'); // false（继承属性）
```

### 错误 cause

```javascript
async function fetchData() {
  try {
    const res = await fetch('/api/data');
    if (!res.ok) throw new Error('API Error', { cause: res });
    return await res.json();
  } catch (e) {
    throw new Error('Failed to fetch data', { cause: e });
  }
}

try {
  await fetchData();
} catch (e) {
  console.log(e.message);       // 'Failed to fetch data'
  console.log(e.cause.message); // 'API Error'
}
```

### RegExp 匹配索引（/d 标志）

```javascript
const regex = /(?<letter>a)b(?<rest>c)/d;
const match = regex.exec('abc');
console.log(match.indices);        // [[0, 3], [0, 1], [2, 3]]
console.log(match.indices.groups); // { letter: [0, 1], rest: [2, 3] }
```

## ES2023 新特性

### Array findLast / findLastIndex

```javascript
const users = [
  { name: 'Alice', active: true },
  { name: 'Bob', active: false },
  { name: 'Charlie', active: true }
];

users.findLast(u => u.active);       // { name: 'Charlie', active: true }
users.findLastIndex(u => u.active);  // 2
```

### Array toSorted / toReversed / toSpliced / with

```javascript
const arr = [3, 1, 4, 1, 5];

arr.toSorted();           // [1, 1, 3, 4, 5]（原数组不变）
arr.toReversed();         // [5, 1, 4, 1, 3]（原数组不变）
arr.toSpliced(1, 2, 9);  // [3, 9, 5]（原数组不变）
arr.with(2, 99);          // [3, 1, 99, 1, 5]（替换指定索引）
```

### Hashbang 语法（标准化）

```javascript
#!/usr/bin/env node
// 被当作单行注释解析，ES2023 正式纳入标准
console.log('Hello');
```

## ES2024 新特性

### Object.groupBy / Map.groupBy

```javascript
const inventory = [
  { name: 'asparagus', type: 'vegetables' },
  { name: 'bananas', type: 'fruit' },
  { name: 'cabbage', type: 'vegetables' }
];

const grouped = Object.groupBy(inventory, item => item.type);
// { vegetables: [...], fruit: [...] }

const groupedMap = Map.groupBy(inventory, item => item.type);
// Map { 'vegetables' => [...], 'fruit' => [...] }
```

### Promise.withResolvers

```javascript
function delay(ms) {
  const { promise, resolve, reject } = Promise.withResolvers();
  setTimeout(resolve, ms);
  return promise;
}

// 比 new Promise((resolve, reject) => ...) 更简洁，
// 可以将 resolve/reject 传递到其他作用域
```

### 导入属性（Import Attributes）

```javascript
// 静态导入
import jsonData from './data.json' with { type: 'json' };

// 动态导入
const data = await import('./data.json', { with: { type: 'json' } });
```

### RegExp v 标志（Unicode 字符集操作）

```javascript
// v 标志支持字符集差集、交集、并集
/[\p{Letter}&&\p{ASCII}]/v.test('a');  // true（ASCII 字母）
/[\p{Emoji}--\p{Number}]/v.test('😀'); // true（非数字 Emoji）

// 字符串 Unicode 属性
/\p{RGI_Emoji}/v.test('👨‍👩‍👧');  // true（多码点 Emoji 序列）
```

### ArrayBuffer.prototype.transfer

```javascript
const buffer = new ArrayBuffer(8);
const view = new Uint8Array(buffer);
view[0] = 42;

const transferred = buffer.transfer();
// buffer 已被分离（detached），无法再使用
// transferred 是包含原始数据的新 ArrayBuffer
```

## ES2025 新特性

### 迭代器辅助方法（Iterator Helpers）

```javascript
// 迭代器直接支持 map/filter/take 等方法
function* range(start, end) {
  for (let i = start; i < end; i++) yield i;
}

const result = range(0, 10)
  .filter(x => x % 2 === 0)
  .map(x => x * x)
  .take(3)
  .toArray();
// [0, 4, 16]
```

### Promise.try

```javascript
// 统一处理同步和异步函数，避免 try/catch 与 .catch() 混用
Promise.try(() => JSON.parse(userInput))
  .then(data => process(data))
  .catch(err => console.error('解析失败:', err));
```

### 正则表达式重复具名捕获组

```javascript
// 同名捕获组可以在不同分支中重复使用
const re = /(?<year>\d{4})-(?<month>\d{2})|(?<month>\d{2})\/(?<year>\d{4})/;
const match = re.exec('12/2025');
match.groups; // { year: '2025', month: '12' }
```

## ES2026 新特性（Node.js 26+）

### 装饰器（Decorators）- TC39 Stage 3，Node.js 26+ 原生支持

```javascript
// TC39 Stage 3 装饰器语法（与旧版 TypeScript 实验性装饰器不同）
// 装饰器函数接收 (value, context) 两个参数

function sealed(value, context) {
  if (context.kind === 'class') {
    Object.seal(value);
    Object.seal(value.prototype);
  }
  return value;
}

function logged(value, context) {
  if (context.kind === 'method') {
    return function(...args) {
      console.log(`Calling ${context.name}`);
      return value.apply(this, args);
    };
  }
}

function limited(value, context) {
  // accessor 装饰器：拦截 getter/setter
  if (context.kind === 'accessor') {
    let stored = value.get.call(this);
    return {
      get() { return stored; },
      set(v) { stored = Math.min(v, 100); }
    };
  }
}

@sealed
class BugReport {
  accessor priority = 0;  // accessor 关键字创建自动存取器

  @logged
  print() {
    console.log(`Priority: ${this.priority}`);
  }
}

const report = new BugReport();
report.print();  // Calling print \n Priority: 0
```

### String.prototype.matchAll 改进（ES2026）

```javascript
// matchAll 现在支持非全局正则（自动添加 g 标志）
const str = 'test1 test2 test3';
for (const match of str.matchAll(/test(\d)/)) {
  console.log(match[1]);  // '1', '2', '3'
}
```

### using 关键字（ES2026）

```javascript
// 用于管理需要显式释放的资源，类似 const 但不可重新赋值
// 退出作用域时自动调用资源的 Symbol.dispose

class DatabaseConnection {
  [Symbol.dispose]() {
    console.log('连接已关闭');
  }
}

{
  using conn = new DatabaseConnection();
  console.log('使用连接');
}
// 输出：使用连接 \n 连接已关闭

// 嵌套 using
{
  using a = new Resource('A');
  using b = new Resource('B');
  // 离开作用域时按逆序释放：先 b 后 a
}

// await using：异步资源清理
{
  await using file = await openFile('data.txt');
  // 文件在作用域结束时自动关闭
}
```

### Temporal API（ES2026）

```javascript
// 新的日期/时间 API，比 Date 更强大（不可变、时区感知）

// 获取当前时间
const now = Temporal.Now.instant();
const date = Temporal.Now.plainDateISO();
const time = Temporal.Now.plainTimeISO();
const datetime = Temporal.Now.zonedDateTimeISO();

// 创建具体时间点
const birthday = Temporal.PlainDate.from({ year: 1990, month: 6, day: 15 });
const meeting = Temporal.ZonedDateTime.from({
  year: 2026,
  month: 4,
  day: 21,
  hour: 14,
  minute: 30,
  timeZone: 'Asia/Shanghai'
});

// 计算
const later = now.add({ hours: 2, minutes: 30 });
const diff = meeting.since(now);

// Duration
const duration = Temporal.Duration.from({ hours: 1, minutes: 30 });
const adjusted = date.add(duration);
```

### Error.isError（ES2026）

```javascript
// 安全地检测错误（可在 iframe 之间使用）
try {
  // some operation
} catch (e) {
  if (Error.isError(e)) {
    console.log(e.message);
  }
}

// 比 instanceof Error 更安全，因为跨 realm 的错误 instanceof 可能失败
```

### Array.fromAsync（ES2026）

```javascript
// 从异步迭代器创建数组
async function* generateItems() {
  yield await fetchItem(1);
  yield await fetchItem(2);
  yield await fetchItem(3);
}

const items = await Array.fromAsync(generateItems());

// 也支持同步可迭代对象
const syncItems = await Array.fromAsync([1, 2, 3]);
```

// fromBase64 / toBase64
const bytes = Uint8Array.fromBase64('SGVsbG8=');
const encoded = bytes.toBase64();

// fromHex / toHex
const data = Uint8Array.fromHex('48656c6c6f');
