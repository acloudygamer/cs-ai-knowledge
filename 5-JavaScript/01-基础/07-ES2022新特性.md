# ES2022+ 新特性

## Top-level await

### 基本用法

```javascript
// ES2022 之前：await 只能在 async 函数内使用
// ES2022：在模块顶层直接使用 await

// dynamic.js
const data = await fetch('/api/data').then(r => r.json());
export { data };

// or
export const data = await fetch('/api/config').then(r => r.json());
```

### 应用场景

```javascript
// 1. 模块初始化
const connection = await createDatabaseConnection();

// 2. 条件式初始化
const translations = language === 'zh'
  ? await import('./zh.json')
  : await import('./en.json');

// 3. 回退机制
const config = await fetch('/api/config').catch(() => defaultConfig);
```

## 私有类字段

### 基本语法

```javascript
class BankAccount {
  // 私有字段（# 开头）
  #balance = 0;
  #id;

  constructor(id) {
    this.#id = id;
  }

  deposit(amount) {
    if (amount <= 0) throw new Error('Invalid amount');
    this.#balance += amount;
  }

  withdraw(amount) {
    if (amount > this.#balance) throw new Error('Insufficient funds');
    this.#balance -= amount;
  }

  getBalance() {
    return this.#balance;
  }

  // 私有方法
  #validate(amount) {
    return amount > 0 && Number.isFinite(amount);
  }
}

const account = new BankAccount('ACC001');
account.deposit(100);
// account.#balance  // SyntaxError: Private field
```

### 私有字段检测

```javascript
class MyClass {
  #privateField = 42;

  hasPrivateField() {
    return #privateField in this;
  }
}

const obj = new MyClass();
obj.hasPrivateField();  // true
```

## 静态类字段

```javascript
class Config {
  static DEFAULT_TIMEOUT = 5000;
  static MAX_RETRIES = 3;
  static #instanceCount = 0;  // 私有静态字段

  static {
    // 静态代码块（ES2022）
    console.log('Config class loaded');
    Config.#instanceCount++;
  }

  constructor() {
    Config.#instanceCount++;
  }

  static getInstanceCount() {
    return Config.#instanceCount;
  }
}
```

## at() 方法

### 数组 at()

```javascript
const arr = [1, 2, 3, 4, 5];

arr.at(0);   // 1
arr.at(-1);  // 5（倒数第一个）
arr.at(-2);  // 4（倒数第二个）
arr.at(10);  // undefined（越界不报错）
```

### 字符串 at()

```javascript
const str = 'hello';

str.at(0);   // 'h'
str.at(-1);  // 'o'
```

### TypedArray at()

```javascript
const ints = new Int8Array([1, 2, 3]);
ints.at(-1);  // 3
```

## Object.hasOwn()

```javascript
// Object.hasOwn() 替代 Object.prototype.hasOwnProperty.call()

const obj = { name: 'Alice' };

// 旧方式
Object.prototype.hasOwnProperty.call(obj, 'name');  // true

// 新方式
Object.hasOwn(obj, 'name');  // true
Object.hasOwn(obj, 'toString');  // false（继承属性）

// 更安全：不用担心 obj 是 null/undefined 或重写了 hasOwnProperty
```

## 类静态块

```javascript
class MyService {
  static baseUrl = 'https://api.example.com';
  static token;

  static {
    // 静态初始化代码
    // 可以访问 private 字段
    try {
      this.token = localStorage.getItem('token');
    } catch {
      this.token = null;
    }
  }

  static {
    // 多个静态块按声明顺序执行
    console.log('MyService initialized');
  }
}
```

## 错误-cause（ES2022）

```javascript
// 在错误中链式携带原因
async function fetchData() {
  try {
    const res = await fetch('/api/data');
    if (!res.ok) throw new Error('API Error', { cause: res });
    return await res.json();
  } catch (e) {
    throw new Error('Failed to fetch data', { cause: e });
  }
}

try {
  await fetchData();
} catch (e) {
  console.log(e.message);  // 'Failed to fetch data'
  console.log(e.cause);     // Error: API Error
  console.log(e.cause.cause);  // Response 对象
}
```

## RegExp 匹配索引（/d 标志）

```javascript
// 使用 v 标志启用高级 Unicode 特性
// 匹配索引：d 标志

const str = 'a(b)c';
const regex = /b/d;

const match = regex.exec(str);
console.log(match.indices);  // [[1, 2], [1, 2]]（匹配位置）

// 具名组
const regex2 = /(?<letter>a)b(?<rest>c)/d;
const match2 = regex2.exec('abc');
console.log(match2.indices);     // [[0, 3], [0, 1], [2, 3]]
console.log(match2.indices.groups);  // { letter: [0, 1], rest: [2, 3] }
```

## 总结：新特性速查

| 特性 | ES 版本 | 用途 |
|------|---------|------|
| `class.field` | ES2022 | 私有类字段 |
| `static field` | ES2022 | 静态类字段 |
| `static {}` | ES2022 | 静态代码块 |
| `await at top level` | ES2022 | 顶层 await |
| `.at()` | ES2022 | 数组/字符串索引 |
| `Object.hasOwn()` | ES2022 | 安全属性检测 |
| `error.cause` | ES2022 | 错误链 |
| `regexp /d` | ES2022 | 匹配索引 |
