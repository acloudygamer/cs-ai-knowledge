## 策略模式（Strategy）

### 概念

定义一系列算法，把它们一个个封装起来，并且使它们可以相互替换。

### 实现

```javascript
// 策略接口
class PaymentStrategy {
  pay(amount) {
    throw new Error('pay() must be implemented');
  }
}

// 具体策略
class CreditCardPayment extends PaymentStrategy {
  constructor(cardNumber, cvv) {
    super();
    this.cardNumber = cardNumber;
    this.cvv = cvv;
  }

  pay(amount) {
    console.log(`Paid $${amount} with Credit Card ${this.cardNumber.slice(-4)}`);
    return { success: true, method: 'credit_card' };
  }
}

class PayPalPayment extends PaymentStrategy {
  constructor(email) {
    super();
    this.email = email;
  }

  pay(amount) {
    console.log(`Paid $${amount} with PayPal ${this.email}`);
    return { success: true, method: 'paypal' };
  }
}

class CryptoPayment extends PaymentStrategy {
  constructor(walletAddress) {
    super();
    this.walletAddress = walletAddress;
  }

  pay(amount) {
    console.log(`Paid $${amount} with Crypto ${this.walletAddress.slice(0, 8)}...`);
    return { success: true, method: 'crypto' };
  }
}

// 上下文：使用策略
class ShoppingCart {
  constructor() {
    this.items = [];
    this.paymentStrategy = null;
  }

  addItem(item) {
    this.items.push(item);
  }

  setPaymentStrategy(strategy) {
    this.paymentStrategy = strategy;
  }

  getTotal() {
    return this.items.reduce((sum, item) => sum + item.price, 0);
  }

  checkout() {
    if (!this.paymentStrategy) {
      throw new Error('Payment strategy not set');
    }
    if (this.items.length === 0) {
      throw new Error('Cart is empty');
    }

    const total = this.getTotal();
    return this.paymentStrategy.pay(total);
  }
}

// 使用
const cart = new ShoppingCart();
cart.addItem({ name: 'Widget', price: 29.99 });
cart.addItem({ name: 'Gadget', price: 49.99 });

cart.setPaymentStrategy(new CreditCardPayment('4111111111111111', '123'));
cart.checkout();  // Paid $79.98 with Credit Card 1111

cart.setPaymentStrategy(new PayPalPayment('user@example.com'));
cart.checkout();  // Paid $79.98 with PayPal user@example.com
```

### 表单验证策略

```javascript
// 验证策略接口
class ValidationStrategy {
  validate(value) {
    throw new Error('validate() must be implemented');
  }
}

// 具体策略
class RequiredValidation extends ValidationStrategy {
  validate(value) {
    return value !== null && value !== undefined && value !== '';
  }
}

class EmailValidation extends ValidationStrategy {
  validate(value) {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
  }
}

class LengthValidation extends ValidationStrategy {
  constructor(min, max) {
    super();
    this.min = min;
    this.max = max;
  }

  validate(value) {
    return value.length >= this.min && value.length <= this.max;
  }
}

class PatternValidation extends ValidationStrategy {
  constructor(regex) {
    super();
    this.regex = regex;
  }

  validate(value) {
    return this.regex.test(value);
  }
}

// 表单验证器
class FormValidator {
  constructor() {
    this.rules = new Map();
  }

  addRule(field, strategy) {
    this.rules.set(field, strategy);
    return this;
  }

  validate(data) {
    const errors = [];

    for (const [field, strategy] of this.rules) {
      if (!strategy.validate(data[field])) {
        errors.push({ field, message: `Invalid ${field}` });
      }
    }

    return {
      valid: errors.length === 0,
      errors
    };
  }
}

// 使用
const validator = new FormValidator();
validator
  .addRule('email', new RequiredValidation())
  .addRule('email', new EmailValidation())
  .addRule('username', new RequiredValidation())
  .addRule('username', new LengthValidation(3, 20))
  .addRule('password', new RequiredValidation())
  .addRule('password', new PatternValidation(/^(?=.*[A-Za-z])(?=.*\d).{8,}$/));

const result = validator.validate({
  email: 'invalid',
  username: 'ab',
  password: 'weak'
});

console.log(result);
// { valid: false, errors: [...] }
```

## 观察者模式（Observer）

### 概念

定义对象间的一种一对多的依赖关系，当一个对象的状态发生改变时，所有依赖它的对象都会收到通知。

### 实现

```javascript
// 主题接口
class Subject {
  attach(observer) {}  // 添加观察者
  detach(observer) {}  // 移除观察者
  notify() {}          // 通知所有观察者
}

// 观察者接口
class Observer {
  update(data) {}
}

// 具体主题
class NewsAgency extends Subject {
  constructor() {
    super();
    this.observers = new Set();
    this.latestNews = null;
  }

  attach(observer) {
    this.observers.add(observer);
  }

  detach(observer) {
    this.observers.delete(observer);
  }

  notify() {
    this.observers.forEach(observer => {
      observer.update(this.latestNews);
    });
  }

  setNews(news) {
    console.log(`NewsAgency: Breaking news - ${news}`);
    this.latestNews = news;
    this.notify();
  }
}

// 具体观察者
class NewsChannel extends Observer {
  constructor(name) {
    super();
    this.name = name;
  }

  update(news) {
    console.log(`${this.name} received: ${news}`);
  }
}

class NewsApp extends Observer {
  constructor(userId) {
    super();
    this.userId = userId;
  }

  update(news) {
    console.log(`NewsApp [User ${this.userId}] showing notification: ${news}`);
  }
}

// 使用
const agency = new NewsAgency();

const cnn = new NewsChannel('CNN');
const bbc = new NewsChannel('BBC');
const app = new NewsApp(12345);

agency.attach(cnn);
agency.attach(bbc);
agency.attach(app);

agency.setNews('JavaScript 2024 features released!');
// CNN received: JavaScript 2024 features released!
// BBC received: JavaScript 2024 features released!
// NewsApp [User 12345] showing notification: JavaScript 2024 features released!

agency.detach(bbc);
agency.setNews('New browser released!');
// CNN received: New browser released!
// NewsApp [User 12345] showing notification: New browser released!
```

### 事件系统

```javascript
// 事件发射器
class EventEmitter {
  constructor() {
    this.events = new Map();
  }

  on(event, listener) {
    if (!this.events.has(event)) {
      this.events.set(event, new Set());
    }
    this.events.get(event).add(listener);
    return () => this.off(event, listener);  // 返回取消订阅函数
  }

  off(event, listener) {
    if (this.events.has(event)) {
      this.events.get(event).delete(listener);
    }
  }

  emit(event, ...args) {
    if (this.events.has(event)) {
      this.events.get(event).forEach(listener => {
        listener(...args);
      });
    }
  }

  once(event, listener) {
    const unsubscribe = this.on(event, (...args) => {
      listener(...args);
      unsubscribe();
    });
  }
}

// 使用
const emitter = new EventEmitter();

const unsubscribe = emitter.on('message', (from, content) => {
  console.log(`${from}: ${content}`);
});

emitter.emit('message', 'Alice', 'Hello!');
emitter.emit('message', 'Bob', 'Hi!');

emitter.once('connection', () => {
  console.log('First connection established!');
});

emitter.emit('connection');
emitter.emit('connection');  // 不再触发
```

### Vue 响应式原理（简化版）

```javascript
// 响应式对象
function reactive(obj) {
  constDep = new Set();  // 依赖该属性的订阅者

  return new Proxy(obj, {
    get(target, key) {
      if (currentSubscriber) {
        deps[key] = deps[key] || new Set();
        deps[key].add(currentSubscriber);
      }
      return target[key];
    },

    set(target, key, value) {
      target[key] = value;
      if (deps[key]) {
        deps[key].forEach(subscriber => subscriber());
      }
      return true;
    }
  });
}

// 订阅者
let currentSubscriber = null;

function watch(getter, callback) {
  const effect = () => {
    currentSubscriber = effect;
    getter();
    currentSubscriber = null;
  };
  effect();
}

// 使用
const data = reactive({ count: 0, name: 'Alice' });

watch(
  () => data.count * 2,
  (value) => console.log(`Count doubled: ${value}`)
);

watch(
  () => data.name.toUpperCase(),
  (value) => console.log(`Name uppercased: ${value}`)
);

data.count = 5;   // Count doubled: 10
data.name = 'Bob'; // Name uppercased: BOB
```

## 命令模式（Command）

### 概念

将请求封装成对象，从而允许你参数化客户端、队列或记录请求。

### 实现

```javascript
// 命令接口
class Command {
  execute() {}
  undo() {}
}

// 具体命令
class LightOnCommand extends Command {
  constructor(light) {
    super();
    this.light = light;
  }

  execute() {
    return this.light.on();
  }

  undo() {
    return this.light.off();
  }
}

class LightOffCommand extends Command {
  constructor(light) {
    super();
    this.light = light;
  }

  execute() {
    return this.light.off();
  }

  undo() {
    return this.light.on();
  }
}

class ACCommand extends Command {
  constructor(ac, action) {
    super();
    this.ac = ac;
    this.action = action;
    this.previousState = null;
  }

  execute() {
    this.previousState = this.ac.getState();
    return this.ac[this.action]();
  }

  undo() {
    if (this.previousState) {
      return this.ac.setState(this.previousState);
    }
  }
}

// 接收者
class Light {
  on() {
    console.log('Light is ON');
    return 'on';
  }

  off() {
    console.log('Light is OFF');
    return 'off';
  }
}

class AirConditioner {
  constructor() {
    this.state = 'off';
  }

  on() {
    this.state = 'on';
    console.log('AC is ON');
    return this.state;
  }

  off() {
    this.state = 'off';
    console.log('AC is OFF');
    return this.state;
  }

  setTemp(temp) {
    console.log(`AC temp set to ${temp}`);
  }

  getState() {
    return this.state;
  }

  setState(state) {
    this.state = state;
    console.log(`AC state: ${state}`);
  }
}

// 调用者
class RemoteControl {
  constructor() {
    this.history = [];
    this.undone = [];
  }

  executeCommand(command) {
    const result = command.execute();
    this.history.push(command);
    this.undone = [];  // 新命令清除撤销历史
    return result;
  }

  undo() {
    const command = this.history.pop();
    if (command) {
      command.undo();
      this.undone.push(command);
    }
  }

  redo() {
    const command = this.undone.pop();
    if (command) {
      command.execute();
      this.history.push(command);
    }
  }
}

// 使用
const remote = new RemoteControl();
const light = new Light();
const ac = new AirConditioner();

remote.executeCommand(new LightOnCommand(light));
remote.executeCommand(new LightOnCommand(light));
remote.executeCommand(new ACCommand(ac, 'on'));

remote.undo();  // 撤销 AC 命令
remote.redo();  // 重做 AC 命令
```

### 宏命令

```javascript
// 宏命令：批量执行多个命令
class MacroCommand extends Command {
  constructor(commands) {
    super();
    this.commands = commands;
  }

  execute() {
    this.commands.forEach(cmd => cmd.execute());
  }

  undo() {
    this.commands.slice().reverse().forEach(cmd => cmd.undo());
  }
}

// 使用场景：下班模式
const goHomeCommands = [
  new LightOffCommand(officeLight),
  new ACCommand(officeAC, 'off'),
  new CloseBlindsCommand(blinds)
];

const goHomeMacro = new MacroCommand(goHomeCommands);
remote.executeCommand(goHomeMacro);  // 一键执行所有下班操作
```

### 异步命令队列

```javascript
// 命令队列
class CommandQueue {
  constructor() {
    this.queue = [];
    this.executing = false;
  }

  addCommand(command) {
    this.queue.push(command);
    if (!this.executing) {
      this.processQueue();
    }
  }

  async processQueue() {
    this.executing = true;

    while (this.queue.length > 0) {
      const command = this.queue.shift();
      try {
        await command.execute();
      } catch (error) {
        console.error('Command failed:', error);
      }
    }

    this.executing = false;
  }
}

// 使用
const queue = new CommandQueue();

queue.addCommand(new AsyncDownloadCommand(url1));
queue.addCommand(new AsyncDownloadCommand(url2));
queue.addCommand(new AsyncDownloadCommand(url3));
```
