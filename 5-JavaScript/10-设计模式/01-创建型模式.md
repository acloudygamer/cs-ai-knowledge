## 单例模式（Singleton）

### 概念

确保一个类只有一个实例，并提供全局访问点。

### 实现

```javascript
// 懒汉式（延迟初始化）
class Singleton {
  constructor() {
    if (Singleton.instance) {
      return Singleton.instance;
    }
    this.data = [];
    Singleton.instance = this;
  }

  add(item) {
    this.data.push(item);
  }

  get() {
    return this.data;
  }
}

const s1 = new Singleton();
const s2 = new Singleton();
console.log(s1 === s2);  // true
```

```javascript
// 饿汉式（立即初始化）
class Singleton {
  constructor() {
    this.data = [];
  }

  add(item) {
    this.data.push(item);
  }
}

Singleton.instance = new Singleton();

const s1 = Singleton.instance;
const s2 = Singleton.instance;
console.log(s1 === s2);  // true
```

```javascript
// 闭包实现（保护私有变量）
const Singleton = (function() {
  let instance;

  function init() {
    const data = [];
    return {
      add(item) { data.push(item); },
      get() { return data; }
    };
  }

  return {
    getInstance() {
      if (!instance) {
        instance = init();
      }
      return instance;
    }
  };
})();

const s1 = Singleton.getInstance();
const s2 = Singleton.getInstance();
console.log(s1 === s2);  // true
```

## 工厂模式（Factory）

### 概念

定义创建对象的接口，让子类决定实例化哪个类。

### 简单工厂

```javascript
class User {}
class Admin extends User {}
class Guest extends User {}

function createUser(type) {
  switch (type) {
    case 'admin':
      return new Admin();
    case 'guest':
      return new Guest();
    default:
      return new User();
  }
}

const admin = createUser('admin');
const guest = createUser('guest');
```

### 工厂方法

```javascript
class Dialog {
  createButton() {
    throw new Error('createButton must be implemented');
  }

  render() {
    const button = this.createButton();
    button.render();
    return button;
  }
}

class WindowsDialog extends Dialog {
  createButton() {
    return new WindowsButton();
  }
}

class MacDialog extends Dialog {
  createButton() {
    return new MacButton();
  }
}

// 抽象产品
class Button {
  render() {
    throw new Error('render must be implemented');
  }
}

// 具体产品
class WindowsButton extends Button {
  render() {
    console.log('Rendering Windows button');
  }
}

class MacButton extends Button {
  render() {
    console.log('Rendering Mac button');
  }
}

// 客户端代码
const os = 'mac';
const dialog = os === 'windows' ? new WindowsDialog() : new MacDialog();
dialog.render();
```

### 抽象工厂

```javascript
// 抽象工厂：创建一系列相关对象
class GUIFactory {
  createButton() { throw new Error('Not implemented'); }
  createCheckbox() { throw new Error('Not implemented'); }
}

class WindowsFactory extends GUIFactory {
  createButton() { return new WindowsButton(); }
  createCheckbox() { return new WindowsCheckbox(); }
}

class MacFactory extends GUIFactory {
  createButton() { return new MacButton(); }
  createCheckbox() { return new MacCheckbox(); }
}

// 客户端
function createUI(factory) {
  const button = factory.createButton();
  const checkbox = factory.createCheckbox();
  return { button, checkbox };
}

const ui = createUI(new WindowsFactory());
```

### 应用场景

```javascript
// 消息工厂
class Message {
  constructor(content) {
    this.content = content;
    this.timestamp = Date.now();
  }
}

class TextMessage extends Message {
  constructor(content) {
    super(content);
    this.type = 'text';
  }
}

class ImageMessage extends Message {
  constructor(content, url) {
    super(content);
    this.type = 'image';
    this.url = url;
  }
}

class MessageFactory {
  static create(type, content, extra = {}) {
    switch (type) {
      case 'text':
        return new TextMessage(content);
      case 'image':
        return new ImageMessage(content, extra.url);
      default:
        throw new Error(`Unknown message type: ${type}`);
    }
  }
}

const msg = MessageFactory.create('image', 'Photo', { url: '/photo.jpg' });
```

## 建造者模式（Builder）

### 概念

将一个复杂对象的构建与它的表示分离。

### 实现

```javascript
// 指挥者
class Director {
  construct(builder) {
    builder.step1();
    builder.step2();
    return builder.getResult();
  }
}

// 建造者接口
class Builder {
  step1() {}
  step2() {}
  getResult() {}
}

// 具体建造者
class CarBuilder extends Builder {
  constructor() {
    super();
    this.car = {};
  }

  step1() {
    this.car.body = 'sedan';
  }

  step2() {
    this.car.engine = 'v6';
  }

  getResult() {
    return this.car;
  }
}

class SUVBuilder extends Builder {
  constructor() {
    super();
    this.car = {};
  }

  step1() {
    this.car.body = 'suv';
  }

  step2() {
    this.car.engine = 'v8';
  }

  getResult() {
    return this.car;
  }
}

// 使用
const director = new Director();
const sedan = director.construct(new CarBuilder());
const suv = director.construct(new SUVBuilder());
```

```javascript
// 流式 API 建造者
class QueryBuilder {
  constructor() {
    this.query = {
      select: [],
      from: '',
      where: [],       // { field, operator, value } 格式
      params: [],     // 参数化查询参数
      orderBy: [],
      limit: null
    };
  }

  select(...fields) {
    this.query.select.push(...fields);
    return this;  // 返回 this 支持链式调用
  }

  from(table) {
    this.query.from = table;
    return this;
  }

  // 参数化查询，防止 SQL 注入
  where(field, operator, value) {
    this.query.where.push({ field, operator });
    this.query.params.push(value);
    return this;
  }

  orderBy(field, direction = 'ASC') {
    this.query.orderBy.push(`${field} ${direction}`);
    return this;
  }

  limit(n) {
    this.query.limit = n;
    return this;
  }

  build() {
    let sql = `SELECT ${this.query.select.join(', ')} FROM ${this.query.from}`;

    if (this.query.where.length) {
      const conditions = this.query.where.map((cond, i) => {
        const paramIndex = this.query.params[i];
        return `${cond.field} ${cond.operator} ${typeof paramIndex === 'number' ? paramIndex : `'${paramIndex}'`}`;
      });
      sql += ` WHERE ${conditions.join(' AND ')}`;
    }

    if (this.query.orderBy.length) {
      sql += ` ORDER BY ${this.query.orderBy.join(', ')}`;
    }

    if (this.query.limit) {
      sql += ` LIMIT ${this.query.limit}`;
    }

    return sql;
  }
}

// 使用
const sql = new QueryBuilder()
  .select('id', 'name', 'email')
  .from('users')
  .where('status', '=', 'active')
  .where('age', '>', 18)
  .orderBy('name', 'ASC')
  .limit(10)
  .build();

console.log(sql);
// SELECT id, name, email FROM users WHERE status = 'active' AND age > 18 ORDER BY name ASC LIMIT 10
```

```javascript
// HTTP 请求建造者
class HttpRequest {
  constructor() {
    this.options = {
      method: 'GET',
      headers: {},
      body: null,
      timeout: 5000
    };
  }

  method(value) {
    this.options.method = value;
    return this;
  }

  url(value) {
    this.options.url = value;
    return this;
  }

  header(key, value) {
    this.options.headers[key] = value;
    return this;
  }

  body(data) {
    this.options.body = JSON.stringify(data);
    this.options.headers['Content-Type'] = 'application/json';
    return this;
  }

  timeout(ms) {
    this.options.timeout = ms;
    return this;
  }

  async send() {
    const { url, ...rest } = this.options;
    return fetch(url, rest);
  }
}

// 使用
const response = await new HttpRequest()
  .url('https://api.example.com/users')
  .method('POST')
  .header('Authorization', 'Bearer token')
  .body({ name: 'Alice', email: 'alice@example.com' })
  .timeout(10000)
  .send();
```
