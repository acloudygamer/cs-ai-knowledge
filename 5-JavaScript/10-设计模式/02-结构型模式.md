## 装饰器模式（Decorator）

### 概念

动态地给对象添加一些额外的职责。

### 实现

```javascript
// 基础组件
class Coffee {
  cost() {
    return 5;
  }

  description() {
    return 'Coffee';
  }
}

// 装饰器基类
class CoffeeDecorator {
  constructor(coffee) {
    this.coffee = coffee;
  }

  cost() {
    return this.coffee.cost();
  }

  description() {
    return this.coffee.description();
  }
}

// 具体装饰器
class Milk extends CoffeeDecorator {
  cost() {
    return this.coffee.cost() + 1.5;
  }

  description() {
    return this.coffee.description() + ', Milk';
  }
}

class Sugar extends CoffeeDecorator {
  cost() {
    return this.coffee.cost() + 0.5;
  }

  description() {
    return this.coffee.description() + ', Sugar';
  }
}

class Whip extends CoffeeDecorator {
  cost() {
    return this.coffee.cost() + 2;
  }

  description() {
    return this.coffee.description() + ', Whip';
  }
}

// 使用
let myCoffee = new Coffee();
console.log(myCoffee.description(), '$' + myCoffee.cost());
// Coffee $5

myCoffee = new Milk(myCoffee);
console.log(myCoffee.description(), '$' + myCoffee.cost());
// Coffee, Milk $6.5

myCoffee = new Sugar(myCoffee);
console.log(myCoffee.description(), '$' + myCoffee.cost());
// Coffee, Milk, Sugar $7
```

### ES7 装饰器语法

```javascript
// 装饰器（需要 Babel 或 TypeScript 支持）
function readonly(target, name, descriptor) {
  descriptor.writable = false;
  return descriptor;
}

function log(target, name, descriptor) {
  const original = descriptor.value;
  descriptor.value = function(...args) {
    console.log(`Calling ${name}(${args.join(', ')})`);
    return original.apply(this, args);
  };
  return descriptor;
}

class Circle {
  @readonly
  PI = 3.14;

  @log
  area(radius) {
    return this.PI * radius * radius;
  }
}

const c = new Circle();
c.area(5);  // Calling area(5)
// c.PI = 3;  // Error: Cannot assign to readonly property
```

## 适配器模式（Adapter）

### 概念

将一个类的接口转换成客户希望的另外一个接口。

### 实现

```javascript
// 现有接口（旧系统）
class OldPaymentSystem {
  processPayment(amount, userId) {
    console.log(`Processing $${amount} for user ${userId}`);
    return { transactionId: 'OLD_123' };
  }
}

// 新接口（第三方库）
class NewPaymentAPI {
  pay(paymentDetails) {
    console.log(`Paying ${paymentDetails.amount} via ${paymentDetails.method}`);
    return Promise.resolve({ id: 'NEW_456', status: 'success' });
  }
}

// 适配器
class PaymentAdapter {
  constructor() {
    this.newAPI = new NewPaymentAPI();
  }

  processPayment(amount, userId) {
    // 转换旧接口参数为新接口格式
    const paymentDetails = {
      amount: amount,
      method: 'credit_card',
      userId: userId,
      currency: 'USD'
    };

    // 调用新接口并转换返回值
    return this.newAPI.pay(paymentDetails).then(result => ({
      transactionId: result.id,
      status: result.status
    }));
  }
}

// 客户端代码
async function checkout(amount, userId) {
  const adapter = new PaymentAdapter();
  const result = await adapter.processPayment(amount, userId);
  console.log('Transaction:', result.transactionId);
}
```

### 适配不同数据格式

```javascript
// 适配不同格式的 API 响应
class ExternalUserService {
  fetchUser(id) {
    // 返回 XML 格式（模拟）
    return Promise.resolve({
      getElementsByTagName: (tag) => {
        if (tag === 'name') return ['John Doe'];
        if (tag === 'email') return ['john@example.com'];
        return [];
      }
    });
  }
}

// 适配器：XML -> JSON
class UserServiceAdapter {
  constructor() {
    this.service = new ExternalUserService();
  }

  async getUser(id) {
    const xmlResponse = await this.service.fetchUser(id);

    // 转换为统一格式
    return {
      name: xmlResponse.getElementsByTagName('name')[0],
      email: xmlResponse.getElementsByTagName('email')[0]
    };
  }
}

// 客户端代码
const adapter = new UserServiceAdapter();
adapter.getUser(1).then(user => {
  console.log(user);  // { name: 'John Doe', email: 'john@example.com' }
});
```

## 代理模式（Proxy）

### 概念

为其他对象提供一种代理以控制对这个对象的访问。

### 实现

```javascript
// 真实对象
class Image {
  constructor(url) {
    this.url = url;
    console.log(`Loading image from ${url}`);
  }

  display() {
    console.log(`Displaying image: ${this.url}`);
  }
}

// 代理对象
class ImageProxy {
  constructor(url) {
    this.url = url;
    this.realImage = null;
  }

  display() {
    // 懒加载：只有真正需要时才加载真实对象
    if (!this.realImage) {
      this.realImage = new Image(this.url);
    }
    this.realImage.display();
  }
}

// 使用
const proxy = new ImageProxy('https://example.com/large-image.jpg');
console.log('Proxy created');  // 真实图片未加载
proxy.display();  // Loading image from https://example.com/large-image.jpg
                   // Displaying image: https://example.com/large-image.jpg
```

### 保护代理

```javascript
// 主题接口
class BankAccount {
  constructor(balance) {
    this.balance = balance;
  }

  withdraw(amount) {
    if (amount > this.balance) {
      throw new Error('Insufficient funds');
    }
    this.balance -= amount;
    return this.balance;
  }

  getBalance() {
    return this.balance;
  }
}

// 代理：权限控制
class BankAccountProxy {
  constructor(account, userRole) {
    this.account = account;
    this.userRole = userRole;
  }

  withdraw(amount) {
    // 权限检查
    if (this.userRole !== 'admin' && amount > 1000) {
      throw new Error('Authorization required for amounts over $1000');
    }
    return this.account.withdraw(amount);
  }

  getBalance() {
    // 所有人都可以查看余额
    return this.account.getBalance();
  }
}

// 使用
const account = new BankAccount(5000);
const userProxy = new BankAccountProxy(account, 'user');
const adminProxy = new BankAccountProxy(account, 'admin');

userProxy.withdraw(500);   // OK
userProxy.withdraw(2000);  // Error: Authorization required
adminProxy.withdraw(2000); // OK
```

### 虚拟代理（延迟初始化）

```javascript
// 虚拟代理：延迟大数据结构初始化
class HeavyData {
  constructor() {
    // 模拟昂贵的初始化
    this.data = Array.from({ length: 100000 }, (_, i) => ({
      id: i,
      content: `Data ${i}`
    }));
    console.log('HeavyData initialized');
  }

  query(predicate) {
    return this.data.filter(predicate);
  }
}

class HeavyDataProxy {
  constructor() {
    this.instance = null;
  }

  getInstance() {
    if (!this.instance) {
      this.instance = new HeavyData();
    }
    return this.instance;
  }

  query(predicate) {
    // 懒加载到真正调用时才初始化
    return this.getInstance().query(predicate);
  }
}

// 使用
const proxy = new HeavyDataProxy();
console.log('Proxy created');  // HeavyData 未初始化

const results = proxy.query(item => item.id < 10);  // HeavyData initialized
console.log(results.length);  // 10
```

### 缓存代理

```javascript
// 缓存代理：存储计算结果
class Calculator {
  calculate(expression) {
    console.log(`Calculating: ${expression}`);
    // 模拟耗时计算
    return eval(expression);
  }
}

class CalculatorProxy {
  constructor() {
    this.calculator = new Calculator();
    this.cache = new Map();
  }

  calculate(expression) {
    if (this.cache.has(expression)) {
      console.log(`Cache hit for: ${expression}`);
      return this.cache.get(expression);
    }

    const result = this.calculator.calculate(expression);
    this.cache.set(expression, result);
    return result;
  }
}

// 使用
const proxy = new CalculatorProxy();
proxy.calculate('2 + 2');     // Calculating: 2 + 2
proxy.calculate('2 + 2');     // Cache hit for: 2 + 2
proxy.calculate('3 * 3');     // Calculating: 3 * 3
```
