# 进程与集群

## 进程基础

### process 对象

```javascript
// 进程信息
console.log(process.pid);           // 进程 ID
console.log(process.platform);      // 'win32', 'linux', 'darwin'
console.log(process.arch);           // 'x64', 'arm64'
console.log(process.version);        // Node.js 版本
console.log(process.env.NODE_ENV);   // 环境变量

// 进程内存
console.log(process.memoryUsage());
// {
//   rss: 27156480,        // 物理内存
//   heapTotal: 5742592,   // V8 堆总量
//   heapUsed: 2418352,    // V8 堆使用
//   external: 1104074,    // C++ 对象内存
//   arrayBuffers: 7168    // ArrayBuffer 内存
// }

// CPU 使用
console.log(process.cpuUsage());
// { user: 120000, system: 45000 }

// 运行时长
console.log(process.uptime());  // 秒
```

### 进程事件

```javascript
process.on('exit', (code) => {
  console.log(`进程退出，代码: ${code}`);
});

process.on('beforeExit', (code) => {
  console.log('即将退出，代码:', code);
});

process.on('uncaughtException', (err) => {
  console.error('未捕获异常:', err);
  process.exit(1);
});

process.on('unhandledRejection', (reason, promise) => {
  console.error('未处理的 Promise 拒绝:', reason);
});

// 信号监听
process.on('SIGTERM', () => {
  console.log('收到 SIGTERM 信号');
  process.exit(0);
});

process.on('SIGINT', () => {
  console.log('收到 Ctrl+C');
  process.exit(0);
});
```

### 进程控制

```javascript
// 退出
process.exit(0);   // 正常退出
process.exit(1);   // 异常退出

// 杀死进程
process.kill(pid, 'SIGTERM');

// 禁止退出
process.exitCode = 1;

// 优雅关闭
process.on('SIGTERM', () => {
  server.close(() => {
    console.log('服务器已关闭');
    process.exit(0);
  });

  // 强制退出（如果 30 秒后还没关闭）
  setTimeout(() => {
    process.exit(1);
  }, 30000);
});
```

## child_process

### spawn

```javascript
const { spawn } = require('child_process');

// 异步生成子进程
const child = spawn('node', ['child.js'], {
  cwd: './src',           // 工作目录
  env: { ...process.env, NODE_ENV: 'production' },
  stdio: 'pipe'          // 'pipe' | 'inherit' | 'ignore'
});

// stdout
child.stdout.on('data', (data) => {
  console.log('stdout:', data.toString());
});

// stderr
child.stderr.on('data', (data) => {
  console.error('stderr:', data.toString());
});

child.on('exit', (code) => {
  console.log('子进程退出，代码:', code);
});

child.on('error', (err) => {
  console.error('启动失败:', err);
});

// 杀死子进程
child.kill('SIGTERM');
```

### exec

```javascript
const { exec } = require('child_process');

// 执行命令并获取输出
exec('ls -la', { shell: true }, (error, stdout, stderr) => {
  if (error) {
    console.error('执行错误:', error);
    return;
  }
  console.log('stdout:', stdout);
  console.error('stderr:', stderr);
});

// exec 文件同步
const { execSync } = require('child_process');

try {
  const output = execSync('node --version', { encoding: 'utf-8' });
  console.log(output);
} catch (error) {
  console.error('执行失败:', error);
}
```

### execFile

```javascript
const { execFile } = require('child_process');

// 直接执行可执行文件
execFile('node', ['--version'], (error, stdout) => {
  if (error) throw error;
  console.log(stdout);  // v20.10.0
});
```

### fork

```javascript
// parent.js
const { fork } = require('child_process');

const child = fork('./child.js', {
  stdio: 'pipe'
});

child.send({ type: 'hello' }, (err) => {
  if (err) console.error('发送失败:', err);
});

child.on('message', (msg) => {
  console.log('收到子进程消息:', msg);
});

child.on('exit', (code) => {
  console.log('子进程退出:', code);
});

// child.js
process.on('message', (msg) => {
  console.log('收到父进程消息:', msg);

  // 发送消息给父进程
  process.send({ type: 'response', data: 'Hello from child' });
});

process.on('disconnect', () => {
  console.log('与父进程断开连接');
});
```

## 进程间通信（IPC）

### 双向通信

```javascript
// main.js
const { spawn } = require('child_process');

const child = spawn('node', ['worker.js'], {
  stdio: [null, null, null, 'ipc']
});

child.on('message', (msg) => {
  console.log('主进程收到:', msg);
  if (msg.type === 'ready') {
    child.send({ type: 'task', data: 'process this' });
  }
});

child.on('exit', (code) => {
  console.log('Worker 退出:', code);
});

// worker.js
process.send({ type: 'ready' });

process.on('message', async (msg) => {
  if (msg.type === 'task') {
    const result = await processTask(msg.data);
    process.send({ type: 'result', data: result });
  }
});

async function processTask(data) {
  return data.toUpperCase();
}
```

### 使用 send 并处理返回值

```javascript
// main.js
const { fork } = require('child_process');

const child = fork('./compute.js');

// 异步消息
child.send({ numbers: [1, 2, 3, 4, 5] });

child.on('message', (msg) => {
  if (msg.type === 'result') {
    console.log('计算结果:', msg.sum);
  }
});

// compute.js
process.on('message', (msg) => {
  const sum = msg.numbers.reduce((a, b) => a + b, 0);
  process.send({ type: 'result', sum });
});
```

## 集群（cluster）

### 基本用法

```javascript
const cluster = require('cluster');
const http = require('http');
const numCPUs = require('os').cpus().length;

if (cluster.isMaster) {
  console.log(`主进程 ${process.pid} 运行中`);

  // 衍生工作进程
  for (let i = 0; i < numCPUs; i++) {
    cluster.fork();
  }

  // 监听工作进程退出
  cluster.on('exit', (worker, code, signal) => {
    console.log(`工作进程 ${worker.process.pid} 退出`);
    // 重新启动（生产环境应检查退出原因）
    cluster.fork();
  });
} else {
  // 工作进程运行服务器
  http.createServer((req, res) => {
    res.writeHead(200);
    res.end(`Hello from ${process.pid}`);
  }).listen(8000);

  console.log(`工作进程 ${process.pid} 启动`);
}
```

### 进程间共享端口

```javascript
const cluster = require('cluster');
const http = require('http');

if (cluster.isMaster) {
  // 不使用 fork()，而是手动启动
  const server = http.createServer((req, res) => {
    res.end(`Handled by ${process.pid}`);
  });

  // 主进程监听端口
  server.listen(3000, () => {
    console.log(`服务器运行在 3000，进程 ${process.pid}`);
  });

  // fork 工作进程时，它们会自动共享同一个 socket
  for (let i = 0; i < 2; i++) {
    const worker = cluster.fork();

    // 将 server 的 file descriptor 发送给工作进程
    worker.send('server', server);
  }
} else {
  // 工作进程
  let server;

  process.on('message', (msg, server) => {
    if (msg === 'server' && server) {
      server.on('request', (req, res) => {
        res.writeHead(200);
        res.end(`Response from worker ${process.pid}`);
      });
    }
  });
}
```

### PM2 集群模式

```javascript
// ecosystem.config.js
module.exports = {
  apps: [{
    name: 'api-server',
    script: './server.js',
    instances: 'max',           // 最大实例数
    exec_mode: 'cluster',       // 集群模式
    env_production: {
      NODE_ENV: 'production'
    },
    wait_ready: true,           // 等待 process.send('ready')
    kill_timeout: 5000
  }]
};

// server.js
const http = require('http');

const server = http.createServer((req, res) => {
  res.end(`Handled by ${process.pid}`);
});

server.listen(3000, () => {
  console.log(`Server running on ${process.pid}`);
  process.send('ready');  // 通知 PM2 已就绪
});

process.on('SIGTERM', () => {
  console.log('收到 SIGTERM，关闭中...');
  server.close(() => {
    console.log('服务器已关闭');
    process.exit(0);
  });
});
```

## Node.js Streams

### 流的类型

```javascript
const { Readable, Writable, Transform, Duplex, pipeline } = require('stream');
```

### Readable 流

```javascript
// 模式 1: flowing（流动）模式
const fs = require('fs');
const readStream = fs.createReadStream('file.txt');

readStream.on('data', (chunk) => {
  console.log('收到数据:', chunk.toString());
});

readStream.on('end', () => {
  console.log('读取完成');
});

readStream.on('error', (err) => {
  console.error('读取错误:', err);
});

// 模式 2: pause（暂停）模式
readStream.pause();

process.stdin.on('data', (chunk) => {
  process.stdout.write(chunk);
  process.stdin.resume();
});
```

### Writable 流

```javascript
const fs = require('fs');
const writeStream = fs.createWriteStream('output.txt');

writeStream.write('Hello, ');
writeStream.write('World!\n');
writeStream.end('End of file');

writeStream.on('finish', () => {
  console.log('写入完成');
});

writeStream.on('error', (err) => {
  console.error('写入错误:', err);
});
```

### Transform 流

```javascript
const { Transform } = require('stream');

// 转换流：大写转换
const upperCase = new Transform({
  transform(chunk, encoding, callback) {
    this.push(chunk.toString().toUpperCase());
    callback();
  }
});

process.stdin
  .pipe(upperCase)
  .pipe(process.stdout);
```

### Duplex 流

```javascript
const { Duplex } = require('stream');

// Duplex: 同时可读可写（如 TCP socket）
const duplex = new Duplex({
  read(size) {
    this.push('data from read');
  },
  write(chunk, encoding, callback) {
    console.log('writing:', chunk.toString());
    callback();
  }
});
```

### pipeline

```javascript
const { pipeline } = require('stream');
const fs = require('fs');
const zlib = require('zlib');

// 使用 pipeline（自动处理错误）
pipeline(
  fs.createReadStream('input.txt'),
  zlib.createGzip(),
  fs.createWriteStream('input.txt.gz'),
  (err) => {
    if (err) {
      console.error('Pipeline 失败:', err);
    } else {
      console.log('Pipeline 完成');
    }
  }
);
```

### 对象模式流

```javascript
const { Transform } = require('stream');

// 对象模式
const jsonParser = new Transform({
  objectMode: true,
  transform(chunk, encoding, callback) {
    try {
      const obj = JSON.parse(chunk.toString());
      this.push(obj);
      callback();
    } catch (err) {
      callback(err);
    }
  }
});

process.stdin
  .pipe(jsonParser)
  .on('data', (obj) => {
    console.log('Parsed object:', obj);
  })
  .on('error', (err) => {
    console.error('JSON 解析错误:', err.message);
  });
```

## 进程管理工具

### 进程守护（supervisor）

```javascript
// supervisor.js
const { spawn } = require('child_process');
const path = require('path');

let child = null;
let restarting = false;

function startProcess() {
  child = spawn('node', ['server.js'], {
    cwd: path.resolve(__dirname),
    stdio: 'inherit'
  });

  child.on('exit', (code, signal) => {
    if (!restarting) {
      console.log(`进程退出: ${code}，重启中...`);
      setTimeout(startProcess, 1000);
    }
  });

  child.on('error', (err) => {
    console.error('进程错误:', err);
  });
}

startProcess();

process.on('SIGTERM', () => {
  restarting = true;
  child.kill();
  process.exit(0);
});
```

### 自动重启

```javascript
// auto-restart.js
const { fork } = require('child_process');
const chokidar = require('chokidar');

let worker = null;

function startWorker() {
  if (worker) {
    worker.kill('SIGTERM');
  }

  worker = fork('worker.js');

  worker.on('exit', (code) => {
    if (code !== 0 && !worker._suicide) {
      console.log('Worker 异常退出，重启中...');
      setTimeout(startWorker, 1000);
    }
  });

  worker.on('error', (err) => {
    console.error('Worker 错误:', err);
  });
}

// 监听文件变化
const watcher = chokidar.watch('./src', {
  ignored: /(^|[\/\\])\../,
  persistent: true
});

watcher.on('change', (file) => {
  console.log(`文件变化: ${file}，重启 worker`);
  worker._suicide = true;
  startWorker();
});

startWorker();
```

## 实际应用

### 1. 子进程处理 CPU 密集任务

```javascript
// main.js
const { fork } = require('child_process');

function calculateInWorker(data) {
  return new Promise((resolve, reject) => {
    const worker = fork('calculator.js');

    worker.on('message', resolve);
    worker.on('error', reject);
    worker.on('exit', (code) => {
      if (code !== 0) reject(new Error(`Worker exited with ${code}`));
    });

    worker.send(data);
  });
}

// calculator.js
process.on('message', (data) => {
  const result = heavyComputation(data);
  process.send(result);
});

function heavyComputation(n) {
  // CPU 密集计算
  return Array.from({ length: n }, (_, i) => i * i)
    .reduce((sum, val) => sum + val, 0);
}
```

### 2. 流式处理大文件

```javascript
const fs = require('fs');
const { pipeline } = require('stream');
const zlib = require('zlib');

function processLargeFile(input, output) {
  return pipeline(

      }
    }

function createUpperCaseStream() {
  return new Transform({
    transform(chunk, encoding, callback) {
      this.push(chunk.toString().toUpperCase());
      callback();
    }
  });
}

processLargeFile('largefile.txt', 'output.txt');
```

### 3. 零拷贝文件传输

```javascript
const fs = require('fs');
const { spawn } = require('child_process');

function sendFileViaSCP(localPath, remotePath, host) {
  return new Promise((resolve, reject) => {
    const scp = spawn('scp', [localPath, `${host}:${remotePath}`]);

    scp.on('close', (code) => {
      if (code === 0) {
        resolve();
      } else {
        reject(new Error(`SCP 失败，代码: ${code}`));
      }
    });

    scp.on('error', reject);
  });
}
```

## 总结

| API | 用途 |
|-----|------|
| `process.pid` | 进程 ID |
| `process.exit()` | 退出进程 |
| `process.kill()` | 杀死进程 |
| `child_process.spawn()` | 启动子进程 |
| `child_process.exec()` | 执行 shell 命令 |
| `child_process.fork()` | 启动 Node.js 子进程 |
| `cluster.isMaster` | 是否主进程 |
| `cluster.fork()` | 衍生工作进程 |
| `Readable/Writable` | 流接口 |
| `Transform` | 转换流 |
| `pipeline()` | 连接流 |
