# Node.js 数据库操作

## 数据库分类

### SQL vs NoSQL

| 特性 | SQL | NoSQL |
|------|-----|-------|
| 数据模型 | 表格、关系 | 文档、键值、图 |
| 事务 | ACID | BASE |
| 扩展性 | 垂直扩展 | 水平扩展 |
| 典型产品 | PostgreSQL, MySQL | MongoDB, Redis, Cassandra |
| 适用场景 | 结构化数据、复杂查询 | 灵活Schema、高并发 |

```javascript
const { Pool } = require('pg');

const pool = new Pool({
  host: 'localhost',
  port: 5432,
  database: 'mydb',
  user: 'username',
  password: 'password',
  max: 20,  // 连接池最大连接数
  idleTimeoutMillis: 30000,
  connectionTimeoutMillis: 2000,
});

// 查询
async function query(text, params) {
  const start = Date.now();
  const result = await pool.query(text, params);
  const duration = Date.now() - start;
  console.log('Executed query', { text, duration, rows: result.rowCount });
  return result;
}

// 使用
async function main() {
  const result = await query('SELECT * FROM users WHERE id = $1', [1]);
  console.log(result.rows);

  // 释放回连接池
  await pool.end();
}

main();
```

### 事务

```javascript
async function transferMoney(fromId, toId, amount) {
  const client = await pool.connect();

  try {
    await client.query('BEGIN');  // 开始事务

    // 扣款
    await client.query(
      'UPDATE accounts SET balance = balance - $1 WHERE id = $2',
      [amount, fromId]
    );

    // 加款
    await client.query(
      'UPDATE accounts SET balance = balance + $1 WHERE id = $2',
      [amount, toId]
    );

    await client.query('COMMIT');  // 提交事务
    return { success: true };
  } catch (error) {
    await client.query('ROLLBACK');  // 回滚事务
    throw error;
  } finally {
    client.release();  // 释放客户端回连接池
  }
}
```

```javascript
// 批量插入
async function bulkInsertUsers(users) {
  const values = users.map((u, i) =>
    `($${i * 2 + 1}, $${i * 2 + 2})`
  ).join(', ');

  const params = users.flatMap(u => [u.name, u.email]);

  const result = await pool.query(
    `INSERT INTO users (name, email) VALUES ${values} RETURNING *`,
    params
  );

  return result.rows;
}

// 批量更新
async function bulkUpdateStatus(ids, status) {
  const placeholders = ids.map((_, i) => `$${i + 1}`).join(', ');

  const result = await pool.query(
    `UPDATE users SET status = $${ids.length + 1} WHERE id IN (${placeholders}) RETURNING *`,
    [...ids, status]
  );

  return result.rows;
}
```

```javascript
async function createUser(user) {

async function getUsers(filter = {}, options = {}) {
  const { limit = 10, skip = 0, sort = { createdAt: -1 } } = options;

  return users
    .find(filter)
    .sort(sort)
    .skip(skip)
    .limit(limit)
    .toArray();

async function updateUser(id, updates) {
  const result = await users.findOneAndUpdate(
    { _id: new ObjectId(id) },
    { $set: { ...updates, updatedAt: new Date() } },
    { returnDocument: 'after' }
  );

    { _id: new ObjectId(id) },
    { $inc: { age: 1 }, $set: { updatedAt: new Date() } }
  );
```

### 聚合管道

```javascript
// 聚合管道示例：用户订单统计
async function getUserOrderStats() {
  const result = await db.collection('orders').aggregate([
    // 1. 连接用户表
    {
      $lookup: {
        from: 'users',
        localField: 'userId',
        foreignField: '_id',
        as: 'user'
      }
    },
    // 2. 展开用户数组
    { $unwind: '$user' },
    // 3. 按用户分组
    {
      $group: {
        _id: '$user._id',
        userName: { $first: '$user.name' },
        totalOrders: { $sum: 1 },
        totalSpent: { $sum: '$amount' },
        avgOrderValue: { $avg: '$amount' }
      }
    },
    // 4. 排序
    { $sort: { totalSpent: -1 } },
    // 5. 限制
    { $limit: 10 }
  ]).toArray();

  return result;
}

// 复杂聚合：月度销售趋势
async function getMonthlySales(year) {
  return db.collection('orders').aggregate([
    {
      $match: {
        createdAt: {
          $gte: new Date(`${year}-01-01`),
          $lt: new Date(`${year + 1}-01-01`)
        }
      }
    },
    {
      $group: {
        _id: { month: { $month: '$createdAt' } },
        revenue: { $sum: '$amount' },
        orders: { $sum: 1 }
      }
    },
    { $sort: { '_id.month': 1 } }
  ]).toArray();
}
```

### 分布式锁

```javascript
// 获取锁
async function acquireLock(lockKey, ttlMs = 10000) {
  const lockValue = Date.now().toString();

  // SET NX EX 原子操作
  const acquired = await client.set(lockKey, lockValue, {
    NX: true,  // Only set if Not eXists
    EX: Math.ceil(ttlMs / 1000)  // 过期时间（秒）
  });

  return acquired ? lockValue : null;
}

// 释放锁（Lua 脚本保证原子性）
async function releaseLock(lockKey, lockValue) {
  const script = `
    if redis.call("get", KEYS[1]) == ARGV[1] then
      return redis.call("del", KEYS[1])
    else
      return 0
    end
  `;

  const result = await client.eval(script, {
    keys: [lockKey],
    arguments: [lockValue]
  });

  return result === 1;
}

// 使用锁
async function doWithLock(lockKey, fn) {
  const lockValue = await acquireLock(lockKey);

  if (!lockValue) {
    throw new Error('Failed to acquire lock');
  }

  try {
    return await fn();
  } finally {
    await releaseLock(lockKey, lockValue);
  }
}

// 使用
async function processPayment(orderId) {
  return doWithLock(`lock:order:${orderId}`, async () => {
    // 业务逻辑
  });
}
```

### 消息队列

```javascript
// 简单队列
class RedisQueue {
  constructor(queueName) {
    this.queue = `${queueName}:queue`;
    this.processing = `${queueName}:processing`;
    this.client = createClient();
  }

  async connect() {
    await this.client.connect();
  }

  async enqueue(job) {
    await this.client.lPush(this.queue, JSON.stringify({
      id: Date.now().toString(),
      data: job,
      addedAt: new Date().toISOString()
    }));
  }

  async dequeue() {
    // BRPOP 阻塞等待
    const result = await this.client.brPop(this.queue, 0);
    if (result) {
      const job = JSON.parse(result.element);
      await this.client.lPush(this.processing, result.element);
      return job;
    }
    return null;
  }

  async complete(jobId) {
    // 从处理中移除
    const items = await this.client.lRange(this.processing, 0, -1);
    for (const item of items) {
      const job = JSON.parse(item);
      if (job.id === jobId) {
        await this.client.lRem(this.processing, 1, item);
        break;
      }
    }
  }

  async fail(jobId) {
    // 失败后重新入队
    const items = await this.client.lRange(this.processing, 0, -1);
    for (const item of items) {
      const job = JSON.parse(item);
      if (job.id === jobId) {
        await this.client.lRem(this.processing, 1, item);
        await this.client.lPush(this.queue, item);
        break;
      }
    }
  }
}

// 使用
async function worker() {
  const queue = new RedisQueue('emails');
  await queue.connect();

  while (true) {
    const job = await queue.dequeue();
    if (job) {
      try {
        await sendEmail(job.data);
        await queue.complete(job.id);
      } catch (error) {
        await queue.fail(job.id);
      }
    }
  }
}
```

```javascript
// 分页查询
async function getPosts(page = 1, pageSize = 10) {
  const skip = (page - 1) * pageSize;

  const [posts, total] = await prisma.$transaction([
    prisma.post.findMany({
      where: { published: true },
      skip,
      take: pageSize,

    total,
    page,
    pageSize,
```

## 连接池最佳实践

```javascript
// 统一数据库管理
class Database {
  constructor() {
    this.pool = null;
  }

  async connect() {
    this.pool = new Pool({
      host: process.env.DB_HOST,
      port: process.env.DB_PORT,
      database: process.env.DB_NAME,
      user: process.env.DB_USER,
      password: process.env.DB_PASSWORD,
      max: 20,
      idleTimeoutMillis: 30000
    });

    // 测试连接
    const client = await this.pool.connect();
    client.release();

    return this;
  }

  async query(text, params) {
    const start = Date.now();
    try {
      const result = await this.pool.query(text, params);
      const duration = Date.now() - start;
      if (duration > 1000) {
        console.warn(`Slow query (${duration}ms):`, text);
      }
      return result;
    } catch (error) {
      console.error('Query error:', text, error);
      throw error;
    }
  }

  async transaction(fn) {
    const client = await this.pool.connect();
    try {
      await client.query('BEGIN');
      const result = await fn(client);
      await client.query('COMMIT');
      return result;
    } catch (error) {
      await client.query('ROLLBACK');
      throw error;
    } finally {
      client.release();
    }
  }

  async close() {
    if (this.pool) {
      await this.pool.end();
    }
  }
}

// 导出单例
module.exports = new Database();
```
