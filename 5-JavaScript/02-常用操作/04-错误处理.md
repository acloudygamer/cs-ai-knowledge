# 错误处理

## try/catch/finally

```javascript
try {
  // 可能抛出错误的代码
  const data = JSON.parse(userInput);
  doSomething(data);
} catch (error) {
  // 处理错误
  console.error('Error:', error.message);
  console.error('Stack:', error.stack);
} finally {
  // 总是执行（清理工作）
  closeConnection();
}

// 捕获错误类型
try {
  mightThrow();
} catch (error) {
  if (error instanceof TypeError) {
    // 类型错误
  } else if (error instanceof RangeError) {
    // 范围错误
  } else {
    throw error;  // 重新抛出未知错误
  }
}
```

## 自定义错误

```javascript
// 自定义错误类
class ValidationError extends Error {
  constructor(message, field) {
    super(message);
    this.name = 'ValidationError';
    this.field = field;
    Error.captureStackTrace(this, this.constructor);
  }
}

// 使用
function validateUser(user) {
  if (!user.name) {
    throw new ValidationError('Name is required', 'name');
  }
  if (user.age < 0) {
    throw new ValidationError('Age cannot be negative', 'age');
  }
}

try {
  validateUser({ name: '', age: -1 });
} catch (error) {
  if (error instanceof ValidationError) {
    console.error(`${error.field}: ${error.message}`);
  } else {
    throw error;
  }
}

// Promise 拒绝处理
Promise.reject(new Error('Oops'))
  .catch(error => console.error(error.message));
```

## async/await 错误处理

```javascript
// try/catch 处理 async 函数
async function fetchData(url) {
  try {
    const response = await fetch(url);
    if (!response.ok) {
      throw new Error(`HTTP error: ${response.status}`);
    }
    return await response.json();
  } catch (error) {
    console.error('Fetch failed:', error.message);
    throw error;  // 重新抛出让调用者处理
  }
}

// 并行错误处理
async function fetchAll(urls) {
  const results = await Promise.allSettled(
    urls.map(url => fetch(url).then(r => r.json()))
  );

  results.forEach((result, index) => {
    if (result.status === 'rejected') {
      console.error(`URL ${urls[index]} failed:`, result.reason);
    }
  });

  return results.filter(r => r.status === 'fulfilled').map(r => r.value);
}

// 全局未捕获 Promise 拒绝
window.addEventListener('unhandledrejection', event => {
  console.error('Unhandled promise rejection:', event.reason);
});
```

## 错误边界（React 风格）

```javascript
// 模拟错误边界模式
class ErrorBoundary {
  constructor() {
    this.error = null;
  }

  tryCatch(fn) {
    try {
      return { success: true, value: fn() };
    } catch (error) {
      return { success: false, error };
    }
  }

  componentDidCatch(error, errorInfo) {
    console.error('Error:', error);
    console.error('Error Info:', errorInfo);
    this.error = error;
  }

  render() {
    if (this.error) {
      return `<div>Something went wrong: ${this.error.message}</div>`;
    }
    return null;
  }
}

// 使用
const boundary = new ErrorBoundary();
const result = boundary.tryCatch(() => {
  throw new Error('Test error');
});

if (!result.success) {
  console.error('Caught error:', result.error.message);
}
```

## 最佳实践

| 场景 | 推荐方式 |
|------|---------|
| 同步代码 | try/catch |
| Promise 链 | .catch() |
| async/await | try/catch |
| 并行请求 | Promise.allSettled |
| 未知错误 | 顶层 unhandledrejection 监听 |

### 常见错误类型

```javascript
// 错误类型判断
try {
  // 代码
} catch (error) {
  if (error instanceof TypeError) {
    // 类型错误：通常是 undefined 调用方法
  } else if (error instanceof ReferenceError) {
    // 引用错误：变量未定义
  } else if (error instanceof SyntaxError) {
    // 语法错误：代码写错了
  } else if (error instanceof RangeError) {
    // 范围错误：数值超出范围
  } else {
    // 自定义错误或未知错误
    throw error;
  }
}
```
