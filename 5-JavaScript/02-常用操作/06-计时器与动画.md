# 计时器与动画

## setTimeout 与 setInterval

### setTimeout

```javascript
// 基本用法
const timeoutId = setTimeout(() => {
  console.log('2秒后执行');
}, 2000);

// 取消定时器
clearTimeout(timeoutId);

// 带参数
setTimeout((name, age) => {
  console.log(`${name} is ${age} years old`);
}, 1000, 'Alice', 25);
```

### setInterval

```javascript
// 基本用法
const intervalId = setInterval(() => {
  console.log('每秒执行一次');
}, 1000);

// 取消定时器
clearInterval(intervalId);

// 倒计时示例
let count = 10;
const timer = setInterval(() => {
  console.log(count--);
  if (count === 0) {
    clearInterval(timer);
    console.log('倒计时结束');
  }
}, 1000);
```

## requestAnimationFrame

### 基本概念

requestAnimationFrame 会在下一次重绘之前调用指定的回调函数，比 setInterval 更适合动画场景。

```javascript
// 动画循环
function animate(timestamp) {
  console.log('当前时间戳:', timestamp);
  // 执行动画逻辑
  requestAnimationFrame(animate);
}

const animationId = requestAnimationFrame(animate);

// 取消动画
cancelAnimationFrame(animationId);
```

### 动画示例：平滑移动

```javascript
function moveElement(element, targetX, duration) {
  const startX = element.getBoundingClientRect().left;
  const startTime = performance.now();

  function step(currentTime) {
    const elapsed = currentTime - startTime;
    const progress = Math.min(elapsed / duration, 1);

    // 缓动函数（ease-out）
    const easeOut = 1 - Math.pow(1 - progress, 3);
    const currentX = startX + (targetX - startX) * easeOut;

    element.style.transform = `translateX(${currentX}px)`;

    if (progress < 1) {
      requestAnimationFrame(step);
    }
  }

  requestAnimationFrame(step);
}

// 使用
const box = document.querySelector('.box');
moveElement(box, 500, 1000);
```

## 动画性能优化

### 使用 transform 和 opacity

```javascript
// 推荐：使用 transform 和 opacity
element.style.transform = 'translateX(100px)';
element.style.opacity = '0.5';

// 避免：改变布局属性
element.style.width = '200px';    // 触发重排
element.style.backgroundColor = 'red';  // 触发重绘
```

### will-change 提示

```javascript
// 告诉浏览器即将发生变换，优化渲染
element.style.willChange = 'transform';
element.style.transform = 'translateX(100px)';

// 动画结束后清除
element.addEventListener('transitionend', () => {
  element.style.willChange = 'auto';
});
```

## 帧率控制

### 节流动画

```javascript
// 使用 requestAnimationFrame 实现节流
function throttle(callback, fps = 60) {
  const interval = 1000 / fps;
  let lastTime = 0;

  return function(...args) {
    const now = performance.now();
    if (now - lastTime >= interval) {
      lastTime = now;
      callback.apply(this, args);
    }
  };
}

// 使用
const throttledScroll = throttle((event) => {
  console.log('滚动位置:', window.scrollY);
}, 30);

window.addEventListener('scroll', throttledScroll);
```

## 典型应用场景

### 防抖与节流

```javascript
// 防抖：在事件停止触发后延迟执行
function debounce(fn, delay) {
  let timer = null;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => {
      fn.apply(this, args);
    }, delay);
  };
}

// 节流：固定频率执行
function throttle(fn, interval) {
  let lastTime = 0;
  return function(...args) {
    const now = Date.now();
    if (now - lastTime >= interval) {
      lastTime = now;
      fn.apply(this, args);
    }
  };
}

// 使用场景
const handleSearch = debounce((query) => {
  console.log('搜索:', query);
}, 300);

const handleResize = throttle(() => {
  console.log('窗口尺寸:', window.innerWidth);
}, 100);
```

### 延迟加载图片

```javascript
function lazyLoadImages() {
  const images = document.querySelectorAll('img[data-src]');

  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const img = entry.target;
        img.src = img.dataset.src;
        img.removeAttribute('data-src');
        observer.unobserve(img);
      }
    });
  });

  images.forEach(img => observer.observe(img));
}

// 启动
lazyLoadImages();
```
