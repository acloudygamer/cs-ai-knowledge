# TypeScript 类型高级特性

## 类型守卫（Type Guards）

### 自定义类型守卫

```typescript
// 基本类型守卫函数
function isString(value: unknown): value is string {
  return typeof value === 'string';
}

function isNumber(value: unknown): value is number {
  return typeof value === 'number';
}

function isArray(value: unknown): value is unknown[] {
  return Array.isArray(value);
}

// 使用
function process(value: unknown) {
  if (isString(value)) {
    console.log(value.toUpperCase());  // value 被收窄为 string
  }
}
```

### 类型守卫与联合类型

```typescript
interface Dog {
  kind: 'dog';
  bark(): void;
}

interface Cat {
  kind: 'cat';
  meow(): void;
}

interface Bird {
  kind: 'bird';
  chirp(): void;
}

type Animal = Dog | Cat | Bird;

// 类型守卫函数
function isDog(animal: Animal): animal is Dog {
  return animal.kind === 'dog';
}

function isCat(animal: Animal): animal is Cat {
  return animal.kind === 'cat';
}

function speak(animal: Animal) {
  if (isDog(animal)) {
    animal.bark();  // animal 被收窄为 Dog
  } else if (isCat(animal)) {
    animal.meow();  // animal 被收窄为 Cat
  } else {
    animal.chirp();  // animal 被收窄为 Bird
  }
}

// switch 模式（可辨识联合）
function speakSwitch(animal: Animal) {
  switch (animal.kind) {
    case 'dog':
      animal.bark();
      break;
    case 'cat':
      animal.meow();
      break;
    case 'bird':
      animal.chirp();
      break;
  }
}
```

### in 操作符守卫

```typescript
interface Admin {
  role: 'admin';
  permissions: string[];
}

interface User {
  role: 'user';
  email: string;
}

function isAdmin(obj: Admin | User): obj is Admin {
  return 'permissions' in obj;
}

function handle(obj: Admin | User) {
  if (isAdmin(obj)) {
    console.log(obj.permissions);  // Admin
  } else {
    console.log(obj.email);  // User
  }
}
```

### unknown 与类型收窄

```typescript
// unknown 是类型安全的 any
function parseJSON(json: string): unknown {
  return JSON.parse(json);
}

const data = parseJSON('{"name": "Alice"}');

// typeof 守卫
if (typeof data === 'string') {
  console.log(data.toUpperCase());
} else if (typeof data === 'number') {
  console.log(data.toFixed(2));
} else if (typeof data === 'object' && data !== null) {
  console.log(Object.keys(data));
}
```

### never 类型与穷尽检查

```typescript
// never 类型：永不返回的值
function assertNever(value: never): never {
  throw new Error(`Unexpected value: ${JSON.stringify(value)}`);
}

type Shape = Circle | Square | Triangle;

function area(shape: Shape): number {
  switch (shape.kind) {
    case 'circle':
      return Math.PI * shape.radius ** 2;
    case 'square':
      return shape.side ** 2;
    case 'triangle':
      return (shape.base * shape.height) / 2;
    default:
      // 穷尽检查：如果忘记处理某个类型，编译时会报错
      return assertNever(shape);
  }
}
```

## 映射类型（Mapped Types）

### 基本映射类型

```typescript
// 所有属性变为可选
type Partial<T> = {
  [P in keyof T]?: T[P];
};

// 所有属性变为必需
type Required<T> = {
  [P in keyof T]-?: T[P];  // -? 移除可选标记
};

// 所有属性变为只读
type Readonly<T> = {
  readonly [P in keyof T]: T[P];
};

// 选取部分属性
type Pick<T, K extends keyof T> = {
  [P in K]: T[P];
};

// 排除部分属性
type Omit<T, K extends keyof T> = {
  [P in Exclude<keyof T, K>]: T[P];
};
```

### 键重新映射

```typescript
// TypeScript 4.1+ 键重新映射
type Getters<T> = {
  [P in keyof T as `get${Capitalize<string & P>}`]: () => T[P];
};

interface Person {
  name: string;
  age: number;
  email: string;
}

type PersonGetters = Getters<Person>;
// {
//   getName: () => string;
//   getAge: () => number;
//   getEmail: () => string;
// }

// 过滤属性
type RemoveKind<T> = {
  [P in keyof T as Exclude<P, 'kind'>]: T[P];
};

interface Circle {
  kind: 'circle';
  radius: number;
}

type KindlessCircle = RemoveKind<Circle>;
// { radius: number }
```

### 条件映射

```typescript
// 根据属性类型选择
type Stringify<T> = {
  [P in keyof T]: T[P] extends string ? string : never;
};

interface User {
  name: string;
  age: number;
  email: string;
}

type UserStrings = Stringify<User>;
// { name: string; age: never; email: string }

// 可选属性映射
type AutoOptional<T> = {
  [P in keyof T]?: T[P];
} & {
  [P in keyof T as `${string & P}?`]: T[P];
};
```

### 递归映射类型

```typescript
// 深度只读
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object
    ? T[P] extends Function
      ? T[P]
      : DeepReadonly<T[P]>
    : T[P];
};

interface Config {
  db: {
    host: string;
    port: number;
  };
  cache: {
    redis: {
      host: string;
      port: number;
    };
  };
}

type ReadonlyConfig = DeepReadonly<Config>;
// 所有嵌套属性都变为 readonly

// 深度可选
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

type PartialConfig = DeepPartial<Config>;
// 所有嵌套属性都变为可选
```

## 条件类型（Conditional Types）

### 基本语法

```typescript
// T extends U ? X : Y
type IsString<T> = T extends string ? true : false;

type A = IsString<string>;   // true
type B = IsString<number>;   // false

// 联合类型分发
type ToArray<T> = T extends any ? T[] : never;

type C = ToArray<string | number>;  // string[] | number[]

// 阻止分发：用元组包裹
type ToArrayNonDist<T> = [T] extends [any] ? T[] : never;

type D = ToArrayNonDist<string | number>;  // (string | number)[]
```

### 常用内置条件类型

```typescript
// Exclude<T, U> - 从 T 中排除可赋值给 U 的类型
type E = Exclude<string | number | boolean, string | boolean>;  // number

// Extract<T, U> - 从 T 中提取可赋值给 U 的类型
type F = Extract<string | number | boolean, string | boolean>;  // string | boolean

// NonNullable<T> - 排除 null 和 undefined
type G = NonNullable<string | null | undefined>;  // string

// ReturnType<T> - 提取函数返回类型
function greet() { return 'hello'; }
type H = ReturnType<typeof greet>;  // string

// Parameters<T> - 提取函数参数类型
type I = Parameters<typeof greet>;  // []

// InstanceType<T> - 提取构造函数实例类型
class User {}
type J = InstanceType<typeof User>;  // User
```

### infer 关键字

```typescript
// infer：在条件类型中推断类型

// 解包 Promise
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T;

type A = UnwrapPromise<Promise<string>>;  // string
type B = UnwrapPromise<number>;          // number

// 解包数组元素
type ElementType<T> = T extends (infer U)[] ? U : T;

type C = ElementType<string[]>;   // string
type D = ElementType<number[]>;   // number

// 解包函数返回类型（自定义 ReturnType）
type MyReturnType<T extends (...args: any) => any> =
  T extends (...args: any) => infer R ? R : never;

function fetchUser() { return { id: 1, name: 'Alice' }; }
type User = MyReturnType<typeof fetchUser>;  // { id: number; name: string; }

// 解包构造函数
type ConstructorInstance<T> =
  T extends new (...args: any) => infer Instance ? Instance : never;

class Person {}
type P = ConstructorInstance<typeof Person>;  // Person
```

### 条件类型与映射

```typescript
// 动态属性类型
type DynamicProps<T, U> = {
  [P in keyof T]: T[P] extends U ? P : never;
}[keyof T];

interface User {
  name: string;
  age: number;
  email: string;
}

type StringKeys = DynamicProps<User, string>;  // 'name' | 'email'
type NumberKeys = DynamicProps<User, number>;  // 'age'

interface Status {
  pending: 'pending';
  active: 'active';
  done: 'done';
}

type StatusValue = Values<Status>;  // 'pending' | 'active' | 'done'
```

## 模板字面量类型（Template Literal Types）

### 基本用法

```typescript
// 字符串字面量组合
type Direction = 'left' | 'right' | 'up' | 'down';
type EventName = `on${Capitalize<Direction>}`;
// 'onLeft' | 'onRight' | 'onUp' | 'onDown'

// 提取类型部分
type PropEventSource<T> = {
  on<K extends string & keyof T>(eventName: `on${K}`, handler: (value: T[K]) => void): void;
};

interface Person {
  name: string;
  age: number;
}

const personEvents: PropEventSource<Person> = {
  on(event, handler) {
    // ...
  }

personEvents.on('onName', (value) => console.log(value));  // value 是 string
personEvents.on('onAge', (value) => console.log(value));   // value 是 number
```

### 工具类型实现

```typescript
// 字符串首字母大写
type Capitalize<T extends string> = T extends `${infer First}${infer Rest}`
  ? `${Uppercase<First>}${Rest}`
  : T;

// Uncapitalize
type Uncapitalize<T extends string> = T extends `${infer First}${infer Rest}`
  ? `${Lowercase<First>}${Rest}`
  : T;

type A = Capitalize<'hello'>;   // 'Hello'
type B = Uncapitalize<'Hello'>;  // 'hello'

// 驼峰转下划线
type ToSnakeCase<T extends string> = T extends `${infer Part}${infer Rest}`
  ? Part extends Uppercase<Part>
    ? `_${Lowercase<Part>}${ToSnakeCase<Rest>}`
    : `${Part}${ToSnakeCase<Rest>}`
  : T;

type C = ToSnakeCase<'helloWorld'>;  // 'hello_world'
type D = ToSnakeCase<'APIClient'>;   // 'api_client'
```

## 分布式杂异类型（Discriminated Unions）

### 基本模式

```typescript
// 可辨识联合：使用共同的字面量类型作为辨识标签
interface SuccessResponse {
  status: 'success';
  data: any;
}

interface ErrorResponse {
  status: 'error';
  message: string;
  code: number;
}

interface LoadingResponse {
  status: 'loading';
}

type ApiResponse = SuccessResponse | ErrorResponse | LoadingResponse;

function handleResponse(response: ApiResponse) {
  switch (response.status) {
    case 'success':
      console.log(response.data);  // response 是 SuccessResponse
      break;
    case 'error':
      console.log(response.message, response.code);  // response 是 ErrorResponse
      break;
    case 'loading':
      console.log('Loading...');  // response 是 LoadingResponse
      break;
  }
}
```

### 复杂可辨识联合

```typescript
// 状态机
type OrderState =
  | { state: 'pending'; orderId: string }
  | { state: 'processing'; orderId: string; startTime: Date }
  | { state: 'shipped'; orderId: string; trackingNumber: string }
  | { state: 'delivered'; orderId: string; deliveryTime: Date }
  | { state: 'cancelled'; orderId: string; reason: string };

function trackOrder(order: OrderState) {
  const { state, orderId } = order;

  switch (state) {
    case 'pending':
      return `Order ${orderId} is pending`;

    case 'processing':
      return `Order ${orderId} processing since ${order.startTime}`;

    case 'shipped':
      return `Order ${orderId} shipped with tracking ${order.trackingNumber}`;

    case 'delivered':
      return `Order ${orderId} delivered at ${order.deliveryTime}`;

    case 'cancelled':
      return `Order ${orderId} cancelled: ${order.reason}`;
  }
}
```

### 链式可辨识联合

```typescript
// Result 类型（类似 Rust）
type Ok<T, E = never> = {
  success: true;
  data: T;
  error?: never;
};

type Err<E, T = never> = {
  success: false;
  data?: never;
  error: E;
};

type Result<T, E = Error> = Ok<T, E> | Err<E, T>;

async function fetchUser(id: number): Promise<Result<User, ApiError>> {
  try {
    const response = await fetch(`/api/users/${id}`);
    if (!response.ok) {
      return {
        success: false,
        error: { code: response.status, message: response.statusText }
      };
    }
    const data = await response.json();
    return { success: true, data };
  } catch (e) {
    return { success: false, error: { code: -1, message: String(e) } };
  }
}

// 使用
async function main() {
  const result = await fetchUser(1);

  if (result.success) {
    console.log(result.data.name);  // TypeScript 知道这是 User
  } else {
    console.error(result.error.message);  // TypeScript 知道这是 ApiError
  }
}
```

## 装饰器与反射元数据

### 反射元数据（Reflect Metadata）

```bash
npm install reflect-metadata
```

```typescript
import 'reflect-metadata';

// 定义元数据键
const METADATA_KEY = {
  REQUIRED: 'required',
  FORMAT: 'format',
  RANGE: 'range'
};

// 属性装饰器：定义格式
function format(formatString: string) {
  return Reflect.metadata(METADATA_KEY.FORMAT, formatString);
}

// 参数装饰器：定义必需
function required(target: any, propertyKey: string | symbol, parameterIndex: number) {
  const existingRequired: Set<number> =
    Reflect.getMetadata(METADATA_KEY.REQUIRED, target, propertyKey) || new Set();

  existingRequired.add(parameterIndex);
  Reflect.defineMetadata(METADATA_KEY.REQUIRED, existingRequired, target, propertyKey);
}

// 验证装饰器
function validate(target: any, propertyKey: string | symbol, descriptor: PropertyDescriptor) {
  const original = descriptor.value;

  descriptor.value = function(...args: any[]) {
    const requiredParams: Set<number> =
      Reflect.getMetadata(METADATA_KEY.REQUIRED, target, propertyKey) || new Set();

    // 验证必需参数
    for (const index of requiredParams) {
      if (args[index] === undefined) {
        throw new Error(`Parameter at index ${index} is required`);
      }
    }

    return original.apply(this, args);
  };

  return descriptor;
}

// 使用
class UserService {
  @validate
  createUser(
    @required name: string,
    @format('email') email: string
  ) {
    return { name, email };
  }
}

const service = new UserService();
service.createUser('Alice', 'alice@example.com');  // OK
// service.createUser(undefined, 'alice@example.com');  // Error: Parameter at index 0 is required
```

// 强制类型转换
function unsafeCast<T, U>(value: T): U {

### 类型断言与安全

```typescript
// 类型谓词（更安全的断言）
function isNonNull<T>(value: T): value is NonNullable<T> {
  return value !== null && value !== undefined;
}

function assertIsDefined<T>(value: T, message?: string): asserts value is NonNullable<T> {
  if (value === null || value === undefined) {
    throw new Error(message ?? 'Value is required');
  }
}

function process(value: string | null | undefined) {
  assertIsDefined(value);
  console.log(value.toUpperCase());  // value 现在是 string
}
```

### 条件类型组合

```typescript
// 联合类型属性提取
type UnionToIntersection<U> =
  (U extends any ? (k: U) => void : never) extends ((k: infer I) => void) ? I : never;

type A = UnionToIntersection<{ a: string } | { b: number }>;
// { a: string } & { b: number }

// 获取函数参数的最后一项
type LastArgument<T extends (...args: any) => any> =
  T extends (...args: infer Args) => any
    ? Args extends [...any, infer Last]
      ? Last
      : never
    : never;

function fn(a: string, b: number, c: boolean) {}
type C = LastArgument<typeof fn>;  // boolean
```
