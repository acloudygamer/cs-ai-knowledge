# TypeScript 泛型深入

## 泛型基础

### 泛型函数

```typescript
// 基本泛型函数
function identity<T>(arg: T): T {
  return arg;
}

identity<string>('hello');  // 显式指定
identity(42);               // 类型推断

// 多个类型参数
function pair<K, V>(key: K, value: V): [K, V] {
  return [key, value];
}

pair<string, number>('age', 25);
pair('name', 'Alice');  // 推断为 pair<string, string>
```

### 泛型接口

```typescript
interface Container<T> {
  value: T;
  get(): T;
  set(value: T): void;
}

interface Pair<K, V> {
  key: K;
  value: V;
}

class Box<T> implements Container<T> {
  constructor(public value: T) {}
  get(): T { return this.value; }
  set(value: T): void { this.value = value; }
}
```

### 泛型类

```typescript
class Stack<T> {
  private items: T[] = [];

  push(item: T): void {
    this.items.push(item);
  }

  pop(): T | undefined {
    return this.items.pop();
  }

  peek(): T | undefined {
    return this.items[this.items.length - 1];
  }

  isEmpty(): boolean {
    return this.items.length === 0;
  }
}

const stack = new Stack<number>();
stack.push(1);
stack.push(2);
```

## 泛型约束

### extends 约束

```typescript
// 必须有特定属性
interface HasLength {
  length: number;
}

function logLength<T extends HasLength>(arg: T): T {
  console.log(arg.length);
  return arg;
}

logLength('hello');     // OK
logLength([1, 2, 3]);  // OK
// logLength(123);     // Error

// 约束为另一类型的键
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key];
}

const user = { name: 'Alice', age: 30 };
getProperty(user, 'name');  // string
// getProperty(user, 'email');  // Error
```

### 多重约束

```typescript
// 继承多个约束
interface Printable {
  print(): void;
}

interface Serializable {
  serialize(): string;
}

function process<T extends Printable & Serializable>(obj: T): void {
  obj.print();
  console.log(obj.serialize());
}
```

## 条件类型

### 基本用法

```typescript
// T extends U ? X : Y
type IsString<T> = T extends string ? true : false;

type A = IsString<string>;  // true
type B = IsString<number>;  // false

// 提取元素类型
type ElementType<T> = T extends (infer U)[] ? U : T;

type C = ElementType<string[]>;  // string
type D = ElementType<number>;     // number
```

### 分布式条件类型

```typescript
// 联合类型会自动分发
type ToArray<T> = T extends any ? T[] : never;

type E = ToArray<string | number>;  // string[] | number[]

// 不想分发时，用元组包裹
type ToArrayNonDist<T> = [T] extends [any] ? T[] : never;
type F = ToArrayNonDist<string | number>;  // (string | number)[]
```

### 实际应用

```typescript
// ReturnType：提取函数返回类型
type MyReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

function greet() { return 'hello'; }
type G = MyReturnType<typeof greet>;  // string

// Parameters：提取函数参数类型
type MyParameters<T> = T extends (...args: infer P) => any ? P : never;

type H = MyParameters<typeof greet>;  // []

// UnwrapPromise：解包 Promise
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T;

type I = UnwrapPromise<Promise<string>>;  // string
type J = UnwrapPromise<number>;           // number
```

## 映射类型

### 基本映射

```typescript
// 将所有属性变为可选
type Partial<T> = {
  [P in keyof T]?: T[P];
};

// 将所有属性变为必需
type Required<T> = {
  [P in keyof T]-?: T[P];  // -? 移除可选
};

// 将所有属性变为只读
type Readonly<T> = {
  readonly [P in keyof T]: T[P];
};

// 选取部分属性
type Pick<T, K extends keyof T> = {
  [P in K]: T[P];
};

// 排除部分属性
type Omit<T, K extends keyof T> = {
  [P in Exclude<keyof T, K>]: T[P];
};
```

### 自定义映射

```typescript
// 将所有属性变为字符串类型
type Stringify<T> = {
  [P in keyof T]: string;
};

// 将所有嵌套属性展开
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object
    ? DeepReadonly<T[P]>
    : T[P];
};

// 键值互换
type Flip<T extends Record<string, string | number>> = {
  [P in keyof T]: `${T[P]}`;
};

type K = Flip<{ name: 'Alice'; age: 25 }>;  // { Alice: 'name'; 25: 'age' }
```

## 模板字面量类型

### 基本用法

```typescript
type EventName = 'click' | 'focus' | 'blur';
type HandlerName = `on${Capitalize<EventName>}`;
// 'onClick' | 'onFocus' | 'onBlur'

// 提取类型
type PropEventSource<T> = {
  on<K extends string & keyof T>(eventName: K, handler: (value: T[K]) => void): void;
};

interface Person {
  name: string;
  age: number;
}
```

### 组合用法

```typescript
// CSS 单位
type CSSUnit = 'px' | 'em' | 'rem' | '%';
type CSSProperty = `width` | `height` | `margin` | `padding`;
type CSSValue = `${CSSProperty}: ${number}${CSSUnit}`;

// API 路径
type ApiVersion = 'v1' | 'v2';
type ApiPath = `/api/${ApiVersion}/${string}`;

// 事件类型
type MouseEvent = `on${'Click' | 'MouseEnter' | 'MouseLeave'}`;
```

## 泛型模式与最佳实践

### 泛型默认类型

```typescript
interface Response<T = any> {
  code: number;
  data: T;
  message: string;
}

type UserResponse = Response;        // data: any
type UserResponse2 = Response<User>;  // data: User
```

### 泛型递归

```typescript
// 深度扁平化
type Flatten<T> = T extends (infer U)[]
  ? Flatten<U>
  : T;

type A = Flatten<number[][]>;        // number
type B = Flatten<string[][][]>;      // string

// DeepPartial
type DeepPartial<T> = {
  [P in keyof T]?: T[P] extends object ? DeepPartial<T[P]> : T[P];
};

type C = DeepPartial<{ user: { name: string; age: number } }>;
// { user?: { name?: string; age?: number } | undefined }
```

### 泛型与 new

```typescript
// 构造函数类型
interface Class<T> {
  new(...args: any[]): T;
}

function create<T>(ctor: Class<T>, ...args: any[]): T {
  return new ctor(...args);
}

class Person {
  constructor(public name: string) {}
}

create(Person, 'Alice');  // Person
```

### 泛型别名 vs 接口

```typescript
// 泛型别名：适合复杂类型组合
type Result<T, E = Error> = { ok: true; value: T } | { ok: false; error: E };

// 接口：适合定义结构，支持合并
interface Cache {
  get(key: string): any;
}
interface Cache {
  set(key: string, value: any): void;
}
// 合并后同时具有 get 和 set
```
