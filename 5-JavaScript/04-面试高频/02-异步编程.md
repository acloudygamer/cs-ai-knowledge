# 异步编程面试题

## Promise

### Q: Promise 的三种状态？

```javascript
// pending（待定）：初始状态
// fulfilled（已兑现）：操作成功
// rejected（已拒绝）：操作失败

const promise = new Promise((resolve, reject) => {
  // pending
  if (success) resolve(data);
  else reject(error);
});
// fulfilled 或 rejected（一旦改变，不可逆）
```

### Q: Promise.all、Promise.race、Promise.allSettled 的区别？

```javascript
// Promise.all：所有都成功才成功，一个失败则整体失败
const results = await Promise.all([
  fetch('/api/users').then(r => r.json()),
  fetch('/api/posts').then(r => r.json())
]);
// 第一个失败会立即 rejected

// Promise.race：返回最先完成（无论成功/失败）的结果
const result = await Promise.race([
  fetch('/api/fast'),
  new Promise((_, reject) =>
    setTimeout(() => reject(new Error('Timeout')), 5000)
  )
]);

// Promise.allSettled：所有都完成后返回结果（无论成功/失败）
const results = await Promise.allSettled([
  fetch('/api/1'),
  fetch('/api/2')
]);
// 每个结果都有 { status: 'fulfilled' | 'rejected', value/reason }
results.forEach(r => {
  if (r.status === 'fulfilled') console.log(r.value);
  else console.log(r.reason);
});

// Promise.any：返回最先成功的（忽略失败）
const first = await Promise.any([
  fetch('/api/1'),
  fetch('/api/2')
]);
// 全部失败才 rejected（AggregateError）
```

### Q: 如何实现一个 Promise？

```javascript
class MyPromise {
  constructor(executor) {
    this.state = 'pending';
    this.value = undefined;
    this.handlers = [];

    const resolve = (value) => {
      if (this.state !== 'pending') return;
      this.state = 'fulfilled';
      this.value = value;
      this.handlers.forEach(this.#handle);
    };

    const reject = (reason) => {
      if (this.state !== 'pending') return;
      this.state = 'rejected';
      this.value = reason;
      this.handlers.forEach(this.#handle);
    };

    try {
      executor(resolve, reject);
    } catch (e) {
      reject(e);
    }
  }

  #handle = ({ onFulfilled, onRejected, resolve, reject }) => {
    if (this.state === 'fulfilled') {
      onFulfilled?.(this.value);
    } else if (this.state === 'rejected') {
      onRejected?.(this.value);
    } else {
      this.handlers.push({ onFulfilled, onRejected, resolve, reject });
    }
  };

  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      this.#handle({
        onFulfilled: (v) => {
          try {
            resolve(onFulfilled?.(v));
          } catch (e) { reject(e); }
        },
        onRejected: (e) => {
          try {
            resolve(onRejected?.(e));
          } catch (err) { reject(err); }
        },
        resolve, reject
      });
    });
  }

  catch(onRejected) {
    return this.then(null, onRejected);
  }

  finally(onFinally) {
    return this.then(
      v => { onFinally(); return v; },
      e => { onFinally(); throw e; }
    );
  }
}
```

## async/await

### Q: async/await 的错误处理方式？

```javascript
// 方式 1：try/catch
async function fetchData() {
  try {
    const res = await fetch('/api/data');
    const data = await res.json();
    return data;
  } catch (error) {
    console.error('Error:', error.message);
  }
}

// 方式 2：.catch()
const data = await fetch('/api/data')
  .then(r => r.json())
  .catch(error => console.error(error));

// 方式 3：封装为 Result 模式
async function to(promise) {
  try {
    const data = await promise;
    return [data, null];
  } catch (error) {
    return [null, error];
  }
}

const [data, error] = await to(fetch('/api/data').then(r => r.json()));
```

### Q: 并发请求如何实现？

```javascript
// 串行（一个个来）
async function serial() {
  const users = await fetchUsers();
  const posts = await fetchPosts();
  // 等待一个完成再执行下一个
}

// 并发（同时请求）
async function parallel() {
  const [users, posts] = await Promise.all([
    fetchUsers(),
    fetchPosts()
  ]);
}

// 竞态处理：取消旧请求
let controller;
async function search(query) {
  controller?.abort();  // 取消上一个
  controller = new AbortController();
  const res = await fetch(`/api/search?q=${query}`, {
    signal: controller.signal
  });
}
```

### Q: async/await 相对于 Promise 的优势？

```javascript
// 1. 代码更直观，像同步代码
// Promise
function fetchData() {
  return fetch('/api/data')
    .then(r => r.json())
    .then(data => {
      return fetch(`/api/user/${data.userId}`)
        .then(r => r.json())
        .then(user => ({ data, user }));
    });
}

// async/await
async function fetchData() {
  const dataRes = await fetch('/api/data');
  const data = await dataRes.json();
  const userRes = await fetch(`/api/user/${data.userId}`);
  const user = await userRes.json();
  return { data, user };
}

// 2. 调试更容易（能在 await 处设置断点）
// 3. 错误堆栈更清晰
```

## 事件循环进阶

### Q: setTimeout(fn, 0) 会立即执行吗？

```javascript
console.log('1');

setTimeout(() => console.log('2'), 0);

Promise.resolve().then(() => console.log('3'));

console.log('4');
// 输出：1, 4, 3, 2
// setTimeout 虽然延迟 0，但仍是宏任务，排在微任务之后
```

### Q: 什么是微任务？有哪些？

```javascript
// 微任务
Promise.then/catch/finally
queueMicrotask()
async/await
MutationObserver
Object.observe（已废弃）

// 宏任务
setTimeout
setInterval
setImmediate（Node.js）
I/O
UI rendering

// 每次事件循环：
// 1. 执行所有微任务
// 2. 执行一个宏任务
// 3. 重复
```

### Q: 下面代码的输出顺序？

```javascript
async function async1() {
  console.log('async1 start');     // 2
  await async2();
  console.log('async1 end');       // 6
}

async function async2() {
  console.log('async2');          // 3
}

console.log('script start');      // 1

setTimeout(() => console.log('setTimeout'), 0);  // 8

async1();

new Promise(resolve => {
  console.log('promise1');        // 4
  resolve();
}).then(() => {
  console.log('promise2');       // 7
});

console.log('script end');        // 5

// 输出：script start, async1 start, async2, promise1, script end,
//       async1 end, promise2, setTimeout
```
