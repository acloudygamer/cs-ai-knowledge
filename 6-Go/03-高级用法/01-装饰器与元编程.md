# 装饰器与元编程

## 高阶函数（装饰器模式）

### 基本装饰器

```go
// 函数装饰器
func logger(fn func(int) int) func(int) int {
    return func(x int) int {
        start := time.Now()
        result := fn(x)
        fmt.Printf("执行耗时: %v\n", time.Since(start))
        return result
    }
}

func slowFunc(n int) int {
    time.Sleep(time.Second)
    return n * 2
}

decorated := logger(slowFunc)
decorated(5)
```

### 泛型装饰器（Go 1.18+）

```go
// 泛型装饰器，适用于任意函数签名
func withRetry[T any](fn func() T, maxRetries int) func() T {
    return func() T {
        var lastErr error
        for i := 0; i < maxRetries; i++ {
            result := fn()
            // 如果函数返回值为 any 类型且可能返回 error，需要用接口类型包装
            if r, ok := any(result).(error); ok && r != nil {
                lastErr = r
                continue
            }
            return result
        }
        if lastErr != nil {
            panic(lastErr)
        }
        // 无错误时返回零值
        var zero T
        return zero
    }
}
```

## 使用 defer 做装饰器

### 入口/出口追踪

```go
func trace(name string) func() {
    start := time.Now()
    fmt.Printf("enter %s\n", name)
    return func() {
        fmt.Printf("exit %s [%v]\n", name, time.Since(start))
    }
}

func doSomething() {
    defer trace("doSomething")()
    // ... 业务逻辑
}
```

### 错误处理装饰器

```go
func withError(fn func() error) func() error {
    return func() error {
        return fn()
    }
}

// 使用
defer func() {
    if err := recover(); err != nil {
        fmt.Printf("panic recovered: %v\n", err)
    }
}()
```

### 编写自定义 generate 工具

```go
// cmd/codegen/main.go
package main

import (
    "fmt"
    "os"
    "text/template"
)

func main() {
    if len(os.Args) < 2 {
        fmt.Fprintln(os.Stderr, "Usage: codegen <output>")
        os.Exit(1)
    }

    tmpl := template.Must(template.New("").Parse(`// Code generated. DO NOT EDIT.

package {{.Package}}

func New{{.Type}}() *{{.Type}} {
    return &{{.Type}}{}
}
`))

    data := struct {
        Package string
        Type    string
    }{
        Package: "main",
        Type:    "MyStruct",
    }

    f, _ := os.Create(os.Args[1])
    defer f.Close()
    tmpl.Execute(f, data)
}
```

## 泛型接口设计

### 策略模式

```go
type Serializer[T any] interface {
    Serialize(T) ([]byte, error)
    Deserialize([]byte) (T, error)
}

type JSONSerializer[T any] struct{}

func (j JSONSerializer[T]) Serialize(v T) ([]byte, error) {
    return json.Marshal(v)
}

func (j JSONSerializer[T]) Deserialize(data []byte) (T, error) {
    var v T
    return v, json.Unmarshal(data, &v)
}

// 使用
var serializer Serializer[User] = JSONSerializer[User]{}
data, _ := serializer.Serialize(User{Name: "Tom"})
```

### 泛型 Repository

```go
type Repository[T any, ID comparable] interface {
    FindByID(ID) (*T, error)
    Save(*T) (*T, error)
    Delete(ID) error
}

type InMemoryRepo[T any, ID comparable] struct {
    data map[ID]*T
}

func NewInMemoryRepo[T any, ID comparable]() *InMemoryRepo[T, ID] {
    return &InMemoryRepo[T, ID]{data: make(map[ID]*T)}
}

func (r *InMemoryRepo[T, ID]) FindByID(id ID) (*T, error) {
    return r.data[id], nil
}

func (r *InMemoryRepo[T, ID]) Save(t *T) (*T, error) {
    r.data[ID(0)] = t // 简化示例
    return t, nil
}
```

## 常用设计模式实现

### 装饰器模式

```go
type DataStore interface {
    Get(key string) (string, error)
}

type KVStore struct{}

func (k KVStore) Get(key string) (string, error) {
    return "value for " + key, nil
}

// 缓存装饰器
type CacheStore struct {
    store DataStore
    cache map[string]string
}

func NewCacheStore(store DataStore) *CacheStore {
    return &CacheStore{store: store, cache: make(map[string]string)}
}

func (c *CacheStore) Get(key string) (string, error) {
    if v, ok := c.cache[key]; ok {
        return v, nil
    }
    v, err := c.store.Get(key)
    if err == nil {
        c.cache[key] = v
    }
    return v, err
}
```

### 建造者模式

```go
type Config struct {
    host string
    port int
    user string
    pass string
}

type ConfigBuilder struct {
    config Config
}

func NewConfigBuilder() *ConfigBuilder {
    return &ConfigBuilder{Config: Config{port: 8080}}
}

func (b *ConfigBuilder) Host(host string) *ConfigBuilder {
    b.config.host = host
    return b
}

func (b *ConfigBuilder) Port(port int) *ConfigBuilder {
    b.config.port = port
    return b
}

func (b *ConfigBuilder) User(user, pass string) *ConfigBuilder {
    b.config.user = user
    b.config.pass = pass
    return b
}

func (b *ConfigBuilder) Build() *Config {
    return &b.config
}

// 使用
config := NewConfigBuilder().
    Host("localhost").
    Port(9090).
    User("admin", "secret").
    Build()
```
