# 装饰器与元编程

## 高阶函数（装饰器模式）

### 基本装饰器

```go
// 函数装饰器
func logger(fn func(int) int) func(int) int {
    return func(x int) int {
        start := time.Now()
        result := fn(x)
        fmt.Printf("执行耗时: %v\n", time.Since(start))
        return result
    }
}

func slowFunc(n int) int {
    time.Sleep(time.Second)
    return n * 2
}

decorated := logger(slowFunc)
decorated(5)
```

### 泛型装饰器（Go 1.18+）

```go
// 泛型装饰器，适用于任意函数签名
func withRetry[T any](fn func() T, maxRetries int) func() T {
    return func() T {
        var lastErr error
        for i := 0; i < maxRetries; i++ {
            result := fn()
            // 如果函数返回值为 any 类型且可能返回 error，需要用接口类型包装
            if r, ok := any(result).(error); ok && r != nil {
                lastErr = r
                continue
            }
            return result
        }
        if lastErr != nil {
            panic(lastErr)
        }
        // 无错误时返回零值
        var zero T
        return zero
    }
}
```

## 使用 defer 做装饰器

### 入口/出口追踪

```go
func trace(name string) func() {
    start := time.Now()
    fmt.Printf("enter %s\n", name)
    return func() {
        fmt.Printf("exit %s [%v]\n", name, time.Since(start))
    }
}

func doSomething() {
    defer trace("doSomething")()
    // ... 业务逻辑
}
```

### 错误处理装饰器

```go
func withError(fn func() error) func() error {
    return func() error {
        return fn()
    }
}

// 使用
defer func() {
    if err := recover(); err != nil {
        fmt.Printf("panic recovered: %v\n", err)
    }
}()
```

## 泛型（Go 1.18+）

### 泛型函数

```go
// 泛型 Map
func Map[T, U any](s []T, fn func(T) U) []U {
    result := make([]U, len(s))
    for i, v := range s {
        result[i] = fn(v)
    }
    return result
}

nums := []int{1, 2, 3}
strs := Map(nums, func(n int) string {
    return fmt.Sprintf("num:%d", n)
})

// 泛型 Filter
func Filter[T any](s []T, pred func(T) bool) []T {
    result := make([]T, 0)
    for _, v := range s {
        if pred(v) {
            result = append(result, v)
        }
    }
    return result
}

// 泛型 Reduce
func Reduce[T, U any](s []T, init U, fn func(U, T) U) U {
    result := init
    for _, v := range s {
        result = fn(result, v)
    }
    return result
}
```

### 泛型结构体

```go
type Stack[T any] struct {
    items []T
}

func (s *Stack[T]) Push(item T) { s.items = append(s.items, item) }

func (s *Stack[T]) Pop() T {
    item := s.items[len(s.items)-1]
    s.items = s.items[:len(s.items)-1]
    return item
}

func (s *Stack[T]) IsEmpty() bool { return len(s.items) == 0 }

// 使用
stack := &Stack[int]{}
stack.Push(1)
stack.Push(2)
fmt.Println(stack.Pop()) // 2
```

### 泛型约束

```go
// 使用 interface{} 作为约束
type Number interface {
    int | int32 | int64 | float32 | float64
}

func Sum[T Number](nums []T) T {
    var sum T
    for _, n := range nums {
        sum += n
    }
    return sum
}

// 使用 comparable 约束
func Contains[T comparable](s []T, target T) bool {
    for _, v := range s {
        if v == target {
            return true
        }
    }
    return false
}

// 使用 Ordered 约束（Go 1.21+）
func Min[T cmp.Ordered](a, b T) T {
    if a < b {
        return a
    }
    return b
}
```

## 反射（reflect）

### 基本概念

```go
import "reflect"

// TypeOf 获取类型信息
var x int = 10
t := reflect.TypeOf(x)
fmt.Println(t.Name())        // int
fmt.Println(t.Kind())        // int
fmt.Println(t.Size())        // 8
fmt.Println(t.Align())       // 8

// ValueOf 获取值信息
v := reflect.ValueOf(x)
fmt.Println(v.Int())         // 10
fmt.Println(v.Type())        // int
```

### 反射操作结构体

```go
type User struct {
    Name string `json:"name" validate:"required"`
    Age  int    `json:"age"`
}

func inspectStruct() {
    user := User{Name: "Tom", Age: 30}

    t := reflect.TypeOf(user)
    v := reflect.ValueOf(user)

    // 遍历字段
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        value := v.Field(i)
        fmt.Printf("%s: %v (%s)\n", field.Name, value, field.Type)
    }

    // 通过名字获取字段
    nameField, _ := t.FieldByName("Name")
    fmt.Println(nameField.Tag.Get("json"))  // name
    fmt.Println(nameField.Tag.Get("validate"))  // required
}
```

### 动态调用方法

```go
type Calculator struct{}

func (c Calculator) Add(a, b int) int { return a + b }
func (c Calculator) Multiply(a, b int) int { return a * b }

func callMethod() {
    calc := Calculator{}
    v := reflect.ValueOf(calc)

    // 获取方法
    addMethod := v.MethodByName("Add")
    if addMethod.IsValid() {
        args := []reflect.Value{reflect.ValueOf(10), reflect.ValueOf(20)}
        results := addMethod.Call(args)
        fmt.Println(results[0].Int()) // 30
    }
}
```

### 创建和修改值

```go
// 创建新实例
slice := reflect.MakeSlice(reflect.SliceOf(reflect.TypeOf(0)), 0, 10)
slice = reflect.Append(slice, reflect.ValueOf(1))
fmt.Println(slice.Interface())  // [1]

// 修改可导出字段
func setField() {
    user := User{Name: "Tom", Age: 30}
    v := reflect.ValueOf(&user).Elem()

    nameField := v.FieldByName("Name")
    if nameField.CanSet() {
        nameField.SetString("Jerry")
    }
    fmt.Println(user.Name) // Jerry
}
```

### 类型断言的反射方式

```go
func typeSwitch(x any) {
    v := reflect.ValueOf(x)
    switch v.Kind() {
    case reflect.Int:
        fmt.Println("int:", v.Int())
    case reflect.String:
        fmt.Println("string:", v.String())
    case reflect.Slice:
        fmt.Println("slice, len:", v.Len())
    default:
        fmt.Println("unknown type")
    }
}
```

### 编写自定义 generate 工具

```go
// cmd/codegen/main.go
package main

import (
    "fmt"
    "os"
    "text/template"
)

func main() {
    if len(os.Args) < 2 {
        fmt.Fprintln(os.Stderr, "Usage: codegen <output>")
        os.Exit(1)
    }

    tmpl := template.Must(template.New("").Parse(`// Code generated. DO NOT EDIT.

package {{.Package}}

func New{{.Type}}() *{{.Type}} {
    return &{{.Type}}{}
}
`))

    data := struct {
        Package string
        Type    string
    }{
        Package: "main",
        Type:    "MyStruct",
    }

    f, _ := os.Create(os.Args[1])
    defer f.Close()
    tmpl.Execute(f, data)
}
```

## 泛型接口设计

### 策略模式

```go
type Serializer[T any] interface {
    Serialize(T) ([]byte, error)
    Deserialize([]byte) (T, error)
}

type JSONSerializer[T any] struct{}

func (j JSONSerializer[T]) Serialize(v T) ([]byte, error) {
    return json.Marshal(v)
}

func (j JSONSerializer[T]) Deserialize(data []byte) (T, error) {
    var v T
    return v, json.Unmarshal(data, &v)
}

// 使用
var serializer Serializer[User] = JSONSerializer[User]{}
data, _ := serializer.Serialize(User{Name: "Tom"})
```

### 泛型 Repository

```go
type Repository[T any, ID comparable] interface {
    FindByID(ID) (*T, error)
    Save(*T) (*T, error)
    Delete(ID) error
}

type InMemoryRepo[T any, ID comparable] struct {
    data map[ID]*T
}

func NewInMemoryRepo[T any, ID comparable]() *InMemoryRepo[T, ID] {
    return &InMemoryRepo[T, ID]{data: make(map[ID]*T)}
}

func (r *InMemoryRepo[T, ID]) FindByID(id ID) (*T, error) {
    return r.data[id], nil
}

func (r *InMemoryRepo[T, ID]) Save(t *T) (*T, error) {
    r.data[ID(0)] = t // 简化示例
    return t, nil
}
```

## 常用设计模式实现

### 装饰器模式

```go
type DataStore interface {
    Get(key string) (string, error)
}

type KVStore struct{}

func (k KVStore) Get(key string) (string, error) {
    return "value for " + key, nil
}

// 缓存装饰器
type CacheStore struct {
    store DataStore
    cache map[string]string
}

func NewCacheStore(store DataStore) *CacheStore {
    return &CacheStore{store: store, cache: make(map[string]string)}
}

func (c *CacheStore) Get(key string) (string, error) {
    if v, ok := c.cache[key]; ok {
        return v, nil
    }
    v, err := c.store.Get(key)
    if err == nil {
        c.cache[key] = v
    }
    return v, err
}
```

### 建造者模式

```go
type Config struct {
    host string
    port int
    user string
    pass string
}

type ConfigBuilder struct {
    config Config
}

func NewConfigBuilder() *ConfigBuilder {
    return &ConfigBuilder{Config: Config{port: 8080}}
}

func (b *ConfigBuilder) Host(host string) *ConfigBuilder {
    b.config.host = host
    return b
}

func (b *ConfigBuilder) Port(port int) *ConfigBuilder {
    b.config.port = port
    return b
}

func (b *ConfigBuilder) User(user, pass string) *ConfigBuilder {
    b.config.user = user
    b.config.pass = pass
    return b
}

func (b *ConfigBuilder) Build() *Config {
    return &b.config
}

// 使用
config := NewConfigBuilder().
    Host("localhost").
    Port(9090).
    User("admin", "secret").
    Build()
```
