# 插件系统

Go 的插件系统允许在运行时动态加载和卸载代码模块。本专题涵盖 Go 1.8 引入的插件机制以及常见的插件架构设计模式。

---

## Go 插件基础

Go 插件基于 Go 编译为共享对象（.so 文件），使用 plugin 包在运行时加载。

### 构建插件

```go
// plugin/hello.go
package main

import "fmt"

var Version = "1.0.0"

func Greet(name string) string {
    return fmt.Sprintf("Hello, %s!", name)
}

// Init 在插件加载时自动调用
func Init() {
    fmt.Println("Plugin initialized!")
}
```

编译为插件：
```bash
go build -buildmode=plugin -o hello.so hello.go
```

### 加载插件

```go
import (
    "plugin"
)

func loadPlugin(path string) (plugin.Symbol, error) {
    p, err := plugin.Open(path)
    if err != nil {
        return nil, fmt.Errorf("open plugin: %w", err)
    }

    sym, err := p.Lookup("Greet")
    if err != nil {
        return nil, fmt.Errorf("lookup Greet: %w", err)
    }

    return sym, nil
}
```

### 使用插件函数

```go
func usePlugin() {
    sym, err := loadPlugin("./hello.so")
    if err != nil {
        panic(err)
    }

    // 类型断言
    greet, ok := sym.(func(string) string)
    if !ok {
        panic("Greet function not found")
    }

    result := greet("World")
    fmt.Println(result)  // Hello, World!
}
```

### 访问插件变量

```go
func accessPluginVar() {
    p, err := plugin.Open("./hello.so")
    if err != nil {
        panic(err)
    }

    // 查找导出的变量
    version, err := p.Lookup("Version")
    if err != nil {
        panic(err)
    }

    ver, ok := version.(string)
    if !ok {
        panic("Version is not a string")
    }

    fmt.Printf("Plugin version: %s\n", ver)
}
```

## 插件接口设计

### 定义插件接口

```go
// plugin/plugin.go
package main

// Plugin 插件接口
type Plugin interface {
    Name() string
    Init() error
    Execute(ctx context.Context) error
}

// 注册函数类型
type RegisterFunc func() (Plugin, error)
```

### 实现插件

```go
// plugins/greeter/greeter.go
package main

import (
    "context"
    "fmt"
)

type Greeter struct{}

func (g *Greeter) Name() string { return "greeter" }

func (g *Greeter) Init() error {
    fmt.Println("Greeter plugin initializing")
    return nil
}

func (g *Greeter) Execute(ctx context.Context) error {
    fmt.Println("Greeter says: Hello!")
    return nil
}

func Register() (Plugin, error) {
    return &Greeter{}, nil
}
```

编译：
```bash
go build -buildmode=plugin -o greeter.so plugins/greeter/greeter.go
```

### 插件管理器

```go
import (
    "plugin"
    "sync"
)

type PluginManager struct {
    mu      sync.RWMutex
    plugins map[string]plugin.Plugin
    libs    map[string]*plugin.Plugin
}

func NewPluginManager() *PluginManager {
    return &PluginManager{
        plugins: make(map[string]plugin.Plugin),
        libs:    make(map[string]*plugin.Plugin),
    }
}

func (pm *PluginManager) Load(path string) error {
    pm.mu.Lock()
    defer pm.mu.Unlock()

    // 加载共享库
    p, err := plugin.Open(path)
    if err != nil {
        return fmt.Errorf("load plugin: %w", err)
    }

    // 查找注册函数
    registerSym, err := p.Lookup("Register")
    if err != nil {
        return fmt.Errorf("lookup Register: %w", err)
    }

    register, ok := registerSym.(func() (Plugin, error))
    if !ok {
        return fmt.Errorf("invalid Register function signature")
    }

    // 调用注册函数
    plug, err := register()
    if err != nil {
        return fmt.Errorf("register plugin: %w", err)
    }

    // 初始化
    if err := plug.Init(); err != nil {
        return fmt.Errorf("init plugin: %w", err)
    }

    // 存储
    pm.plugins[plug.Name()] = plug
    pm.libs[path] = p

    return nil
}

func (pm *PluginManager) Get(name string) (plugin.Plugin, bool) {
    pm.mu.RLock()
    defer pm.mu.RUnlock()
    p, ok := pm.plugins[name]
    return p, ok
}

func (pm *PluginManager) List() []string {
    pm.mu.RLock()
    defer pm.mu.RUnlock()

    names := make([]string, 0, len(pm.plugins))
    for name := range pm.plugins {
        names = append(names, name)
    }
    return names
}

func (pm *PluginManager) Unload(name string) error {
    pm.mu.Lock()
    defer pm.mu.Unlock()

    plug, ok := pm.plugins[name]
    if !ok {
        return fmt.Errorf("plugin not found: %s", name)
    }

    // 清理资源
    if closer, ok := plug.(io.Closer); ok {
        closer.Close()
    }

    delete(pm.plugins, name)
    return nil
}
```

## 插件热加载

### 热加载机制

```go
import (
    "plugin"
    "sync"
    "time"
)

type HotReloader struct {
    pm         *PluginManager
    mu         sync.RWMutex
    watchPaths map[string]string  // name -> path
    ticker     *time.Ticker
}

func NewHotReloader(pm *PluginManager) *HotReloader {
    return &HotReloader{
        pm:         pm,
        watchPaths: make(map[string]string),
    }
}

func (hr *HotReloader) Watch(name, path string) {
    hr.mu.Lock()
    defer hr.mu.Unlock()
    hr.watchPaths[name] = path
}

func (hr *HotReloader) Start(ctx context.Context) {
    hr.ticker = time.NewTicker(5 * time.Second)
    go func() {
        for {
            select {
            case <-ctx.Done():
                return
            case <-hr.ticker.C:
                hr.checkUpdates()
            }
        }
    }()
}

func (hr *HotReloader) checkUpdates() {
    hr.mu.RLock()
    defer hr.mu.RUnlock()

    for name, path := range hr.watchPaths {
        // 检查文件修改时间
        info, err := os.Stat(path)
        if err != nil {
            continue
        }

        // 如果有更新，重新加载
        if hr.shouldReload(name, info.ModTime()) {
            if err := hr.pm.Unload(name); err != nil {
                fmt.Printf("unload error: %v\n", err)
            }
            if err := hr.pm.Load(path); err != nil {
                fmt.Printf("reload error: %v\n", err)
            }
        }
    }
}

func (hr *HotReloader) shouldReload(name string, modTime time.Time) bool {
    // 实现文件变化检测逻辑
    return false
}
```

## 插件隔离

### 资源限制

```go
import (
    "syscall"
)

func setResourceLimits() {
    // 限制内存
    syscall.Setrlimit(syscall.RLIMIT_AS, &syscall.Rlimit{
        Cur: 1024 * 1024 * 100,  // 100MB
        Max: 1024 * 1024 * 200,  // 200MB
    })
}
```

### 插件错误处理

```go
func safeCall(plug plugin.Plugin, name string) (err error) {
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("plugin panic: %v", r)
        }
    }()

    switch p := plug.(type) {
    case interface{ Execute(ctx context.Context) error }:
        err = p.Execute(context.Background())
    default:
        err = fmt.Errorf("plugin does not implement Execute")
    }

    return
}
```

## 插件版本管理

### 版本接口

```go
type Versionable interface {
    Version() string
    CompatibleWith(version string) bool
}
```

### 插件注册表

```go
type PluginRegistry struct {
    mu       sync.RWMutex
    plugins  map[string]Versionable
    metadata map[string]PluginMeta
}

type PluginMeta struct {
    Path      string
    LoadedAt  time.Time
    Checksum  string
}

func (r *PluginRegistry) Register(name string, plug Versionable, meta PluginMeta) error {
    r.mu.Lock()
    defer r.mu.Unlock()

    if existing, ok := r.plugins[name]; ok {
        if !existing.CompatibleWith(plug.Version()) {
            return fmt.Errorf("incompatible version: %s vs %s", existing.Version(), plug.Version())
        }
    }

    r.plugins[name] = plug
    r.metadata[name] = meta
    return nil
}
```

## 实际应用场景

### 1. 可插拔处理器

```go
type Handler interface {
    Handle(ctx context.Context, req interface{}) (interface{}, error)
    CanHandle(req interface{}) bool
}

// 注册处理器
type HandlerRegistry struct {
    handlers []Handler
}

func (r *HandlerRegistry) Register(h Handler) {
    r.handlers = append(r.handlers, h)
}

func (r *HandlerRegistry) Handle(ctx context.Context, req interface{}) (interface{}, error) {
    for _, h := range r.handlers {
        if h.CanHandle(req) {
            return h.Handle(ctx, req)
        }
    }
    return nil, fmt.Errorf("no handler for %T", req)
}
```

## 注意事项

### 插件限制

1. **仅 Linux/macOS**：Windows 不支持 Go 插件
2. **同 Go 版本**：主程序和插件必须使用相同 Go 版本编译
3. **相同构建标签**：插件和主程序必须使用相同的编译标签
4. **插件依赖冲突**：可能导致符号冲突

### 调试插件

```go
// 在插件中导出调试端点
func DebugInfo() map[string]interface{} {
    return map[string]interface{}{
        "goroutines": runtime.NumGoroutine(),
        "memory":     runtime.MemStats{},
    }
}
```
