# 设计模式

Go 没有类继承，惯用设计模式与其他 OOP 语言不同。本章介绍 Go 中常用的设计模式实现。

---

## 单例模式

### sync.Once 方式（推荐）

```go
type Manager struct {
    name string
}

var (
    instance *Manager
    once     sync.Once
)

func GetManager() *Manager {
    once.Do(func() {
        instance = &Manager{name: "default"}
    })
    return instance
}
```

### 互斥锁方式

```go
type Database struct {
    conn string
}

var (
    instance *Database
    mu      sync.RWMutex
)

func GetDatabase() *Database {
    mu.RLock()
    if instance != nil {
        mu.RUnlock()
        return instance
    }
    mu.RUnlock()

    mu.Lock()
    defer mu.Unlock()
    if instance == nil {  // 双重检查
        instance = &Database{conn: "localhost:5432"}
    }
    return instance
}
```

## 工厂模式

### 简单工厂

```go
type Notification interface {
    Send(msg string) error
}

type EmailNotification struct{}
func (e *EmailNotification) Send(msg string) error {
    fmt.Printf("Email: %s\n", msg)
    return nil
}

type SMSNotification struct{}
func (s *SMSNotification) Send(msg string) error {
    fmt.Printf("SMS: %s\n", msg)
    return nil
}

func NewNotification(t string) Notification {
    switch t {
    case "email":
        return &EmailNotification{}
    case "sms":
        return &SMSNotification{}
    default:
        return nil
    }
}

notif := NewNotification("email")
notif.Send("Hello")
```

### 构造函数工厂

```go
type Server struct {
    host string
    port int
}

type ServerOption func(*Server)

func WithHost(host string) ServerOption {
    return func(s *Server) { s.host = host }
}

func WithPort(port int) ServerOption {
    return func(s *Server) { s.port = port }
}

func NewServer(opts ...ServerOption) *Server {
    s := &Server{host: "localhost", port: 8080}
    for _, opt := range opts {
        opt(s)
    }
    return s
}

server := NewServer(WithHost("0.0.0.0"), WithPort(3000))
```

## 建造者模式

### 链式调用

```go
type Query struct {
    table   string
    fields  []string
    where   string
    orderBy string
    limit   int
}

type QueryBuilder struct {
    query Query
}

func (b *QueryBuilder) Table(name string) *QueryBuilder {
    b.query.table = name
    return b
}

func (b *QueryBuilder) Select(fields ...string) *QueryBuilder {
    b.query.fields = fields
    return b
}

func (b *QueryBuilder) Where(cond string) *QueryBuilder {
    b.query.where = cond
    return b
}

func (b *QueryBuilder) OrderBy(field string) *QueryBuilder {
    b.query.orderBy = field
    return b
}

func (b *QueryBuilder) Limit(n int) *QueryBuilder {
    b.query.limit = n
    return b
}

func (b *QueryBuilder) Build() Query {
    return b.query
}

query := (&QueryBuilder{}).
    Table("users").
    Select("id", "name").
    Where("age > 18").
    OrderBy("name").
    Limit(10).
    Build()
```

## 策略模式

### 接口策略

```go
type PaymentStrategy interface {
    Pay(amount float64) error
}

type CreditCardPayment struct {
    CardNumber string
}

func (c *CreditCardPayment) Pay(amount float64) error {
    fmt.Printf("Paid %.2f with credit card %s\n", amount, c.CardNumber)
    return nil
}

type PayPalPayment struct {
    Email string
}

func (p *PayPalPayment) Pay(amount float64) error {
    fmt.Printf("Paid %.2f with PayPal %s\n", amount, p.Email)
    return nil
}

type Checkout struct {
    strategy PaymentStrategy
}

func (c *Checkout) SetStrategy(s PaymentStrategy) {
    c.strategy = s
}

func (c *Checkout) Pay(amount float64) error {
    return c.strategy.Pay(amount)
}

checkout := &Checkout{}
checkout.SetStrategy(&CreditCardPayment{CardNumber: "1234-5678"})
checkout.Pay(99.99)
```

## 观察者模式

### 事件系统

```go
type Event struct {
    Type    string
    Payload any
}

type Observer interface {
    OnEvent(e Event)
}

type Subject struct {
    observers []Observer
    mu        sync.RWMutex
}

func (s *Subject) Subscribe(o Observer) {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.observers = append(s.observers, o)
}

func (s *Subject) Unsubscribe(o Observer) {
    s.mu.Lock()
    defer s.mu.Unlock()
    for i, obs := range s.observers {
        if obs == o {
            s.observers = append(s.observers[:i], s.observers[i+1:]...)
            return
        }
    }
}

func (s *Subject) Notify(e Event) {
    s.mu.RLock()
    defer s.mu.RUnlock()
    for _, o := range s.observers {
        o.OnEvent(e)
    }
}

// 使用
type Logger struct{}
func (l *Logger) OnEvent(e Event) {
    fmt.Printf("[Logger] Got event: %s\n", e.Type)
}
```

## 装饰器模式

### 函数装饰器

```go
type Handler func(req string) string

func Logger(h Handler) Handler {
    return func(req string) string {
        fmt.Println("Before:", req)
        resp := h(req)
        fmt.Println("After:", resp)
        return resp
    }
}

func Auth(h Handler) Handler {
    return func(req string) string {
        if !isAuthenticated() {
            return "Unauthorized"
        }
        return h(req)
    }
}

handler := Logger(Auth(func(req string) string {
    return "Response: " + req
}))
```

### 结构体装饰器

```go
type DataStore interface {
    Get(key string) string
}

type InMemoryStore struct {
    data map[string]string
}

func (s *InMemoryStore) Get(key string) string {
    return s.data[key]
}

type CachedStore struct {
    store  DataStore
    cache  sync.Map
}

func NewCachedStore(s DataStore) *CachedStore {
    return &CachedStore{store: s}
}

func (c *CachedStore) Get(key string) string {
    if v, ok := c.cache.Load(key); ok {
        return v.(string)
    }
    v := c.store.Get(key)
    c.cache.Store(key, v)
    return v
}
```

## 责任链模式

```go
type Handler interface {
    SetNext(Handler)
    Handle(ctx context.Context) error
}

type Middleware func(next Handler) Handler

func Chain(h Handler, middlewares ...Middleware) Handler {
    for i := len(middlewares) - 1; i >= 0; i-- {
        h = middlewares[i](h)
    }
    return h
}

// 具体实现
type LoggingHandler struct {
    next Handler
}

func (h *LoggingHandler) SetNext(next Handler) { h.next = next }
func (h *LoggingHandler) Handle(ctx context.Context) error {
    fmt.Println("Logging...")
    if h.next != nil {
        return h.next.Handle(ctx)
    }
    return nil
}

type AuthHandler struct {
    next Handler
}

func (h *AuthHandler) SetNext(next Handler) { h.next = next }
func (h *AuthHandler) Handle(ctx context.Context) error {
    fmt.Println("Authenticating...")
    if h.next != nil {
        return h.next.Handle(ctx)
    }
    return nil
}
```

## 命令模式

```go
type Command interface {
    Execute() error
}

type CopyCommand struct {
    src, dst string
}

func (c *CopyCommand) Execute() error {
    fmt.Printf("Copying from %s to %s\n", c.src, c.dst)
    return nil
}

type PasteCommand struct {
    dst string
}

func (p *PasteCommand) Execute() error {
    fmt.Printf("Pasting to %s\n", p.dst)
    return nil
}

type Editor struct {
    commands []Command
}

func (e *Editor) ExecuteCommand(cmd Command) {
    if err := cmd.Execute(); err != nil {
        fmt.Println("Error:", err)
        return
    }
    e.commands = append(e.commands, cmd)
}

func (e *Editor) Undo() {
    if len(e.commands) == 0 {
        return
    }
    fmt.Println("Undoing last command")
    e.commands = e.commands[:len(e.commands)-1]
}

editor := &Editor{}
editor.ExecuteCommand(&CopyCommand{src: "a.txt", dst: "b.txt"})
editor.ExecuteCommand(&PasteCommand{dst: "c.txt"})
```

## 适配器模式

```go
// 第三方接口（不可改）
type LegacyStorage struct{}

func (ls *LegacyStorage) SaveData(data []byte) {
    fmt.Println("Legacy storage: saving", len(data), "bytes")
}

// 目标接口
type Storage interface {
    Save(key string, data []byte) error
}

// 适配器
type StorageAdapter struct {
    legacy *LegacyStorage
}

func NewStorageAdapter() *StorageAdapter {
    return &StorageAdapter{legacy: &LegacyStorage{}}
}

func (a *StorageAdapter) Save(key string, data []byte) error {
    fmt.Printf("Adapter: saving %s (%d bytes)\n", key, len(data))
    a.legacy.SaveData(data)
    return nil
}

// 使用
var store Storage = NewStorageAdapter()
store.Save("file.txt", []byte("hello"))
```

## 模板方法模式

### 使用结构体组合

```go
type DataProcessor struct{}

func (p *DataProcessor) Process(data []byte) ([]byte, error) {
    validated, err := p.Validate(data)
    if err != nil {
        return nil, err
    }
    transformed := p.Transform(validated)
    return p.Save(transformed)
}

func (p *DataProcessor) Validate(data []byte) ([]byte, error) {
    if len(data) == 0 {
        return nil, errors.New("empty data")
    }
    return data, nil
}

func (p *DataProcessor) Transform(data []byte) []byte {
    return bytes.ToUpper(data)
}

func (p *DataProcessor) Save(data []byte) ([]byte, error) {
    fmt.Printf("Saving %d bytes\n", len(data))
    return data, nil
}

// 特化
type JSONProcessor struct {
    DataProcessor
}

func (p *JSONProcessor) Transform(data []byte) []byte {
    // JSON 特定转换
    return bytes.Trim(data, "{}")
}
```

## 代理模式

```go
type Service interface {
    DoSomething() string
}

type RealService struct{}

func (s *RealService) DoSomething() string {
    time.Sleep(100 * time.Millisecond)
    return "Real result"
}

type CachedService struct {
    real    Service
    cache   string
    cacheMu sync.RWMutex
}

func NewCachedService(s Service) *CachedService {
    return &CachedService{real: s}
}

func (s *CachedService) DoSomething() string {
    s.cacheMu.RLock()
    if s.cache != "" {
        defer s.cacheMu.RUnlock()
        return s.cache
    }
    s.cacheMu.RUnlock()

    s.cacheMu.Lock()
    defer s.cacheMu.Unlock()
    s.cache = s.real.DoSomething()
    return s.cache
}
```

## 状态模式

```go
type State interface {
    Enter()
    Exit()
    Do(ctx *Context)
}

type Context struct {
    current State
}

func NewContext() *Context {
    return &Context{}
}

func (c *Context) Transition(state State) {
    c.current.Exit()
    c.current = state
    c.current.Enter()
}

func (c *Context) Do() {
    c.current.Do(c)
}

// 状态实现
type DraftState struct{}

func (s *DraftState) Enter() { fmt.Println("Enter Draft") }
func (s *DraftState) Exit()  { fmt.Println("Exit Draft") }
func (s *DraftState) Do(ctx *Context) {
    fmt.Println("Draft: editing...")
    ctx.Transition(&PublishedState{})
}

type PublishedState struct{}

func (s *PublishedState) Enter() { fmt.Println("Enter Published") }
func (s *PublishedState) Exit()  { fmt.Println("Exit Published") }
func (s *PublishedState) Do(ctx *Context) {
    fmt.Println("Published: serving content...")
}

ctx := NewContext()
ctx.Transition(&DraftState{})
ctx.Do()
```

## 生成器 vs 工厂 vs 模板方法

| 模式 | 适用场景 | Go 实现 |
|------|----------|---------|
| 工厂 | 创建同类不同参数的对象 | 函数返回接口 |
| 建造者 | 复杂对象，多种配置 | 链式方法 |
| 模板方法 | 算法框架固定，步骤可变 | 结构体组合 |

---

## 惯用 Go 模式小结

Go 的设计模式强调：

1. **组合优于继承** - 用嵌套结构体代替类继承
2. **接口优于抽象类** - 小接口定义行为
3. **函数优于类** - 装饰器用函数实现
4. **显式依赖注入** - 构造函数传依赖

这些模式在 Go 标准库中广泛使用，如 `http.Handler`、`io.Reader/Writer`、`database/sql`。
