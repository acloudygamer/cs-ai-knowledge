## TCP 编程

### TCP 服务端

```go
import (
    "bufio"
    "errors"
    "fmt"
    "io"
    "net"
)

func tcpServer() {
    listener, err := net.Listen("tcp", ":8080")
    if err != nil {
        panic(err)
    }
    defer listener.Close()

    fmt.Println("TCP server listening on :8080")

    for {
        conn, err := listener.Accept()
        if err != nil {
            fmt.Printf("Accept error: %v\n", err)
            continue
        }

        go handleTCPConn(conn)
    }
}

func handleTCPConn(conn net.Conn) {
    defer conn.Close()

    reader := bufio.NewReader(conn)
    for {
        message, err := reader.ReadString('\n')
        if err != nil {
            if !errors.Is(err, io.EOF) {
                fmt.Printf("Read error: %v\n", err)
            }
            return
        }

        fmt.Printf("Received: %s", message)

        // 回显
        conn.Write([]byte("Echo: " + message))
    }
}
```

### TCP 客户端

```go
func tcpClient() {

    if err != nil {

    }
```

## UDP 编程

### UDP 服务端

```go
import (
    "fmt"
    "net"
)

func udpServer() {
    addr, err := net.ResolveUDPAddr("udp", ":8080")
    if err != nil {
        panic(err)
    }

    conn, err := net.ListenUDP("udp", addr)
    if err != nil {
        panic(err)
    }
    defer conn.Close()

    fmt.Println("UDP server listening on :8080")

    buf := make([]byte, 1024)
    for {
        n, remoteAddr, err := conn.ReadFromUDP(buf)
        if err != nil {
            fmt.Printf("Read error: %v\n", err)
            continue
        }

        fmt.Printf("Received from %s: %s\n", remoteAddr, string(buf[:n]))

        // 发送响应
        conn.WriteToUDP([]byte("Hello, UDP client!"), remoteAddr)
    }
}
```

### UDP 客户端

```go
func udpClient() {
    conn, err := net.Dial("udp", "localhost:8080")

    n, err := conn.Read(buf)
    if err != nil {
        panic(err)

}
```

### UDP 连接池（多播场景）

```go
func udpMulticast() {
    // 加入多播组
    conn, err := net.ListenPacket("udp", "239.0.0.1:9999")
    if err != nil {
        panic(err)
    }
    defer conn.Close()

    pc, err := net.ListenPacket("udp", ":0")
    if err != nil {
        panic(err)
    }
    defer pc.Close()

    groupAddr, err := net.ResolveUDPAddr("udp", "239.0.0.1:9999")
    if err != nil {
        panic(err)
    }

    // 设置多播接口
    if err := pc.(*net.UDPConn).JoinGroup(nil, groupAddr); err != nil {
        panic(err)
    }
}
```

### WebSocket 服务端

```go
import (
    "github.com/gorilla/websocket"
    "net/http"
)

var upgrader = websocket.Upgrader{
    ReadBufferSize:  1024,
    WriteBufferSize: 1024,
    CheckOrigin: func(r *http.Request) bool {
        return true  // 生产环境应该验证 Origin
    },
}

func wsHandler(w http.ResponseWriter, r *http.Request) {
    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil {
        panic(err)
    }
    defer conn.Close()

    for {
        messageType, msg, err := conn.ReadMessage()
        if err != nil {
            if websocket.IsUnexpectedCloseError(err, websocket.CloseGoingAway, websocket.CloseAbnormalClosure) {
                fmt.Printf("WebSocket error: %v\n", err)
            }
            break
        }

        fmt.Printf("Received: %s\n", msg)

        // 发送文本消息
        if err := conn.WriteMessage(messageType, []byte("Echo: "+string(msg))); err != nil {
            panic(err)
        }
    }
}

func wsServer() {
    http.HandleFunc("/ws", wsHandler)
    http.ListenAndServe(":8080", nil)
}
```


### WebSocket 读写超时

```go
import (
    "time"
    "github.com/gorilla/websocket"
)

func wsWithTimeout(conn *websocket.Conn) {
    for {
        // 设置读取超时
        conn.SetReadDeadline(time.Now().Add(60 * time.Second))

        _, msg, err := conn.ReadMessage()
        if err != nil {
            if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
                fmt.Println("Read timeout")
            }
            break
        }

        // 处理消息...

        // 设置写入超时
        conn.SetWriteDeadline(time.Now().Add(30 * time.Second))
        if err := conn.WriteMessage(websocket.TextMessage, msg); err != nil {
            break
        }
    }
}
```

## HTTP 服务器

### 基础 HTTP 服务器

```go
import (
    "fmt"
    "net/http"
)

func httpServer() {
    http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        fmt.Fprintf(w, "Hello, %s!\n", r.URL.Path)
    })

    http.HandleFunc("/headers", func(w http.ResponseWriter, r *http.Request) {
        for k, v := range r.Header {
            fmt.Fprintf(w, "%s: %v\n", k, v)
        }
    })

    http.ListenAndServe(":8080", nil)
}
```

### 自定义 HTTP 服务器

```go
import (
    "context"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "time"
)

func customHTTPServer() {
    server := &http.Server{
        Addr:         ":8080",
        Handler:      myHandler(),
        ReadTimeout:  15 * time.Second,
        WriteTimeout: 15 * time.Second,
        IdleTimeout:  60 * time.Second,
        MaxHeaderBytes: 1 << 20,  // 1MB
    }

    // 优雅关闭
    go func() {
        if err := server.ListenAndServe(); err != http.ErrServerClosed {
            panic(err)
        }
    }()

    // 等待中断信号
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
    <-sigChan

    // 优雅关闭
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()

    if err := server.Shutdown(ctx); err != nil {
        panic(err)
    }
}
```

### 自定义 TCP 连接池

```go
import (
    "sync"
    "time"
)

type Pool struct {
    mu       sync.Mutex
    conns    []*Conn
    maxSize  int
    minSize  int
    factory  func() (*Conn, error)
}

type Conn struct {
    net.Conn
    created time.Time
}

func (p *Pool) Get() (*Conn, error) {
    p.mu.Lock()
    defer p.mu.Unlock()

    // 复用现有连接
    if len(p.conns) > 0 {
        conn := p.conns[len(p.conns)-1]
        p.conns = p.conns[:len(p.conns)-1]
        return conn, nil
    }

    // 创建新连接
    return p.factory()
}

func (p *Pool) Put(conn *Conn) {
    p.mu.Lock()
    defer p.mu.Unlock()

    if len(p.conns) < p.maxSize {
        p.conns = append(p.conns, conn)
    } else {
        conn.Close()
    }
}
```

## 常见模式

### 粘包处理

```go
import (
    "bufio"
    "encoding/binary"
    "bytes"
)

// 定长消息
func fixedLengthHandler(conn net.Conn) {
    const msgLen = 4
    buf := make([]byte, msgLen)

    for {
        _, err := conn.Read(buf)
        if err != nil {
            return
        }

        length := binary.BigEndian.Uint32(buf)
        msg := make([]byte, length)
        conn.Read(msg)

        fmt.Printf("Received: %s\n", string(msg))
    }
}

// 分隔符消息
func delimitedHandler(conn net.Conn) {
    reader := bufio.NewReader(conn)
    for {
        line, err := reader.ReadString('\n')
        if err != nil {
            return
        }
        fmt.Printf("Received: %s\n", line)
    }
}

// 长度前缀消息（TLV 格式）
func tlvHandler(conn net.Conn) {
    for {
        // 读取长度（4 字节）
        lenBuf := make([]byte, 4)
        if _, err := conn.Read(lenBuf); err != nil {
            return
        }
        length := binary.BigEndian.Uint32(lenBuf)

        // 读取数据
        data := make([]byte, length)
        if _, err := conn.Read(data); err != nil {
            return
        }

        fmt.Printf("Received: %s\n", string(data))
    }
}
```

```go
import (
    "time"

type HeartbeatConn struct {
    net.Conn
    timeout  time.Duration
    heartbeat time.Duration
}

func (c *HeartbeatConn) Read(p []byte) (n int, err error) {
    c.Conn.SetReadDeadline(time.Now().Add(c.timeout))
    return c.Conn.Read(p)
}

func (c *HeartbeatConn) startHeartbeat() {
    ticker := time.NewTicker(c.heartbeat)
    defer ticker.Stop()

    for range ticker.C {
        if _, err := c.Write([]byte("ping")); err != nil {
            return
        }
    }
}
```

## 性能优化

### Nagle 算法

```go
// 禁用 Nagle 算法（低延迟场景）
conn, err := net.Dial("tcp", "localhost:8080")
if err != nil {
    panic(err)
}

// 设置 TCP 连接选项
tcpConn := conn.(*net.TCPConn)
tcpConn.SetNoDelay(true)  // 禁用 Nagle
tcpConn.SetWriteBuffer(1024 * 1024)
tcpConn.SetReadBuffer(1024 * 1024)
```

### epoll 和 kqueue

Go runtime 自动使用 epoll（Linux）或 kqueue（macOS/BSD）处理大量连接，无需手动管理。

```go
// 设置最大打开文件描述符
// Linux: ulimit -n 65535

// Go 中无需特殊代码，runtime 自动处理
func highConcurrencyServer() {
    listener, err := net.Listen("tcp", ":8080")
    if err != nil {
        panic(err)
    }

    for {
        conn, err := listener.Accept()
        if err != nil {
            continue
        }
        go handleConn(conn)  // goroutine per connection
    }
}
```
