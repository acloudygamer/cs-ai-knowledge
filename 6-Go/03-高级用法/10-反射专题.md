# 反射专题

## 概述

反射（Reflection）是 Go 在运行时检查和操作对象的能力。通过 `reflect` 包，程序可以：
- 动态获取变量的类型和值
- 动态调用方法
- 动态修改变量值
- 检查结构体字段标签

**警告**：反射开销较大（比直接操作慢 10-100 倍），生产代码应谨慎使用。

## reflect.Type 与 reflect.Value

### 基本概念

```go
import "reflect"

// TypeOf 获取变量的类型信息
funcType := reflect.TypeOf(42)
fmt.Println(funcType)        // int
fmt.Println(funcType.Kind()) // int
fmt.Println(funcType.Name()) // int

// ValueOf 获取变量的值信息
val := reflect.ValueOf(42)
fmt.Println(val)        // 42
fmt.Println(val.Type()) // int
fmt.Println(val.Kind()) // int
```

### Type 接口核心方法

```go
func inspectType(t reflect.Type) {
    // 基本信息
    fmt.Printf("Name: %s, Kind: %s\n", t.Name(), t.Kind())

    // Kind 枚举值
    switch t.Kind() {
    case reflect.Struct:
        fmt.Printf("Fields: %d\n", t.NumField())
    case reflect.Slice:
        fmt.Printf("Element: %s\n", t.Elem())
    case reflect.Map:
        fmt.Printf("Key: %s, Value: %s\n", t.Key(), t.Elem())
    case reflect.Func:
        fmt.Printf("NumIn: %d, NumOut: %d\n", t.NumIn(), t.NumOut())
    }

    // 嵌套类型
    if t.Kind() == reflect.Ptr {
        fmt.Printf("Elem: %s\n", t.Elem())
    }
}
```

### Value 接口核心方法

```go
func inspectValue(v reflect.Value) {
    // 获取实际值（需要转换）
    if v.Kind() == reflect.Int {
        i := v.Int()  // int64
        fmt.Println(i)
    }

    if v.Kind() == reflect.String {
        s := v.String()
        fmt.Println(s)
    }

    // 设置值（需要可寻址）
    if v.CanSet() {
        v.SetInt(100)
    }
}
```

### Kind vs Type

```go
type User struct {
    Name string
    Age  int
}

func demo() {
    var u User
    t := reflect.TypeOf(u)
    v := reflect.ValueOf(u)

    fmt.Println(t.Name())              // User（类型名）
    fmt.Println(t.Kind())              // struct（种类）
    fmt.Println(v.Kind())              // struct

    // 指针和值的 Kind 不同
    up := &u
    fmt.Println(reflect.TypeOf(up).Kind())   // ptr
    fmt.Println(reflect.TypeOf(up).Elem())   // User
    fmt.Println(reflect.ValueOf(up).Kind())  // ptr
}
```

## 结构体操作

### 获取字段信息

```go
type User struct {
    Name string `json:"name" validate:"required"`
    Age  int    `json:"age"`
    email string // 未导出字段
}

func inspectStruct() {
    u := User{Name: "Alice", Age: 30}

    t := reflect.TypeOf(u)

    // 遍历所有字段
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        fmt.Printf("Field %d: %s, Type: %s, Tag: %v\n",
            i, field.Name, field.Type, field.Tag)
    }

    // 按名查找（支持导出字段）
    if nameField, ok := t.FieldByName("Name"); ok {
        fmt.Printf("Name field: %s, Tag: %s\n", nameField.Name, nameField.Tag)
    }

    // 获取未导出字段（需要通过指针）
    up := &User{Name: "Bob", Age: 25}
    tp := reflect.TypeOf(up).Elem()
    for i := 0; i < tp.NumField(); i++ {
        field := tp.Field(i)
        if field.PkgPath != "" { // 未导出字段
            fmt.Printf("Unexported: %s\n", field.Name)
        }
    }
}
```

### 读取字段标签

```go
func readTags() {
    u := User{Name: "Alice", Age: 30}

    t := reflect.TypeOf(u)

    // 获取 JSON 标签
    nameField, _ := t.FieldByName("Name")
    jsonTag := nameField.Tag.Get("json")
    fmt.Printf("JSON tag: %s\n", jsonTag)  // name

    validateTag := nameField.Tag.Get("validate")
    fmt.Printf("Validate tag: %s\n", validateTag)  // required

    // 遍历所有标签
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        json := field.Tag.Get("json")
        if json != "" {
            fmt.Printf("Field %s -> JSON: %s\n", field.Name, json)
        }
    }
}
```

### 动态设置字段值

```go
func setField() {
    u := User{Name: "Alice", Age: 30}

    // 需要指针才能修改
    v := reflect.ValueOf(&u).Elem()

    // 按名设置
    nameField := v.FieldByName("Name")
    if nameField.CanSet() {
        nameField.SetString("Bob")
    }

    // 按索引设置
    ageField := v.Field(1) // Age 字段
    if ageField.CanSet() {
        ageField.SetInt(25)
    }

    fmt.Println(u)  // {Bob 25}
}
```

## 动态方法调用

### 调用函数

```go
func add(a, b int) int {
    return a + b
}

func greet(name string) string {
    return "Hello, " + name
}

func callFunc() {
    // 获取函数值
    addVal := reflect.ValueOf(add)
    greetVal := reflect.ValueOf(greet)

    // 调用函数
    // func (v Value) Call(in []Value) []Value
    args := []reflect.Value{
        reflect.ValueOf(10),
        reflect.ValueOf(20),
    }
    result := addVal.Call(args)
    fmt.Println(result[0].Int())  // 30

    // 调用 greet
    greetArgs := []reflect.Value{reflect.ValueOf("Alice")}
    greetResult := greetVal.Call(greetArgs)
    fmt.Println(greetResult[0].String())  // Hello, Alice
}
```

### 调用方法

```go
type User struct {
    Name string
    Age  int
}

func (u User) Greet() string {
    return fmt.Sprintf("Hi, I'm %s", u.Name)
}

func (u User) SetAge(age int) {
    u.Age = age
}

func callMethod() {
    u := User{Name: "Alice", Age: 30}

    // 获取方法值（需要值或指针）
    v := reflect.ValueOf(u)

    // 方法名获取
    greetMethod := v.MethodByName("Greet")
    result := greetMethod.Call(nil)
    fmt.Println(result[0].String())  // Hi, I'm Alice

    // 通过指针调用可修改的方法
    up := &u
    vp := reflect.ValueOf(up)

    setAgeMethod := vp.MethodByName("SetAge")
    setAgeMethod.Call([]reflect.Value{reflect.ValueOf(25)})

    fmt.Println(u.Age)  // 仍然是 30，因为是值接收者
}
```

### 动态创建结构体

```go
func createStruct() {
    // 创建新的结构体实例
    userType := reflect.TypeOf(User{})
    newUser := reflect.New(userType).Elem()

    // 设置字段
    nameField := newUser.FieldByName("Name")
    if nameField.CanSet() {
        nameField.SetString("Bob")
    }

    ageField := newUser.FieldByName("Age")
    if ageField.CanSet() {
        ageField.SetInt(20)
    }

    // 转换为具体类型
    user := newUser.Interface().(User)
    fmt.Println(user)  // {Bob 20}
}
```

## 集合操作

### 切片操作

```go
func sliceOps() {
    // 创建切片类型
    sliceType := reflect.TypeOf([]int{})
    fmt.Println(sliceType.Kind())  // slice
    fmt.Println(sliceType.Elem())  // int

    // 创建切片值
    slice := reflect.MakeSlice(sliceType, 0, 10)
    slice = reflect.Append(slice, reflect.ValueOf(1))
    slice = reflect.AppendSlice(slice, []int{2, 3, 4})

    // 转换为具体切片
    ints := slice.Interface().([]int)
    fmt.Println(ints)  // [1 2 3 4]
}
```

### Map 操作

```go
func mapOps() {
    // 创建 map 类型
    mapType := reflect.TypeOf(map[string]int{})
    fmt.Println(mapType.Kind())  // map
    fmt.Println(mapType.Key())    // string
    fmt.Println(mapType.Elem())   // int

    // 创建 map 值
    m := reflect.MakeMap(mapType)

    // 设置值
    key := reflect.ValueOf("age")
    val := reflect.ValueOf(30)
    m.SetMapIndex(key, val)

    // 读取值
    result := m.MapIndex(key)
    fmt.Println(result.Int())  // 30

    // 遍历
    keys := m.MapKeys()
    for _, k := range keys {
        v := m.MapIndex(k)
        fmt.Printf("%s: %d\n", k.String(), v.Int())
    }

    // 转换为具体 map
    m2 := m.Interface().(map[string]int)
    fmt.Println(m2)  // map[age:30]
}
```

### 动态分配

func dynamicAlloc() {

## 接口与类型断言

### 检查接口实现

```go
type Writer interface {
    Write([]byte) (int, error)
}

type Logger struct{}

func (l Logger) Write(data []byte) (int, error) {
    fmt.Println(string(data))
    return len(data), nil
}

func checkInterface() {
    var w Writer = Logger{}

    // 检查类型
    t := reflect.TypeOf(w)
    fmt.Printf("Type: %s, Kind: %s\n", t.Name(), t.Kind())

    // 检查是否实现了接口
    writerType := reflect.TypeOf((*Writer)(nil)).Elem()
    fmt.Printf("Logger implements Writer: %v\n",
        reflect.PtrTo(t).Implements(writerType))

    // 检查指针接收者
    loggerPtrType := reflect.TypeOf(&Logger{})
    fmt.Printf("*Logger implements Writer: %v\n",
        loggerPtrType.Implements(writerType))
}
```

### 动态类型断言

```go
func typeAssert() {
    var i interface{} = "hello"

    // 方法1：直接断言（危险）
    // s := i.(string)  // panic 如果失败

    // 方法2：安全断言
    s, ok := i.(string)
    if ok {
        fmt.Println(s)
    }

    // 方法3：使用 reflect
    v := reflect.ValueOf(i)
    if v.Kind() == reflect.String {
        fmt.Println(v.String())
    }

    // 动态类型切换
    var x interface{} = 42
    switch v := x.(type) {
    case int:
        fmt.Printf("int: %d\n", v)
    case string:
        fmt.Printf("string: %s\n", v)
    default:
        fmt.Printf("unknown: %T\n", v)
    }
}
```

## 反射最佳实践

### 性能优化

```go
// 慢：每次都反射
func slowGetField(v interface{}, name string) interface{} {
    r := reflect.ValueOf(v)
    f := r.FieldByName(name)
    return f.Interface()
}

// 快：缓存反射结果
type fieldCache struct {
    sync.Map
    index map[reflect.Type]map[string]int
}

func (c *fieldCache) getIndex(t reflect.Type, name string) (int, bool) {
    typeFields, ok := c.index[t]
    if !ok {
        typeFields = make(map[string]int)
        for i := 0; i < t.NumField(); i++ {
            typeFields[t.Field(i).Name] = i
        }
        c.index[t] = typeFields
    }
    idx, ok := typeFields[name]
    return idx, ok
}

// 最快：泛型（Go 1.18+）
func getField[T any](v *T, name string) interface{} {
    // 编译时确定类型，无反射开销
    r := reflect.ValueOf(v).Elem()
    f := r.FieldByName(name)
    return f.Interface()
}
```

### 安全反射

```go
// 安全获取可设置的值
func safeSet(v reflect.Value, val interface{}) bool {
    if !v.CanSet() {
        return false
    }
    if val == nil {
        v.SetZero()
        return true
    }
    // 类型匹配检查
    if !v.Type().AssignableTo(reflect.TypeOf(val)) {
        return false
    }
    v.Set(reflect.ValueOf(val))
    return true
}

// 安全获取字段值
func safeGetField(v interface{}, fieldName string) (interface{}, error) {
    r := reflect.ValueOf(v)
    if r.Kind() != reflect.Struct {
        return nil, fmt.Errorf("not a struct")
    }
    f := r.FieldByName(fieldName)
    if !f.IsValid() {
        return nil, fmt.Errorf("field not found: %s", fieldName)
    }
    return f.Interface(), nil
}
```

### 常见模式

```go
// 1. 深拷贝
func deepCopy(src interface{}) interface{} {
    v := reflect.ValueOf(src)
    return deepCopyValue(v).Interface()
}

func deepCopyValue(v reflect.Value) reflect.Value {
    switch v.Kind() {
    case reflect.Slice:
        newSlice := reflect.MakeSlice(v.Type(), v.Len(), v.Cap())
        for i := 0; i < v.Len(); i++ {
            newSlice.Index(i).Set(deepCopyValue(v.Index(i)))
        }
        return newSlice
    case reflect.Struct:
        newStruct := reflect.New(v.Type()).Elem()
        for i := 0; i < v.NumField(); i++ {
            newStruct.Field(i).Set(deepCopyValue(v.Field(i)))
        }
        return newStruct
    default:
        return v
    }
}

// 2. 结构体比较
func compareStructs(a, b interface{}) bool {
    va := reflect.ValueOf(a)
    vb := reflect.ValueOf(b)
    if va.Type() != vb.Type() {
        return false
    }
    for i := 0; i < va.NumField(); i++ {
        if !reflect.DeepEqual(va.Field(i).Interface(), vb.Field(i).Interface()) {
            return false
        }
    }
    return true
}

// 3. 零值创建
func createZero(v interface{}) interface{} {
    t := reflect.TypeOf(v)
    if t.Kind() == reflect.Ptr {
        t = t.Elem()
    }
    return reflect.New(t).Elem().Interface()
}
```

## 反射与代码生成

### 简化 JSON 处理

```go
type Person struct {
    Name    string `json:"name"`
    Age     int    `json:"age"`
    Address string `json:"address,omitempty"`
}

func autoMapStruct(src, dst interface{}) error {
    sv := reflect.ValueOf(src).Elem()
    dv := reflect.ValueOf(dst).Elem()

    if sv.Type() != dv.Type() {
        return fmt.Errorf("type mismatch")
    }

    for i := 0; i < sv.NumField(); i++ {
        sf := sv.Type().Field(i)
        jsonTag := sf.Tag.Get("json")
        if jsonTag == "" || jsonTag == "-" {
            continue
        }

        df := dv.FieldByName(sf.Name)
        if !df.IsValid() || !df.CanSet() {
            continue
        }

        df.Set(sv.Field(i))
    }
    return nil
}
```

### 通用验证器

```go
func validate(v interface{}) error {
    rv := reflect.ValueOf(v).Elem()
    rt := rv.Type()

    for i := 0; i < rt.NumField(); i++ {
        field := rt.Field(i)
        value := rv.Field(i)

        validateTag := field.Tag.Get("validate")
        if validateTag == "" {
            continue
        }

        // 检查 required
        if strings.Contains(validateTag, "required") {
            if field.Type.Kind() == reflect.String && value.String() == "" {
                return fmt.Errorf("field %s is required", field.Name)
            }
            if field.Type.Kind() == reflect.Int && value.Int() == 0 {
                return fmt.Errorf("field %s is required", field.Name)
            }
        }

        // 检查 min
        if minTag := field.Tag.Get("min"); minTag != "" {
            if field.Type.Kind() == reflect.Int {
                min, _ := strconv.Atoi(minTag)
                if int(value.Int()) < min {
                    return fmt.Errorf("field %s must be >= %d", field.Name, min)
                }
            }
        }
    }
    return nil
}

func usage() {
    p := Person{Name: "Alice", Age: 20}
    if err := validate(p); err != nil {
        fmt.Println(err)
    }
}
```

### 数据库映射

```go
type User struct {
    ID   int    `db:"id"`
    Name string `db:"name"`
    Age  int    `db:"age"`
}

func scanToStruct(rows *sql.Rows, dest interface{}) error {
    dv := reflect.ValueOf(dest).Elem()
    dt := dv.Type()

    columns, _ := rows.Columns()
    values := make([]interface{}, len(columns))

    for i, col := range columns {
        // 查找对应的字段
        for j := 0; j < dt.NumField(); j++ {
            if dt.Field(j).Tag.Get("db") == col {
                values[i] = dv.Field(j).Addr().Interface()
                break
            }
        }
    }

    return rows.Scan(values...)
}
```

## 反射使用场景

| 场景 | 推荐方案 | 原因 |
|------|----------|------|
| JSON 编组 | encoding/json | 内部使用反射，已优化 |
| ORM 映射 | 代码生成 | 避免运行时反射开销 |
| 依赖注入 | 反射/代码生成 | 框架内部使用 |
| 配置绑定 | 泛型或反射 | 可选，配置通常启动时加载 |
| 测试框架 | 反射 | 动态调用是测试的核心 |
| RPC 框架 | 代码生成 | 性能和类型安全 |

## 常见错误

```go
// 错误1：ValueOf 返回的是副本，无法修改
func wrong1() {
    x := 10
    v := reflect.ValueOf(x)
    v.SetInt(20)  // panic: unaddressable value
}

// 正确：使用指针
func right1() {
    x := 10
    v := reflect.ValueOf(&x).Elem()
    v.SetInt(20)  // x = 20
}

// 错误2：未检查 CanSet
func wrong2() {
    u := User{Name: "Alice"}
    v := reflect.ValueOf(u)
    v.FieldByName("Name").SetString("Bob")  // panic: unaddressable
}

// 错误3：类型不匹配
func wrong3() {
    var s string = "hello"
    v := reflect.ValueOf(&s).Elem()
    v.SetInt(123)  // panic: int is not assignable to string
}
```

func traceReflect(prefix string, v reflect.Value) {
    fmt.Printf("%s: Type=%s Kind=%s Value=%v\n",
        prefix, v.Type(), v.Kind(), v)

    if v.Kind() == reflect.Struct {
        for i := 0; i < v.NumField(); i++ {
