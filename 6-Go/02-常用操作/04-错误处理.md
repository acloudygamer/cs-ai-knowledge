# 错误处理

### 解决什么问题
Go 使用显式错误返回而非异常，调用者必须处理错误，保证程序健壮性。

### 核心概念
- error 接口：内置接口，只有一个 Error() string 方法
- 多返回值：函数返回 (result, error) 模式
- 错误包装：fmt.Errorf with %w 保留错误链
- errors.Is/As：检查错误链中的特定错误（Go 1.20）
- errors.AsType：泛型、类型安全的错误解包（<latest> 版本新增）

### 怎么用

## error 接口

```go
type error interface {
    Error() string
}

// 创建错误
err := errors.New("这是一个错误")
err = fmt.Errorf("格式化错误: %s", "details")

// 自定义错误类型
type MyError struct {
    Code    int
    Message string
}
func (e *MyError) Error() string {
    return fmt.Sprintf("code:%d, message:%s", e.Code, e.Message)
}
```

## 多返回值错误处理模式

```go
// 简单处理
result, err := doSomething()
if err != nil {
    // 错误处理
}
fmt.Println(result)

// 快速返回
if err := doThing(); err != nil {
    return err  // 或 return nil, err
}

// 忽略错误（下划线，但慎用）
_, err = doSomething()

// 错误断言和类型判断
err := doSomething()
if e, ok := err.(*MyError); ok {
    fmt.Println(e.Code)
}
```

## panic / recover（异常控制）

```go
func safeCall() {
    defer func() {
        if r := recover(); r != nil {
            fmt.Println("recovered:", r)
        }
    }()

    panic("something bad happened")
    // 不会执行到这里
}

// panic vs error：仅用于真正不可恢复的错误
// 大部分情况用 error 返回
```

## 错误链和 Wrap（Go 1.20+）

```go
import "fmt"

// 错误包装
err := fmt.Errorf("failed: %w", os.ErrPermission)
if errors.Is(err, os.ErrPermission) {
    fmt.Println("权限错误")
}

// 错误断言
var e *MyError
if errors.As(err, &e) {
    fmt.Println(e.Code)
}

// 多层包装
err := fmt.Errorf("load config: %w", fmt.Errorf("read file: %w", os.ErrNotExist))
```

### 哨兵错误（Sentinel Errors）

预定义错误常量，用于公开的错误检查。

```go
import "errors"

// 标准库哨兵错误
errors.Is(err, os.ErrNotExist)
errors.Is(err, os.ErrPermission)
errors.Is(err, io.EOF)

// 自定义哨兵错误
var (
    ErrNotFound     = errors.New("resource not found")
    ErrUnauthorized = errors.New("unauthorized")
    ErrInvalidInput = errors.New("invalid input")
)

// 使用
if errors.Is(err, ErrNotFound) {
    // 资源未找到处理
}
```

### 自定义错误类型

```go
// 领域特定错误
type ValidationError struct {
    Field   string
    Message string
}

func (e *ValidationError) Error() string {
    return fmt.Sprintf("validation failed on %s: %s", e.Field, e.Message)
}

// 错误检查
err := doSomething()
var validationErr *ValidationError
if errors.As(err, &validationErr) {
    fmt.Printf("Field: %s, Message: %s\n",
        validationErr.Field, validationErr.Message)
}
```

### errors.Is vs errors.As

```go
// errors.Is - 检查错误链中是否有匹配的错误
// 适用于哨兵错误和 wrapped errors
if errors.Is(err, os.ErrPermission) {
    log.Println("权限不足")
}

// errors.As - 在错误链中查找特定类型
// 适用于自定义错误类型
var validationErr *ValidationError
if errors.As(err, &validationErr) {
    fmt.Println(validationErr.Field)
}

// errors.AsType - 泛型、类型安全的错误解包 (<latest> 版本新增)
if err := errors.AsType[ValidationError](err); err != nil {
    fmt.Println(err.Field, err.Message)
}
```

### 错误处理最佳实践

```go
// 好：包装错误并附带上下文
func LoadConfig(path string) (*Config, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, fmt.Errorf("load config %s: %w", path, err)
    }

    var cfg Config
    if err := json.Unmarshal(data, &cfg); err != nil {
        return nil, fmt.Errorf("parse config %s: %w", path, err)
    }

    return &cfg, nil
}

// 差：不要忽略错误
result, _ := doSomething()  // 永远不要这样做

// 好：处理错误或说明原因
result, err := doSomething()
if err != nil {
    return err
}
// 如果真的无关紧要，可以用 _ 但要加注释说明
_ = writer.Close() // 最大努力清理
```

### 错误包装层级

```go
// 多层包装示例
func loadConfig(path string) (*Config, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, fmt.Errorf("load config: %w", err)
    }

    var cfg Config
    if err := json.Unmarshal(data, &cfg); err != nil {
        return nil, fmt.Errorf("parse config %s: %w", path, err)
    }

    if err := validateConfig(&cfg); err != nil {
        return nil, fmt.Errorf("validate config: %w", err)
    }

    return &cfg, nil
}

// 调用链错误检查
func main() {
    err := loadConfig("config.json")
    if err != nil {
        // 检查具体错误类型
        if errors.Is(err, os.ErrNotExist) {
            fmt.Println("config file not found")
        } else if strings.Contains(err.Error(), "parse") {
            fmt.Println("invalid config format")
        } else {
            fmt.Printf("config error: %v\n", err)
        }
    }
}
```

### 错误上下文传播

```go
// 领域特定错误
type DomainError struct {
    Err       error
    Domain    string
    Operation string
    Entity    string
    ID        string
}

func (e *DomainError) Error() string {
    return fmt.Sprintf("%s.%s: %s [%s] - %v",
        e.Domain, e.Operation, e.Entity, e.ID, e.Err)
}

func (e *DomainError) Unwrap() error {
    return e.Err
}

// 创建领域错误
func notFoundError(domain, entity, id string) error {
    return &DomainError{
        Err:       os.ErrNotExist,
        Domain:    domain,
        Operation: "get",
        Entity:    entity,
        ID:        id,
    }
}

// 处理领域错误
func handleDomainError(err error) {
    var domainErr *DomainError
    if errors.As(err, &domainErr) {
        fmt.Printf("Domain: %s, Operation: %s, Entity: %s, ID: %s\n",
            domainErr.Domain, domainErr.Operation, domainErr.Entity, domainErr.ID)
    }
}
```

### 批量错误处理

```go
import "errors"

// 错误集合
type Errors []error

func (e Errors) Error() string {
    if len(e) == 0 {
        return ""
    }
    if len(e) == 1 {
        return e[0].Error()
    }
    return fmt.Sprintf("%d errors: %v", len(e), []error(e))
}

// 并行执行收集错误
func parallelWithErrors() error {
    var errs Errors
    var mu sync.Mutex

    tasks := []func() error{
        func() error { return nil },
        func() error { return errors.New("task 2 failed") },
        func() error { return errors.New("task 3 failed") },
    }

    var wg sync.WaitGroup
    for _, task := range tasks {
        wg.Add(1)
        go func(t func() error) {
            defer wg.Done()
            if err := t(); err != nil {
                mu.Lock()
                errs = append(errs, err)
                mu.Unlock()
            }
        }(task)
    }

    wg.Wait()

    if len(errs) > 0 {
        return errs
    }
    return nil
}

// errors.Join（Go 1.20+）
func joinErrors() error {
    err1 := errors.New("first error")
    err2 := errors.New("second error")

    return errors.Join(err1, err2)
}

func checkJoinedErrors() {
    err := joinErrors()
    err1 := errors.New("first error")
    err2 := errors.New("second error")

    if errors.Is(err, err1) {
        fmt.Println("contains first error")
    }
    if errors.Is(err, err2) {
        fmt.Println("contains second error")
    }
}
```

### 错误恢复和防御

```go
import "runtime"

// 安全执行函数
func SafeCall(fn func()) (err error) {
    defer func() {
        if r := recover(); r != nil {
            buf := make([]byte, 4096)
            n := runtime.Stack(buf, false)
            err = fmt.Errorf("panic recovered: %v\n%s", r, string(buf[:n]))
        }
    }()

    fn()
    return nil
}

// 带返回值的版本
func SafeCallWithResult[T any](fn func() T) (result T, err error) {
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("panic: %v", r)
        }
    }()

    return fn(), nil
}

// 使用示例
func example() {
    result, err := SafeCallWithResult(func() int {
        // 可能 panic 的代码
        return 42
    })
}
```

### 错误重试机制

```go
import "time"

// 重试配置
type RetryConfig struct {
    MaxAttempts int
    Backoff     time.Duration
    MaxBackoff  time.Duration
    Multiplier  float64
}

// 重试函数
func WithRetry[T any](
    fn func() (T, error),
    config RetryConfig,
) (T, error) {
    var result T
    var lastErr error
    backoff := config.Backoff

    for attempt := 0; attempt < config.MaxAttempts; attempt++ {
        result, lastErr = fn()
        if lastErr == nil {
            return result, nil
        }

        if attempt < config.MaxAttempts-1 {
            time.Sleep(backoff)
            backoff = time.Duration(float64(backoff) * config.Multiplier)
            if backoff > config.MaxBackoff {
                backoff = config.MaxBackoff
            }
        }
    }

    return result, fmt.Errorf("after %d attempts: %w", config.MaxAttempts, lastErr)
}

// 使用
func retryExample() {
    config := RetryConfig{
        MaxAttempts: 3,
        Backoff:     100 * time.Millisecond,
        MaxBackoff:  5 * time.Second,
        Multiplier:  2.0,
    }

    result, err := WithRetry(func() (int, error) {
        return doSomething()
    }, config)
}
```

### 错误降级处理

```go
// 降级处理模式
func WithFallback[T any](
    primary func() (T, error),
    fallback func() T,
) T {
    result, err := primary()
    if err == nil {
        return result
    }

    fmt.Printf("primary failed: %v, using fallback\n", err)
    return fallback()
}

// 缓存降级
type CacheWithFallback struct {
    cache  map[string]any
    source func(key string) (any, error)
}

func (c *CacheWithFallback) Get(key string) (any, error) {
    // 尝试从缓存获取
    if val, ok := c.cache[key]; ok {
        return val, nil
    }

    // 缓存失败，从源获取
    val, err := c.source(key)
    if err != nil {
        return nil, err
    }

    c.cache[key] = val
    return val, nil
}
```

### 错误链工具函数

```go
// 错误链遍历
func WalkErrorChain(err error, fn func(error) bool) {
    for e := err; e != nil; e = errors.Unwrap(e) {
        if !fn(e) {
            return
        }
    }
}

// 获取最底层错误
func RootError(err error) error {
    for {
        u := errors.Unwrap(err)
        if u == nil {
            return err
        }
        err = u
    }
}

// 检查错误类型
func IsErrorOfType[T error](err error) bool {
    var target T
    return errors.As(err, &target)
}

// 格式化完整错误链
func FormatErrorChain(err error) string {
    var lines []string
    WalkErrorChain(err, func(e error) bool {
        lines = append(lines, fmt.Sprintf("  - %v", e))
        return true
    })
    return fmt.Sprintf("error chain:\n%s", strings.Join(lines, "\n"))
}
```

### 错误处理与 context

```go
import "context"

func withErrorContext(ctx context.Context, operation string) context.Context {
    return context.WithValue(ctx, contextKey("operation"), operation)
}

func runWithContext(ctx context.Context) error {
    op, _ := ctx.Value(contextKey("operation")).(string)

    err := doOperation()
    if err != nil {
        return fmt.Errorf("%s: %w", op, err)
    }

    return nil
}

// 错误自动添加 context
type contextError struct {
    err      error
    op       string
    duration time.Duration
}

func (e *contextError) Error() string {
    return fmt.Sprintf("%s: %v (took %v)", e.op, e.err, e.duration)
}

func (e *contextError) Unwrap() error {
    return e.err
}
```

## 错误处理最佳实践总结

| 场景 | 推荐做法 |
|------|---------|
| 标准错误 | 使用 `fmt.Errorf("context: %w", err)` |
| 公开 API | 定义哨兵错误 `var ErrXXX = errors.New("xxx")` |
| 内部错误 | 使用自定义错误类型添加上下文 |
| 外部错误 | 通过 errors.Is/As 检查 |
| 批量操作 | 使用 errors.Join 收集多个错误 |
| 不可恢复错误 | 使用 panic，仅在顶层 recover |
| 异步任务 | 传递 error 通道而非 panic |
