# 认证授权

## JWT（JSON Web Token）

### 安装

```bash
go get github.com/golang-jwt/jwt/v5
```

### JWT 结构

```
Header.Payload.Signature
```

- **Header**: 算法和类型
- **Payload**: 声明（claims）
- **Signature**: 签名验证

### 基本使用

```go
import (
    "fmt"
    "time"
    "github.com/golang-jwt/jwt/v5"
)

// JWT Claims 定义
type Claims struct {
    UserID   int64  `json:"user_id"`
    Username string `json:"username"`
    Role     string `json:"role"`
    jwt.RegisteredClaims
}

// 生成 Token
func generateToken(userID int64, username, role string) (string, error) {
    claims := Claims{
        UserID:   userID,
        Username: username,
        Role:     role,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(time.Now().Add(24 * time.Hour)),
            IssuedAt:  jwt.NewNumericDate(time.Now()),
            NotBefore: jwt.NewNumericDate(time.Now()),
            Issuer:    "my-app",
            Subject:   fmt.Sprintf("%d", userID),
        },
    }

    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString([]byte("your-secret-key"))
}

// 解析验证 Token
func parseToken(tokenString string) (*Claims, error) {
    token, err := jwt.ParseWithClaims(tokenString, &Claims{}, func(token *jwt.Token) (interface{}, error) {
        // 验证签名算法
        if _, ok := token.Method.(*jwt.SigningMethodHMAC); !ok {
            return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])
        }
        return []byte("your-secret-key"), nil
    })

    if err != nil {
        return nil, err
    }

    if claims, ok := token.Claims.(*Claims); ok && token.Valid {
        return claims, nil
    }

    return nil, fmt.Errorf("invalid token")
}
```

### Refresh Token

```go
type TokenPair struct {
    AccessToken  string
    RefreshToken string
}

func generateTokenPair(userID int64) (*TokenPair, error) {
    now := time.Now()

    // Access Token（短期）
    accessClaims := Claims{
        UserID: userID,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(now.Add(15 * time.Minute)),
            IssuedAt:  jwt.NewNumericDate(now),
            Issuer:    "my-app",
        },
    }

    accessToken := jwt.NewWithClaims(jwt.SigningMethodHS256, accessClaims)
    accessTokenString, err := accessToken.SignedString([]byte("access-secret"))

    // Refresh Token（长期）
    refreshClaims := jwt.RegisteredClaims{
        Subject:   fmt.Sprintf("%d", userID),
        ExpiresAt: jwt.NewNumericDate(now.Add(7 * 24 * time.Hour)),
        IssuedAt:  jwt.NewNumericDate(now),
        Issuer:    "my-app-refresh",
    }

    refreshToken := jwt.NewWithClaims(jwt.SigningMethodHS256, refreshClaims)
    refreshTokenString, err := refreshToken.SignedString([]byte("refresh-secret"))

    return &TokenPair{
        AccessToken:  accessTokenString,
        RefreshToken: refreshTokenString,
    }, err
}
```

### JWT 中间件

```go
import (
    "net/http"
    "strings"
    "github.com/golang-jwt/jwt/v5"
)

func JWTAuthMiddleware(secret string) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            authHeader := r.Header.Get("Authorization")
            if authHeader == "" {
                http.Error(w, "missing authorization header", http.StatusUnauthorized)
                return
            }

            // Bearer token
            parts := strings.SplitN(authHeader, " ", 2)
            if len(parts) != 2 || parts[0] != "Bearer" {
                http.Error(w, "invalid authorization header", http.StatusUnauthorized)
                return
            }

            tokenString := parts[1]

            token, err := jwt.ParseWithClaims(tokenString, &Claims{}, func(token *jwt.Token) (interface{}, error) {
                return []byte(secret), nil
            })

            if err != nil || !token.Valid {
                http.Error(w, "invalid token", http.StatusUnauthorized)
                return
            }

            claims, ok := token.Claims.(*Claims)
            if !ok {
                http.Error(w, "invalid claims", http.StatusUnauthorized)
                return
            }

            // 将用户信息放入 context
            ctx := r.Context()
            ctx = context.WithValue(ctx, "userID", claims.UserID)
            ctx = context.WithValue(ctx, "username", claims.Username)
            ctx = context.WithValue(ctx, "role", claims.Role)

            next.ServeHTTP(w, r.WithContext(ctx))
        })
    }
}

// 从 context 获取用户信息
func getUserIDFromContext(ctx context.Context) int64 {
    if userID, ok := ctx.Value("userID").(int64); ok {
        return userID
    }
    return 0
}
```

### Authorization Code Flow

```go
// 1. 重定向到授权服务器
func authorizationCodeHandler(w http.ResponseWriter, r *http.Request) {
    authURL := "https://oauth-provider.com/authorize"
    clientID := "your-client-id"
    redirectURI := "https://your-app.com/callback"
    scope := "read write"
    state := generateRandomState()

    url := fmt.Sprintf("%s?client_id=%s&redirect_uri=%s&response_type=code&scope=%s&state=%s",
        authURL, clientID, redirectURI, scope, state)

    // 保存 state 用于验证
    http.SetCookie(w, &http.Cookie{
        Name:  "oauth_state",
        Value: state,
        HttpOnly: true,
        Secure: true,
    })

    http.Redirect(w, r, url, http.StatusTemporaryRedirect)
}

// 2. 回调处理
func callbackHandler(w http.ResponseWriter, r *http.Request) {
    // 验证 state
    cookie, _ := r.Cookie("oauth_state")
    if cookie == nil || cookie.Value != r.URL.Query().Get("state") {
        http.Error(w, "invalid state", http.StatusBadRequest)
        return
    }

    code := r.URL.Query().Get("code")
    if code == "" {
        http.Error(w, "missing code", http.StatusBadRequest)
        return
    }

    // 交换 access token
    token, err := exchangeCodeForToken(code)
    if err != nil {
        http.Error(w, "failed to exchange token", http.StatusInternalServerError)
        return
    }

    // 使用 token 获取用户信息
    userInfo, err := getUserInfo(token.AccessToken)
    if err != nil {
        http.Error(w, "failed to get user info", http.StatusInternalServerError)
        return
    }

    // 创建本地会话
    createSession(w, userInfo)
}

func exchangeCodeForToken(code string) (*TokenResponse, error) {
    data := map[string]string{
        "grant_type":    "authorization_code",
        "code":          code,
        "client_id":     "your-client-id",
        "client_secret": "your-client-secret",
        "redirect_uri":  "https://your-app.com/callback",
    }

    resp, err := http.PostForm("https://oauth-provider.com/token", data)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var token TokenResponse
    json.NewDecoder(resp.Body).Decode(&token)
    return &token, nil
}

type TokenResponse struct {
    AccessToken  string `json:"access_token"`
    TokenType    string `json:"token_type"`
    ExpiresIn    int    `json:"expires_in"`
    RefreshToken string `json:"refresh_token"`
    Scope        string `json:"scope"`
}
```

### PKCE（Proof Key for Code Exchange）

```go
import (
    "crypto/rand"
    "crypto/sha256"
    "encoding/base64"
)

func generateCodeVerifier() string {
    b := make([]byte, 32)
    _, err := rand.Read(b)
    if err != nil {
        return "" // 实际生产中应处理错误
    }
    return base64.RawURLEncoding.EncodeToString(b)
}

func generateCodeChallenge(verifier string) string {
    h := sha256.New()
    h.Write([]byte(verifier))
    return base64.RawURLEncoding.EncodeToString(h.Sum(nil))
}

// 使用 PKCE 的授权 URL
func authURLWithPKCE() string {
    verifier := generateCodeVerifier()
    challenge := generateCodeChallenge(verifier)

    // 保存 verifier 用于后续验证
    storeVerifier(verifier)

    return fmt.Sprintf("%s?client_id=%s&response_type=code&redirect_uri=%s&code_challenge=%s&code_challenge_method=S256",
        authURL, clientID, redirectURI, challenge)
}

// Token 交换时提供 verifier
func exchangeToken(code, verifier string) (*TokenResponse, error) {
    data := map[string]string{
        "grant_type":    "authorization_code",
        "code":          code,
        "client_id":     "your-client-id",
        "code_verifier": verifier,
    }
    // ...
}
```

## 密码处理

### 密码哈希（bcrypt）

```bash
go get golang.org/x/crypto/bcrypt
```

```go
import "golang.org/x/crypto/bcrypt"

const bcryptCost = 12

// 哈希密码
func hashPassword(password string) (string, error) {
    bytes, err := bcrypt.GenerateFromPassword([]byte(password), bcryptCost)
    return string(bytes), err
}

// 验证密码
func checkPassword(password, hash string) bool {
    err := bcrypt.CompareHashAndPassword([]byte(hash), []byte(password))
    return err == nil
}

// 密码强度验证
func validatePasswordStrength(password string) error {
    if len(password) < 8 {
        return fmt.Errorf("password must be at least 8 characters")
    }
    if len(password) > 128 {
        return fmt.Errorf("password must be less than 128 characters")
    }

    hasLower := false
    hasUpper := false
    hasDigit := false
    hasSpecial := false

    for _, c := range password {
        switch {
        case 'a' <= c && c <= 'z':
            hasLower = true
        case 'A' <= c && c <= 'Z':
            hasUpper = true
        case '0' <= c && c <= '9':
            hasDigit = true
        default:
            hasSpecial = true
        }
    }

    if !hasLower || !hasUpper || !hasDigit {
        return fmt.Errorf("password must contain lowercase, uppercase and digit")
    }

    return nil
}
```

### 密码重置

```go
import (
    "crypto/rand"
    "encoding/hex"

// 生成重置令牌
func generateResetToken() (string, error) {
    b := make([]byte, 32)
    _, err := rand.Read(b)
    if err != nil {
        return "", err
    }
    return hex.EncodeToString(b), nil
}

// 存储重置请求（使用 Redis，设置过期时间）
func storeResetToken(userID int64, token string) error {
    key := fmt.Sprintf("password_reset:%s", token)
    return redis.Set(key, userID, 1*time.Hour)
}

// 验证并使用重置令牌
func useResetToken(token, newPassword string) error {
    key := fmt.Sprintf("password_reset:%s", token)
    userIDStr, err := redis.Get(key)
    if err != nil {
        return fmt.Errorf("invalid or expired reset token")
    }

    userID, _ := strconv.ParseInt(userIDStr, 10, 64)

    // 删除已使用的令牌
    redis.Del(key)

    // 更新密码
    hash, _ := hashPassword(newPassword)
    return userRepo.UpdatePassword(userID, hash)
```

## 基于角色的访问控制（RBAC）

### 角色定义

```go
type Role string

const (
    RoleAdmin  Role = "admin"
    RoleUser   Role = "user"
    RoleGuest  Role = "guest"
)

type Permission string

const (
    PermReadUsers   Permission = "read:users"
    PermWriteUsers  Permission = "write:users"
    PermDeleteUsers Permission = "delete:users"
    PermReadOrders  Permission = "read:orders"
    PermWriteOrders Permission = "write:orders"
)

// 角色权限映射
var rolePermissions = map[Role][]Permission{
    RoleAdmin: {
        PermReadUsers, PermWriteUsers, PermDeleteUsers,
        PermReadOrders, PermWriteOrders,
    },
    RoleUser: {
        PermReadUsers,           // 只能读自己
        PermReadOrders, PermWriteOrders,
    },
    RoleGuest: {
        PermReadOrders,          // 只能读订单
    },
}
```

### 权限检查

```go
import "context"

type Authorizer struct {
    rolePermissions map[Role][]Permission
}

func NewAuthorizer() *Authorizer {
    return &Authorizer{
        rolePermissions: rolePermissions,
    }
}

func (a *Authorizer) HasPermission(role Role, perm Permission) bool {
    perms, ok := a.rolePermissions[role]
    if !ok {
        return false
    }

    for _, p := range perms {
        if p == perm {
            return true
        }
    }
    return false
}

// 权限检查中间件
func RequirePermission(authorizer *Authorizer, perm Permission) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            role := r.Context().Value("role").(Role)

            if !authorizer.HasPermission(role, perm) {
                http.Error(w, "forbidden", http.StatusForbidden)
                return
            }

            next.ServeHTTP(w, r)
        })
    }
}

// 使用示例
r.Group(func(r chi.Router) {
    r.Use(JWTAuthMiddleware(secret))

    r.Get("/users", RequirePermission(authorizer, PermReadUsers)(listUsersHandler))
    r.Post("/users", RequirePermission(authorizer, PermWriteUsers)(createUserHandler))
    r.Delete("/users/{id}", RequirePermission(authorizer, PermDeleteUsers)(deleteUserHandler))
})
```

## API Key 认证

### API Key 生成与验证

```go
import (
    "crypto/rand"
    "crypto/sha256"
    "encoding/hex"
)

// 生成 API Key
func generateAPIKey() (key, hash string, err error) {
    bytes := make([]byte, 32)
    _, err = rand.Read(bytes)
    if err != nil {
        return "", "", err
    }
    key = hex.EncodeToString(bytes)

    // 存储哈希而非原始 key
    h := sha256.Sum256([]byte(key))
    hash = hex.EncodeToString(h[:])

    return key, hash, nil
}

// 验证 API Key
func validateAPIKey(providedKey, storedHash string) bool {
    h := sha256.Sum256([]byte(providedKey))
    providedHash := hex.EncodeToString(h[:])
    return providedHash == storedHash
}

// API Key 中间件
func APIKeyAuthMiddleware(getHash func(string) (string, bool)) func(http.Handler) http.Handler {
    return func(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
            apiKey := r.Header.Get("X-API-Key")
            if apiKey == "" {
                apiKey = r.URL.Query().Get("api_key")
            }

            if apiKey == "" {
                http.Error(w, "missing API key", http.StatusUnauthorized)
                return
            }

            hash, found := getHash(apiKey)
            if !found || !validateAPIKey(apiKey, hash) {
                http.Error(w, "invalid API key", http.StatusUnauthorized)
                return
            }

            next.ServeHTTP(w, r)
        })
    }
}
```
