# 配置管理

## Viper（推荐）

Viper 是 Go 最流行的配置管理库，支持多种格式（JSON、TOML、YAML、ENV），支持远程配置（etcd、Consul），支持环境变量绑定。

### 安装

```bash
go get github.com/spf13/viper
```

### 基本用法

```go
import (
    "fmt"
    "github.com/spf13/viper"
)

func main() {
    // 设置配置文件名（不含扩展名）
    viper.SetConfigName("config")
    viper.SetConfigType("yaml")

    // 添加搜索路径
    viper.AddConfigPath("./configs")
    viper.AddConfigPath("$HOME/.app")
    viper.AddConfigPath("/etc/app/")

    // 或直接设置配置内容
    viper.SetConfigFile("config.yaml")

    // 读取配置
    if err := viper.ReadInConfig(); err != nil {
        panic(fmt.Errorf("配置读取失败: %w", err))
    }

    // 获取值
    dbHost := viper.GetString("database.host")
    dbPort := viper.GetInt("database.port")
    debug := viper.GetBool("app.debug")

    // 带默认值的获取
    timeout := viper.GetDuration("server.timeout") // 默认 0
    logLevel := viper.GetString("log.level")       // 默认 ""

    fmt.Println(dbHost, dbPort, debug, timeout, logLevel)
}
```

### 配置优先级（从高到低）

```go
import "strings"

func loadConfig() error {
    // 1. 命令行 flags
    //    --config /path/to/config.yaml
    viper.SetConfigFile("/path/to/config.yaml")

    // 2. 环境变量（会覆盖配置文件）
    //    APP_DEBUG=true
    //    APP_DATABASE_HOST=localhost
    viper.AutomaticEnv()
    viper.SetEnvPrefix("APP")           // 前缀：APP_DEBUG
    viper.SetEnvKeyReplacer(strings.NewReplacer(".", "_"))

    // 3. 当前值（程序中 Set 的值）
    viper.Set("database.host", "override-host")

    // 4. 配置文件
    return viper.ReadInConfig()
}
```

### 绑定嵌套配置到结构体

```go
// 定义配置结构
type Config struct {
    App      AppConfig
    Database DatabaseConfig
    Redis    RedisConfig
    Log      LogConfig
    JWT      JWTConfig
}

type AppConfig struct {
    Name  string
    Debug bool
    Port  int
}

type DatabaseConfig struct {
    Host            string
    Port            int
    User            string
    Password        string
    Name            string
    MaxOpenConns    int `mapstructure:"max_open_conns"`
    MaxIdleConns    int `mapstructure:"max_idle_conns"`
    ConnMaxLifetime int `mapstructure:"conn_max_lifetime"`
}

type RedisConfig struct {
    Host     string
    Port     int
    Password string `mapstructure:"password"`
    DB       int    `mapstructure:"db"`
}

type LogConfig struct {
    Level  string
    Format string
    Output string
}

type JWTConfig struct {
    Secret      string `mapstructure:"secret"`
    ExpiryHours int    `mapstructure:"expiry_hours"`
}

// 使用 viper.Unmarshal
func loadConfig() (*Config, error) {
    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        return nil, fmt.Errorf("配置反序列化失败: %w", err)
    }
    return &cfg, nil
}
```

### 远程配置（etcd / Consul）

```go
import (
    "github.com/spf13/viper"
    "github.com/spf13/viper/remote"
)

// 从 etcd 读取远程配置
func loadRemoteConfig() error {
    viper.SetConfigType("json")

    // WatchRemoteConfig(path string, channel <-chan *RemoteResponse, delay time.Duration) error
    err := viper.WatchRemoteConfig(
        "etcd://localhost:2379/config/app.json",
        nil, // nil channel 表示不使用 channel 模式
        0,   // 0 表示立即返回
    )

    return err
}

// 从 Vault 读取
func loadFromVault() error {
    client, _ := api.NewClient(&api.Config{
        Address: "http://localhost:8200",
    })

    secret, err := client.KVv2("secret").Get(context.Background(), "myapp/config")
    if err != nil {
        return err
    }

    for k, v := range secret.Data {
        viper.Set(k, v)
    }

    return nil
}
```

### 热重载配置

```go
import (
    "fmt"
    "github.com/fsnotify/fsnotify"
    "github.com/spf13/viper"
)

func watchConfig() error {
    viper.WatchConfig()

    viper.OnConfigChange(func(e fsnotify.Event) {
        fmt.Println("配置已更改:", e.Name)

        // 重新读取关键配置
        newValue := viper.GetString("some.key")
        fmt.Println("新值:", newValue)
    })

    return nil
}
```

## YAML 解析（gopkg.in/yaml.v3）

当只需要解析 YAML 而不需要 Viper 的全部功能时。

### 安装

```bash
go get gopkg.in/yaml.v3
```

### 基本用法

```go
import (
    "fmt"
    "gopkg.in/yaml.v3"
)

type Config struct {
    App      AppConfig      `yaml:"app"`
    Database DatabaseConfig `yaml:"database"`
}

type AppConfig struct {
    Name string `yaml:"name"`
    Port int    `yaml:"port"`
}

type DatabaseConfig struct {
    Host string `yaml:"host"`
    Port int    `yaml:"port"`
}

func parseYAML(data []byte) (*Config, error) {
    var cfg Config
    if err := yaml.Unmarshal(data, &cfg); err != nil {
        return nil, err
    }
    return &cfg, nil
}

func main() {
    yamlData := `
app:
  name: my-service
  port: 8080
database:
  host: localhost
  port: 5432
`
    cfg, err := parseYAML([]byte(yamlData))
    if err != nil {
        panic(err)
    }
    fmt.Printf("%+v\n", cfg)
}
```

### YAML 序列化

```go
func toYAML(cfg *Config) ([]byte, error) {
    return yaml.Marshal(cfg)
}
```

## TOML 解析（github.com/BurntSushi/toml）

### 安装

```bash
go get github.com/BurntSushi/toml
```

### 基本用法

```go
import "github.com/BurntSushi/toml"

// 定义结构（与 toml key 对应）
type Config struct {
    Title   string
    Owner   OwnerInfo
    Database Database struct {
        Server        string
        Ports         []int
        ConnectionMax int
        Enabled       bool
    }
}

type OwnerInfo struct {
    Name string
    Email string
}

// 解析文件
func loadConfig(path string) (*Config, error) {
    var cfg Config
    _, err := toml.DecodeFile(path, &cfg)
    if err != nil {
        return nil, err
    }
    return &cfg, nil
}

// 从字符串解析
func parseTOML(data string) (*Config, error) {
    var cfg Config
    _, err := toml.Decode(data, &cfg)
    return &cfg, err
}
```

## 环境变量

### os.ExpandEnv 展开

```go
import (
    "os"
    "strings"
)

// 配置中引用环境变量
// DB_PASSWORD=${DB_PASSWORD}
func expandEnv(s string) string {
    return os.ExpandEnv(s)
}

// 或手动处理缺失的环境变量
func getEnvOrDefault(key, defaultVal string) string {
    if val := os.Getenv(key); val != "" {
        return val
    }
    return defaultVal
}
```

## 敏感信息处理

```go
import (
    "crypto/aes"
    "crypto/cipher"
    "encoding/base64"
    "fmt"
)

// 1. 环境变量加密存储
// 生产环境使用 KMS 或 vault

// 2. 密钥轮换支持
type SecretProvider struct {
    currentKey []byte
    olderKey   []byte
}

func (p *SecretProvider) Decrypt(ciphertext string) (string, error) {
    // 尝试用当前密钥解密
    if plaintext, err := p.decryptWithKey(ciphertext, p.currentKey); err == nil {
        return plaintext, nil
    }

    // 尝试用旧密钥解密（用于密钥轮换）
    return p.decryptWithKey(ciphertext, p.olderKey)
}

// 3. 不打印敏感配置
type SafeConfig struct {
    Password string `json:"-"`
    Token    string `json:"-"`
}
```

## 最佳实践

```go
// 1. 统一配置加载入口
func InitConfig(path string) (*Config, error) {
    viper.SetConfigFile(path)

    // 设置默认值
    viper.SetDefault("app.port", 8080)
    viper.SetDefault("log.level", "info")

    // 绑定环境变量
    viper.AutomaticEnv()
    viper.SetEnvPrefix("APP")
    viper.SetEnvKeyReplacer(strings.NewReplacer(".", "_"))

    if err := viper.ReadInConfig(); err != nil {
        return nil, err
    }

    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        return nil, err
    }

    return &cfg, nil
}

// 2. 验证必填配置
func validateConfig(cfg *Config) error {
    if cfg.Database.Host == "" {
        return fmt.Errorf("DATABASE_HOST is required")
    }
    if cfg.JWT.Secret == "" {
        return fmt.Errorf("JWT_SECRET is required")
    }
    if len(cfg.JWT.Secret) < 32 {
        return fmt.Errorf("JWT_SECRET must be at least 32 characters")
    }
    return nil
}

// 3. 配置版本管理
// 配置文件添加 version 字段，便于迁移
type Config struct {
    Version int `mapstructure:"version"`
}
```
