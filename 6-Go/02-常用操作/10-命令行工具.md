# 命令行工具

Go 标准库提供了丰富的命令行工具支持：flag 用于命令行参数解析，os/exec 用于执行外部命令。本专题涵盖这两个标准库以及 cobra 等流行第三方库。

---

## flag 标准库

flag 是 Go 标准库的命令行参数解析包，支持多种参数类型。

### 基本用法

```go
package main

import (
    "flag"
    "fmt"
)

func basicFlag() {
    // 字符串参数
    name := flag.String("name", "world", "your name")
    // 短选项（需要手动绑定）
    namePtr := flag.String("n", "default", "short for name")

    // 整数参数
    port := flag.Int("port", 8080, "server port")
    age := flag.Int("age", 0, "your age")

    // 布尔参数
    verbose := flag.Bool("v", false, "verbose mode")
    debug := flag.Bool("d", false, "debug mode")

    // 解析命令行参数
    flag.Parse()

    fmt.Printf("Name: %s, Port: %d, Age: %d, Verbose: %v\n",
        *name, *port, *age, *verbose)
}
```

### 绑定已声明变量

```go
func bindVariable() {
    var (
        host string
        port int
        maxConn int
    )

    // 使用 StringVar/IntVar 等绑定到已声明变量
    flag.StringVar(&host, "host", "localhost", "server host")
    flag.IntVar(&port, "port", 8080, "server port")
    flag.IntVar(&maxConn, "max-conn", 100, "max connections")

    flag.Parse()

    fmt.Printf("host=%s port=%d max-conn=%d\n", host, port, maxConn)
}
```

### 位置参数和未命名参数

```go
func positionalArgs() {
    flag.Parse()

    // 获取位置参数（非选项参数）
    args := flag.Args()
    fmt.Printf("Args: %v\n", args)

    // 获取第 N 个位置参数
    if len(args) > 0 {
        fmt.Printf("First arg: %s\n", args[0])
    }

    // NFlag 返回已解析的选项数量
    fmt.Printf("Parsed %d flags\n", flag.NFlag())
}
```

### 显示用法信息

```go
func usageInfo() {
    flag.Usage = func() {
        fmt.Fprintf(flag.CommandLine.Output(), "Usage: %s [options] [args]\n\nOptions:\n", os.Args[0])
        flag.PrintDefaults()
    }

    flag.Parse()
}
```

### 自定义选项类型

```go
import (
    "flag"
    "fmt"
    "net"
)

func customType() {
    var ip net.IP
    flag.Var(&ipVal{ip: &ip}, "server", "server IP address")

    flag.Parse()
    fmt.Printf("Server IP: %s\n", ip.String())
}

type ipVal struct {
    ip *net.IP
}

func (v *ipVal) Set(s string) error {
    parsed := net.ParseIP(s)
    if parsed == nil {
        return fmt.Errorf("invalid IP: %s", s)
    }
    *v.ip = parsed
    return nil
}

func (v *ipVal) String() string {
    if v.ip == nil {
        return ""
    }
    return v.ip.String()
}
```

## os/exec 标准库

os/exec 提供了执行外部命令的能力。

### 执行简单命令

```go
import (
    "bytes"
    "fmt"
    "os/exec"
)

func execBasic() {
    // 执行 ls 命令
    cmd := exec.Command("ls", "-la")

    // 获取输出
    output, err := cmd.Output()
    if err != nil {
        fmt.Printf("Error: %v\n", err)
        return
    }

    fmt.Printf("Output:\n%s\n", string(output))
}
```

### 带参数的命令

```go
func execWithArgs() {
    // 方式一：Command 参数分开
    cmd := exec.Command("go", "version")

    // 方式二：通过 Shell 执行复杂命令（注意安全问题）
    // cmd := exec.Command("sh", "-c", "ls -la | grep go")

    output, err := cmd.Output()
    if err != nil {
        fmt.Printf("Error: %v\n", err)
        return
    }

    fmt.Printf("Go version: %s", string(output))
}
```

### 获取 stderr

```go
func execWithStderr() {
    cmd := exec.Command("ls", "/nonexistent")

    // CombinedOutput 同时获取 stdout 和 stderr
    output, err := cmd.CombinedOutput()
    if err != nil {
        fmt.Printf("Error (includes stderr): %s\nOutput: %s\n", err, string(output))
    }
}
```

### 设置环境变量和工作目录

```go
func execWithEnv() {
    cmd := exec.Command("go", "env", "GOPATH")

        "DEBUG=true",

    if err != nil {
```


### 流式输出（实时）

```go
func execStreaming() {
    cmd := exec.Command("ping", "-c", "5", "localhost")
    cmd.Stdout = os.Stdout  // 直接输出到标准输出
    cmd.Stderr = os.Stderr

    if err := cmd.Run(); err != nil {
        fmt.Printf("Error: %v\n", err)
    }
}
```

### 执行管道命令

```go
func execPipeline() {
    // 第一个命令：ps aux
    cmd1 := exec.Command("ps", "aux")
    // 第二个命令：grep go
    cmd2 := exec.Command("grep", "go")

    // 连接管道
    cmd2.Stdin, _ = cmd1.StdoutPipe()

    var buf bytes.Buffer
    cmd2.Stdout = &buf

    // 同时启动两个命令
    if err := cmd1.Start(); err != nil {
        fmt.Printf("cmd1 start error: %v\n", err)
        return
    }
    if err := cmd2.Start(); err != nil {
        fmt.Printf("cmd2 start error: %v\n", err)
        return
    }

    // 等待完成
    cmd1.Wait()
    if err := cmd2.Wait(); err != nil {
        fmt.Printf("cmd2 wait error: %v\n", err)
        return
    }

    fmt.Printf("Filtered output:\n%s\n", buf.String())
}
```

### 查找可执行文件

### 命令超时和取消

```go
import (
    "context"
    "time"
)

func execWithTimeout() {
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
    defer cancel()

    cmd := exec.CommandContext(ctx, "sleep", "10")

    if err := cmd.Run(); err != nil {
        if ctx.Err() == context.DeadlineExceeded {
            fmt.Println("Command timed out")
        } else {
            fmt.Printf("Command error: %v\n", err)
        }
        return
    }

    fmt.Println("Command completed")
}
```

## cobra 第三方库

cobra 是 Go 最流行的 CLI 框架，被 Docker、Kubernetes 等项目使用。

### 安装

```bash
go get github.com/spf13/cobra@latest
```

### 基础结构

```go
package main

import (
    "fmt"
    "github.com/spf13/cobra"
)

var rootCmd = &cobra.Command{
    Use:   "myapp",
    Short: "My CLI application",
    Long:  `A longer description that spans multiple lines.`,
}

func main() {
    if err := rootCmd.Execute(); err != nil {
        fmt.Println(err)
    }
}
```

### 添加子命令

```go
// version 命令
var versionCmd = &cobra.Command{
    Use:   "version",

    Run: func(cmd *cobra.Command, args []string) {

    },
}

// serve 命令
var serveCmd = &cobra.Command{
    Use:   "serve",

var port int

func init() {
```

## flag vs cobra vs kingpin

| 特性 | flag | cobra | kingpin |
|------|------|-------|---------|
| 复杂度 | 低 | 高 | 中 |
| 子命令 | 无 | 支持 | 支持 |
| Bash 补全 | 无 | 原生支持 | 支持 |
| 验证 | 手动 | 手动 | 自动 |
| 最佳场景 | 简单脚本 | 复杂 CLI | 复杂 CLI |

## 最佳实践

### 1. 组合 flag 和 cobra

```go
func cobraWithFlags() *cobra.Command {
    var (
        port  int
        host  string
        debug bool
    )

    cmd := &cobra.Command{
        Use:   "serve",
        Short: "Start server",
        Run: func(cmd *cobra.Command, args []string) {
            fmt.Printf("Server: host=%s port=%d debug=%v\n", host, port, debug)
        },
    }

    cmd.Flags().IntVarP(&port, "port", "p", 8080, "port")
    cmd.Flags().StringVarP(&host, "host", "H", "localhost", "host")
    cmd.Flags().BoolVarP(&debug, "debug", "d", false, "debug mode")

    return cmd
}
```

### 2. 安全的 shell 命令执行

```go
func safeExec() error {
    // 避免 shell=True，直接执行命令和参数
    cmd := exec.Command("go", "build", "-o", "app", "main.go")

    // 设置安全的环境变量
    cmd.Env = []string{
        "GOPATH=" + os.Getenv("GOPATH"),
        "GO111MODULE=on",
    }

    return cmd.Run()
}
```

### 3. 命令超时包装

```go
func execWithTimeoutCtx(cmd *exec.Cmd, timeout time.Duration) error {
    ctx, cancel := context.WithTimeout(context.Background(), timeout)
    defer cancel()

    cmd.Cancel = func() error {
        return cmd.Process.Kill()
    }

    go func() {
        <-ctx.Done()
        cmd.Cancel()
    }()

    return cmd.Run()
}
```

### 4. 统一错误输出

```go
import (
    "github.com/spf13/cobra"
)

func runE(cmd *cobra.Command, args []string) error {
    err := doSomething()
    if err != nil {
        return fmt.Errorf("operation failed: %w", err)
    }
    return nil
}
```
