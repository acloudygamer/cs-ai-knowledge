# 数据库操作

## database/sql（标准库）

Go 标准库提供 `database/sql`，支持所有兼容 DB-API 2.0 的数据库驱动。

### 驱动注册

```go
import (
    "database/sql"
    _ "github.com/go-sql-driver/mysql"    // MySQL
    // _ "github.com/lib/pq"                 // PostgreSQL
    // _ "github.com/mattn/go-sqlite3"      // SQLite
)
```

### 连接数据库

```go
db, err := sql.Open("mysql", "user:password@tcp(localhost:3306)/mydb?parseTime=true")
if err != nil {
    log.Fatal(err)
}
defer db.Close()

// 设置连接池
db.SetMaxOpenConns(25)              // 最大打开连接数
db.SetMaxIdleConns(5)              // 最大空闲连接数
db.SetConnMaxLifetime(5 * time.Minute)  // 连接最大生命周期

// 验证连接
if err := db.Ping(); err != nil {
    log.Fatal(err)
}
```

### 查询（SELECT）

```go
// 单行查询
var name string
var age int
err := db.QueryRow("SELECT name, age FROM users WHERE id = ?", id).
    Scan(&name, &age)
if err == sql.ErrNoRows {
    // 没找到记录
}
fmt.Println(name, age)

// 多行查询
rows, err := db.Query("SELECT id, name, email FROM users WHERE age > ?", 18)
if err != nil {
    log.Fatal(err)
}
defer rows.Close()

for rows.Next() {
    var id int
    var name, email string
    if err := rows.Scan(&id, &name, &email); err != nil {
        log.Fatal(err)
    }
    fmt.Printf("%d: %s <%s>\n", id, name, email)
}
if err := rows.Err(); err != nil {
    log.Fatal(err)
}
```

### 插入/更新/删除

```go
// 插入并获取自增 ID
result, err := db.Exec(
    "INSERT INTO users (name, email) VALUES (?, ?)",
    "Tom", "tom@example.com",
)
if err != nil {
    log.Fatal(err)
}
id, _ := result.LastInsertId()

// 更新
affected, _ := result.RowsAffected()
_, err = db.Exec(
    "UPDATE users SET email = ? WHERE id = ?",
    "new@example.com", id,
)

// 删除
_, err = db.Exec("DELETE FROM users WHERE id = ?", id)
```

### 预处理语句

```go
// 预处理语句（适合重复执行）
stmt, err := db.Prepare("INSERT INTO users (name, email) VALUES (?, ?)")
if err != nil {
    log.Fatal(err)
}
defer stmt.Close()

// 多次执行
stmt.Exec("Alice", "alice@example.com")
stmt.Exec("Bob", "bob@example.com")
```

### 事务

```go
tx, err := db.Begin()
if err != nil {
    log.Fatal(err)
}
defer tx.Rollback()  // 如果没 Commit，会 Rollback

_, err = tx.Exec("UPDATE accounts SET balance = balance - ? WHERE id = ?", 100, 1)
if err != nil {
    tx.Rollback()
    return
}

_, err = tx.Exec("UPDATE accounts SET balance = balance + ? WHERE id = ?", 100, 2)
if err != nil {
    tx.Rollback()
    return
}

if err := tx.Commit(); err != nil {
    log.Fatal(err)
}
```

### 空值处理

```go
var name sql.NullString
var age sql.NullInt64

row := db.QueryRow("SELECT name, age FROM users WHERE id = ?", id)
if err := row.Scan(&name, &age); err != nil {
    log.Fatal(err)
}

if name.Valid {
    fmt.Println(name.String)
} else {
    fmt.Println("name is NULL")
}
```

## GORM（ORM 框架）

GORM 是 Go 最流行的 ORM 库。

### 安装

```bash
go get -u gorm.io/gorm
go get -u gorm.io/driver/mysql
```

### 基本模型

```go
type User struct {
    ID        uint      `gorm:"primaryKey"`
    Name      string    `gorm:"size:255;not null"`
    Email     string    `gorm:"uniqueIndex;size:255"`
    Age       int       `gorm:"default:0"`
    CreatedAt time.Time
    UpdatedAt time.Time
}

// 自动迁移
db.AutoMigrate(&User{})
```

### CRUD 操作

```go
// 创建
user := User{Name: "Tom", Email: "tom@example.com", Age: 30}
result := db.Create(&user)
fmt.Println(user.ID)

// 批量创建
users := []User{
    {Name: "Alice", Email: "alice@example.com"},
    {Name: "Bob", Email: "bob@example.com"},
}
db.Create(&users)

// 查询
var user User
db.First(&user, 1)           // 按主键查找
db.First(&user, "email = ?", "tom@example.com")

var users []User
db.Where("age > ?", 18).Find(&users)

// 更新
db.Model(&user).Update("Email", "new@example.com")
db.Where("age < ?", 18).Model(&User{}).Update("age", 18)

// 删除
db.Delete(&user, 1)
```

### 关联关系

```go
type CreditCard struct {
    ID      uint
    Number  string
    UserID  uint
}

type User struct {
    ID           uint
    Name         string
    CreditCards  []CreditCard `gorm:"foreignKey:UserID"`
}

// 预加载
var user User
db.Preload("CreditCards").First(&user, 1)
```

### 事务

```go
db.Transaction(func(tx *gorm.DB) error {
    if err := tx.Create(&User{Name: "Tom"}).Error; err != nil {
        return err
    }
    if err := tx.Create(&User{Name: "Alice"}).Error; err != nil {
        return err
    }
    return nil
})
```

## sqlx（轻量级扩展）

sqlx 在 `database/sql` 基础上提供结构体映射，比 GORM 更轻量。

### 安装

```bash
go get github.com/jmoiron/sqlx
```

### 基本用法

```go
import "github.com/jmoiron/sqlx"

type User struct {
    ID    uint      `db:"id"`
    Name  string    `db:"name"`
    Email string    `db:"email"`
}

// 查询到结构体
var user User
err := db.Get(&user, "SELECT * FROM users WHERE id = ?", 1)

// 查询到切片
var users []User
err = db.Select(&users, "SELECT * FROM users WHERE age > ?", 18)

// 命名参数
_, err = db.NamedExec(`INSERT INTO users (name, email) VALUES (:name, :email)`,
    map[string]interface{}{
        "name":  "Tom",
        "email": "tom@example.com",
    })
```

## Redis 操作

### go-redis 客户端

```bash
go get github.com/redis/go-redis/v9
```

### 基本操作

```go
import "github.com/redis/go-redis/v9"

client := redis.NewClient(&redis.Options{
    Addr:     "localhost:6379",
    Password: "",
    DB:       0,
})
ctx := context.Background()

// String
err := client.Set(ctx, "key", "value", 0).Err()
val, err := client.Get(ctx, "key").Result()

// Hash
client.HSet(ctx, "user:1", map[string]interface{}{
    "name":  "Tom",
    "email": "tom@example.com",
})
user, _ := client.HGetAll(ctx, "user:1").Result()

// List
client.LPush(ctx, "queue", "task1")
task, _ := client.RPop(ctx, "queue").Result()

// Set
client.SAdd(ctx, "tags", "go", "backend")
tags, _ := client.SMembers(ctx, "tags").Result()

// 过期时间
client.Set(ctx, "token", "abc123", 30*time.Minute)
```

### 分布式锁

```go
func AcquireLock(ctx context.Context, client *redis.Client, key string, ttl time.Duration) (bool, error) {
    ok, err := client.SetNX(ctx, key, "locked", ttl).Result()
    return ok, err
}

func ReleaseLock(ctx context.Context, client *redis.Client, key string) error {
    return client.Del(ctx, key).Err()
}
```

### 缓存模式

```go
func GetUserCached(ctx context.Context, client *redis.Client, db *sql.DB, id string) (*User, error) {
    cacheKey := "user:" + id

    // 先查缓存
    cached, err := client.Get(ctx, cacheKey).Result()
    if err == nil {
        var user User
        json.Unmarshal([]byte(cached), &user)
        return &user, nil
    }

    // 缓存未命中，查数据库
    var user User
    err = db.Get(&user, "SELECT * FROM users WHERE id = ?", id)
    if err != nil {
        return nil, err
    }

    // 写入缓存
    data, _ := json.Marshal(user)
    client.Set(ctx, cacheKey, data, 5*time.Minute)

    return &user, nil
}
```

## 数据库连接池调优

```go
// 查看连接池状态
stats := db.Stats()
fmt.Printf("Open: %d, InUse: %d, Idle: %d\n",
    stats.OpenConnections, stats.InUse, stats.Idle)

// 常见问题
// 1. 连接泄漏：确保 defer db.Close()，使用 defer rows.Close()
// 2. 慢查询：使用 EXPLAIN 分析，添加索引
// 3. 连接数过多：调整 MaxOpenConns/MaxIdleConns
```

## 最佳实践

```go
// 1. 使用 ? 占位符（防止 SQL 注入）
db.Query("SELECT * FROM users WHERE id = ?", id)

// 2. 始终检查错误
if err != nil {
    return nil, fmt.Errorf("query failed: %w", err)
}

// 3. 释放资源
rows, err := db.Query("SELECT ...")
if err != nil {
    return err
}
defer rows.Close()

// 4. 使用短事务
tx, err := db.Begin()
if err != nil {
    return err
}
defer tx.Rollback()
// ... 操作 ...
tx.Commit()

// 5. 错误处理：sql.ErrNoRows
var user User
err := db.Get(&user, "SELECT ... WHERE id = ?", id)
if err == sql.ErrNoRows {
    // 处理未找到
}
```
