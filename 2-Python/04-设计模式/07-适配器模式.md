# 适配器模式

将一个类的接口转换成客户期望的另一个接口，使原本接口不兼容的类可以一起工作。

## 类适配器

```python
# 目标接口
class Target:
    def request(self) -> str:
        return "Target: Default behavior"

# 需要适配的类
class Adaptee:
    def specific_request(self) -> str:
        return ".eetpadA eht fo roivaheb laicepS"

# 类适配器（多重继承）
class ClassAdapter(Adaptee, Target):
    def request(self) -> str:
        # 翻转字符串来适配
        return self.specific_request()[::-1]

# 使用
target = Target()
print(target.request())  # Target: Default behavior

adapter = ClassAdapter()
print(adapter.request())  # Special behavior for the target
```

## 对象适配器（组合）

```python
class Target:
    def request(self) -> str:
        return "Target: Default behavior"

class Adaptee:
    def specific_request(self) -> str:
        return ".eetpadA eht fo roivaheb laicepS"

class ObjectAdapter(Target):
    def __init__(self, adaptee: Adaptee):
        self._adaptee = adaptee

    def request(self) -> str:
        return self._adaptee.specific_request()[::-1]

# 使用
adaptee = Adaptee()
adapter = ObjectAdapter(adaptee)
print(adapter.request())  # Special behavior for the target
```

## 实际应用：支付网关适配

```python
# 我们的应用期望的支付接口
class PaymentGateway:
    def charge(self, amount: float, currency: str) -> dict:
        raise NotImplementedError

# 第三方支付 SDK
class StripeSDK:
    def make_payment(self, amount_cents: int, currency: str) -> dict:
        return {
            "status": "succeeded",
            "transaction_id": "stripe_tx_123",
            "amount_cents": amount_cents
        }

class PayPalSDK:
    def send_payment(self, amount: float, currency: str) -> dict:
        return {
            "result": "success",
            "tx_id": "pp_tx_456",
            "money": amount
        }

# 适配器
class StripeAdapter(PaymentGateway):
    def __init__(self, stripe_sdk: StripeSDK):
        self._stripe = stripe_sdk

    def charge(self, amount: float, currency: str) -> dict:
        amount_cents = int(amount * 100)
        result = self._stripe.make_payment(amount_cents, currency)
        return {
            "success": result["status"] == "succeeded",
            "transaction_id": result["transaction_id"],
            "amount": amount,
            "currency": currency
        }

class PayPalAdapter(PaymentGateway):
    def __init__(self, paypal_sdk: PayPalSDK):
        self._paypal = paypal_sdk

    def charge(self, amount: float, currency: str) -> dict:
        result = self._paypal.send_payment(amount, currency)
        return {
            "success": result["result"] == "success",
            "transaction_id": result["tx_id"],
            "amount": amount,
            "currency": currency
        }

# 客户端代码
def process_payment(gateway: PaymentGateway, amount: float):
    result = gateway.charge(amount, "USD")
    print(f"Payment {result['transaction_id']}: {'Success' if result['success'] else 'Failed'}")

# 使用
stripe = StripeAdapter(StripeSDK())
paypal = PayPalAdapter(PayPalSDK())

process_payment(stripe, 99.99)
# Payment stripe_tx_123: Success

process_payment(paypal, 49.99)
# Payment pp_tx_456: Success
```

## 文件格式适配器

```python
import json
import xml.etree.ElementTree as ET

# 目标接口
class DataProcessor:
    def process(self, data: str) -> dict:
        raise NotImplementedError

# XML 解析器
class XMLParser:
    def parse(self, xml_string: str) -> dict:
        root = ET.fromstring(xml_string)
        return {child.tag: child.text for child in root}

# CSV 解析器
class CSVParser:
    def parse(self, csv_string: str) -> dict:
        lines = csv_string.strip().split("\n")
        headers = lines[0].split(",")
        values = lines[1].split(",")
        return dict(zip(headers, values))

# 适配器
class XMLAdapter(DataProcessor):
    def __init__(self):
        self._parser = XMLParser()

    def process(self, data: str) -> dict:
        return self._parser.parse(data)

class CSVAdapter(DataProcessor):
    def __init__(self):
        self._parser = CSVParser()

    def process(self, data: str) -> dict:
        return self._parser.parse(data)

# 使用
xml_data = """<user><name>Alice</name><age>30</age></user>"""
csv_data = """name,age\nBob,25"""

processor: DataProcessor = XMLAdapter()
print(processor.process(xml_data))  # {'name': 'Alice', 'age': '30'}

processor = CSVAdapter()
print(processor.process(csv_data))  # {'name': 'Bob', 'age': '25'}
```

## 双向适配器

```python
# 两个不兼容的接口互相适配
class OldAPI:
    def get_user_data(self) -> dict:
        return {"user_name": "Alice", "user_age": 30}

class NewAPI:
    def fetch_user(self) -> dict:
        return {"name": "Bob", "age": 25}

class BidirectionalAdapter:
    def __init__(self, api):
        self._api = api

    # 将 OldAPI 适配为 NewAPI 接口
    def fetch_user(self):
        data = self._api.get_user_data()
        return {
            "name": data.get("user_name", ""),
            "age": data.get("user_age", 0)
        }

    # 将 NewAPI 适配为 OldAPI 接口
    def get_user_data(self):
        data = self._api.fetch_user()
        return {
            "user_name": data.get("name", ""),
            "user_age": data.get("age", 0)
        }

# 使用
old_api = OldAPI()
new_style = BidirectionalAdapter(old_api)
print(new_style.fetch_user())  # {'name': 'Alice', 'age': 30}
```
