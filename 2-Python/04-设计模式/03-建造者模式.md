# 建造者模式

将一个复杂对象的构建与它的表示分离，使同样的构建过程可以创建不同的表示。

## 基本实现

```python
class House:
    def __init__(self):
        self.walls = None
        self.roof = None
        self.windows = None
        self.door = None
        self.garden = None

    def __repr__(self):
        return (f"House(walls={self.walls}, roof={self.roof}, "
                f"windows={self.windows}, door={self.door}, garden={self.garden})")

class HouseBuilder:
    def __init__(self):
        self.house = House()

    def build_walls(self, material: str):
        self.house.walls = material
        return self  # 返回 self 支持链式调用

    def build_roof(self, material: str):
        self.house.roof = material
        return self

    def build_windows(self, count: int):
        self.house.windows = count
        return self

    def build_door(self, style: str):
        self.house.door = style
        return self

    def build_garden(self, has_garden: bool):
        self.house.garden = has_garden
        return self

    def get_result(self):
        return self.house

# 使用
house = (HouseBuilder()
         .build_walls("brick")
         .build_roof("tile")
         .build_windows(4)
         .build_door("oak")
         .build_garden(True)
         .get_result())
print(house)
# House(walls=brick, roof=tile, windows=4, door=oak, garden=True)
```

## 带导演类的实现

```python
class HouseDirector:
    def __init__(self, builder: HouseBuilder):
        self.builder = builder

    def build_simple_house(self):
        return (self.builder
                .build_walls("wood")
                .build_roof("shingle")
                .build_windows(2)
                .build_door("pine")
                .build_garden(False)
                .get_result())

    def build_luxury_house(self):
        return (self.builder
                .build_walls("stone")
                .build_roof("copper")
                .build_windows(8)
                .build_door("mahogany")
                .build_garden(True)
                .get_result())

# 使用
simple_builder = HouseBuilder()
director = HouseDirector(simple_builder)
simple_house = director.build_simple_house()

luxury_builder = HouseBuilder()
director = HouseDirector(luxury_builder)
luxury_house = director.build_luxury_house()
```

## 使用 dataclasses 简化

```python
from dataclasses import dataclass, field

@dataclass
class Pizza:
    size: str = ""
    cheese: bool = False
    pepperoni: bool = False
    mushrooms: bool = False
    olives: bool = False

    def __repr__(self):
        return (f"Pizza(size={self.size!r}, cheese={self.cheese}, "
                f"pepperoni={self.pepperoni}, mushrooms={self.mushrooms}, "
                f"olives={self.olives})")

class PizzaBuilder:
    def __init__(self, size: str):
        self.pizza = Pizza(size=size)

    def add_cheese(self):
        self.pizza.cheese = True
        return self

    def add_pepperoni(self):
        self.pizza.pepperoni = True
        return self

    def add_mushrooms(self):
        self.pizza.mushrooms = True
        return self

    def add_olives(self):
        self.pizza.olives = True
        return self

    def build(self):
        return self.pizza

# 使用
pizza = (PizzaBuilder("large")
         .add_cheese()
         .add_pepperoni()
         .add_mushrooms()
         .build())
print(pizza)
# Pizza(size='large', cheese=True, pepperoni=True, mushrooms=True, olives=False)
```

## 链式调用的注意事项

```python
# 错误示例：可变默认参数
class BadBuilder:
    def __init__(self):
        self.data = []  # 共享的可变状态

    def add_item(self, item):
        self.data.append(item)
        return self

    def build(self):
        return self.data

# 正确示例：每次构建返回新实例
class GoodBuilder:
    def __init__(self, data=None):
        self.data = data or []  # 避免可变默认参数

    def add_item(self, item):
        new_builder = GoodBuilder(self.data + [item])
        return new_builder

    def build(self):
        return self.data
```
