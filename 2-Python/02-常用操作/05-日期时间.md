# 日期时间处理

## datetime 模块

```python
from datetime import datetime, date, time, timedelta
import zoneinfo

# 当前时间
now = datetime.now()
utc_now = datetime.now(zoneinfo.UTC)

# 创建特定时间
dt = datetime(2024, 6, 15, 14, 30, 0)
print(dt)  # 2024-06-15 14:30:00

# 日期和时间互转
d = dt.date()
t = dt.time()

# 组合 datetime
dt = datetime.combine(date(2024, 6, 15), time(14, 30))
```

## date 对象

```python
from datetime import date

# 当前日期
today = date.today()
print(today)  # 2024-06-15

# 创建特定日期
d = date(2024, 12, 25)

# 日期属性
print(d.year)   # 2024
print(d.month)  # 12
print(d.day)    # 25

# 格式化
print(d.strftime("%Y-%m-%d"))  # 2024-12-25
print(d.strftime("%B %d, %Y"))  # December 25, 2024
print(d.strftime("%a, %d %b %Y"))  # Wed, 25 Dec 2024
```

## time 对象

```python
from datetime import time

# 创建时间
t = time(14, 30, 45)
print(t.hour)    # 14
print(t.minute)  # 30
print(t.second)  # 45

# 格式化
print(t.strftime("%H:%M:%S"))  # 14:30:45
print(t.strftime("%I:%M %p"))  # 02:30 PM
```

## timedelta（时间差）

```python
from datetime import datetime, timedelta

# 创建时间差
delta = timedelta(days=7, hours=12)
print(delta.total_seconds())  # 648000.0

# 加减法
dt1 = datetime(2024, 6, 15)
dt2 = dt1 + timedelta(days=30)
dt3 = dt2 - timedelta(weeks=1)

# 计算日期间隔
start = datetime(2024, 1, 1)
end = datetime(2024, 12, 31)
duration = end - start
print(duration.days)      # 365
print(duration.total_seconds())  # 31536000.0

# 比较
delta1 = timedelta(hours=2)
delta2 = timedelta(minutes=120)
print(delta1 == delta2)  # True
```

## 时区处理

```python
from datetime import datetime, timezone
import zoneinfo

# 固定时区
utc = timezone.utc
pst = timezone(timedelta(hours=-8))

# 创建带时区的时间
dt_utc = datetime(2024, 6, 15, 14, 30, tzinfo=utc)
dt_pst = datetime(2024, 6, 15, 14, 30, tzinfo=pst)

# 时区转换
tokyo = zoneinfo.ZoneInfo("Asia/Tokyo")
dt_tokyo = dt_utc.astimezone(tokyo)
print(dt_tokyo)  # 2024-06-15 23:30:00+09:00

# 本地时间
local_dt = datetime.now(zoneinfo.ZoneInfo("Asia/Shanghai"))
print(local_dt.tzname())  # CST
```

## 字符串解析

```python
from datetime import datetime

# 字符串转 datetime
dt = datetime.strptime("2024-06-15 14:30:00", "%Y-%m-%d %H:%M:%S")
print(dt)  # 2024-06-15 14:30:00

# 常用格式
datetime.strptime("Jun 15, 2024", "%b %d, %Y")
datetime.strptime("15/06/2024", "%d/%m/%Y")
datetime.strptime("2024-06-15T14:30:00Z", "%Y-%m-%dT%H:%M:%SZ")

# ISO 格式
dt_iso = datetime.fromisoformat("2024-06-15T14:30:00+08:00")

# datetime 转字符串
dt = datetime(2024, 6, 15, 14, 30)
print(dt.strftime("%Y-%m-%d %H:%M:%S"))  # 2024-06-15 14:30:00
print(dt.isoformat())  # 2024-06-15T14:30:00
```

## 时间戳互转

```python
from datetime import datetime, timezone

# datetime 转时间戳（秒）
dt = datetime(2024, 6, 15, 14, 30, tzinfo=timezone.utc)
timestamp = dt.timestamp()
print(timestamp)  # 1718456400.0

# 时间戳转 datetime
ts = 1718456400
dt = datetime.fromtimestamp(ts, tz=timezone.utc)
print(dt)  # 2024-06-15T14:30:00+00:00

# UTC 时间戳
from datetime import datetime as dt_module, timezone
print(dt_module.now(timezone.utc).timestamp())  # 当前 UTC 时间戳（非废弃方式）
```

## time 模块（墙钟时间）

```python
import time

# 纪元时间（1970-01-01 起）
print(time.time())  # 1718456400.123

# 睡眠
time.sleep(1)  # 睡眠 1 秒

# 格式化（当前时间）
print(time.strftime("%Y-%m-%d %H:%M:%S"))  # 当前本地时间
print(time.strftime("%Y-%m-%d %H:%M:%S", time.gmtime()))  # UTC

# 结构化时间
t = time.localtime()
print(t.tm_year, t.tm_mon, t.tm_mday)
```

## 常用场景

```python
from datetime import datetime, timedelta, timezone
import zoneinfo

# 1. 日期范围
start = datetime(2024, 1, 1)
end = datetime(2024, 12, 31)
current = start
while current <= end:
    print(current.strftime("%Y-%m-%d"))
    current += timedelta(days=1)

# 2. 相对时间（如"3天前"）
now = datetime.now()
three_days_ago = now - timedelta(days=3)

# 3. 下个月的第一天
def first_of_next_month(year: int, month: int):
    if month == 12:
        return datetime(year + 1, 1, 1)
    return datetime(year, month + 1, 1)

# 4. 工作日计算
from datetime import datetime, timedelta

def add_business_days(start_date: datetime, days: int) -> datetime:
    current = start_date
    while days > 0:
        current += timedelta(days=1)
        if current.weekday() < 5:  # 周一到周五
            days -= 1
    return current

# 5. 年龄计算
def calculate_age(birth_date: datetime) -> int:
    today = datetime.today()
    age = today.year - birth_date.year
    if (today.month, today.day) < (birth_date.month, birth_date.day):
        age -= 1
    return age
```

## 常见陷阱

```python
# 陷阱1：混淆 naive 和 aware datetime
from datetime import datetime, timezone

naive = datetime(2024, 6, 15)  # 无时区信息
aware = datetime(2024, 6, 15, tzinfo=timezone.utc)  # 带时区

# naive 和 aware 不能直接比较
# aware - naive  # TypeError

# 陷阱2：夏令时边界问题
import zoneinfo
tz = zoneinfo.ZoneInfo("America/New_York")
# 2024-03-10 02:00:00 不存在（夏令时跳变）
# 使用 zoneinfo 会自动处理

# 陷阱3：timedelta 月份计算
from dateutil.relativedelta import relativedelta
from datetime import datetime

# timedelta 不支持月份计算
dt = datetime(2024, 1, 31)
# dt + timedelta(months=1)  # 错误

# 正确做法
dt + relativedelta(months=1)  # 2024-02-29
```
