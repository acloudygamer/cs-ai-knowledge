# 网络请求

## requests

### 基本用法

```python
import requests

# GET 请求
response = requests.get("https://api.example.com/users")
response.raise_for_status()                 # 失败时抛异常
data = response.json()                       # 解析 JSON

# 带参数
params = {"page": 1, "limit": 10}
response = requests.get("https://api.example.com/users", params=params)

# POST 请求
payload = {"name": "Alice", "email": "alice@example.com"}
response = requests.post("https://api.example.com/users", json=payload)
```

### 进阶用法

```python
# 自定义请求头
url = "https://api.example.com/upload"
headers = {"Authorization": "Bearer token123"}
response = requests.get(url, headers=headers)

# 上传文件
files = {"file": open("report.pdf", "rb")}
response = requests.post("https://api.example.com/upload", files=files)

# 设置超时（避免无限等待）
response = requests.get(url, timeout=5)

# Session（保持会话、复用连接）
session = requests.Session()
session.headers.update({"Authorization": "Bearer token"})
response = session.get("https://api.example.com/profile")
```

### 错误处理

```python
import requests
from requests.exceptions import RequestException, Timeout, ConnectionError

def fetch_data(url, retries=3):
    """带重试的请求封装"""
    for attempt in range(retries):
        try:
            response = requests.get(url, timeout=10)
            response.raise_for_status()
            return response.json()
        except Timeout:
            print(f"请求超时 (尝试 {attempt + 1}/{retries})")
        except ConnectionError:
            print(f"连接失败 (尝试 {attempt + 1}/{retries})")
        except RequestException as e:
            print(f"请求失败: {e}")
            raise
    raise Exception(f"达到最大重试次数 {retries}")
```

### 实践场景

```python
# 下载文件（流式处理大文件）
def download_file(url, dest_path):
    with requests.get(url, stream=True) as response:
        response.raise_for_status()
        with open(dest_path, "wb") as f:
            for chunk in response.iter_content(chunk_size=8192):
                f.write(chunk)

# RESTful API 完整示例
class APIClient:
    def __init__(self, base_url, token):
        self.base_url = base_url.rstrip("/")
        self.session = requests.Session()
        self.session.headers.update({"Authorization": f"Bearer {token}"})

    def get_users(self, page=1, limit=10):
        return self.session.get(
            f"{self.base_url}/users",
            params={"page": page, "limit": limit}
        ).json()

    def create_user(self, name, email):
        return self.session.post(
            f"{self.base_url}/users",
            json={"name": name, "email": email}
        ).json()
```

## httpx（异步支持）

### 基础用法

```python
import httpx
import asyncio

# 同步用法（类似 requests）
client = httpx.Client()
response = client.get("https://api.example.com/users")

# 异步用法
async def fetch_all(urls):
    async with httpx.AsyncClient() as client:
        tasks = [client.get(url) for url in urls]
        responses = await asyncio.gather(*tasks)
        return [r.json() for r in responses]
```

### 超时配置

```python
# 全局超时

    connect=5.0,    # 连接超时

    write=10.0,    # 写入超时

)
```

### 指数退避重试

```python
import asyncio
import httpx

async def fetch_with_retry(
    url: str,
    max_retries: int = 3,
    base_delay: float = 1.0
) -> dict:
    """指数退避重试"""
    async with httpx.AsyncClient() as client:
        for attempt in range(max_retries):
            try:
                response = await client.get(url)
                response.raise_for_status()
                return response.json()
            except (httpx.HTTPError, httpx.TimeoutException) as e:
                if attempt == max_retries - 1:
                    raise
                delay = base_delay * (2 ** attempt)
                print(f"请求失败，{delay}秒后重试 ({attempt + 1}/{max_retries})")
                await asyncio.sleep(delay)
```

### 异步并发请求

```python
import asyncio
import httpx

async def concurrent_requests(urls: list[str]) -> list[dict]:
    """并发请求多个 URL"""
    async with httpx.AsyncClient(
        timeout=30.0,
        limits=httpx.Limits(max_connections=100)
    ) as client:
        tasks = [
            client.get(url, headers={"Accept": "application/json"})
            for url in urls
        ]
        responses = await asyncio.gather(*tasks, return_exceptions=True)

        results = []
        for i, response in enumerate(responses):
            if isinstance(response, Exception):
                print(f"请求 {urls[i]} 失败: {response}")
                results.append(None)
            else:
                results.append(response.json())
        return results

# 带优先级的请求
async def prioritized_requests():
    """高优先级先执行，低优先级批量执行"""
    async with httpx.AsyncClient() as client:
        # 高优先级
        high_priority = [
            client.get("https://api.example.com/users"),
            client.get("https://api.example.com/config"),
        ]

        # 先执行高优先级
        high_results = await asyncio.gather(*high_priority)

        # 低优先级（基于结果）
        low_priority = [
            client.get(f"https://api.example.com/users/{uid}/details")
            for uid in range(100)
        ]
        low_results = await asyncio.gather(*low_priority)

        return high_results, low_results
```

async def websocket_client():

        await ws.send("Hello!")

async def chat_client():
    uri = "ws://localhost:8765/chat"
    async with websockets.connect(uri) as ws:
        # 发送消息
        await ws.send('{"message": "Hello", "user": "Alice"}')

        # 接收消息（带超时）
        try:
            message = await asyncio.wait_for(ws.recv(), timeout=30.0)
            print(f"收到: {message}")
        except asyncio.TimeoutError:
            print("接收超时")

# 服务器
async def websocket_server():
    async def echo(websocket, path):
        async for message in websocket:
            await websocket.send(f"Echo: {message}")

    await websockets.serve(echo, "localhost", 8765)
    print("WebSocket 服务器启动")
```

### 断线重连

```python
import asyncio
import websockets

async def resilient_client(uri: str, max_retries: int = 5):
    """带断线重连的 WebSocket 客户端"""
    for attempt in range(max_retries):
        try:
            async with websockets.connect(uri) as ws:
                print(f"已连接 (尝试 {attempt + 1})")
                while True:
                    message = await ws.recv()
                    print(f"收到: {message}")
        except websockets.exceptions.ConnectionClosed:
            delay = min(2 ** attempt, 60)  # 指数退避，最大 60 秒
            print(f"连接断开，{delay}秒后重连...")
            await asyncio.sleep(delay)
        except Exception as e:
            print(f"错误: {e}")
            break
    print("达到最大重试次数")
```

## httpx + WebSocket 组合

```python
import asyncio
import json
import httpx
import websockets

async def api_and_websocket_demo():
    """同时使用 REST API 和 WebSocket"""
    async with httpx.AsyncClient() as http_client:
        # REST API 获取初始数据
        response = await http_client.get("https://api.example.com/state")
        initial_state = response.json()
        print(f"初始状态: {initial_state}")

    # WebSocket 接收更新
    uri = "wss://api.example.com/stream"
    async with websockets.connect(uri) as ws:
        async for message in ws:
            data = json.loads(message)
            print(f"更新: {data}")

## 常见问题

| 问题 | 解决方案 |
|------|----------|
| SSL 证书错误 | `verify=False`（测试环境）或指定证书路径 |
| 代理设置 | `proxies={"http": "http://proxy:8080"}` |
| 重定向过多 | `allow_redirects=False` |
| 中文乱码 | 使用 `response.encoding = "utf-8"` |
