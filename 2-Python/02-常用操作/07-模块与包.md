# 模块与包

## import 基础

```python
# 导入整个模块
import math
print(math.pi)

# 导入特定内容
from math import pi, sqrt
print(pi)
print(sqrt(16))

# 别名
import numpy as np
from pandas import DataFrame as DF

# 相对导入（包内使用）
from . import sibling_module          # 相对当前包
from ..parent import ParentModule     # 相对上级包
from ...grandparent import GrandParent # 相对上上级包
```

## 模块结构

```python
# my_module.py
"""模块文档字符串"""

# 模块级变量
MODULE_VERSION = "1.0.0"

# 模块级常量
MAX_RETRIES = 3

# 模块初始化代码
print("Module loaded!")  # 慎用，仅用于调试

# 函数定义
def greet(name: str) -> str:
    return f"Hello, {name}!"

# 类定义
class MyClass:
    pass

# 模块级执行（作为脚本时）
if __name__ == "__main__":
    # 仅在直接运行此模块时执行
    print(greet("World"))
```

## 包结构

```
mypackage/
├── __init__.py          # 包初始化
├── __main__.py          # python -m mypackage 入口
├── submodule_a.py       # 子模块
├── submodule_b.py       # 子模块
└── subpackage/
    ├── __init__.py
    └── module.py
```

## __init__.py

```python
# __init__.py 的作用

# 1. 标记目录为包（Python 3.3+ 可省略，但仍推荐）
# 无内容也保留此文件

# 2. 控制导出内容
# from .module import * 的默认行为
__all__ = ["ClassA", "function_b"]  # 限制 * 导出的内容

# 3. 包级别初始化
import logging
logging.basicConfig(level=logging.INFO)

# 4. 简化导入（子模块提升到包级别）
from .submodule import (
    useful_class as Class,
    useful_func as func,
)

# 使用时可以直接：
# from mypackage import Class, func
```

## __main__.py

```python
# __main__.py - 包的可执行入口

# python -m mypackage 时执行此文件
def main():
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("name")
    args = parser.parse_args()
    print(f"Hello, {args.name}!")


if __name__ == "__main__":
    main()
```

## importlib 动态导入

```python
import importlib

# 动态导入
module_name = "math"
math = importlib.import_module(module_name)
print(math.pi)

# 动态导入子模块
collections = importlib.import_module("collections.abc")

# 重新加载模块（开发时有用）
import importlib
import my_module
importlib.reload(my_module)
```

## 相对导入注意事项

```python
# 相对导入只能在包内使用
# 错误示例：作为主脚本直接运行
# python mypackage/submodule.py
# 会报错：attempted relative import with no known parent package

# 正确做法
# python -m mypackage.submodule

# 或在包外使用
# from mypackage import submodule
```

## 包内导入最佳实践

```python
# mypackage/main.py
from . import utils      # 导入同包模块
from .utils import helper # 直接导入具体函数
from .. import config    # 导入父包

# 绝对导入（更清晰，推荐）
from mypackage.utils import helper

# 相对导入（包内移动时更稳定）
from .utils import helper
```

## __name__ 与 __all__

```python
# __name__ 变量
# 模块直接运行时: __name__ == "__main__"
# 被导入时: __name__ == "模块名"

# __all__ 控制 from module import *
__all__ = [
    "public_function",
    "PublicClass",
    "_private_not_exported",  # 不在 * 中导出，但可直接导入
]

def public_function():
    pass

def _private_function():
    pass

class PublicClass:
    pass
```

## 常见陷阱

```python
# 陷阱1：循环导入
# a.py
# from b import b_func
# b_func()
#
# b.py
# from a import a_func
# a_func()
# 解决：延迟导入、在函数内导入、使用 __init__.py 重构

# 陷阱2：模块级可变默认值
# mymodule.py
# default_items = []  # 所有导入共享同一列表
# 解决：使用 None，在函数内初始化

# 陷阱3：相对导入位置错误
# from . import module  # 在包内可以
# import module          # 在包外可以
# 但不能混用

# 陷阱4：忽略 __pycache__
# Python 3.8+ 默认创建 __pycache__
# 不要删除，它加速模块加载

# 陷阱5：__init__.py 副作用
# 在 __init__.py 中执行过多初始化代码
# 会拖慢所有子模块的导入速度
```
