# 正则表达式

## re 模块基础

```python
import re

# match - 从字符串开头匹配
result = re.match(r'\d+', '123abc')
print(result.group())  # '123'

# search - 查找第一个匹配
result = re.search(r'\d+', 'abc123def')
print(result.group())  # '123'

# findall - 返回所有匹配列表
result = re.findall(r'\d+', 'a1b2c3')
print(result)  # ['1', '2', '3']

# finditer - 返回迭代器（省内存）
for match in re.finditer(r'\d+', 'a1b2c3'):
    print(match.group(), match.start())
```

## 常用元字符

```python
# . 任意字符（除换行）
re.findall(r'a.c', 'abc adc a1c')  # ['abc', 'adc', 'a1c']

# \d 数字
re.findall(r'\d{4}', 'a1234b')  # ['1234']

# \w 单词字符（字母、数字、下划线）
re.findall(r'\w+', 'hello_world!')  # ['hello_world']

# \s 空白字符
re.split(r'\s+', 'a b  c')  # ['a', 'b', 'c']

# ^ $ 开头结尾
re.match(r'^abc', 'abcdef')  # 匹配
re.search(r'abc$', 'defabc')  # 匹配

# [] 字符集
re.findall(r'[aeiou]', 'hello')  # ['e', 'o']
re.findall(r'[a-zA-Z]', 'Hello123')  # ['H', 'e', 'l', 'l', 'o']

# | 或
re.findall(r'cat|dog', 'I have a cat and a dog')  # ['cat', 'dog']
```

## 量词（数量词）

```python
# * 0或多个（贪婪）
re.findall(r'a*', 'aaaa')  # ['aaaa']

# + 1或多个
re.findall(r'a+', 'aab')  # ['aa']

# ? 0或1个
re.findall(r'colou?r', 'color colour')  # ['color', 'colour']

# {n} 精确n个
re.findall(r'\d{4}', '2024-01-01')  # ['2024', '0101']

# {n,m} n到m个（贪婪）
re.findall(r'\d{2,4}', '123456')  # ['1234', '56']

# {n,m}? 非贪婪（最短匹配）
re.findall(r'<.*?>', '<a><b>')  # ['<a>', '<b>']
re.findall(r'<.*>', '<a><b>')   # ['<a><b>']（贪婪）
```

## 分组与捕获

```python
import re

# () 捕获分组
match = re.search(r'(\d{4})-(\d{2})-(\d{2})', '2024-06-15')
print(match.group())      # 2024-06-15
print(match.group(1))     # 2024
print(match.group(2))     # 06
print(match.group(3))     # 15
print(match.groups())    # ('2024', '06', '15')

# 命名分组 (?P<name>...)
match = re.search(r'(?P<year>\d{4})-(?P<month>\d{2})-(?P<day>\d{2})', '2024-06-15')
print(match.groupdict())  # {'year': '2024', 'month': '06', 'day': '15'}

# 非捕获分组 (?:...)
re.findall(r'(?:ab)+', 'ababab')  # ['ababab']（不创建捕获组）

# \1 \2 引用分组
re.search(r'(\w+)\s+\1', 'the the')  # 匹配重复单词
# ('the', ' the') - \1 引用前面的 the
```

## 零宽断言

```python
# (?=...) 正前瞻（后面是什么，但不消耗）
re.findall(r'\w+(?=\.)', 'hello.world.csv')  # ['world', 'csv']

# (?!...) 负前瞻（后面不是什么）
re.findall(r'\d+(?!\.)', '123abc 456')  # ['123', '456']

# (?<=...) 正后顾（前面是什么，不消耗）
re.findall(r'(?<=\$)\d+', '$100 $200')  # ['100', '200']

# (?<!...) 负后顾（前面不是什么）
re.findall(r'(?<!\$)\d+', '$100 200')  # ['100', '200']
```

## 编译与标志

```python
import re

# 编译（重复使用同一模式时更高效）
pattern = re.compile(r'\d+')
pattern.findall('a1b2c3')  # ['1', '2', '3']
pattern.search('a1b2c3')  # Match object

# 常用标志
re.IGNORECASE  # i - 忽略大小写
re.MULTILINE   # m - ^ $ 匹配每行
re.DOTALL      # s - . 匹配换行
re.VERBOSE     # x - 允许注释和空白

# 示例
re.findall(r'hello', 'HELLO Hello hello', re.IGNORECASE)  # ['HELLO', 'Hello', 'hello']

# 编译时指定标志
pattern = re.compile(r'^hello', re.MULTILINE | re.IGNORECASE)
```

## 替换与分割

```python
import re

# sub - 替换
text = '2024-06-15'
result = re.sub(r'-', '/', text)
print(result)  # 2024/06/15

# 使用分组替换
text = 'John: 30, Jane: 25'
result = re.sub(r'(\w+): (\d+)', r'\2 - \1', text)
print(result)  # 30 - John, 25 - Jane

# 回调函数替换
def upperize(match):
    return match.group(1).upper()

re.sub(r'(hello)', upperize, 'say hello world')  # 'say HELLO world'

# subn - 返回替换次数
result, count = re.subn(r'\d+', 'X', 'a1b2c3')
print(count)  # 3

# split - 分割
re.split(r'[,;]', 'a,b;c,d')  # ['a', 'b', 'c', 'd']
```

## 常用模式速查

```python
import re

# 邮箱
email = r'^[\w.-]+@[\w.-]+\.\w+$'
re.match(email, 'user@example.com')  # 匹配

# URL
url = r'https?://[\w.-]+(?:/[\w./-]*)?'
re.findall(url, 'Visit https://example.com/path')

# IP 地址
ip = r'\b(?:\d{1,3}\.){3}\d{1,3}\b'
re.findall(ip, 'Server: 192.168.1.1')

# 手机号（中国）
phone = r'1[3-9]\d{9}'
re.match(phone, '13812345678')

# 身份证号（中国）
id_card = r'\d{17}[\dXx]'

# 信用卡号（简单版）
card = r'\d{4}[- ]?\d{4}[- ]?\d{4}[- ]?\d{4}'

# 密码（8位，含数字字母）
password = r'^(?=.*\d)(?=.*[a-zA-Z])[\w]{8,}$'

# HTML 标签
tag = r'<(\w+)>(.*?)</\1>'
```

## Match 对象

```python
import re

match = re.search(r'(\w+)\s+(\d+)', 'hello 123')

# group / groups
match.group()    # 'hello 123'（整体）
match.group(1)  # 'hello'
match.group(2)  # '123'
match.groups()  # ('hello', '123')

# span / start / end
match.span()    # (0, 9)
match.start()   # 0
match.end()     # 9

# 布尔判断
if match:
    print("Matched!")

# 遍历所有分组
for i, group in enumerate(match.groups(), 1):
    print(f"Group {i}: {group}")
```

## 常用场景

```python
import re

# 1. 提取数据
text = 'Name: John, Age: 30, City: Beijing'
pattern = r'Name: (\w+), Age: (\d+), City: (\w+)'
m = re.search(pattern, text)
if m:
    name, age, city = m.groups()

# 2. 验证输入
def validate_date(text: str) -> bool:
    return bool(re.fullmatch(r'\d{4}-\d{2}-\d{2}', text))

# 3. 清理文本
def clean_html(text: str) -> str:
    return re.sub(r'<[^>]+>', '', text)

# 4. 分割 camelCase
def split_camel(text: str) -> list[str]:
    return re.findall(r'[A-Z]?[a-z]+|[A-Z]+', text)

split_camel('camelCaseText')  # ['camel', 'Case', 'Text']

# 5. 去重相邻字符
def remove_dupes(text: str) -> str:
    return re.sub(r'(.)\1+', r'\1', text)

remove_dupes('helloo')  # 'helo'
```

## 常见陷阱

```python
import re

# 陷阱1：贪婪匹配导致意外
re.findall(r'<.*>', '<a><b>')  # ['<a><b>']（贪婪，匹配整个字符串）
re.findall(r'<.*?>', '<a><b>')  # ['<a>', '<b>']（非贪婪）

# 陷阱2：字符集内未转义
re.findall(r'[.*]', 'a.b*c')  # ['.', '*']（正确：字符集内大部分元字符不需要转义）
re.findall(r'[\d]', 'a1b2')   # ['1', '2']

# 陷阱3：编译 vs 非编译（重复使用同一模式）
# 错误：每次调用都重新编译
for _ in range(1000):
    re.search(pattern, text)

# 正确：预编译
compiled = re.compile(pattern)
for _ in range(1000):
    compiled.search(text)

# 陷阱4：.\n 不匹配换行
re.findall(r'a.*b', 'a\nb')    # None
re.findall(r'a.*b', 'a\nb', re.DOTALL)  # ['a\nb']

# 陷阱5：re.match 只匹配开头
re.match(r'\d+', 'abc123')  # None（从开头匹配数字，失败）
re.search(r'\d+', 'abc123')  # Match（从任意位置）
```
