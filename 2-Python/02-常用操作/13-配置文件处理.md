# 配置文件处理

Python 项目中常用的配置文件格式包括 JSON、TOML、YAML、INI 等。本专题聚焦于 TOML（Python 3.11+ 原生支持）和广泛使用的 YAML。

## TOML 配置文件

### TOML 基础

TOML（Tom's Obvious, Minimal Language）是一种轻量级配置文件格式，Python 3.11 起内置支持。

```toml
# config.toml
[database]
host = "localhost"
port = 5432
name = "myapp"

[database.credentials]
username = "admin"
password = "secret"

[server]
host = "0.0.0.0"
port = 8000
workers = 4

[logging]
level = "INFO"
format = "%(asctime)s - %(name)s - %(levelname)s - %(message)s"

[[plugins]]
name = "auth"
enabled = true

[[plugins]]
name = "cache"
enabled = false
```

### Python 读取 TOML

```python
import tomllib  # Python 3.11 起内置

# 从文件读取
with open("config.toml", "rb") as f:
    config = tomllib.load(f)

print(config["database"]["host"])          # "localhost"
print(config["database"]["port"])          # 5432
print(config["plugins"][0]["name"])        # "auth"

# 使用 get 方法安全访问嵌套配置
db_host = config.get("database", {}).get("host", "127.0.0.1")
print(db_host)  # "localhost"
```

### 写入 TOML 文件

```python
import tomllib
try:
    import tomli_w
except ImportError:
    # 需要安装：pip install tomli-w
    tomli_w = None


def write_config(config: dict, filepath: str) -> None:
    """写入 TOML 配置文件"""
    if tomli_w is None:
        raise ImportError("需要安装 tomli-w: pip install tomli-w")

    with open(filepath, "wb") as f:
        tomli_w.dump(config, f)


config = {
    "database": {
        "host": "db.example.com",
        "port": 5432,
        "name": "production",
        "credentials": {
            "username": "admin",
        }
    },
    "server": {
        "workers": 8
    }
}

write_config(config, "config.toml")
```

### 完整配置管理类

```python
import tomllib
from pathlib import Path
from typing import Any


class Config:
    """TOML 配置管理类"""

    def __init__(self, filepath: str | Path):
        self.filepath = Path(filepath)
        self._data: dict = {}
        self.load()

    def load(self) -> None:
        """从文件加载配置"""
        if not self.filepath.exists():
            raise FileNotFoundError(f"配置文件不存在: {self.filepath}")
        with open(self.filepath, "rb") as f:
            self._data = tomllib.load(f)

    def get(self, key: str, default: Any = None) -> Any:
        """获取配置值，支持点号分隔的键"""
        keys = key.split(".")
        value = self._data
        for k in keys:
            if isinstance(value, dict) and k in value:
                value = value[k]
            else:
                return default
        return value

    def section(self, name: str) -> dict:
        """获取配置节"""
        return self._data.get(name, {})

    def __getitem__(self, key: str) -> Any:
        """支持 config["database.host"] 语法"""
        value = self.get(key)
        if value is None:
            raise KeyError(f"配置键不存在: {key}")
        return value

    def __contains__(self, key: str) -> bool:
        return self.get(key) is not None


# 使用
config = Config("config.toml")
print(config["database.host"])           # "localhost"
print(config.get("server.workers", 4))  # 4（默认值）
```

## YAML 配置文件

### YAML 基础

YAML（YAML Ain't Markup Language）是一种人类可读的数据序列化格式，适合配置文件。

```yaml
# config.yaml
database:
  host: localhost
  port: 5432
  name: myapp
  credentials:
    username: admin
    password: secret

server:
  host: 0.0.0.0
  port: 8000
  workers: 4

logging:
  level: INFO
  handlers:
    - console
    - file
  file:
    path: /var/log/app.log
    max_bytes: 10485760
    backup_count: 5

plugins:
  - name: auth
    enabled: true
  - name: cache
    enabled: false
```

### PyYAML 使用

```python
import yaml

# 加载 YAML
with open("config.yaml", "r") as f:
    config = yaml.safe_load(f)

print(config["database"]["host"])        # "localhost"
print(config["plugins"][0]["name"])      # "auth"

# 安全加载（推荐，防止代码注入）
# yaml.safe_load() 只加载基本 Python 类型
# yaml.unsafe_load() 或 yaml.load() 可以加载任意 Python 对象
```

### 自定义类型加载

```python
import yaml
from datetime import datetime


# 自定义类型处理
class ConfigLoader(yaml.SafeLoader):
    pass


def datetime_constructor(loader, node):
    """解析日期时间字符串"""
    value = loader.construct_scalar(node)
    return datetime.fromisoformat(value)


ConfigLoader.add_constructor("!datetime", datetime_constructor)


yaml_content = """
created_at: !datetime "2024-01-15T10:30:00"
name: test
"""

# 使用
data = yaml.load(yaml_content, Loader=ConfigLoader)
print(data["created_at"])  # 2024-01-15 10:30:00
```

## 环境变量与配置

### 配置分层

```python
import os
from pathlib import Path
import yaml


def load_config(env: str = None) -> dict:
    """分层配置加载：默认配置 -> 环境配置 -> 环境变量"""
    # 基础配置
    config_path = Path("config.yaml")
    if config_path.exists():
        with open(config_path) as f:
            config = yaml.safe_load(f) or {}
    else:
        config = {}

    # 环境特定配置
    if env:
        env_config_path = Path(f"config.{env}.yaml")
        if env_config_path.exists():
            with open(env_config_path) as f:
                env_config = yaml.safe_load(f) or {}
                config = merge_config(config, env_config)

    # 环境变量覆盖
    if "DATABASE_HOST" in os.environ:
        config.setdefault("database", {})["host"] = os.environ["DATABASE_HOST"]

    return config


def merge_config(base: dict, override: dict) -> dict:
    """深度合并配置"""
    result = base.copy()
    for key, value in override.items():
        if key in result and isinstance(result[key], dict) and isinstance(value, dict):
            result[key] = merge_config(result[key], value)
        else:
            result[key] = value
    return result


# 使用
# config = load_config("production")
# config = load_config(os.getenv("ENV", "development"))
```

## INI 文件处理

### configparser

```python
import configparser

# 创建配置
config = configparser.ConfigParser()

config["DEFAULT"] = {
    "debug": "false",
    "log_level": "INFO"
}

config["database"] = {
    "host": "localhost",
    "port": "5432",
    "name": "myapp"
}

config["server"]["host"] = "0.0.0.0"
config["server"]["port"] = "8000"

# 写入文件
with open("config.ini", "w") as f:
    config.write(f)

# 读取文件
config = configparser.ConfigParser()
config.read("config.ini")

print(config["database"]["host"])           # "localhost"
print(config.get("database", "port"))        # "5432"
print(config.get("database", "timeout", fallback=30))  # 30

# 获取整数值
port = config.getint("database", "port")
print(port)  # 5432

# 获取布尔值
debug = config.getboolean("DEFAULT", "debug")
print(debug)  # False
```

## 配置模式验证

### Pydantic 配置验证

```python
from pydantic import BaseModel, Field, field_validator
from typing import Optional
import yaml


class DatabaseConfig(BaseModel):
    """数据库配置模型"""
    host: str = "localhost"
    port: int = Field(default=5432, ge=1, le=65535)
    name: str
    credentials: dict = Field(default_factory=dict)

    @field_validator("host")
    @classmethod
    def validate_host(cls, v):
        if not v:
            raise ValueError("Host cannot be empty")
        return v


class ServerConfig(BaseModel):
    """服务器配置模型"""
    host: str = "0.0.0.0"
    port: int = Field(default=8000, ge=1, le=65535)
    workers: int = Field(default=4, ge=1)


class AppConfig(BaseModel):
    """应用完整配置"""
    database: DatabaseConfig
    server: ServerConfig = Field(default_factory=ServerConfig)
    debug: bool = False


def load_config_with_validation(filepath: str) -> AppConfig:
    """加载并验证配置"""
    with open(filepath) as f:
        data = yaml.safe_load(f)
    return AppConfig(**data)
```

## 敏感信息处理

### 环境变量和 secrets

```python
import os
from pathlib import Path
import yaml


def load_config_with_secrets(config: dict) -> dict:
    """处理敏感配置值"""
    # 支持 ${ENV_VAR} 或 ${ENV_VAR:default} 语法
    def resolve_value(value):
        if isinstance(value, str):
            if value.startswith("${") and value.endswith("}"):
                env_expr = value[2:-1]
                if ":" in env_expr:
                    env_name, default = env_expr.split(":", 1)
                    return os.getenv(env_name, default)
                else:
                    return os.getenv(env_expr, "")
            return value
        elif isinstance(value, dict):
            return {k: resolve_value(v) for k, v in value.items()}
        elif isinstance(value, list):
            return [resolve_value(item) for item in value]
        return value

    return resolve_value(config)


config_yaml = """
database:
  host: ${DB_HOST:localhost}
  port: ${DB_PORT:5432}
  password: ${DB_PASSWORD}
"""

config = yaml.safe_load(config_yaml)
config = load_config_with_secrets(config)
print(config["database"]["host"])  # 从环境变量或 localhost
print(config["database"]["password"])  # 从环境变量
```

## JSON vs TOML vs YAML 对比

| 特性 | JSON | TOML | YAML |
|------|------|------|------|
| Python 内置 | 3.0+ | 3.11+ | 需要 pyyaml |
| 人类可读性 | 一般 | 优秀 | 优秀 |
| 注释支持 | 否 | 是 | 是 |
| 嵌套结构 | 是 | 是 | 是 |
| 多行字符串 | 麻烦 | 是 | 是 |
| 常见用途 | API, 配置文件 | 现代项目配置 | 项目配置, Docker |

## 完整示例

### 项目配置管理系统

```python
import os
import yaml
from pathlib import Path
from typing import Any
from dataclasses import dataclass, field


@dataclass
class DatabaseConfig:
    """数据库配置"""
    host: str = "localhost"
    port: int = 5432
    name: str = "myapp"
    username: str = "admin"
    password: str = ""
    pool_size: int = 10

    def connection_string(self) -> str:
        return f"postgresql://{self.username}:***@{self.host}:{self.port}/{self.name}"


@dataclass
class ServerConfig:
    """服务器配置"""
    host: str = "0.0.0.0"
    port: int = 8000
    workers: int = 4
    debug: bool = False


@dataclass
class LoggingConfig:
    """日志配置"""
    level: str = "INFO"
    file_path: str = "/var/log/app.log"
    max_bytes: int = 10 * 1024 * 1024
    backup_count: int = 5


@dataclass
class AppConfig:
    """应用完整配置"""
    database: DatabaseConfig = field(default_factory=DatabaseConfig)
    server: ServerConfig = field(default_factory=ServerConfig)
    logging: LoggingConfig = field(default_factory=LoggingConfig)

    @classmethod
    def from_yaml(cls, filepath: str | Path) -> "AppConfig":
        """从 YAML 文件加载"""
        filepath = Path(filepath)
        if not filepath.exists():
            raise FileNotFoundError(f"配置文件不存在: {filepath}")

        with open(filepath) as f:
            data = yaml.safe_load(f)

        return cls(
            database=DatabaseConfig(**data.get("database", {})),
            server=ServerConfig(**data.get("server", {})),
            logging=LoggingConfig(**data.get("logging", {})),
        )

    def to_yaml(self, filepath: str | Path) -> None:
        """保存到 YAML 文件"""
        filepath = Path(filepath)
        data = {
            "database": {
                "host": self.database.host,
                "port": self.database.port,
                "name": self.database.name,
                "username": self.database.username,
                "password": self.database.password,
                "pool_size": self.database.pool_size,
            },
            "server": {
                "host": self.server.host,
                "port": self.server.port,
                "workers": self.server.workers,
                "debug": self.server.debug,
            },
            "logging": {
                "level": self.logging.level,
                "file_path": self.logging.file_path,
                "max_bytes": self.logging.max_bytes,
                "backup_count": self.logging.backup_count,
            }
        }
        with open(filepath, "w") as f:
            yaml.dump(data, f, default_flow_style=False)


# 使用
config = AppConfig.from_yaml("config.yaml")
print(config.database.connection_string())
config.server.debug = True
config.to_yaml("config_modified.yaml")
```
