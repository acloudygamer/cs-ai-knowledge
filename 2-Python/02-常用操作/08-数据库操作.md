# 数据库操作

## sqlite3（内置轻量级数据库）

### 基础操作

```python
import sqlite3
from contextlib import contextmanager

# 连接数据库（文件不存在则自动创建）
conn = sqlite3.connect("app.db")
cursor = conn.cursor()

# 创建表
cursor.execute("""
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        email TEXT UNIQUE,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
""")

# 插入数据（参数化查询防 SQL 注入）
cursor.execute(
    "INSERT INTO users (name, email) VALUES (?, ?)",
    ("Alice", "alice@example.com")
)
conn.commit()

# 查询数据
cursor.execute("SELECT * FROM users WHERE id = ?", (1,))
row = cursor.fetchone()  # 返回元组
print(row)  # (1, 'Alice', 'alice@example.com', '2024-01-01 12:00:00')

# 查询转为字典
cursor.execute("SELECT * FROM users")
columns = [col[0] for col in cursor.description]
rows = [dict(zip(columns, row)) for row in cursor.fetchall()]
```

### 上下文管理器封装

```python
@contextmanager
def get_db_connection(db_path: str = "app.db"):
    """安全的数据库连接上下文管理器"""
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row  # 支持字典访问
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()

# 使用
with get_db_connection() as conn:
    cursor = conn.cursor()
    cursor.execute("SELECT * FROM users")
    for row in cursor.fetchall():
        print(dict(row))  # 字典形式访问
```

### 事务与回滚

```python
def transfer_funds(from_id: int, to_id: int, amount: float):
    """转账操作演示事务"""
    with get_db_connection() as conn:
        cursor = conn.cursor()
        try:
            # 扣除转出账户余额
            cursor.execute(
                "UPDATE accounts SET balance = balance - ? WHERE id = ?",
                (amount, from_id)
            )
            # 增加转入账户余额
            cursor.execute(
                "UPDATE accounts SET balance = balance + ? WHERE id = ?",
                (amount, to_id)
            )
            # conn.commit() 由上下文管理器自动执行
        except Exception as e:
            # 异常时自动 rollback
            raise ValueError(f"转账失败: {e}")
```

### 批量操作

```python
# 批量插入
users = [
    ("Bob", "bob@example.com"),
    ("Charlie", "charlie@example.com"),
    ("Diana", "diana@example.com"),
]
cursor.executemany(
    "INSERT INTO users (name, email) VALUES (?, ?)",
    users
)
conn.commit()

# 批量更新
cursor.executemany(
    "UPDATE users SET created_at = datetime('now') WHERE id = ?",
    [(1,), (2,), (3,)]
)
conn.commit()
```

### 引擎与会话

```python
from sqlalchemy import create_engine, Column, Integer, String, DateTime
from sqlalchemy.orm import sessionmaker, declarative_base
from datetime import datetime, timezone

# SQLite 数据库
engine = create_engine("sqlite:///app.db", echo=False)

# 创建基类
Base = declarative_base()

# 定义模型
class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, autoincrement=True)
    name = Column(String(100), nullable=False)
    email = Column(String(100), unique=True)
    created_at = Column(DateTime, default=lambda: datetime.now(timezone.utc))

    def __repr__(self):
        return f"<User(id={self.id}, name='{self.name}')>"

# 创建表
Base.metadata.create_all(engine)

# 创建会话
Session = sessionmaker(bind=engine)
session = Session()
```

### CRUD 操作

```python
# 创建
user = User(name="Alice", email="alice@example.com")
session.add(user)
session.commit()
print(user.id)  # 自动生成主键

# 读取
user = session.query(User).filter_by(id=1).first()
users = session.query(User).filter(User.email.like("%@example.com")).all()

# 更新
user.name = "Alice Smith"
session.commit()

# 删除
session.delete(user)
session.commit()
```

### 查询构建器

```python
from sqlalchemy import and_, or_, desc, func

# 条件查询
results = session.query(User).filter(
    and_(
        User.id > 1,
        or_(User.name.like("A%"), User.email.contains("example"))
    )
).order_by(desc(User.created_at)).limit(10).all()

# 聚合查询
count = session.query(func.count(User.id)).scalar()
avg_id = session.query(func.avg(User.id)).scalar()

# 分组查询
stats = session.query(
    User.name,
    func.count(User.id)
).group_by(User.name).all()
```

### 关系与外键

```python
from sqlalchemy import ForeignKey
from sqlalchemy.orm import relationship

class Post(Base):
    __tablename__ = "posts"

    id = Column(Integer, primary_key=True)
    title = Column(String(200), nullable=False)
    user_id = Column(Integer, ForeignKey("users.id"))
    user = relationship("User", back_populates="posts")

class User(Base):
    # ...
    posts = relationship("Post", back_populates="user", cascade="all, delete-orphan")

# 使用
user = session.query(User).first()
for post in user.posts:
    print(post.title)
```

### 连接池与性能

```python
# 连接池配置
engine = create_engine(
    "sqlite:///app.db",
    pool_size=5,
    max_overflow=10,
    pool_pre_ping=True,  # 连接前检测
    echo=False
)

# 批量插入优化
session.bulk_insert_mappings(User, [
    {"name": f"User{i}", "email": f"user{i}@example.com"}
    for i in range(1000)
])
session.commit()
```

## 常见问题

| 问题 | 解决方案 |
|------|----------|
| 并发写入冲突 | 使用 `WAL` 模式：`PRAGMA journal_mode=WAL` |
| 大数据量查询 | 使用 `yield_per()` 流式处理 |
| SQL 注入 | 始终使用参数化查询 |
| 连接未关闭 | 使用上下文管理器确保释放 |
| 事务失败 | 明确 `rollback()` 并重新尝试 |
