# 文件操作

## open / with

```python
# 基础 open
f = open("file.txt", "r", encoding="utf-8")
content = f.read()
f.close()

# with 自动关闭（推荐）
with open("file.txt", "r", encoding="utf-8") as f:
    content = f.read()

# 逐行读取
with open("file.txt", "r") as f:
    for line in f:
        print(line.strip())

# 写入
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("Hello\n")
    f.write("World\n")

# 追加
with open("log.txt", "a") as f:
    f.write("New entry\n")

# 模式: r/rb/w/wb/a/ab/+/r+/w+
```

## pathlib

```python
from pathlib import Path

p = Path(".") / "data" / "file.txt"  # 路径拼接

# 读取文件
content = p.read_text(encoding="utf-8")
bytes_data = p.read_bytes()

# 写入文件
p.write_text("Hello", encoding="utf-8")
p.write_bytes(b"binary data")

# 目录操作
Path("output").mkdir(exist_ok=True)          # 创建目录
Path("output").mkdir(parents=True, exist_ok=True)  # 递归创建
list(Path(".").glob("*.txt"))               # 匹配文件
list(Path(".").rglob("*.py"))               # 递归匹配

# 路径属性
p.exists()                                  # 是否存在
p.is_file()                                 # 是否文件
p.is_dir()                                  # 是否目录
p.is_symlink()                              # 是否符号链接
p.parent                                    # 父目录
p.stem                                      # 文件名（无扩展名）
p.suffix                                    # 扩展名
p.name                                      # 完整文件名

# 解析路径
p.absolute()                                # 绝对路径
p.resolve()                                 # 解析符号链接
p.relative_to(Path("/home/user"))           # 相对路径
```

## CSV 操作

```python
import csv
from pathlib import Path

# 读取 CSV
with open("data.csv", "r", newline="", encoding="utf-8") as f:
    reader = csv.DictReader(f)  # 字典形式读取
    for row in reader:
        print(row["name"], row["email"])

# 写入 CSV
data = [
    {"name": "Alice", "email": "alice@example.com", "age": 30},
    {"name": "Bob", "email": "bob@example.com", "age": 25},
]
with open("output.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(f, fieldnames=["name", "email", "age"])
    writer.writeheader()
    writer.writerows(data)

# 使用 pathlib
Path("output.csv").write_text("name,email\nAlice,alice@example.com\n")
```

### CSV 高级选项

```python
# 自定义分隔符
reader = csv.DictReader(f, delimiter=";")
writer = csv.DictWriter(f, delimiter="\t")

# 处理引号
writer = csv.DictWriter(f, quoting=csv.QUOTE_ALL)
writer = csv.DictWriter(f, quoting=csv.QUOTE_MINIMAL)

# 大文件流式处理
def process_large_csv(filepath: Path):
    with open(filepath, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for i, row in enumerate(reader):
            # 处理每一行
            yield row
            if i % 10000 == 0:
                print(f"已处理 {i} 行")
```

## pickle（对象序列化）

```python
import pickle
from pathlib import Path

# 序列化对象
data = {"name": "Alice", "scores": [90, 85, 88]}
with open("data.pkl", "wb") as f:
    pickle.dump(data, f)

# 反序列化
with open("data.pkl", "rb") as f:
    loaded = pickle.load(f)

# 使用 pathlib
bytes_data = pickle.dumps(data)
loaded = pickle.loads(bytes_data)
```

### 安全注意

```python
# pickle 安全问题：永远不反序列化不可信来源
# 不要这样做！
# with open("untrusted.pkl", "rb") as f:
#     obj = pickle.load(f)  # 可能执行恶意代码

# 安全替代：JSON 或其他格式
import json
json_data = json.dumps(data)
obj = json.loads(json_data)
```

## 临时文件

```python
import tempfile
from pathlib import Path

# 临时文件
with tempfile.NamedTemporaryFile(mode="w", suffix=".txt", delete=True) as f:
    f.write("Temporary content")
    f.flush()
    temp_path = f.name
    print(f"临时文件: {temp_path}")

# 临时目录
with tempfile.TemporaryDirectory() as tmpdir:
    temp_file = Path(tmpdir) / "output.txt"
    temp_file.write_text("Data")
    print(f"临时目录: {tmpdir}")

# 自定义位置
temp_dir = tempfile.mkdtemp(prefix="myapp_")
```

## 文件锁定

```python
import fcntl  # Unix
import msvcrt  # Windows

# Unix 文件锁定
def lock_fileunix(path: Path):
    with open(path, "r+") as f:
        fcntl.flock(f.fileno(), fcntl.LOCK_EX)
        try:
            # 临界区操作
            yield f
        finally:
            fcntl.flock(f.fileno(), fcntl.LOCK_UN)

# Windows 文件锁定
def lock_file_windows(path: Path):
    with open(path, "r+") as f:
        msvcrt.locking(f.fileno(), msvcrt.LK_LOCK, 0)
        try:
            yield f
        finally:
            msvcrt.locking(f.fileno(), msvcrt.LK_UNLCK, 0)

# 跨平台方案
try:
    import fcntl
    lock_file = lock_fileunix
except ImportError:
    lock_file = lock_file_windows
```

## 文件批量操作

```python
from pathlib import Path
import shutil

# 递归复制目录
shutil.copytree("src_dir", "dst_dir")

# 复制文件
shutil.copy2("file.txt", "backup.txt")  # 保留元数据

# 移动/重命名
shutil.move("old.txt", "new.txt")

# 递归删除
shutil.rmtree("temp_dir")

# 查找文件
def find_files(directory: Path, pattern: str = "*.py"):
    return list(directory.rglob(pattern))

# 批量重命名
for f in Path(".").glob("*.txt"):
    new_name = f.stem + "_backup" + f.suffix
    f.rename(f.parent / new_name)

# 文件大小统计
total_size = sum(f.stat().st_size for f in Path(".").rglob("*.py"))
print(f"总大小: {total_size / 1024:.2f} KB")
```

## 常见问题

| 问题 | 解决方案 |
|------|----------|
| 中文路径乱码 | 使用 `encoding="utf-8"` |
| 大文件读取慢 | 使用 `mmap` 内存映射 |
| 文件被占用 | 使用 `FileNotFoundError` 或 `PermissionError` 处理 |
| 跨平台路径 | 使用 `pathlib.Path` 自动处理 |
| CSV 中文乱码 | 指定 `encoding="utf-8-sig"`（带 BOM）|
```
