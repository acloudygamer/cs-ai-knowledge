# 参数化测试

使用不同输入多次运行同一测试。

## @pytest.mark.parametrize

```python
import pytest

@pytest.mark.parametrize("input,expected", [
    (1, 1),
    (2, 4),
    (3, 9),
    (4, 16),
    (5, 25),
])
def test_square(input, expected):
    assert input ** 2 == expected


@pytest.mark.parametrize("a,b,expected", [
    (1, 2, 3),
    (0, 0, 0),
    (-1, 1, 0),
    (100, 200, 300),
])
def test_addition(a, b, expected):
    assert a + b == expected


@pytest.mark.parametrize("email", [
    "user@example.com",
    "test@test.org",
    "admin@company.co.uk",
])
def test_valid_email_format(email):
    assert "@" in email
    assert email.count("@") == 1
```

## 多参数组合

```python
# 方式 1：笛卡尔积
@pytest.mark.parametrize("x", [1, 2, 3])
@pytest.mark.parametrize("y", [10, 20])
def test_combinations(x, y):
    # 运行 3 * 2 = 6 次
    assert x + y > 0

# 方式 2：元组参数
@pytest.mark.parametrize("x,y,expected", [
    (1, 10, 11),
    (2, 20, 22),
    (3, 30, 33),
])
def test_add(x, y, expected):
    assert x + y == expected
```

## skipif 参数化

```python
import sys

@pytest.mark.parametrize(
    ("version", "expected"),
    [
        ("3.8", {"typed_dict": True, "dataclass": False}),
        ("3.9", {"walrus": True, "zoneinfo": False}),
        ("3.10", {"match_case": True, "except_star": False}),
        ("3.11", {"exception_group": True, "task_started": False}),
    ],
)
def test_version_features(version, expected):
    major, minor = map(int, version.split("."))
    current = (sys.version_info.major, sys.version_info.minor)
    assert current >= (major, minor)


@pytest.mark.skipif(
    sys.version_info < (3, 10),
    reason="Requires Python 3.10+"
)
@pytest.mark.parametrize("pattern,text,matches", [
    ("match", "match word", True),
    ("match", "no match", False),
])
def test_pattern_matching(pattern, text, matches):
    import re
    result = re.search(pattern, text) is not None
    assert result == matches
```

## indirect 参数化

```python
@pytest.fixture
def http_client(base_url):
    return Client(base_url)

@pytest.mark.parametrize("base_url", [

def test_api_endpoints(http_client):

```

## 生成参数

```python
import pytest

def generate_test_cases():
    """生成大量测试用例"""
    return [
        {"input": i, "expected": i * i}
        for i in range(100)
    ]

@pytest.mark.parametrize("case", generate_test_cases())
def test_square_table(case):
    assert case["input"] ** 2 == case["expected"]
```

## ids 参数

```python
@pytest.mark.parametrize("value,expected", [
    (1, 2),
    (2, 4),
    (3, 6),
], ids=["one_twos", "two_fours", "three_sixes"])
def test_doubling(value, expected):
    assert value * 2 == expected

# 自定义 ids 函数
def pytest_param_id(val):
    if isinstance(val, dict):
        return val.get("name", str(val))

@pytest.mark.parametrize("val", [
    {"name": "test1", "value": 1},
    {"name": "test2", "value": 2},
], ids=pytest_param_id)
def test_with_named_params(val):
    assert val["value"] > 0
```

## Fixture 作为参数源

```python
@pytest.fixture(params=[1, 2, 3])
def number(request):
    return request.param

def test_number_processing(number):
    result = process(number)
    assert result > 0


# 组合 fixture 和 parametrize
@pytest.fixture(params=["json", "xml", "csv"])
def data_format(request):
    return request.param

@pytest.mark.parametrize("compression", [None, "gzip", "bz2"])
def test_data_export(data_format, compression):
    result = export(data_format, compression)
    assert result is not None
```

## 负向测试参数化

```python
@pytest.mark.parametrize("invalid_input,expected_error", [
    ("", ValueError),
    ("abc", ValueError),
    ("-1", ValueError),
    ("0", ValueError),
    ("1001", ValueError),
])
def test_invalid_age(invalid_input, expected_error):
    with pytest.raises(expected_error):
        parse_age(invalid_input)


@pytest.mark.parametrize("email", [
    "",
    "@",
    "noat",
    "@nodomain",
    "spaces in@email.com",
], ids=["empty", "just_at", "no_domain", "no_tld", "with_spaces"])
def test_invalid_emails(email):
    with pytest.raises(ValueError):
        validate_email(email)
```
