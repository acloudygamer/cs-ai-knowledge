# 迭代器与生成器

## 迭代器协议

```python
# Python 迭代器必须实现 __iter__ 和 __next__
class Counter:
    def __init__(self, max_value):
        self.max = max_value
        self.current = 0

    def __iter__(self):
        return self

    def __next__(self):
        if self.current >= self.max:
            raise StopIteration
        value = self.current
        self.current += 1
        return value

# 使用
counter = Counter(3)
for num in counter:
    print(num, end=" ")
# 0 1 2
```

## 迭代器 vs 可迭代对象

```python
# 可迭代对象：实现了 __iter__，返回迭代器
# 迭代器：实现了 __iter__ 和 __next__

# list 是可迭代对象，不是迭代器
my_list = [1, 2, 3]
iterator = iter(my_list)  # 获取迭代器
print(next(iterator))  # 1
print(next(iterator))  # 2
print(next(iterator))  # 3
# next(iterator)  # StopIteration
```

## 生成器函数

```python
# yield 使函数成为生成器
def countdown(n):
    while n > 0:
        yield n
        n -= 1

# 使用
for num in countdown(5):
    print(num, end=" ")
# 5 4 3 2 1

# 生成器是迭代器
gen = countdown(3)
print(next(gen))  # 3
print(next(gen))  # 2
print(next(gen))  # 1
# next(gen)  # StopIteration
```

## yield from

```python
def flatten(nested):
    for item in nested:
        if isinstance(item, (list, tuple)):
            yield from flatten(item)
        else:
            yield item

nested = [1, [2, [3, 4], 5], 6]
print(list(flatten(nested)))  # [1, 2, 3, 4, 5, 6]


def chain(*iterables):
    for it in iterables:
        yield from it

print(list(chain([1, 2], [3, 4], [5, 6])))  # [1, 2, 3, 4, 5, 6]
```

## 生成器方法

```python
def echo():
    while True:
        received = yield
        print(received)

gen = echo()
next(gen)  # 启动生成器
gen.send("Hello")  # 打印 Hello
gen.send("World")  # 打印 World
gen.close()  # 关闭生成器

# throw - 向生成器抛出异常
def gen_with_error():
    try:
        yield 1
    except ValueError:
        print("Caught ValueError")
        yield 2
    yield 3

g = gen_with_error()
print(next(g))  # 1
print(g.throw(ValueError))  # Caught ValueError\n2
print(next(g))  # 3
```

## itertools 模块

```python
import itertools

# 无限迭代器
count = itertools.count(start=1, step=2)  # 1, 3, 5, 7, ...
cycle = itertools.cycle(['A', 'B'])  # A, B, A, B, ...

compress = itertools.compress('ABCDEF', [1, 0, 1, 0, 1, 1])  # A, C, E, F

takewhile = itertools.takewhile(lambda x: x < 3, [1, 2, 3, 4])  # 1, 2
filterfalse = itertools.filterfalse(lambda x: x % 2, [1, 2, 3, 4])  # 2, 4
```

## 实际应用

```python
# 大文件处理
def read_lines(filename, chunk_size=1024):
    with open(filename, 'r') as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            yield chunk

# 批量处理
def batch_process(items, batch_size=100):
    batch = []
    for item in items:
        batch.append(item)
        if len(batch) == batch_size:
            yield batch
            batch = []
    if batch:
        yield batch

# 管道处理
def pipeline(data):
    return (
```

## 生成器 vs 列表

| 特性 | 生成器 | 列表 |
|------|--------|------|
| 内存 | 按需生成，低内存 | 一次性创建，高内存 |
| 速度 | 迭代时逐个计算 | 一次性计算 |
| 重复使用 | 只能迭代一次 | 可多次使用 |
| 索引 | 不支持 | 支持 |
| 长度 | 未知（无限可） | 已知 |

```python
# 需要多次使用时转为列表
data = [x for x in generator_func()]  # 一次迭代
print(data[0])  # OK
print(data[0])  # OK

# 只需迭代时使用生成器
result = sum(x for x in large_range())  # 低内存
```

## 高级生成器模式

### 生成器.send()

```python
# 向生成器发送值，可实现双向通信
def counter():
    total = 0
    while True:
        increment = yield total  # yield 返回接收到的值
        if increment is not None:
            total += increment
        else:
            total += 1

gen = counter()
print(next(gen))  # 0 (启动生成器)
print(gen.send(5))  # 5 (发送 5，total += 5)
print(gen.send(3))  # 8 (发送 3，total += 3)
print(next(gen))  # 9 (发送 None，total += 1)

# 初始化生成器 with 初始值
def echo(initial=None):
    value = initial
    while True:
        value = yield value
```

### 生成器.throw()

```python
# 向生成器抛出异常
def gen_with_error():
    try:
        yield 1
        yield 2
    except ValueError as e:
        yield f"Caught: {e}"
        yield 3

g = gen_with_error()
print(next(g))  # 1
print(g.throw(ValueError, ValueError("test error")))  # Caught: test error
print(next(g))  # 3
```

### 生成器.close()

```python
# 关闭生成器，会在下次 yield 时抛出 StopIteration
def infinite():
    i = 0
    while True:
        yield i
        i += 1

gen = infinite()
for _ in range(5):
    print(next(gen), end=" ")
gen.close()  # 关闭生成器
# next(gen)  # StopIteration
```

## 上下文管理器与生成器

```python
# 生成器实现上下文管理器（早期 Python 风格）
from contextlib import contextmanager

@contextmanager
def managed_resource(name):
    print(f"Acquiring {name}")
    resource = f"{name}_obj"
    try:
        yield resource
    finally:
        print(f"Releasing {name}")

with managed_resource("DB") as res:
    print(f"Using {res}")

# 输出:
# Acquiring DB
# Using DB_obj
# Releasing DB

# 嵌套上下文管理器
@contextmanager
def transaction(connection):
    connection.begin()
    try:
        yield connection
        connection.commit()
    except Exception:
        connection.rollback()
        raise
```

## itertools 进阶用法

```python
import itertools

# 迭代器组合
def take(n, iterable):
    """取前 n 个元素"""
    return itertools.islice(iterable, n)

def drop(n, iterable):
    """跳过前 n 个元素"""
    return itertools.islice(iterable, n, None)

def batched(iterable, n):
    """将迭代器分成 n 个元素的组"""
    it = iter(iterable)
    while batch := tuple(itertools.islice(it, n)):
        yield batch

# 排列组合生成器
def powerset(iterable):
    """生成所有子集"""
    s = list(iterable)
    return itertools.chain.from_iterable(
        itertools.combinations(s, r) for r in range(len(s) + 1)
    )

print(list(powerset([1, 2, 3])))
# [(), (1,), (2,), (3,), (1,2), (1,3), (2,3), (1,2,3)]

# 滑动窗口
def window(iterable, n=2):
    """生成 n 大小的滑动窗口"""
    it = iter(iterable)
    window = tuple(itertools.islice(it, n))
    if len(window) == n:
        yield window
    for item in it:
        window = window[1:] + (item,)
        yield window

print(list(window([1, 2, 3, 4], 3)))
# [(1, 2, 3), (2, 3, 4)]

# 迭代器分区
def partition(pred, iterable):
    """根据条件分割迭代器"""
    t1, t2 = itertools.tee(
        (pred(item), item) for item in iterable
    )
    return (
        (item for pred, item in t1 if pred),
        (item for pred, item in t2 if not pred)
    )

even, odd = partition(lambda x: x % 2 == 0, range(10))
print(list(even))  # [0, 2, 4, 6, 8]
print(list(odd))   # [1, 3, 5, 7, 9]
```

## 异步生成器

```python
import asyncio

# 异步生成器 - 在 async 函数中使用 yield
async def async_counter(max_value):
    for i in range(max_value):
        await asyncio.sleep(0.1)  # 模拟异步操作
        yield i

async def main():
    async for i in async_counter(5):
        print(i)

asyncio.run(main())

# 异步生成器是异步可迭代对象
async def async_fetch(urls):
    for url in urls:
        await asyncio.sleep(0.1)
        yield f"Fetched: {url}"

async def main():
    async for result in async_fetch(["url1", "url2", "url3"]):
        print(result)

asyncio.run(main())

# 异步迭代器（手动实现）
class AsyncIterator:
    def __init__(self, data):
        self.data = data
        self.index = 0

    def __aiter__(self):
        return self

    async def __anext__(self):
        if self.index >= len(self.data):
            raise StopAsyncIteration
        await asyncio.sleep(0.1)
        value = self.data[self.index]
        self.index += 1
        return value
```

## 生成器最佳实践

```python
# 1. 使用生成器处理大文件
def read_large_file(filepath, chunk_size=8192):
    """内存高效地读取大文件"""
    with open(filepath, 'rb') as f:
        while chunk := f.read(chunk_size):
            yield chunk

# 2. 生成器管道
def process_pipeline(data):
    return (
        item
        for item in data
        if item.is_valid()
        if item.process()
        if item.save()
    )

# 3. 记忆化生成器（生成器版本斐波那契）
def fibonacci_gen():
    """生成斐波那契数列（内存高效）"""
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

# 4. 无限序列生成器
def primes_gen():
    """生成素数序列"""
    yield 2
    n = 3
    while True:
        if all(n % i for i in range(3, int(n ** 0.5) + 1, 2)):
            yield n
        n += 2

# 5. 交叉合并多个迭代器
def interleave(*iterables):
    """交叉合并多个迭代器"""
    queues = [iter(it) for it in iterables]
    while queues:
        for q in queues[:]:
            try:
                yield next(q)
            except StopIteration:
                queues.remove(q)

print(list(interleave([1, 4, 7], [2, 5, 8], [3, 6, 9])))
# [1, 2, 3, 4, 5, 6, 7, 8, 9]
```

## 生成器性能优化

```python
# 1. 使用生成器代替列表推导避免内存峰值
# 差: [x**2 for x in range(10**7)]  # 创建 10M 元素列表
# 好: (x**2 for x in range(10**7))  # 惰性求值

# 2. itertools.islice 优化切片操作
import itertools
data = range(1000)
result = list(itertools.islice(data, 100))  # 比 list(data)[:100] 高效

# 3. 使用 yield from 委托生成器（避免手动迭代）
def flattern(nested):
    """扁平化嵌套结构"""
    for item in nested:
        if isinstance(item, (list, tuple)):
            yield from flattern(item)  # 委托给子生成器
        else:
            yield item

# 4. 生成器链式调用
def map_gen(func, iterable):
    yield from (func(x) for x in iterable)

def filter_gen(predicate, iterable):
    yield from (x for x in iterable if predicate(x))

# 使用
result = filter_gen(lambda x: x > 0,
             map_gen(lambda x: x * 2, range(-5, 6)))
print(list(result))  # [2, 4, 6, 8, 10]
```
