# 迭代器与生成器

## 迭代器协议

```python
# Python 迭代器必须实现 __iter__ 和 __next__
class Counter:
    def __init__(self, max_value):
        self.max = max_value
        self.current = 0

    def __iter__(self):
        return self

    def __next__(self):
        if self.current >= self.max:
            raise StopIteration
        value = self.current
        self.current += 1
        return value

# 使用
counter = Counter(3)
for num in counter:
    print(num, end=" ")
# 0 1 2
```

## 迭代器 vs 可迭代对象

```python
# 可迭代对象：实现了 __iter__，返回迭代器
# 迭代器：实现了 __iter__ 和 __next__

# list 是可迭代对象，不是迭代器
my_list = [1, 2, 3]
iterator = iter(my_list)  # 获取迭代器
print(next(iterator))  # 1
print(next(iterator))  # 2
print(next(iterator))  # 3
# next(iterator)  # StopIteration
```

## 生成器函数

```python
# yield 使函数成为生成器
def countdown(n):
    while n > 0:
        yield n
        n -= 1

# 使用
for num in countdown(5):
    print(num, end=" ")
# 5 4 3 2 1

# 生成器是迭代器
gen = countdown(3)
print(next(gen))  # 3
print(next(gen))  # 2
print(next(gen))  # 1
# next(gen)  # StopIteration
```

## 生成器表达式

```python
# 类似列表推导式，但使用圆括号
gen = (x ** 2 for x in range(5))
print(list(gen))  # [0, 1, 4, 9, 16]

# 使用场景：避免创建大列表
sum(x ** 2 for x in range(1000000))  # 不会创建百万元素列表
```

## yield from

```python
def flatten(nested):
    for item in nested:
        if isinstance(item, (list, tuple)):
            yield from flatten(item)
        else:
            yield item

nested = [1, [2, [3, 4], 5], 6]
print(list(flatten(nested)))  # [1, 2, 3, 4, 5, 6]


def chain(*iterables):
    for it in iterables:
        yield from it

print(list(chain([1, 2], [3, 4], [5, 6])))  # [1, 2, 3, 4, 5, 6]
```

## 生成器方法

```python
def echo():
    while True:
        received = yield
        print(received)

gen = echo()
next(gen)  # 启动生成器
gen.send("Hello")  # 打印 Hello
gen.send("World")  # 打印 World
gen.close()  # 关闭生成器

# throw - 向生成器抛出异常
def gen_with_error():
    try:
        yield 1
    except ValueError:
        print("Caught ValueError")
        yield 2
    yield 3

g = gen_with_error()
print(next(g))  # 1
print(g.throw(ValueError))  # Caught ValueError\n2
print(next(g))  # 3
```

## itertools 模块

```python
import itertools

# 无限迭代器
count = itertools.count(start=1, step=2)  # 1, 3, 5, 7, ...
cycle = itertools.cycle(['A', 'B'])  # A, B, A, B, ...
repeat = itertools.repeat(10, times=3)  # 10, 10, 10

# 有限迭代器
accumulate = itertools.accumulate([1, 2, 3, 4])  # 1, 3, 6, 10
chain = itertools.chain([1, 2], ['a', 'b'])  # 1, 2, a, b
compress = itertools.compress('ABCDEF', [1, 0, 1, 0, 1, 1])  # A, C, E, F
dropwhile = itertools.dropwhile(lambda x: x < 3, [1, 2, 3, 4])  # 3, 4
takewhile = itertools.takewhile(lambda x: x < 3, [1, 2, 3, 4])  # 1, 2
filterfalse = itertools.filterfalse(lambda x: x % 2, [1, 2, 3, 4])  # 2, 4
islice = itertools.islice(range(10), 0, 10, 2)  # 0, 2, 4, 6, 8
zip_longest = itertools.zip_longest('ABCD', 'xy', fillvalue='z')  # A, x / B, y / C, z / D, z

# 组合迭代器
product = itertools.product([1, 2], repeat=2)  # (1,1), (1,2), (2,1), (2,2)
permutations = itertools.permutations('ABC', 2)  # AB, AC, BA, BC, CA, CB
combinations = itertools.combinations('ABC', 2)  # AB, AC, BC
combinations_with_replacement = itertools.combinations_with_replacement('ABC', 2)  # AA, AB, AC, BB, BC, CC
```

## 实际应用

```python
# 大文件处理
def read_lines(filename, chunk_size=1024):
    with open(filename, 'r') as f:
        while True:
            chunk = f.read(chunk_size)
            if not chunk:
                break
            yield chunk

# 批量处理
def batch_process(items, batch_size=100):
    batch = []
    for item in items:
        batch.append(item)
        if len(batch) == batch_size:
            yield batch
            batch = []
    if batch:
        yield batch

# 管道处理
def pipeline(data):
    return (
        item
        for item in data
        if item.is_valid()
        if item.process()
        if item.save()
    )

# 斐波那契生成器
def fibonacci():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

# 质数生成器
def primes():
    n = 2
    while True:
        if all(n % i for i in range(2, int(n ** 0.5) + 1)):
            yield n
        n += 1
```

## 生成器 vs 列表

| 特性 | 生成器 | 列表 |
|------|--------|------|
| 内存 | 按需生成，低内存 | 一次性创建，高内存 |
| 速度 | 迭代时逐个计算 | 一次性计算 |
| 重复使用 | 只能迭代一次 | 可多次使用 |
| 索引 | 不支持 | 支持 |
| 长度 | 未知（无限可） | 已知 |

```python
# 需要多次使用时转为列表
data = [x for x in generator_func()]  # 一次迭代
print(data[0])  # OK
print(data[0])  # OK

# 只需迭代时使用生成器
result = sum(x for x in large_range())  # 低内存
```
