## def / lambda

### 函数定义

```python

def greet(name: str) -> str:
    """问候函数"""

    return a + b

# 无返回值（返回 None）

```

### lambda 表达式

```python

square = lambda x: x ** 2
add = lambda x, y: x + y

# lambda 不支持类型注解语法，需通过变量标注或 Callable

numbers = [1, 2, 3, 4, 5]

filtered = list(filter(lambda x: x % 2 == 0, numbers))
```

## 参数类型

### 位置参数和关键字参数

```python
# 位置参数（必须参数）
def func(a, b, c):
    return a + b + c

func(1, 2, 3)  # 正确
func(1, 2)     # TypeError
func(a=1, b=2, c=3)  # 可以用关键字参数

# 关键字参数
func(a=1, b=2, c=3)
func(1, 2, c=3)  # 混合使用
```

### 默认参数

```python
def func(a, b=10, c=20):
    return a + b + c

func(1)              # a=1, b=10, c=20
func(1, c=30)        # a=1, b=10, c=30
func(1, 5, 10)      # a=1, b=5, c=10
func(1, b=5)        # a=1, b=5, c=20

# 重要：默认参数应该是不可变对象
# 错误示例
def bad_func(items=[]):  # 可变默认参数！
    items.append(1)
    return items

# 正确示例
def good_func(items=None):
    if items is None:
        items = []
    items.append(1)
    return items
```

### *args 和 **kwargs

```python
# *args（可变位置参数）
def sum_all(*args):
    return sum(args)

sum_all(1, 2, 3, 4)  # 10
sum_all()             # 0

# **kwargs（可变关键字参数）
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="Alice", age=25, city="Beijing")
# name: Alice
# age: 25
# city: Beijing

# 组合使用
def func(*args, **kwargs):
    print(f"Args: {args}")
    print(f"Kwargs: {kwargs}")

func(1, 2, 3, name="test")
# Args: (1, 2, 3)
# Kwargs: {'name': 'test'}

# 参数解包
args = (1, 2, 3)
kwargs = {"a": 4, "b": 5}
func(*args, **kwargs)
# Args: (1, 2, 3)
# Kwargs: {'a': 4, 'b': 5}
```

### 仅关键字参数

```python
# 使用 * 分隔强制关键字参数
def func(a, *, b, c):
    """b 和 c 必须是关键字参数"""
    return a + b + c

func(1, b=2, c=3)  # 正确
func(1, 2, 3)      # TypeError

# Python 3.10+ 支持位置关键字
def func(a, /, b, *, c):
    """a 必须位置参数，c 必须关键字参数"""
    return a + b + c

func(1, 2, c=3)  # 正确
func(1, b=2, c=3)  # 正确
```

## 作用域 LEGB

### 作用域规则

```python
# LEGB: Local -> Enclosing -> Global -> Built-in

x = "global"  # global

def outer():
    x = "enclosing"

    def inner():
        x = "local"      # local 作用域
        print(x)         # 打印 local

    inner()
    print(x)             # 打印 enclosing

outer()
print(x)                 # 打印 global
```

### global 和 nonlocal

```python
# global 声明使用全局变量
counter = 0

def increment():
    global counter      # 声明使用全局变量
    counter += 1

# nonlocal 声明使用外层变量
def outer():
    count = 0

    def inner():
        nonlocal count   # 声明使用外层变量
        count += 1

    inner()
    print(count)  # 1
```

# 内置作用域可以通过 builtins 模块访问
import builtins

## 装饰器基础

### 简单装饰器

```python
import functools

def my_decorator(func):
    @functools.wraps(func)  # 保留原函数元信息
    def wrapper(*args, **kwargs):
        print("Before call")
        result = func(*args, **kwargs)
        print("After call")
        return result
    return wrapper

@my_decorator
def say_hello(name):
    """Say hello to someone"""
    return f"Hello, {name}!"

print(say_hello("Alice"))
# Before call
# Hello, Alice!
# After call

print(say_hello.__name__)  # say_hello（保留原函数名）
print(say_hello.__doc__)   # Say hello to someone
```

### 带参数的装饰器

```python
def repeat(times):
    """重复调用指定次数"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(3)
def greet(name):
    print(f"Hello, {name}!")

greet("Alice")
# Hello, Alice!
# Hello, Alice!
# Hello, Alice!
```

### 类装饰器

```python
class CountCalls:
    """统计函数调用次数"""
    def __init__(self, func):
        self.func = func
        self.call_count = 0
        functools.update_wrapper(self, func)

    def __call__(self, *args, **kwargs):
        self.call_count += 1
        print(f"{self.func.__name__} called {self.call_count} times")
        return self.func(*args, **kwargs)

@CountCalls
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)
```

## 生成器

### yield 语句

```python
def count_up_to(n):
    """生成器函数"""
    count = 1
    while count <= n:
        yield count
        count += 1

# 使用生成器
gen = count_up_to(5)
print(next(gen))  # 1
print(next(gen))  # 2
print(list(gen))  # [3, 4, 5]

# 在循环中使用
for num in count_up_to(3):
    print(num)  # 1, 2, 3
```

## 匿名函数进阶

### map 和 filter

```python
numbers = [1, 2, 3, 4, 5]

# map - 对每个元素应用函数
squared = list(map(lambda x: x ** 2, numbers))
print(squared)  # [1, 4, 9, 16, 25]

# filter - 过滤元素
evens = list(filter(lambda x: x % 2 == 0, numbers))
print(evens)  # [2, 4]

# 结合使用
result = sum(map(lambda x: x ** 2, filter(lambda x: x % 2 == 0, numbers)))
print(result)  # 4 + 16 = 20
```

### reduce

```python
from functools import reduce

numbers = [1, 2, 3, 4, 5]

# reduce - 累积计算
total = reduce(lambda x, y: x + y, numbers)
print(total)  # 15

product = reduce(lambda x, y: x * y, numbers, 1)  # 带初始值
print(product)  # 120

# 求最大值
max_val = reduce(lambda a, b: a if a > b else b, numbers)
print(max_val)  # 5
```

### sorted 的 key

```python
students = [
    {"name": "Alice", "grade": 90},
    {"name": "Bob", "grade": 85},
    {"name": "Charlie", "grade": 92}
]

# 按成绩排序
sorted_students = sorted(students, key=lambda x: x["grade"])
print([s["name"] for s in sorted_students])  # ['Bob', 'Alice', 'Charlie']

# 降序排序
sorted_students = sorted(students, key=lambda x: x["grade"], reverse=True)
print([s["name"] for s in sorted_students])  # ['Charlie', 'Alice', 'Bob']
```

## 递归

### 基本递归

```python
def factorial(n: int) -> int:
    """阶乘"""
    if n <= 1:
        return 1
    return n * factorial(n - 1)

# 斐波那契数列
def fibonacci(n: int) -> int:
    """斐波那契数列"""
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)
```

### 尾递归优化

Python 不支持尾递归优化，但可以用改写避免栈溢出：

```python
def factorial_tail(n: int, acc: int = 1) -> int:
    """尾递归形式的阶乘"""
    if n <= 1:
        return acc
    return factorial_tail(n - 1, n * acc)

def fibonacci_iter(n: int) -> int:
    """迭代版本避免栈溢出"""
    if n < 2:
        return n
    a, b = 0, 1
    for _ in range(n - 1):
        a, b = b, a + b
    return b
```

## 偏函数

### functools.partial

```python
from functools import partial

def power(base, exponent):
    return base ** exponent

# 创建偏函数
square = partial(power, exponent=2)
cube = partial(power, exponent=3)

print(square(5))  # 25
print(cube(5))    # 125

# 固定第一个参数
def multiply(a, b):
    return a * b

double = partial(multiply, 2)
print(double(5))  # 10
```

## 函数作为返回值

### 高阶函数

```python
def make_adder(n):
    """返回一个加 n 的函数"""
    def adder(x):
        return x + n
    return adder

add_5 = make_adder(5)
add_10 = make_adder(10)

print(add_5(3))   # 8
print(add_10(3))  # 13
```

## 常见错误

### 可变默认参数

```python
# 错误
def bad_append(item, items=[]):
    items.append(item)
    return items

# 正确
def good_append(item, items=None):
    if items is None:
        items = []
    items.append(item)
    return items
```

### 忘记返回值

def process_data(data):

result = process_data([1, 2, 3])
