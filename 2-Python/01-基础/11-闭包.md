# 闭包

闭包（Closure）是函数和其引用的外部变量组合形成的整体。闭包允许函数访问定义函数外部作用域中的变量，即使外部函数已经执行完毕。

## 基础概念

### 什么是闭包

```python
def outer_function(x):
    """外部函数"""
    def inner_function(y):
        """内部函数，引用了外部变量 x"""
        return x + y
    return inner_function


# 创建闭包
closure = outer_function(10)
print(closure(5))  # 15
print(closure(3))  # 13

# 闭包捕获的变量
print(closure.__code__.co_freevars)  # ('x',) - 自由变量
```

### 闭包的三个条件

1. 必须有一个嵌套函数
2. 嵌套函数引用了外部作用域的变量
3. 外部函数返回嵌套函数

```python
# 示例：满足闭包条件的函数
def make_multiplier(factor):
    def multiplier(number):
        return number * factor  # 引用了外部变量 factor
    return multiplier  # 返回嵌套函数


times_3 = make_multiplier(3)
times_5 = make_multiplier(5)

print(times_3(10))  # 30
print(times_5(10))  # 50
```

## 闭包 vs 普通函数

### 状态保持

```python
# 普通函数：每次调用独立
def add_plain(x, y):
    return x + y


# 闭包：保持状态
def counter(start=0):
    def step():
        nonlocal start
        start += 1
        return start
    return step


cnt = counter(0)
print(cnt())  # 1
print(cnt())  # 2
print(cnt())  # 3

# 多个闭包独立状态
cnt2 = counter(10)
print(cnt2())  # 11
print(cnt())   # 4（cnt 保持自己的状态）
```

## nonlocal 关键字

### 修改闭包捕获的变量

```python
def make_accumulator():
    total = 0

    def accumulator(value):
        nonlocal total  # 允许修改外部变量
        total += value
        return total

    return accumulator


acc = make_accumulator()
print(acc(10))  # 10
print(acc(5))   # 15
print(acc(3))   # 18
```

### 闭包变量遮蔽

```python
def outer():
    x = "outer"

    def inner():
        nonlocal x
        x = "inner"  # 修改外层变量
        print(f"Inner: {x}")

    inner()
    print(f"Outer: {x}")


outer()
# Inner: inner
# Outer: inner
```

## 闭包的实际应用

### 函数工厂

```python
def make_power_function(exponent):
    """函数工厂：创建指定指数的幂函数"""
    def power(base):
        return base ** exponent
    return power


square = make_power_function(2)
cube = make_power_function(3)
fourth = make_power_function(4)

print(square(5))    # 25
print(cube(5))      # 125
print(fourth(5))    # 625
```

### 配置函数

```python
def make_greeting(prefix, suffix):
    """创建问候语函数"""
    def greet(name):
        return f"{prefix}, {name}{suffix}"
    return greet


hello = make_greeting("Hello", "!")
goodbye = make_greeting("Goodbye", "...")
welcome = make_greeting("Welcome", " to our service.")

print(hello("Alice"))      # Hello, Alice!
print(goodbye("Bob"))      # Goodbye, Bob...
print(welcome("Charlie"))  # Welcome, Charlie to our service.
```

### 记忆化 / 缓存

```python
def memoize(func):
    """记忆化装饰器"""
    cache = {}

    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]

    # 保留原函数信息
    wrapper.cache = cache
    return wrapper


@memoize
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)


print(fibonacci(100))  # 354224848179261915075
print(fibonacci.cache)  # 缓存的中间结果
```

### 延迟求值

```python
def lazy_property(func):
    """延迟属性：只在首次访问时计算"""
    attr_name = f"_lazy_{func.__name__}"

    @property
    def wrapper(self):
        if not hasattr(self, attr_name):
            setattr(self, attr_name, func(self))
        return getattr(self, attr_name)

    return wrapper


class DataProcessor:
    def __init__(self, data):
        self.data = data

    @lazy_property
    def expensive_computation(self):
        """耗时的计算，只有首次访问时执行"""
        print("Computing...")
        return sum(self.data) / len(self.data)


processor = DataProcessor([1, 2, 3, 4, 5])
print(processor.expensive_computation)  # Computing... 3.0
print(processor.expensive_computation)  # 直接返回 3.0（不再打印 Computing）
```

## 闭包陷阱

### 循环闭包

```python
# 错误：所有函数引用同一个变量
def create_functions_wrong():
    funcs = []
    for i in range(3):
        funcs.append(lambda: i)  # i 是可变变量
    return funcs


funcs = create_functions_wrong()
print([f() for f in funcs])  # [2, 2, 2]（都是最终值）

# 正确：使用默认参数捕获当前值
def create_functions_correct():
    funcs = []
    for i in range(3):
        funcs.append(lambda i=i: i)  # i=i 使用默认参数
    return funcs


funcs = create_functions_correct()
print([f() for f in funcs])  # [0, 1, 2]

# 或者使用闭包
def create_functions_closure():
    def make_func(j):
        return lambda: j
    funcs = []
    for i in range(3):
        funcs.append(make_func(i))
    return funcs


funcs = create_functions_closure()
print([f() for f in funcs])  # [0, 1, 2]
```

### 变量生命周期

```python
def outer():
    """外部变量在闭包引用期间不会被垃圾回收"""
    data = [0]  # 模拟大对象

    def inner():
        return len(data)  # 引用外部变量

    print(f"Before: {inner()}")
    result = inner()
    return result, data


# 即使 outer() 返回，data 也不会被回收，因为 inner 还在引用它
result, data = outer()
print(f"Result: {result}")
```

## nonlocal 关键字

Python 3 引入了 `nonlocal` 关键字，允许闭包修改外层作用域的变量：

```python
def counter():
    count = 0

    def inner():
        nonlocal count
        count += 1
        return count

    return inner


c = counter()
print(c())  # 1
print(c())  # 2
print(c())  # 3
```

`nonlocal` 解决了闭包中无法修改外层变量的限制，使代码更加清晰。

## 闭包与内存管理

闭包会捕获其引用的外部变量，这些变量在闭包存在期间不会被垃圾回收：

```python
def outer():
    data = [0]  # 模拟大对象

    def inner():
        return len(data)  # inner 引用 data

    return inner

closure = outer()
# 即使 outer() 返回，data 也不会被回收
# 因为 closure 还在引用它
```

使用 `weakref` 可以避免闭包对对象的强制保留：

```python
import weakref

def make_weak_closure(obj):
    weak_obj = weakref.ref(obj)
    def closure():
        return weak_obj()  # 对象无其他引用时可被回收
    return closure
```

## 总结

闭包的核心价值：

1. **状态保持** - 在函数调用之间保持状态
2. **数据隐藏** - 闭包创建的变量对外部不可见
3. **函数工厂** - 动态生成定制化的函数
4. **延迟执行** - 推迟某些计算直到真正需要时

理解闭包对于掌握高阶函数、函数式编程风格至关重要。
