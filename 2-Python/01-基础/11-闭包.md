# 闭包

闭包（Closure）是函数和其引用的外部变量组合形成的整体。闭包允许函数访问定义函数外部作用域中的变量，即使外部函数已经执行完毕。

## 基础概念

### 什么是闭包

```python
def outer_function(x):
    """外部函数"""
    def inner_function(y):
        """内部函数，引用了外部变量 x"""
        return x + y
    return inner_function


# 创建闭包
closure = outer_function(10)
print(closure(5))  # 15
print(closure(3))  # 13

# 闭包捕获的变量
print(closure.__code__.co_freevars)  # ('x',) - 自由变量
```

### 闭包的三个条件

1. 必须有一个嵌套函数
2. 嵌套函数引用了外部作用域的变量
3. 外部函数返回嵌套函数

```python
# 示例：满足闭包条件的函数
def make_multiplier(factor):
    def multiplier(number):
        return number * factor  # 引用了外部变量 factor
    return multiplier  # 返回嵌套函数


times_3 = make_multiplier(3)
times_5 = make_multiplier(5)

print(times_3(10))  # 30
print(times_5(10))  # 50
```

## 闭包 vs 普通函数

### 状态保持

```python
# 普通函数：每次调用独立
def add_plain(x, y):
    return x + y


# 闭包：保持状态
def counter(start=0):
    def step():
        nonlocal start
        start += 1
        return start
    return step


cnt = counter(0)
print(cnt())  # 1
print(cnt())  # 2
print(cnt())  # 3

# 多个闭包独立状态
cnt2 = counter(10)
print(cnt2())  # 11
print(cnt())   # 4（cnt 保持自己的状态）
```

## nonlocal 关键字

### 修改闭包捕获的变量

```python
def make_accumulator():
    total = 0

    def accumulator(value):
        nonlocal total  # 允许修改外部变量
        total += value
        return total

    return accumulator


acc = make_accumulator()
print(acc(10))  # 10
print(acc(5))   # 15
print(acc(3))   # 18
```

### 闭包变量遮蔽

```python
def outer():
    x = "outer"

    def inner():
        nonlocal x
        x = "inner"  # 修改外层变量
        print(f"Inner: {x}")

    inner()
    print(f"Outer: {x}")


outer()
# Inner: inner
# Outer: inner
```

### 记忆化 / 缓存

```python
def memoize(func):
    """记忆化装饰器"""
    cache = {}

    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]

    # 保留原函数信息
    wrapper.cache = cache
    return wrapper


@memoize
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)


print(fibonacci(100))  # 354224848179261915075
print(fibonacci.cache)  # 缓存的中间结果
```

### 延迟求值

```python
def lazy_property(func):
    """延迟属性：只在首次访问时计算"""
    attr_name = f"_lazy_{func.__name__}"

    @property
    def wrapper(self):
        if not hasattr(self, attr_name):
            setattr(self, attr_name, func(self))
        return getattr(self, attr_name)

    return wrapper


class DataProcessor:
    def __init__(self, data):
        self.data = data

    @lazy_property
    def expensive_computation(self):
        """耗时的计算，只有首次访问时执行"""
        print("Computing...")
        return sum(self.data) / len(self.data)


processor = DataProcessor([1, 2, 3, 4, 5])
print(processor.expensive_computation)  # Computing... 3.0
print(processor.expensive_computation)  # 直接返回 3.0（不再打印 Computing）
```

## 装饰器

### 简单装饰器

```python
import functools


def debug(func):
    """打印函数调用的装饰器"""
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        print(f"Calling {func.__name__}")
        result = func(*args, **kwargs)
        print(f"{func.__name__} returned {result}")
        return result
    return wrapper


@debug
def add(a, b):
    return a + b


add(3, 5)
# Calling add
# add returned 8
```

### 带参数的装饰器

```python
import functools


def repeat(times):
    """重复调用指定次数"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            results = []
            for _ in range(times):
                results.append(func(*args, **kwargs))
            return results
        return wrapper
    return decorator


@repeat(3)
def greet(name):
    return f"Hello, {name}!"


print(greet("Alice"))
# ['Hello, Alice!', 'Hello, Alice!', 'Hello, Alice!']
```

### 类装饰器

```python
class CallCounter:
    """统计函数调用次数"""
    def __init__(self, func):
        self.func = func
        self.count = 0
        functools.update_wrapper(self, func)

    def __call__(self, *args, **kwargs):
        self.count += 1
        print(f"{self.func.__name__} called {self.count} times")
        return self.func(*args, **kwargs)


@CallCounter
def fetch_data(url):
    return f"Data from {url}"


fetch_data("example.com")  # fetch_data called 1 times
fetch_data("example.com")  # fetch_data called 2 times
```

## 闭包陷阱

### 循环闭包

```python
# 错误：所有函数引用同一个变量
def create_functions_wrong():
    funcs = []
    for i in range(3):
        funcs.append(lambda: i)  # i 是可变变量
    return funcs


funcs = create_functions_wrong()
print([f() for f in funcs])  # [2, 2, 2]（都是最终值）

# 正确：使用默认参数捕获当前值
def create_functions_correct():
    funcs = []
    for i in range(3):
        funcs.append(lambda i=i: i)  # i=i 使用默认参数
    return funcs


funcs = create_functions_correct()
print([f() for f in funcs])  # [0, 1, 2]

# 或者使用闭包
def create_functions_closure():
    def make_func(j):
        return lambda: j
    funcs = []
    for i in range(3):
        funcs.append(make_func(i))
    return funcs


funcs = create_functions_closure()
print([f() for f in funcs])  # [0, 1, 2]
```

### 变量生命周期

```python
def outer():
    """外部变量在闭包引用期间不会被垃圾回收"""
    data = [0]  # 模拟大对象

    def inner():
        return len(data)  # 引用外部变量

    print(f"Before: {inner()}")
    result = inner()
    return result, data


# 即使 outer() 返回，data 也不会被回收，因为 inner 还在引用它
result, data = outer()
print(f"Result: {result}")
```

## 完整示例

### 节流装饰器

```python
import time
import functools


def throttle(seconds):
    """节流装饰器：限制函数调用频率"""
    last_call = [0.0]  # 使用列表存储可变变量

    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            now = time.time()
            if now - last_call[0] >= seconds:
                last_call[0] = now
                return func(*args, **kwargs)
            else:
                print(f"Throttled: {func.__name__} called too frequently")
        return wrapper
    return decorator


@throttle(1)
def fetch_data():
    return "Data fetched"


fetch_data()  # 执行
fetch_data()  # 被节流
time.sleep(1)
fetch_data()  # 执行
```

### 防抖装饰器

```python
import time
import functools


def debounce(wait_seconds):
    """防抖装饰器：等待一定时间后执行"""
    def decorator(func):
        timer = [None]  # 使用列表存储定时器

        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            def call_func():
                timer[0] = None
                return func(*args, **kwargs)

            if timer[0] is not None:
                timer[0].cancel()

            timer[0] = threading.Timer(wait_seconds, call_func)
            timer[0].start()

        return wrapper
    return decorator


# 注意：需要 threading 支持
import threading


@debounce(0.5)
def process_input(text):
    print(f"Processing: {text}")


process_input("a")
process_input("ab")
process_input("abc")
# 只输出最后一个 "abc" 的处理结果
```

### 单例模式

```python
def singleton(cls):
    """单例模式装饰器"""
    instances = {}

    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]

    return get_instance


@singleton
class Database:
    def __init__(self, connection_string):
        self.connection_string = connection_string
        print(f"Connected to {connection_string}")


db1 = Database("postgresql://localhost/db1")
db2 = Database("postgresql://localhost/db2")
print(db1 is db2)  # True（同一个实例）
```

### 上下文管理器

```python
from contextlib import contextmanager


@contextmanager
def managed_resource(name):
    """资源管理的上下文管理器"""
    print(f"Acquiring {name}")
    resource = f"Resource-{name}"
    try:
        yield resource
    finally:
        print(f"Releasing {name}")


with managed_resource("database") as res:
    print(f"Using {res}")
# Acquiring database
# Using Resource-database
# Releasing database
```

### 事件系统

```python
class EventSystem:
    """简单的事件系统"""
    def __init__(self):
        self._handlers = {}

    def on(self, event_name):
        """注册事件处理器的装饰器"""
        def decorator(func):
            if event_name not in self._handlers:
                self._handlers[event_name] = []
            self._handlers[event_name].append(func)
            return func
        return decorator

    def emit(self, event_name, *args, **kwargs):
        """触发事件"""
        if event_name in self._handlers:
            for handler in self._handlers[event_name]:
                handler(*args, **kwargs)


events = EventSystem()


@events.on("user_login")
def on_user_login(user_id):
    print(f"User {user_id} logged in")


@events.on("user_login")
def send_welcome_email(user_id):
    print(f"Sending welcome email to user {user_id}")


events.emit("user_login", 12345)
# User 12345 logged in
# Sending welcome email to user 12345
```

## 闭包与内存管理

### 闭包与循环引用

```python
def create_cycle():
    """创建循环引用的闭包"""
    x = []

    def closure():
        x.append(1)  # closure 引用 x，x 也可能引用 closure
        return x

    return closure


# Python 的垃圾回收会处理循环引用
# 但闭包通常不会造成内存泄漏
```

### 弱引用

```python
import weakref


def make_weak_closure(obj):
    """使用弱引用的闭包"""
    weak_obj = weakref.ref(obj)

    def closure():
        if weak_obj() is None:
            return None
        return weak_obj()

    return closure


class Data:
    def __init__(self, value):
        self.value = value


data = Data(42)
weak_closure = make_weak_closure(data)
print(weak_closure().value)  # 42

del data  # 删除原始对象
print(weak_closure())  # None（对象已被回收）
```

## 总结

闭包的核心价值：

1. **状态保持** - 在函数调用之间保持状态
2. **数据隐藏** - 闭包创建的变量对外部不可见
3. **函数工厂** - 动态生成定制化的函数
4. **装饰器基础** - 装饰器本质上是返回函数的函数
5. **延迟执行** - 推迟某些计算直到真正需要时

理解闭包对于掌握装饰器、高阶函数、函数式编程风格至关重要。
