# 面向对象编程

## class 基础

```python
# 定义类
class Dog:
    # 类属性（所有实例共享）
    species = "Canis familiaris"

    # 实例属性（在 __init__ 中定义）
    def __init__(self, name: str, age: int):
        self.name = name
        self.age = age

    # 实例方法
    def bark(self) -> str:
        return f"{self.name} says woof!"

    # repr 用于调试
    def __repr__(self) -> str:
        return f"Dog(name={self.name!r}, age={self.age})"

# 创建实例
my_dog = Dog("Buddy", 3)
print(my_dog.bark())  # Buddy says woof!
print(my_dog.species)  # Canis familiaris

# 类型检查
isinstance(my_dog, Dog)  # True
```

## 继承

```python
# 单继承
class Animal:
    def __init__(self, name: str):
        self.name = name

    def speak(self) -> str:
        raise NotImplementedError


class Cat(Animal):
    def speak(self) -> str:
        return f"{self.name} says meow!"


class Dog(Animal):
    def speak(self) -> str:
        return f"{self.name} says woof!"


cat = Cat("Whiskers")
print(cat.speak())  # Whiskers says meow!

# 多继承（慎用，优先用组合）
class Pet:
    def __init__(self, name: str, owner: str):
        self.name = name
        self.owner = owner


class ShowDog(Dog, Pet):
    """多继承：同时继承 Dog 和 Pet"""

    def __init__(self, name: str, age: int, owner: str, title: str = ""):
        Dog.__init__(self, name, age)
        Pet.__init__(self, name, owner)
        self.title = title

    def bark(self) -> str:
        return f"{self.title}{self.name} competes!".upper()
```

## MRO（方法解析顺序）

```python
# Python 使用 C3 线性化算法确定方法查找顺序
class A:
    def greet(self):
        return "A"


class B(A):
    def greet(self):
        return "B"


class C(A):
    def greet(self):
        return "C"


class D(B, C):
    pass


d = D()
print(d.greet())       # B（MRO: D -> B -> C -> A）
print(D.__mro__)       # 查看完整 MRO
# (<class 'D'>, <class 'B'>, <class 'C'>, <class 'A'>, <class 'object'>)

# super() 按照 MRO 顺序调用
class Base:
    def __init__(self):
        print("Base init")


class Child(Base):
    def __init__(self):
        super().__init__()  # 调用 Base.__init__
        print("Child init")


Child()
# Base init
# Child init
```

## 数据类 dataclass

```python
from dataclasses import dataclass, field
from typing import ClassVar


# 基础 dataclass
@dataclass
class Point:
    x: float
    y: float

    def distance_to(self, other: "Point") -> float:
        return ((self.x - other.x) ** 2 + (self.y - other.y) ** 2) ** 0.5


p1 = Point(1.0, 2.0)
p2 = Point(4.0, 6.0)
print(p1.distance_to(p2))

# 带默认值的字段
@dataclass
class User:
    name: str
    email: str
    active: bool = True
    created_at: str = field(default_factory=lambda: "2024-01-01")


# field(default_factory=...) 用于可变默认值
@dataclass
class ShoppingCart:
    items: list[str] = field(default_factory=list)
    owner: str = ""

    def add(self, item: str):
        self.items.append(item)


# 比较功能（自动生成 __eq__, __lt__ 等）
@dataclass(order=True)
class Employee:
    sort_key: tuple[int, str] = field(compare=False)
    name: str
    salary: int


# frozen=True（不可变对象）
@dataclass(frozen=True)
class ImmutablePoint:
    x: float
    y: float


p = ImmutablePoint(1, 2)
# p.x = 3  # FrozenInstanceError

# post_init（初始化后处理）
@dataclass
class Circle:
    radius: float
    area: float = field(init=False)

    def __post_init__(self):
        import math
        self.area = math.pi * self.radius ** 2
```

## ABC（抽象基类）

```python
from abc import ABC, abstractmethod


class Shape(ABC):
    @abstractmethod
    def area(self) -> float:
        """计算面积"""
        pass

    @abstractmethod
    def perimeter(self) -> float:
        """计算周长"""
        pass

    def describe(self) -> str:
        """具体方法（可选）"""
        return f"A shape with area {self.area()}"


class Rectangle(Shape):
    def __init__(self, width: float, height: float):
        self.width = width
        self.height = height

    def area(self) -> float:
        return self.width * self.height

    def perimeter(self) -> float:
        return 2 * (self.width + self.height)


# Shape()  # TypeError: Can't instantiate abstract class Shape
rect = Rectangle(3, 4)
print(rect.area())  # 12
```

## Protocol（结构子类型）

```python
from typing import Protocol, runtime_checkable


# 定义 Protocol（类似接口）
@runtime_checkable
class Closable(Protocol):
    def close(self) -> None:
        ...


class Resource:
    def close(self) -> None:
        print("Closing resource")


# 静态类型检查器认可，runtime_checkable 允许 isinstance 检查
r = Resource()
isinstance(r, Closable)  # True（因为有 close 方法）


# 函数参数使用 Protocol
def close_all(things: list[Closable]) -> None:
    for thing in things:
        thing.close()


# 复杂 Protocol
class SupportsLessThan(Protocol):
    def __lt__(self, other) -> bool:
        ...


# PEP 695 语法（Python 3.12+，3.14 中进一步改进）
def find_min[T: SupportsLessThan](items: list[T]) -> T | None:
    if not items:
        return None
    min_item = items[0]
    for item in items[1:]:
        if item < min_item:
            min_item = item
    return min_item
```

**兼容写法（Python 3.9+）**：

```python
from typing import TypeVar

T = TypeVar('T', bound=SupportsLessThan)

def find_min(items: list[T]) -> T | None:
    if not items:
        return None
    min_item = items[0]
    for item in items[1:]:
        if item < min_item:
            min_item = item
    return min_item
```

## 组合 vs 继承

```python
# 优先组合，慎用继承
class Engine:
    def start(self):
        print("Engine started")


class Wheel:
    def rotate(self):
        print("Wheel rotating")


# 组合（推荐）
class Car:
    def __init__(self):
        self.engine = Engine()  # 组合
        self.wheels = [Wheel() for _ in range(4)]

    def drive(self):
        self.engine.start()
        for wheel in self.wheels:
            wheel.rotate()


# 继承（仅当"是-a"关系明确时使用）
class ElectricCar(Car):  # 电动车是汽车
    def __init__(self):
        super().__init__()
        self.battery = 100

    def charge(self):
        print("Charging...")
```

## 属性 property

```python
# getter / setter / deleter
class Temperature:
    def __init__(self, celsius: float = 0):
        self._celsius = celsius

    @property
    def celsius(self) -> float:
        return self._celsius

    @celsius.setter
    def celsius(self, value: float):
        self._celsius = value

    @celsius.deleter
    def celsius(self):
        print("Deleting temperature")
        del self._celsius

    # 华氏温度（计算属性）
    @property
    def fahrenheit(self) -> float:
        return self._celsius * 9 / 5 + 32


t = Temperature(25)
print(t.fahrenheit)  # 77.0
t.fahrenheit = 100
print(t.celsius)     # ~37.78
```

## __slots__

```python
# 限制实例属性，减少内存占用
class Point:
    __slots__ = ('x', 'y')

    def __init__(self, x, y):
        self.x = x
        self.y = y


# Point 实例不能再添加其他属性
p = Point(1, 2)
p.z = 3  # AttributeError

# 结合继承（父类有 __slots__，子类也要有才能优化）
class LabeledPoint(Point):
    __slots__ = ('label',)
```

## 常见陷阱

```python
# 陷阱1：可变默认参数
class Node:
    def __init__(self, children=None):  # 错误！
        self.children = children or []

    def add_child(self, child):
        self.children.append(child)


# 正确做法
class Node:
    def __init__(self, children=None):
        self.children = list(children) if children else []

    def add_child(self, child):
        self.children.append(child)


# 陷阱2：类属性 vs 实例属性
class Counter:
    count = 0  # 类属性（共享）

    def __init__(self):
        self.count += 1  # 创建实例属性，遮蔽类属性


c1 = Counter()
print(c1.count)  # 1（实例属性）
print(Counter.count)  # 0（类属性未被修改）

# 正确做法
class Counter:
    count = 0

    def __init__(self):
        Counter.count += 1


# 陷阱3：忘记调用 super().__init__
class Child(Base):
    def __init__(self, extra_param):
        self.extra = extra_param
        # 忘记 super().__init__()，Base 的属性未初始化


# 陷阱4：多继承时的菱形继承歧义
class A:
    def process(self):
        print("A")


class B(A):
    pass


class C(A):
    def process(self):
        print("C")


class D(B, C):
    pass


D().process()  # C（按 MRO，B -> C -> A）
```
