## 异步上下文管理器

```python
class AsyncContext:
    async def __aenter__(self):
        await self.connect()
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        await self.disconnect()

async def main():
    async with AsyncContext() as ctx:
```

## 异步迭代器

```python
class AsyncIterator:
    def __init__(self, data):
        self.data = data
        self.index = 0

    def __aiter__(self):
        return self

    async def __anext__(self):
        if self.index >= len(self.data):
            raise StopAsyncIteration
        value = self.data[self.index]
        self.index += 1
        await asyncio.sleep(0.1)
        return value

async def main():
    async for item in AsyncIterator([1, 2, 3]):
        print(item)
```

## 异步生成器组合

```python
# 异步生成器组合
async def fetch_all(urls):
    async def fetch_one(url):
        await asyncio.sleep(0.1)
        return f"Result: {url}"

    results = []
    async for result in async_gen_map(fetch_one, urls):
        results.append(result)
    return results

async def async_gen_map(async_func, items):
    for item in items:
        yield await async_func(item)

# 使用 asyncio.gather 实现并发
async def fetch_all_fast(urls):
    async def fetch_one(url):
        await asyncio.sleep(0.1)
        return f"Result: {url}"

    return await asyncio.gather(*[fetch_one(url) for url in urls])
```

## 异步上下文管理器进阶

```python
# 资源池模式
class AsyncPool:
    def __init__(self, factory, max_size=10):
        self.factory = factory
        self.max_size = max_size
        self._pool = asyncio.Queue(max_size)
        self._size = 0

    async def acquire(self):
        if self._pool.empty() and self._size < self.max_size:
            resource = await self.factory()
            self._size += 1
            return resource
        return await self._pool.get()

    async def release(self, resource):
        if self._pool.full():
            await resource.close()
            self._size -= 1
        else:
            await self._pool.put(resource)

    async def __aenter__(self):
        self.resource = await self.acquire()
        return self.resource

    async def __aexit__(self, *args):
        await self.release(self.resource)

# 使用
async def create_connection():
    await asyncio.sleep(0.1)
    return f"Connection-{id(object())}"

async def main():
    pool = AsyncPool(create_connection, max_size=5)
    async with pool as conn:
        print(f"Using {conn}")

# 嵌套异步上下文管理器
class OuterContext:
    async def __aenter__(self):
        print("Outer: entering")
        self.inner = InnerContext()
        return await self.inner.__aenter__()

    async def __aexit__(self, *args):
        await self.inner.__aexit__(*args)
        print("Outer: exited")

class InnerContext:
    async def __aenter__(self):
        print("Inner: entering")
        return self

    async def __aexit__(self, *args):
        print("Inner: exited")
```

## asyncio 进阶用法

```python
# 延迟执行任务
async def delayed_task(delay, coro):
    await asyncio.sleep(delay)
    return await coro

# 一次性定时任务
async def run_once():
    loop = asyncio.get_event_loop()
    handle = loop.call_later(
        5.0,
        lambda: print("Delayed execution!")
    )
    return handle

# 任务组和取消
async def cancellable_task(name):
    try:
        for i in range(10):
            await asyncio.sleep(1)
            print(f"{name}: step {i}")
    except asyncio.CancelledError:
        print(f"{name} was cancelled")
        raise

async def main():
    task = asyncio.create_task(cancellable_task("Task1"))
    await asyncio.sleep(3)
    task.cancel()
    try:
        await task
    except asyncio.CancelledError:
        print("Task1 confirmed cancelled")
```

## 异步 I/O 模式

```python
# 异步文件操作（使用 aiofiles）
async def read_file_async(path):
    import aiofiles
    async with aiofiles.open(path, mode='r') as f:
        content = await f.read()
    return content

async def write_file_async(path, content):
    import aiofiles
    async with aiofiles.open(path, mode='w') as f:
        await f.write(content)

# 异步数据库操作（使用 aiomysql）
async def query_db(pool, sql):
    async with pool.acquire() as conn:
        async with conn.cursor() as cursor:
            await cursor.execute(sql)
            return await cursor.fetchall()

# 异步 Redis 操作（使用 aioredis）
async def redis_example():
    import aioredis
    redis = await aioredis.create_redis_pool('redis://localhost')
    await redis.set('key', 'value')
    value = await redis.get('key')
    redis.close()
    return value

# 异步消息队列
async def async_queue_producer(queue):
    for i in range(10):
        await queue.put(i)
        print(f"Produced {i}")
        await asyncio.sleep(0.1)

async def async_queue_consumer(queue):
    while True:
        item = await queue.get()
        if item is None:
            break
        print(f"Consumed {item}")
        await asyncio.sleep(0.2)

async def main():
    queue = asyncio.Queue()
    await asyncio.gather(
        async_queue_producer(queue),
        async_queue_consumer(queue)
    )
```

## 性能优化与调试

```python
import asyncio
import time

# 1. 避免创建过多小任务
async def bad_pattern():
    tasks = [asyncio.create_task(asyncio.sleep(0.1)) for _ in range(10000)]
    await asyncio.gather(*tasks)

async def good_pattern():
    for _ in range(100):
        await asyncio.gather(*[asyncio.sleep(0.1) for _ in range(100)])

# 2. 使用 asyncio.timeout 简化超时处理
async def main():
    async with asyncio.timeout(5.0) as cm:
        await long_running_task()
    print(f"Completed: {cm.expired is None}")

# 3. 调试异步代码
async def main():
    asyncio.set_debug(True)
    loop = asyncio.get_event_loop()
    loop.set_debug(True)
    loop.slow_callback_duration = 0.1
    await asyncio.sleep(1)

# 4. 使用 asyncio.current_task() 获取当前任务
async def nested():
    current = asyncio.current_task()
    print(f"Current task: {current}")

async def main():
    task = asyncio.create_task(nested())
    await task

# 5. 任务完成后自动清理
async def main():
    tasks = []
    for i in range(10):
        task = asyncio.create_task(asyncio.sleep(1))
        tasks.append(task)

    done, pending = await asyncio.wait(tasks)

    for task in done:
        if task.exception():
            print(f"Task failed: {task.exception()}")
```

## 常见错误与解决方案

```python
# 错误 1: 忘记 await 导致任务未执行
async def bad_example():
    task = asyncio.create_task(some_coro())

async def good_example():
    task = asyncio.create_task(some_coro())
    await task

# 错误 2: 在同步函数中调用异步函数
def bad_sync():
    asyncio.run(async_func())

def good_sync():
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    try:
        result = loop.run_until_complete(async_func())
    finally:
        loop.close()

# 错误 3: 混合使用阻塞和异步代码
async def bad_mixed():
    time.sleep(1)

async def good_mixed():
    await asyncio.to_thread(time.sleep, 1)
    await asyncio.sleep(1)

# 错误 4: 资源泄露
async def bad_resource():
    session = aiohttp.ClientSession()
    await session.get("url")

async def good_resource():
    async with aiohttp.ClientSession() as session:
        await session.get("url")

# 错误 5: 任务组中异常处理不当
async def bad_task_group():
    async with asyncio.TaskGroup() as tg:
        tg.create_task(failing_task())
        raise ValueError("early error")

async def good_task_group():
    try:
        async with asyncio.TaskGroup() as tg:
            tg.create_task(failing_task())
    except ExceptionGroup as e:
        print(f"Caught {len(e.exceptions)} exceptions")
```
