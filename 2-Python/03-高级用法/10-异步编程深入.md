## async/await 基础

```python
import asyncio

async def main():
    print("Hello")
    await asyncio.sleep(1)
    print("World")

# 运行
asyncio.run(main())

# 异步生成器
async def async_gen():
    for i in range(3):
        yield i
        await asyncio.sleep(0.1)

async def consume():
    async for value in async_gen():
        print(value)

asyncio.run(consume())
```

## awaitable 对象

```python
# 三种 awaitable：coroutine, Task, Future

# 1. Coroutine
async def coro():
    return "result"

# 2. Task - 调度协程执行
async def main():
    task = asyncio.create_task(coro())
    result = await task
    print(result)

# 3. Future - 低级 awaitable
async def main():
    loop = asyncio.get_event_loop()
    future = loop.create_future()
    # 在另一个地方: future.set_result("done")
    result = await future
```

## 并发执行


## 任务管理

```python
async def worker(name):
    await asyncio.sleep(1)
    return f"{name} done"

async def main():
    # 创建任务
    task = asyncio.create_task(worker("task1"))

    # 等待任务完成
    result = await task

    # 等待多个任务
    tasks = [asyncio.create_task(worker(i)) for i in range(5)]
    results = await asyncio.gather(*tasks)

    # 取消任务
    task.cancel()
    try:
        await task
    except asyncio.CancelledError:
        print("Task cancelled")

    # 等待但不失联
    task = asyncio.create_task(worker("long"))
    done, pending = await asyncio.wait([task], timeout=0.1)
    # pending 中的任务仍在运行
```

## 锁与同步

```python
# asyncio.Lock
async def critical_section(lock):
    async with lock:
        print("Critical section")
        await asyncio.sleep(0.1)

async def main():
    lock = asyncio.Lock()
    await asyncio.gather(
        critical_section(lock),
        critical_section(lock),
    )

# asyncio.Event
async def waiter(event):
    print("Waiting...")
    await event.wait()
    print("Waiter woke up!")

async def setter(event):
    await asyncio.sleep(1)
    event.set()

async def main():
    event = asyncio.Event()
    await asyncio.gather(waiter(event), setter(event))

# asyncio.Condition
async def consumer(condition):
    async with condition:
        await condition.wait()
        print("Consumer notified")

async def producer(condition):
    await asyncio.sleep(1)
    async with condition:
        condition.notify()

# asyncio.Semaphore
async def limited_access(sem):
    async with sem:
        print("Access granted")
        await asyncio.sleep(0.1)

async def main():
    sem = asyncio.Semaphore(2)
    await asyncio.gather(*[limited_access(sem) for _ in range(5)])
```

## 队列

```python
import asyncio

async def producer(queue):
    for i in range(5):
        await queue.put(i)
        print(f"Produced {i}")
        await asyncio.sleep(0.1)
    await queue.put(None)  # 信号结束

async def consumer(queue):
    while True:
        item = await queue.get()
        if item is None:
            break
        print(f"Consumed {item}")
        await asyncio.sleep(0.2)

async def main():
    queue = asyncio.Queue()
    await asyncio.gather(
        producer(queue),
        consumer(queue)
    )

asyncio.run(main())
```

## asyncio 信号处理

```python
import signal
import sys

def handle_signal(sig):
    print(f"Received signal {sig}")

async def main():
    loop = asyncio.get_event_loop()

    # 注册信号处理器
    for sig in (signal.SIGINT, signal.SIGTERM):
        loop.add_signal_handler(
            sig,
            lambda s=sig: handle_signal(s)
        )

    # 或者使用 asyncio.Event
    shutdown_event = asyncio.Event()

    def shutdown_callback():
        shutdown_event.set()

    loop.add_signal_handler(signal.SIGINT, shutdown_callback)
    loop.add_signal_handler(signal.SIGTERM, shutdown_callback)

    print("Running...")
    await shutdown_event.wait()
    print("Shutting down...")

asyncio.run(main())
```

## 运行阻塞代码

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor

def blocking_io():
    # 同步 I/O 操作
    with open("large_file.txt", "r") as f:
        return f.read()

def blocking_cpu():
    # CPU 密集型操作
    return sum(range(10**7))

async def main():
    loop = asyncio.get_event_loop()

    # 在线程池中运行
    with ThreadPoolExecutor() as pool:
        result = await loop.run_in_executor(
            pool, blocking_io
        )
        print(f"Read {len(result)} bytes")

    # CPU 密集型用进程池更合适
    from concurrent.futures import ProcessPoolExecutor
    with ProcessPoolExecutor() as pool:
        result = await loop.run_in_executor(
            pool, blocking_cpu
        )

asyncio.run(main())
```

## 常见陷阱

```python
# 1. 忘记 await
async def bad():
    data = fetch_data()  # 错误：返回协程对象而非结果
    print(data)  # <coroutine object ...>

async def good():
    data = await fetch_data()  # 正确
    print(data)

# 2. 混淆同步和异步
def sync_func():
    return "sync"

async def async_func():
    return "async"

# sync_func() 不需要 await
# async_func() 需要 await

# 3. 死锁 - 不要在协程中无限等待
# 4. 异常未捕获 - 协程异常会传播到 await 处

# 5. 创建任务后不等待
async def bad():
    asyncio.create_task(background_task())  # 任务可能未完成就退出

async def good():
    await asyncio.create_task(background_task())  # 等待完成
```

## 异步迭代器

```python
class AsyncIterator:
    def __init__(self, data):
        self.data = data
        self.index = 0

    def __aiter__(self):
        return self

    async def __anext__(self):
        if self.index >= len(self.data):
            raise StopAsyncIteration
        value = self.data[self.index]
        self.index += 1
        await asyncio.sleep(0.1)  # 异步操作
        return value

async def main():
    async for item in AsyncIterator([1, 2, 3]):
```

## 异步上下文管理器进阶

```python
# 资源池模式
class AsyncPool:
    def __init__(self, factory, max_size=10):
        self.factory = factory
        self.max_size = max_size
        self._pool = asyncio.Queue(max_size)
        self._size = 0

    async def acquire(self):
        if self._pool.empty() and self._size < self.max_size:
            resource = await self.factory()
            self._size += 1
            return resource
        return await self._pool.get()

    async def release(self, resource):
        if self._pool.full():
            # 关闭多余资源
            await resource.close()
            self._size -= 1
        else:
            await self._pool.put(resource)

    async def __aenter__(self):
        self.resource = await self.acquire()
        return self.resource

    async def __aexit__(self, *args):
        await self.release(self.resource)

# 使用
async def create_connection():
    await asyncio.sleep(0.1)
    return f"Connection-{id(object())}"

async def main():
    pool = AsyncPool(create_connection, max_size=5)
    async with pool as conn:
        print(f"Using {conn}")

# 嵌套异步上下文管理器
class OuterContext:
    async def __aenter__(self):
        print("Outer: entering")
        self.inner = InnerContext()
        return await self.inner.__aenter__()

    async def __aexit__(self, *args):
        await self.inner.__aexit__(*args)
        print("Outer: exited")

class InnerContext:
    async def __aenter__(self):
        print("Inner: entering")
        return self

    async def __aexit__(self, *args):
        print("Inner: exited")
```

## asyncio 进阶用法

```python
# 延迟执行任务
async def delayed_task(delay, coro):
    """延迟执行协程"""
    await asyncio.sleep(delay)
    return await coro

# 一次性定时任务
async def run_once():
    loop = asyncio.get_event_loop()
    # call_later 返回 TimerHandle，无需 await
    handle = loop.call_later(
        5.0,  # 5秒后执行
        lambda: print("Delayed execution!")
    )
    return handle

# asyncio.wait_for with 超时
async def main():
    try:
        result = await asyncio.wait_for(
            asyncio.sleep(3),
            timeout=1.0
        )
    except asyncio.TimeoutError:
        print("Task timed out")

# 任务组和取消
async def cancellable_task(name):
    try:
        for i in range(10):
            await asyncio.sleep(1)
            print(f"{name}: step {i}")
    except asyncio.CancelledError:
        print(f"{name} was cancelled")
        raise

async def main():
    task = asyncio.create_task(cancellable_task("Task1"))
```

## 性能优化与调试

```python
import asyncio
import time

# 1. 避免创建过多小任务
async def bad_pattern():
    """错误：创建大量小任务"""
    tasks = [asyncio.create_task(asyncio.sleep(0.1)) for _ in range(10000)]
    await asyncio.gather(*tasks)

async def good_pattern():
    """正确：批量等待"""
    for _ in range(100):
        await asyncio.gather(*[asyncio.sleep(0.1) for _ in range(100)])

# 2. 使用 asyncio.timeout 简化超时处理
async def main():
    async with asyncio.timeout(5.0) as cm:
        await long_running_task()
    print(f"Completed: {cm.expired is None}")

# 3. 调试异步代码
async def main():
    # 启用调试模式
    asyncio.set_debug(True)

    loop = asyncio.get_event_loop()
    loop.set_debug(True)

    # 慢回调警告（>100ms）
    loop.slow_callback_duration = 0.1

    await asyncio.sleep(1)

# 4. 使用 asyncio.current_task() 获取当前任务
async def nested():
    current = asyncio.current_task()
    print(f"Current task: {current}")

async def main():
    task = asyncio.create_task(nested())
    await task

# 5. 任务完成后自动清理
async def main():
    tasks = []
    for i in range(10):
        task = asyncio.create_task(asyncio.sleep(1))
        tasks.append(task)

    # 等待所有任务完成
    done, pending = await asyncio.wait(tasks)

    # 清理已完成的任务
    for task in done:
        if task.exception():
            print(f"Task failed: {task.exception()}")
```

## 常见错误与解决方案

```python
# 错误 1: 忘记 await 导致任务未执行
async def bad_example():
    task = asyncio.create_task(some_coro())  # 协程未执行
    # 任务可能不会运行

async def good_example():
    task = asyncio.create_task(some_coro())
    await task  # 等待完成

# 错误 2: 在同步函数中调用异步函数
def bad_sync():
    asyncio.run(async_func())  # 正确但阻塞

def good_sync():
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    try:
        result = loop.run_until_complete(async_func())
    finally:
        loop.close()

# 错误 3: 混合使用阻塞和异步代码
async def bad_mixed():
    time.sleep(1)  # 阻塞事件循环
    await asyncio.sleep(1)  # 非阻塞

async def good_mixed():
    await asyncio.to_thread(time.sleep, 1)  # 在线程中运行
    await asyncio.sleep(1)

# 错误 4: 资源泄露
async def bad_resource():
    session = aiohttp.ClientSession()  # 未关闭
    await session.get("url")

async def good_resource():
    async with aiohttp.ClientSession() as session:
        await session.get("url")  # 自动关闭

# 错误 5: 任务组中异常处理不当
async def bad_task_group():
    async with asyncio.TaskGroup() as tg:
        tg.create_task(failing_task())
        raise ValueError("early error")  # 导致所有任务被取消

async def good_task_group():
    try:
        async with asyncio.TaskGroup() as tg:
            tg.create_task(failing_task())
    except ExceptionGroup as e:
        print(f"Caught {len(e.exceptions)} exceptions")
```
