# 异步编程深入

## async/await 基础

```python
import asyncio

async def main():
    print("Hello")
    await asyncio.sleep(1)
    print("World")

# 运行
asyncio.run(main())

# 异步生成器
async def async_gen():
    for i in range(3):
        yield i
        await asyncio.sleep(0.1)

async def consume():
    async for value in async_gen():
        print(value)

asyncio.run(consume())
```

## awaitable 对象

```python
# 三种 awaitable：coroutine, Task, Future

# 1. Coroutine
async def coro():
    return "result"

# 2. Task - 调度协程执行
async def main():
    task = asyncio.create_task(coro())
    result = await task
    print(result)

# 3. Future - 低级 awaitable
async def main():
    loop = asyncio.get_event_loop()
    future = loop.create_future()
    # 在另一个地方: future.set_result("done")
    result = await future
```

## 并发执行

```python
async def fetch(url):
    await asyncio.sleep(1)  # 模拟网络请求
    return f"Response from {url}"

async def main():
    # 并发执行
    results = await asyncio.gather(
        fetch("http://api1.com"),
        fetch("http://api2.com"),
        fetch("http://api3.com"),
    )
    print(results)
    # ['Response from http://api1.com', 'Response from http://api2.com', 'Response from http://api3.com']

asyncio.run(main())
```

## 任务管理

```python
async def worker(name):
    await asyncio.sleep(1)
    return f"{name} done"

async def main():
    # 创建任务
    task = asyncio.create_task(worker("task1"))

    # 等待任务完成
    result = await task

    # 等待多个任务
    tasks = [asyncio.create_task(worker(i)) for i in range(5)]
    results = await asyncio.gather(*tasks)

    # 取消任务
    task.cancel()
    try:
        await task
    except asyncio.CancelledError:
        print("Task cancelled")

    # 等待但不失联
    task = asyncio.create_task(worker("long"))
    done, pending = await asyncio.wait([task], timeout=0.1)
    # pending 中的任务仍在运行
```

## 锁与同步

```python
# asyncio.Lock
async def critical_section(lock):
    async with lock:
        print("Critical section")
        await asyncio.sleep(0.1)

async def main():
    lock = asyncio.Lock()
    await asyncio.gather(
        critical_section(lock),
        critical_section(lock),
    )

# asyncio.Event
async def waiter(event):
    print("Waiting...")
    await event.wait()
    print("Waiter woke up!")

async def setter(event):
    await asyncio.sleep(1)
    event.set()

async def main():
    event = asyncio.Event()
    await asyncio.gather(waiter(event), setter(event))

# asyncio.Condition
async def consumer(condition):
    async with condition:
        await condition.wait()
        print("Consumer notified")

async def producer(condition):
    await asyncio.sleep(1)
    async with condition:
        condition.notify()

# asyncio.Semaphore
async def limited_access(sem):
    async with sem:
        print("Access granted")
        await asyncio.sleep(0.1)

async def main():
    sem = asyncio.Semaphore(2)
    await asyncio.gather(*[limited_access(sem) for _ in range(5)])
```

## 队列

```python
import asyncio

async def producer(queue):
    for i in range(5):
        await queue.put(i)
        print(f"Produced {i}")
        await asyncio.sleep(0.1)
    await queue.put(None)  # 信号结束

async def consumer(queue):
    while True:
        item = await queue.get()
        if item is None:
            break
        print(f"Consumed {item}")
        await asyncio.sleep(0.2)

async def main():
    queue = asyncio.Queue()
    await asyncio.gather(
        producer(queue),
        consumer(queue)
    )

asyncio.run(main())
```

## asyncio Streams（网络）

```python
# TCP 服务器
async def handle_client(reader, writer):
    data = await reader.read(100)
    message = data.decode()
    print(f"Received: {message}")

    writer.write(data)
    await writer.drain()
    writer.close()
    await writer.wait_closed()

async def main():
    server = await asyncio.start_server(
        handle_client, '127.0.0.1', 8888
    )
    print("Server started")
    async with server:
        await server.serve_forever()

asyncio.run(main())

# TCP 客户端
async def client():
    reader, writer = await asyncio.open_connection(
        '127.0.0.1', 8888
    )
    writer.write(b"Hello, server!")
    await writer.drain()
    data = await reader.read(100)
    print(f"Received: {data.decode()}")
    writer.close()
    await writer.wait_closed()

asyncio.run(client())
```

## asyncio 信号处理

```python
import signal
import sys

def handle_signal(sig):
    print(f"Received signal {sig}")

async def main():
    loop = asyncio.get_event_loop()

    # 注册信号处理器
    for sig in (signal.SIGINT, signal.SIGTERM):
        loop.add_signal_handler(
            sig,
            lambda s=sig: handle_signal(s)
        )

    # 或者使用 asyncio.Event
    shutdown_event = asyncio.Event()

    def shutdown_callback():
        shutdown_event.set()

    loop.add_signal_handler(signal.SIGINT, shutdown_callback)
    loop.add_signal_handler(signal.SIGTERM, shutdown_callback)

    print("Running...")
    await shutdown_event.wait()
    print("Shutting down...")

asyncio.run(main())
```

## 运行阻塞代码

```python
import asyncio
from concurrent.futures import ThreadPoolExecutor

def blocking_io():
    # 同步 I/O 操作
    with open("large_file.txt", "r") as f:
        return f.read()

def blocking_cpu():
    # CPU 密集型操作
    return sum(range(10**7))

async def main():
    loop = asyncio.get_event_loop()

    # 在线程池中运行
    with ThreadPoolExecutor() as pool:
        result = await loop.run_in_executor(
            pool, blocking_io
        )
        print(f"Read {len(result)} bytes")

    # CPU 密集型用进程池更合适
    from concurrent.futures import ProcessPoolExecutor
    with ProcessPoolExecutor() as pool:
        result = await loop.run_in_executor(
            pool, blocking_cpu
        )

asyncio.run(main())
```

## 常见陷阱

```python
# 1. 忘记 await
async def bad():
    data = fetch_data()  # 错误：返回协程对象而非结果
    print(data)  # <coroutine object ...>

async def good():
    data = await fetch_data()  # 正确
    print(data)

# 2. 混淆同步和异步
def sync_func():
    return "sync"

async def async_func():
    return "async"

# sync_func() 不需要 await
# async_func() 需要 await

# 3. 死锁 - 不要在协程中无限等待
# 4. 异常未捕获 - 协程异常会传播到 await 处

# 5. 创建任务后不等待
async def bad():
    asyncio.create_task(background_task())  # 任务可能未完成就退出

async def good():
    await asyncio.create_task(background_task())  # 等待完成
```

## 异步上下文管理器

```python
class AsyncContext:
    async def __aenter__(self):
        await self.connect()
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        await self.disconnect()

async def main():
    async with AsyncContext() as ctx:
        await ctx.do_something()
```

## 异步迭代器

```python
class AsyncIterator:
    def __init__(self, data):
        self.data = data
        self.index = 0

    def __aiter__(self):
        return self

    async def __anext__(self):
        if self.index >= len(self.data):
            raise StopAsyncIteration
        value = self.data[self.index]
        self.index += 1
        await asyncio.sleep(0.1)  # 异步操作
        return value

async def main():
    async for item in AsyncIterator([1, 2, 3]):
        print(item)
```
