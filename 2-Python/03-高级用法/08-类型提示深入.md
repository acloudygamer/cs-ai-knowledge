# 类型提示深入

## typing 模块核心概念

```python
# 类型别名
from typing import TypeAlias

Vector: TypeAlias = list[float]
Matrix: TypeAlias = list[list[float]]

# 新语法（Python 3.12+）
type Vector = list[float]
type Matrix = list[list[float]]

# Union 和 Optional
from typing import Union, Optional

# Union: 多种可能类型
def process(x: Union[int, str]) -> Union[str, int]:
    return x

# Optional: 相当于 Union[T, None]
def greet(name: Optional[str]) -> str:
    if name is None:
        return "Hello, stranger!"
    return f"Hello, {name}!"

# Python 3.10+ 简写
def process(x: int | str) -> str | int:
    return x

def greet(name: str | None) -> str:
    return f"Hello, {name or 'stranger'}!"
```

## TypeVar 泛型变量

```python
from typing import TypeVar

# 基本用法
T = TypeVar('T')
U = TypeVar('U')

def first(lst: list[T]) -> T | None:
    return lst[0] if lst else None


# 约束类型
OrderedT = TypeVar('OrderedT', int, float, str)

def max_(a: OrderedT, b: OrderedT) -> OrderedT:
    return a if a > b else b


# 类绑定
class Generic(Generic[T]):
    def __init__(self, value: T):
        self.value = value

    def get(self) -> T:
        return self.value
```

## Generic

```python
from typing import Generic, TypeVar

T = TypeVar('T')
K = TypeVar('K')
V = TypeVar('V')


class Container(Generic[T]):
    def __init__(self, item: T):
        self._item = item

    def get(self) -> T:
        return self._item

    def set(self, item: T) -> None:
        self._item = item


class KeyValue(Generic[K, V]):
    def __init__(self, key: K, value: V):
        self.key = key
        self.value = value

    def items(self) -> list[tuple[K, V]]:
        return [(self.key, self.value)]


# 使用
c: Container[int] = Container(42)
kv: KeyValue[str, int] = KeyValue("age", 30)
```

## Protocol（结构子类型）

```python
from typing import Protocol, runtime_checkable


# Protocol 定义"结构化接口"
@runtime_checkable
class Closable(Protocol):
    def close(self) -> None:
        ...


class Resource:
    def close(self) -> None:
        print("Closing")


# runtime_checkable 允许 isinstance 检查
r = Resource()
isinstance(r, Closable)  # True（必须有 close 方法）


# 更复杂的 Protocol
class Drawable(Protocol):
    def draw(self, x: int, y: int) -> None:
        ...


class Shape:
    def draw(self, x: int, y: int) -> None:
        print(f"Drawing at {x}, {y}")


def render(drawable: Drawable) -> None:
    drawable.draw(0, 0)
```

## Literal 文字类型

```python
from typing import Literal

# 限制为特定值
def handle_mode(mode: Literal["read", "write"]) -> None:
    pass


handle_mode("read")   # OK
handle_mode("write")  # OK
handle_mode("append")  # TypeError


# 多个文字值
def print_size(size: Literal["small", "medium", "large"]) -> None:
    print(f"Size: {size}")


# 结合 Union
def process(mode: Literal["r", "w", "a"] | None) -> None:
    pass
```

## TypedDict

```python

class User(TypedDict):
    name: str
    age: int

# 使用
user: User = {"name": "Alice", "age": 30, "email": "alice@example.com"}
```

## NewType

```python
from typing import NewType

# 创建"语义类型"
UserId = NewType('UserId', int)
OrderId = NewType('OrderId', int)

# 类型检查器会区分
def get_user(user_id: UserId) -> str:
    return f"User {user_id}"


uid = UserId(123)
oid = OrderId(123)

get_user(uid)    # OK
get_user(oid)    # TypeError: OrderId 不能赋值给 UserId
get_user(123)    # TypeError: int 不能赋值给 UserId

# NewType 的底层是函数
print(UserId(123))  # 123（运行时返回原值）
```

## Callable

```python
from typing import Callable

# Callable[[参数类型...], 返回类型]
def apply(func: Callable[[int], int], x: int) -> int:
    return func(x)


apply(lambda x: x * 2, 5)  # 10

# 可调用对象
def call_twice(func: Callable[[], None]) -> None:
    func()
    func()


# Callable 带额外参数
def with_retry(func: Callable[..., int], retries: int) -> int:
    for _ in range(retries):
        try:
            return func()
        except:
            pass
    raise RuntimeError("All retries failed")


# 泛型 Callable
F = TypeVar('F', bound=Callable[..., object])
```

## 泛型自动引用

```python
from typing import Generic, TypeVar

T = TypeVar('T')


# 泛型类方法自动引用
class Stack(Generic[T]):
    def __init__(self) -> None:
        self._items: list[T] = []

    def push(self, item: T) -> None:
        self._items.append(item)

    def pop(self) -> T:
        return self._items.pop()

    def peek(self) -> T:
        return self._items[-1]


# self 返回类型
from typing import TypeVar

T = TypeVar('T', bound='MyClass')


class MyClass:
    def copy(self: T) -> T:
        return MyClass()
```

## 类型守卫

```python
from typing import TypeGuard

# TypeGuard 帮助类型收窄
def is_str_list(val: list[object]) -> TypeGuard[list[str]]:
    return all(isinstance(x, str) for x in val)


def process(items: list[object]) -> None:
    if is_str_list(items):
        # 类型检查器知道 items 是 list[str]
        print(items.upper())  # 每个元素都有 .upper()
    else:
        print("Not a string list")


# Overload 用于同函数多签名
from typing import overload


@overload
def process(x: int) -> int: ...
@overload
def process(x: str) -> str: ...


def process(x: int | str) -> int | str:
    return x
```

## 泛型结构

```python
from typing import TypeVar, Generic

# 泛型函数
def identity[T](x: T) -> T:
    return x


# 泛型数据类
from dataclasses import dataclass

@dataclass
class Pair[T, U]:
    first: T
    second: U


# 泛型上下文管理器
from typing import ContextManager
from contextlib import contextmanager


@contextmanager
def managed_resource[T](resource: T) -> Generic[T]:
    try:
        yield resource
    finally:
        pass
```

## 常见模式

```python
# 1. 返回类型为 None 的函数
from typing import NoReturn

def fatal_error() -> NoReturn:
    raise RuntimeError("Fatal error")


# 2. 延迟类型（避免循环导入）
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from .models import User

def get_user(user_id: int) -> "User":  # 引号避免循环导入
    ...


# 3. 泛型映射
from typing import Mapping, Iterable

def sum_values(m: Mapping[str, int]) -> int:
    return sum(m.values())


def to_list(i: Iterable[T]) -> list[T]:
    return list(i)


# 4. 回调类型
from typing import Callable

Callback: TypeAlias = Callable[[int, str], None]
```

## 常见陷阱

```python
# 陷阱1：泛型容器与 list[int] vs List[int]
# Python 3.9+ 直接用 list[int]
def func(x: list[int]) -> None:  # 简洁
    pass


# 陷阱2：Mutable 容器默认值
# 错误
class W:
    def __init__(self, items: list[int] = []):  # 共享同一个列表
        self.items = items


# 正确：使用 None
class W:
    def __init__(self, items: list[int] | None = None):
        self.items = items or []


# 陷阱3：类型注解与实际类型不匹配
# 类型检查器不会阻止运行时错误
x: int = "hello"  # TypeError: str 不能赋值给 int（仅类型检查）


# 陷阱4：None vs Optional
# Optional[str] 等价于 str | None
# 但 None 本身不是类型
def foo(x: None) -> None:  # 只能是 None
    pass


# 陷阱5：忽略泛型参数
# 错误
def process(items: list) -> None:
    for item in items:  # item 类型未知
        print(item)


# 正确
def process(items: list[str]) -> None:
    for item in items:
        print(item)  # item 是 str
```
