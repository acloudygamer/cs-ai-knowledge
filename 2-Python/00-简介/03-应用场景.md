### 机器学习示例

```python
# 使用 scikit-learn 进行分类
from sklearn.ensemble import RandomForestClassifier
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split

# 加载数据
iris = load_iris()
X_train, X_test, y_train, y_test = train_test_split(
    iris.data, iris.target, test_size=0.2
)

# 训练模型
clf = RandomForestClassifier(n_estimators=100)
clf.fit(X_train, y_train)

# 预测
accuracy = clf.score(X_test, y_test)
print(f"Accuracy: {accuracy:.2f}")
```

### 深度学习示例

```python
# 使用 PyTorch 定义神经网络
import torch
import torch.nn as nn

class SimpleNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(784, 128)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(128, 10)

    def forward(self, x):
        x = x.view(-1, 784)
        x = self.fc1(x)
        x = self.relu(x)
        x = self.fc2(x)
        return x

model = SimpleNN()
optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
criterion = nn.CrossEntropyLoss()
```

### FastAPI 示例

```python
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()


class Item(BaseModel):
    name: str
    price: float
    tags: list[str] = []


items: dict[str, Item] = {}


@app.post("/items/{item_id}")
async def create_item(item_id: str, item: Item):
    items[item_id] = item
    return item


@app.get("/items/{item_id}")
async def get_item(item_id: str):
    if item_id not in items:
        raise HTTPException(status_code=404, detail="Item not found")
    return items[item_id]
```

## 自动化与脚本

Python 是编写自动化脚本的首选语言：

### 文件批处理

```python
import os
import shutil
from pathlib import Path

# 批量重命名
def batch_rename(directory: str, pattern: str, replacement: str):
    """批量重命名文件"""
    path = Path(directory)
    for file in path.glob("*"):
        if pattern in file.name:
            new_name = file.name.replace(pattern, replacement)
            file.rename(file.parent / new_name)

# 批量压缩
def batch_compress(directory: str, output: str):
    """将目录压缩为 zip"""
    shutil.make_archive(output, "zip", directory)
```

import scrapy


class QuoteSpider(scrapy.Spider):
    name = "quotes"
    start_urls = ["http://quotes.toscrape.com"]

    def parse(self, response):
        for quote in response.css("div.quote"):
            yield {
                "text": quote.css("span.text::text").get(),
                "author": quote.css("span small::text").get(),
            }

### 异步爬虫

```python
import aiohttp
import asyncio

async def fetch(session, url):
    async with session.get(url) as response:
        return await response.text()

async def main():
    urls = ["https://example.com" for _ in range(10)]
    async with aiohttp.ClientSession() as session:
        htmls = await asyncio.gather(*[fetch(session, url) for url in urls])

asyncio.run(main())
```

|------|----------------|------|
| Web 开发 | 是 | 框架成熟，开发效率高 |
| 数据科学/ML | 是 | 生态最完善 |
| 自动化脚本 | 是 | 语法简洁，跨平台 |
| 系统工具 | 一般 | 可用但不是最佳 |
| 游戏开发 | 一般 | 性能受限 |
| 移动应用 | 否 | 不推荐 |

Python 的设计哲学使其特别适合快速原型开发、数据处理和自动化任务。
