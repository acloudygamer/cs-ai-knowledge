# 观察者模式

观察者模式定义对象间的一对多依赖关系，当主题状态改变时，所有观察者收到通知。适用于事件系统、消息订阅等场景。

## 基本实现

Subject 维护观察者列表，提供 attach/detach/notify 方法。

### 参考样例

```python
from abc import ABC, abstractmethod
from typing import List

class Observer(ABC):
    @abstractmethod
    def update(self, message: str):
        pass

class Subject:
    def __init__(self):
        self._observers: List[Observer] = []

    def attach(self, observer: Observer):
        if observer not in self._observers:
            self._observers.append(observer)

    def detach(self, observer: Observer):
        if observer in self._observers:
            self._observers.remove(observer)

    def notify(self, message: str):
        for observer in self._observers:
            observer.update(message)

class NewsAgency(Subject):
    def __init__(self, name: str):
        super().__init__()
        self.name = name
        self._news = ""

    @property
    def news(self):
        return self._news

    @news.setter
    def news(self, value: str):
        self._news = value
        self.notify(f"{self.name}: {value}")

class NewsChannel(Observer):
    def __init__(self, name: str):
        self.name = name

    def update(self, message: str):
        print(f"{self.name} received: {message}")

# 使用
agency = NewsAgency("CNN")

channel1 = NewsChannel("Channel 1")
channel2 = NewsChannel("Channel 2")

agency.attach(channel1)
agency.attach(channel2)

agency.news = "Breaking: Python 4.0 released!"
# Channel 1 received: CNN: Breaking: Python 4.0 released!
# Channel 2 received: CNN: Breaking: Python 4.0 released!

agency.detach(channel1)

agency.news = "Update: Python 4.0 features revealed"
# Channel 2 received: CNN: Update: Python 4.0 features revealed
```

回调式实现使用 Callable 替代 Observer 接口，更轻量。

### 参考样例

```python
from typing import Callable

class EventManager:
    def __init__(self):
        self._callbacks: dict[str, list[Callable]] = {}

    def subscribe(self, event: str, callback: Callable):
        if event not in self._callbacks:
            self._callbacks[event] = []
        self._callbacks[event].append(callback)

    def unsubscribe(self, event: str, callback: Callable):
        if event in self._callbacks:
            self._callbacks[event].remove(callback)

    def emit(self, event: str, data=None):
        if event in self._callbacks:
            for callback in self._callbacks[event]:
                callback(data)

# 使用
events = EventManager()

def on_click(data):
    print(f"Button clicked: {data}")

def on_hover(data):
    print(f"Button hovered: {data}")

events.subscribe("click", on_click)
events.subscribe("click", on_hover)

events.emit("click", {"button": "submit"})
# Button clicked: {'button': 'submit'}
# Button hovered: {'button': 'submit'}
```

观察者模式实际应用：用户操作触发多系统响应（日志、邮件、分析追踪）。

### 参考样例

```python
from abc import ABC, abstractmethod
from datetime import datetime
from typing import List

class UserObserver(ABC):
    @abstractmethod
    def on_user_action(self, user_id: str, action: str, data: dict):
        pass

class AuditLog(UserObserver):
    def on_user_action(self, user_id: str, action: str, data: dict):
        print(f"[AUDIT] {datetime.now()}: User {user_id} performed {action}")

class EmailNotifier(UserObserver):
    def on_user_action(self, user_id: str, action: str, data: dict):
        if action == "purchase":
            print(f"[EMAIL] Sending receipt to user {user_id}")

class AnalyticsTracker(UserObserver):
    def on_user_action(self, user_id: str, action: str, data: dict):
        print(f"[ANALYTICS] Tracking event: {action} for user {user_id}")

class UserService:
    def __init__(self):
        self._observers: List[UserObserver] = []

    def add_observer(self, observer: UserObserver):
        self._observers.append(observer)

    def remove_observer(self, observer: UserObserver):
        self._observers.remove(observer)

    def _notify(self, user_id: str, action: str, data: dict):
        for observer in self._observers:
            observer.on_user_action(user_id, action, data)

    def register(self, username: str, email: str):
        print(f"User {username} registered")
        self._notify(username, "register", {"email": email})

    def purchase(self, user_id: str, item: str, amount: float):
        print(f"User {user_id} purchased {item} for {amount}")
        self._notify(user_id, "purchase", {"item": item, "amount": amount})

# 使用
service = UserService()
service.add_observer(AuditLog())
service.add_observer(EmailNotifier())
service.add_observer(AnalyticsTracker())

service.register("alice", "alice@example.com")
# [AUDIT] User alice registered
# [ANALYTICS] Tracking event: register for user alice

service.purchase("alice", "Premium Plan", 9.99)
# [AUDIT] User alice purchased Premium Plan for 9.99
# [EMAIL] Sending receipt to user alice
# [ANALYTICS] Tracking event: purchase for user alice
```

## 观察者模式的注意事项

```python
# 1. 避免循环依赖
# Subject 和 Observer 互相引用会导致内存泄漏

# 2. 考虑使用弱引用
import weakref

class WeakObserver:
    def __init__(self, callback):
        self._ref = weakref.ref(callback)

    def update(self, message):
        callback = self._ref()
        if callback:
            callback(message)

# 3. 异步通知避免阻塞
import asyncio

class AsyncSubject:
    async def notify(self, message: str):
        tasks = [asyncio.create_task(obs.update(message))
                 for obs in self._observers]
        await asyncio.gather(*tasks)
```
