# 字符串模式匹配

### 解决什么问题
在文本串中查找模式串的出现位置，是字符串处理的核心问题。典型应用：搜索引擎、DNA序列比对、日志处理、敏感词过滤等。

### 核心概念
- 朴素匹配：O(nm)，暴力遍历
- KMP：O(n+m)，利用已匹配信息避免回溯
- Rabin-Karp：O(n+m) 平均，使用哈希加速
- AC自动机：多模式匹配，O(n+m)

## KMP 算法

### 前缀函数（Failure Function）

对于模式串 p，前缀函数 pi[i] 表示 p[0..i] 中最长相等前后缀长度。

```python
def build_prefix_function(p):
    """
    构建前缀函数
    p: 模式串
    返回: 前缀函数数组 pi
    """
    n = len(p)
    pi = [0] * n
    for i in range(1, n):
        j = pi[i - 1]
        while j > 0 and p[i] != p[j]:
            j = pi[j - 1]
        if p[i] == p[j]:
            j += 1
        pi[i] = j
    return pi


def kmp_search(text, pattern):
    """
    KMP 搜索
    text: 主串
    pattern: 模式串
    返回: 第一个匹配的位置，或 -1
    """
    if not pattern:
        return 0
    if not text:
        return -1

    n, m = len(text), len(pattern)
    pi = build_prefix_function(pattern)
    j = 0  # pattern 的当前匹配长度

    for i in range(n):
        while j > 0 and text[i] != pattern[j]:
            j = pi[j - 1]
        if text[i] == pattern[j]:
            j += 1
        if j == m:
            return i - m + 1  # 匹配成功

    return -1


# 查找所有匹配位置
def kmp_find_all(text, pattern):
    """返回所有匹配位置列表"""
    if not pattern:
        return list(range(len(text) + 1))
    if not text:
        return []

    n, m = len(text), len(pattern)
    pi = build_prefix_function(pattern)
    j = 0
    result = []

    for i in range(n):
        while j > 0 and text[i] != pattern[j]:
            j = pi[j - 1]
        if text[i] == pattern[j]:
            j += 1
        if j == m:
            result.append(i - m + 1)
            j = pi[j - 1]  # 继续查找下一个匹配

    return result
```

- 字符串子串查找
- 重复子串检测（前缀函数值的应用）
- 串的模式匹配问题

```python
# 例：字符串中查找重复出现的子串
def find_repeated_substring(text):
    """
    查找出现多次的子串
    利用前缀函数：如果 n % (n - pi[n-1]) == 0，
    则子串 text[0:n-pi[n-1]] 重复出现
    """
    n = len(text)
    pi = build_prefix_function(text)
    print("前缀函数:", pi)

    # 检查是否有整周期的重复
    k = n - pi[-1]
    if n % k == 0 and n // k > 1:
        repeat_times = n // k
        substring = text[:k]
        print(f"子串 '{substring}' 重复 {repeat_times} 次")
        return substring
    return None
```

## Rabin-Karp 算法

### 滚动哈希

使用多项式哈希，将字符串映射为整数，利用滚动哈希实现 O(1) 窗口滑动。

```python
def rabin_karp(text, pattern, base=256, mod=101):
    """
    Rabin-Karp 字符串匹配
    base: 进制基数
    mod: 模数（取大素数以减少冲突）
    返回: 第一个匹配的位置，或 -1
    """
    if not pattern:
        return 0
    if not text:
        return -1

    n, m = len(text), len(pattern)
    pattern_hash = 0
    text_hash = 0
    h = 1  # base^(m-1) 的值

    # 计算 base^(m-1) % mod
    for _ in range(m - 1):
        h = (h * base) % mod

    # 计算模式串和主串第一个窗口的哈希值
    for i in range(m):
        pattern_hash = (pattern_hash * base + ord(pattern[i])) % mod
        text_hash = (text_hash * base + ord(text[i])) % mod

    # 滑动窗口
    for i in range(n - m + 1):
        # 哈希匹配时再逐字符确认（避免假阳性）
        if pattern_hash == text_hash:
            if text[i:i + m] == pattern:
                return i

        # 计算下一个窗口的哈希值
        if i < n - m:
            text_hash = ((text_hash - ord(text[i]) * h) * base +
                         ord(text[i + m])) % mod
            if text_hash < 0:
                text_hash += mod

    return -1
```

### 暴力对比（用于哈希冲突时确认）

```python
def rabin_karp_robust(text, pattern, base=256, mod=2**31):
    """
    增强版 Rabin-Karp：使用更大的模数减少冲突
    """
    if not pattern:
        return 0
    if not text:
        return -1

    n, m = len(text), len(pattern)
    pattern_hash = 0
    text_hash = 0
    h = pow(base, m - 1, mod)

    for i in range(m):
        pattern_hash = (pattern_hash * base + ord(pattern[i])) % mod
        text_hash = (text_hash * base + ord(text[i])) % mod

    for i in range(n - m + 1):
        if pattern_hash == text_hash:
            # 逐字符确认
            match = True
            for j in range(m):
                if text[i + j] != pattern[j]:
                    match = False
                    break
            if match:
                return i

        if i < n - m:
            text_hash = ((text_hash - ord(text[i]) * h) * base +
                         ord(text[i + m])) % mod

    return -1
```

| 特性 | KMP | Rabin-Karp |

| 空间复杂度 | O(m) | O(1) |
| 预处理时间 | O(m) | O(m) |
| 适用场景 | 精确匹配 | 多模式匹配、哈希验证 |

```python
def repeated_substring_pattern(s):
    """
    判断字符串是否由重复子串构成
    方法：KMP 前缀函数
    """
    n = len(s)
    pi = [0] * n
    for i in range(1, n):
        j = pi[i - 1]
        while j > 0 and s[i] != s[j]:
            j = pi[j - 1]
        if s[i] == s[j]:
            j += 1
        pi[i] = j

    # 如果整个字符串有真前缀等于后缀，且长度能被整除
    k = n - pi[-1]
    return k < n and n % k == 0
```

```python
def rotate_string(s, goal):
    """
    判断 goal 是否为 s 旋转而成
    方法：将 s 拼接自身，再用 KMP 查找 goal
    """
    if len(s) != len(goal):
        return False
    return kmp_search(s + s, goal) != -1
```

## Manacher 算法

O(n) 时间复杂度求最长回文子串。

### 核心思想

- 奇回文和偶回文统一处理：插入分隔符
- 维护右边界最靠右的回文中心及其半径
- 利用已计算信息加速后续计算

```python
def manacher(s):
    """
    Manacher 算法：O(n) 求最长回文子串
    返回: (最长回文子串, 起始位置, 长度)
    """
    if not s:
        return "", -1, 0

    # 插入分隔符，统一处理奇偶回文
    # 例如 "abc" -> "#a#b#c#"
    t = '#' + '#'.join(s) + '#'
    n = len(t)

    # p[i] = 以 t[i] 为中心的回文半径（包含中心）
    p = [0] * n
    c = r = 0  # center, right boundary

    for i in range(n):
        # 计算初始半径
        if i < r:
            mirror = 2 * c - i
            p[i] = min(r - i, p[mirror])

        # 尝试扩展回文
        while i - p[i] - 1 >= 0 and i + p[i] + 1 < n:
            if t[i - p[i] - 1] == t[i + p[i] + 1]:
                p[i] += 1
            else:
                break

        # 更新中心
        if i + p[i] > r:
            c = i
            r = i + p[i]

    # 找最长回文
    max_len = max(p)
    center_idx = p.index(max_len)

    # 转换为原字符串索引
    start = (center_idx - max_len) // 2
    return s[start:start + max_len], start, max_len


def manacher_all(s):
    """
    返回所有回文子串
    """
    if not s:
        return []

    t = '#' + '#'.join(s) + '#'
    n = len(t)
    p = [0] * n
    c = r = 0
    result = []

    for i in range(n):
        if i < r:
            mirror = 2 * c - i
            p[i] = min(r - i, p[mirror])

        while i - p[i] - 1 >= 0 and i + p[i] + 1 < n:
            if t[i - p[i] - 1] == t[i + p[i] + 1]:
                p[i] += 1
            else:
                break

        if i + p[i] > r:
            c = i
            r = i + p[i]

        # 记录回文（去重）
        radius = p[i]
        if radius > 0:
            start = (i - radius) // 2
            length = radius
            result.append((s[start:start + length], start, length))

    return result
```

### Manacher vs 其他方法

| 方法 | 时间复杂度 | 空间复杂度 | 特点 |
|------|-----------|-----------|------|
| 中心扩展 | O(n^2) | O(1) | 简单直观 |
| 动态规划 | O(n^2) | O(n^2) | 可预处理 |
| Manacher | O(n) | O(n) | 最优但稍复杂 |

## Aho-Corasick 自动机 (AC 自动机)

### 核心概念

AC 自动机是一种多模式字符串匹配算法，在 O(n + m + z) 时间内找出所有模式串在文本中的出现位置（n=文本长度，m=模式串总长度，z=匹配次数）。

核心思想：将所有模式串构建为 Trie，然后在 Trie 上构建失败指针（类似 KMP 的前缀函数）。

### 构建过程

1. **插入所有模式串到 Trie**
2. **构建失败指针（Failure Link）**：
   - 根节点的失败指针指向自己
   - 其他节点的失败指针：沿父节点的失败指针向上，直到某个节点有对应字符的边
3. **构建输出指针（Output Link）**：指向最近的可输出匹配的模式节点

### 实现

```python
from collections import deque


def build_ac_automaton(patterns):
    """
    构建 AC 自动机
    patterns: 模式串列表
    返回: (trie, fail_links)
    """
    # 节点列表
    nodes = [{'children': {}, 'fail': 0, 'output': []}]

    for pid, pattern in enumerate(patterns):
        node = 0
        for ch in pattern:
            if ch not in nodes[node]['children']:
                nodes[node]['children'][ch] = len(nodes)
                nodes.append({'children': {}, 'fail': 0, 'output': []})
            node = nodes[node]['children'][ch]
        nodes[node]['output'].append(pid)

    # BFS 构建失败指针
    from collections import deque
    queue = deque()

    # 第一层：直接将第一层节点的 fail 指向根
    for ch, next_node in nodes[0]['children'].items():
        nodes[next_node]['fail'] = 0
        queue.append(next_node)

    while queue:
        current = queue.popleft()

        for ch, next_node in nodes[current]['children'].items():
            queue.append(next_node)

            # 追溯失败指针
            fail = nodes[current]['fail']
            while fail > 0 and ch not in nodes[fail]['children']:
                fail = nodes[fail]['fail']

            nodes[next_node]['fail'] = nodes[fail]['children'].get(ch, 0)

            # 合并输出
            if nodes[nodes[next_node]['fail']]['output']:
                nodes[next_node]['output'].extend(
                    nodes[nodes[next_node]['fail']]['output']
                )

    return nodes


def ac_search(text, patterns):
    """
    AC 自动机多模式匹配
    返回: [(模式串索引, 匹配位置), ...]
    """
    nodes = build_ac_automaton(patterns)
    result = []
    state = 0

    for i, ch in enumerate(text):
        # 沿着失败指针直到有对应转移或到达根
        while state > 0 and ch not in nodes[state]['children']:
            state = nodes[state]['fail']

        state = nodes[state]['children'].get(ch, 0)

        # 输出所有匹配的模式串
        for pattern_id in nodes[state]['output']:
            result.append((pattern_id, i - len(patterns[pattern_id]) + 1))

    return result


# 使用示例
def ac_demo():
    patterns = ["he", "she", "his", "hers"]
    text = "ahishers"

    result = ac_search(text, patterns)
    for pid, pos in result:
        print(f"Pattern '{patterns[pid]}' found at position {pos}")

    # 输出:
    # Pattern 'his' found at position 1
    # Pattern 'she' found at position 2
    # Pattern 'he' found at position 3
    # Pattern 'hers' found at position 4
```

### AC 自动机 vs KMP

| 特性 | KMP | AC 自动机 |
|------|-----|-----------|
| 适用场景 | 单模式匹配 | 多模式匹配 |
| 时间复杂度 | O(n + m) | O(n + m + z) |
| 空间复杂度 | O(m) | O(m σ)，σ 为字符集大小 |
| 实现难度 | 中等 | 较难 |

### 进阶：AC 自动机 + DP

```python
def ac_dp(text, patterns, cost):
    """
    AC 自动机 + DP：带代价的字符串匹配
    cost: 每个模式串匹配的代价
    """
    nodes = build_ac_automaton(patterns)
    n = len(text)

    # dp[i][state] = 到达位置 i、状态为 state 的最小代价
    INF = float('inf')
    dp = [[INF] * len(nodes) for _ in range(n + 1)]
    dp[0][0] = 0

    for i in range(n):
        for state in range(len(nodes)):
            if dp[i][state] == INF:
                continue

            ch = text[i]
            next_state = state

            # 转移
            while next_state > 0 and ch not in nodes[next_state]['children']:
                next_state = nodes[next_state]['fail']
            next_state = nodes[next_state]['children'].get(ch, 0)

            # 不匹配任何模式串
            dp[i + 1][next_state] = min(dp[i + 1][next_state], dp[i][state])

            # 匹配模式串（使用输出指针）
            for pid in nodes[next_state]['output']:
                if i + 1 - len(patterns[pid]) >= 0:
                    # 可以在这里处理匹配事件
```
