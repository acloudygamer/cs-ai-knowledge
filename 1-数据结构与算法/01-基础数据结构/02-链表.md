# 链表

### 解决什么问题
动态数据结构，适用于频繁插入删除、不知道数据规模、需要节省内存的场景。与数组互补，链表 O(1) 插入删除但 O(n) 访问。

### 核心概念
- 单链表：节点有 val 和 next 指针
- 双链表：节点有 prev 和 next，支持双向遍历
- 哨兵节点：简化边界处理
- 快慢指针：检测环、找中点、找倒数第 K 个

### 怎么用

### ListNode 类定义

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
```

## 常用操作

### 反转链表

```python
# 迭代反转
def reverse_list(head):
    prev, curr = None, head
    while curr:
        next_temp = curr.next
        curr.next = prev
        prev = curr
        curr = next_temp
    return prev

# 递归反转
def reverse_recursive(head):
    if not head or not head.next:
        return head
    new_head = reverse_recursive(head.next)
    head.next.next = head
    head.next = None
    return new_head
```

### 环检测 - Floyd算法

```python
def has_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            return True
    return False

# 找环起点
def detect_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            slow = head
            while slow != fast:
                slow = slow.next
                fast = fast.next
            return slow
    return None
```

### 合并有序链表

```python
def merge_two_lists(l1, l2):
    dummy = ListNode(0)
    curr = dummy
    while l1 and l2:
        if l1.val <= l2.val:
            curr.next = l1
            l1 = l1.next
        else:
            curr.next = l2
            l2 = l2.next
        curr = curr.next
    curr.next = l1 or l2
    return dummy.next
```

### 删除倒数第N个节点

```python
def remove_nth_from_end(head, n):
    dummy = ListNode(0, head)
    slow = fast = dummy
    for _ in range(n + 1):
        fast = fast.next
    while fast:
        slow = slow.next
        fast = fast.next
    slow.next = slow.next.next
    return dummy.next
```

## 链表 vs 数组

| 操作 | 链表 | 数组 |
|------|------|------|
| 随机访问 | O(n) | O(1) |
| 头部插入 | O(1) | O(n) |
| 尾部插入 | 摊还 O(1) | O(1) |
| 中间插入 | O(1)* | O(n) |

## 双向链表

双向链表每个节点包含指向前后两个节点的指针，支持 O(1) 的前向和后向遍历。

### 节点定义

```python
class DListNode:
    def __init__(self, val=0, prev=None, next=None):
        self.val = val
        self.prev = prev
        self.next = next
```

### 基本操作

```python
class DoublyLinkedList:
    def __init__(self):
        self.head = DListNode()  # 虚拟头节点
        self.tail = DListNode()  # 虚拟尾节点
        self.head.next = self.tail
        self.tail.prev = self.head
        self.size = 0

    def append(self, val):
        """尾部添加 O(1)"""
        node = DListNode(val)
        node.prev = self.tail.prev
        node.next = self.tail
        self.tail.prev.next = node
        self.tail.prev = node
        self.size += 1

    def prepend(self, val):
        """头部添加 O(1)"""
        node = DListNode(val)
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
        self.size += 1

    def delete(self, node):
        """删除节点 O(1)"""
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1

    def get(self, index):
        """获取节点 O(n)"""
        if index < 0 or index >= self.size:
            return -1
        if index < self.size // 2:
            curr = self.head.next
            for _ in range(index):
                curr = curr.next
        else:
            curr = self.tail.prev
            for _ in range(self.size - index - 1):
                curr = curr.prev
        return curr.val
```

## 循环链表

循环链表是链表的变种，尾节点的 next 指向头节点（单向）或头节点的 prev 指向尾节点（双向）。

### 单向循环链表

```python
class CircularListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class CircularLinkedList:
    def __init__(self):
        self.head = None
        self.size = 0

    def append(self, val):
        """O(n) - 需要找到尾节点"""
        new_node = CircularListNode(val)
        if not self.head:
            self.head = new_node
            new_node.next = self.head  # 指向自己
        else:
            curr = self.head
            while curr.next != self.head:  # 找到尾节点
                curr = curr.next
            curr.next = new_node
            new_node.next = self.head
        self.size += 1

    def traverse(self, k=None):
        """遍历 k 个节点，不指定则遍历全部"""
        if not self.head:
            return []
        result = []
        curr = self.head
        while True:
            result.append(curr.val)
            curr = curr.next
            if k and len(result) >= k:
                break
            if curr == self.head:
                break
        return result
```

### 约瑟夫问题（Josephus）

**LeetCode 1823**: 找到胜利者的编号。

```python
def find_the_winner(n, k):
    """
    模拟循环链表
    时间复杂度: O(n*k) 每次数 k 个，数 n 次
    """
    if n == 1:
        return 1

    # 创建循环链表
    class Node:
        def __init__(self, val):
            self.val = val
            self.next = None

    head = Node(1)
    curr = head
    for i in range(2, n + 1):
        curr.next = Node(i)
        curr = curr.next
    curr.next = head  # 形成环

    # 删除节点直到只剩一个
    count = 0
    while curr.next != curr:
        count += 1
        if count == k - 1:
            # 删除下一个节点
            curr.next = curr.next.next
            count = 0
        curr = curr.next

    return curr.val


def find_the_winner_math(n, k):
    """
    数学解法 - 约瑟夫问题的递推公式
    f(1) = 0
    f(i) = (f(i-1) + k) % i
    最终答案需要 +1 转为 1-indexed
    """
    result = 0
    for i in range(2, n + 1):
        result = (result + k) % i
    return result + 1
```

## 链表技巧总结

### 虚拟头节点

处理头节点边界情况时，使用虚拟头节点可以简化代码。

```python
def remove_elements_dummy(head, val):
    """使用虚拟头节点"""
    dummy = ListNode(0, head)
    curr = dummy

    while curr.next:
        if curr.next.val == val:
            curr.next = curr.next.next
        else:
            curr = curr.next

    return dummy.next
```

### 合并 K 个有序链表

```python
import heapq


def merge_k_lists(lists):
    """
    最小堆优化
    时间复杂度: O(n log k)，n 是总节点数，k 是链表数
    """
    heap = []
    for i, lst in enumerate(lists):
        if lst:
            heapq.heappush(heap, (lst.val, i, lst))

    dummy = ListNode()
    curr = dummy

    while heap:
        val, i, node = heapq.heappop(heap)
        curr.next = node
        curr = curr.next
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))

    return dummy.next
```

### 复杂链表复制

**LeetCode 138**: 复制带随机指针的链表。

```python
def copy_random_pointer(head):
    """
    三步法：
    1. 复制每个节点并插入原节点后面
    2. 设置随机指针
    3. 分离两个链表
    """
    if not head:
        return None

    # Step 1: 复制节点并插入
    curr = head
    while curr:
        new_node = Node(curr.val)
        new_node.next = curr.next
        curr.next = new_node
        curr = new_node.next

    # Step 2: 设置随机指针
    curr = head
    while curr:
        if curr.random:
            curr.next.random = curr.random.next
        curr = curr.next.next

    # Step 3: 分离两个链表
    dummy = head.next
    curr = head
    while curr:
        copy = curr.next
        curr.next = copy.next
        if copy.next:
            copy.next = copy.next.next
        curr = curr.next

    return dummy
```
