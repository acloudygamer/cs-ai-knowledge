# 链表

### ListNode 类定义

```python
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next
```

## 核心要点

- 动态数据结构，节点通过指针连接
- O(1)插入/删除，O(n)访问
- 无需连续内存空间

## 常用操作

### 反转链表

```python
# 迭代反转
def reverse_list(head):
    prev, curr = None, head
    while curr:
        next_temp = curr.next
        curr.next = prev
        prev = curr
        curr = next_temp
    return prev

# 递归反转
def reverse_recursive(head):
    if not head or not head.next:
        return head
    new_head = reverse_recursive(head.next)
    head.next.next = head
    head.next = None
    return new_head
```

### 环检测 - Floyd算法

```python
def has_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            return True
    return False

# 找环起点
def detect_cycle(head):
    slow = fast = head
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            slow = head
            while slow != fast:
                slow = slow.next
                fast = fast.next
            return slow
    return None
```

### 合并有序链表

```python
def merge_two_lists(l1, l2):
    dummy = ListNode(0)
    curr = dummy
    while l1 and l2:
        if l1.val <= l2.val:
            curr.next = l1
            l1 = l1.next
        else:
            curr.next = l2
            l2 = l2.next
        curr = curr.next
    curr.next = l1 or l2
    return dummy.next
```

### 删除倒数第N个节点

```python
def remove_nth_from_end(head, n):
    dummy = ListNode(0, head)
    slow = fast = dummy
    for _ in range(n + 1):
        fast = fast.next
    while fast:
        slow = slow.next
        fast = fast.next
    slow.next = slow.next.next
    return dummy.next
```

## 链表 vs 数组

| 操作 | 链表 | 数组 |
|------|------|------|
| 随机访问 | O(n) | O(1) |
| 头部插入 | O(1) | O(n) |
| 尾部插入 | 摊还 O(1) | O(1) |
| 中间插入 | O(1)* | O(n) |

## 双向链表

双向链表每个节点包含指向前后两个节点的指针，支持 O(1) 的前向和后向遍历。

### 节点定义

```python
class DListNode:
    def __init__(self, val=0, prev=None, next=None):
        self.val = val
        self.prev = prev
        self.next = next
```

### 基本操作

```python
class DoublyLinkedList:
    def __init__(self):
        self.head = DListNode()  # 虚拟头节点
        self.tail = DListNode()  # 虚拟尾节点
        self.head.next = self.tail
        self.tail.prev = self.head
        self.size = 0

    def append(self, val):
        """尾部添加 O(1)"""
        node = DListNode(val)
        node.prev = self.tail.prev
        node.next = self.tail
        self.tail.prev.next = node
        self.tail.prev = node
        self.size += 1

    def prepend(self, val):
        """头部添加 O(1)"""
        node = DListNode(val)
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node
        self.size += 1

    def delete(self, node):
        """删除节点 O(1)"""
        node.prev.next = node.next
        node.next.prev = node.prev
        self.size -= 1

    def get(self, index):
        """获取节点 O(n)"""
        if index < 0 or index >= self.size:
            return -1
        if index < self.size // 2:
            curr = self.head.next
            for _ in range(index):
                curr = curr.next
        else:
            curr = self.tail.prev
            for _ in range(self.size - index - 1):
                curr = curr.prev
        return curr.val
```

## 循环链表

循环链表是链表的变种，尾节点的 next 指向头节点（单向）或头节点的 prev 指向尾节点（双向）。

### 单向循环链表

```python
class CircularListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class CircularLinkedList:
    def __init__(self):
        self.head = None
        self.size = 0

    def append(self, val):
        """O(n) - 需要找到尾节点"""
        new_node = CircularListNode(val)
        if not self.head:
            self.head = new_node
            new_node.next = self.head  # 指向自己
        else:
            curr = self.head
            while curr.next != self.head:  # 找到尾节点
                curr = curr.next
            curr.next = new_node
            new_node.next = self.head
        self.size += 1

    def traverse(self, k=None):
        """遍历 k 个节点，不指定则遍历全部"""
        if not self.head:
            return []
        result = []
        curr = self.head
        while True:
            result.append(curr.val)
            curr = curr.next
            if k and len(result) >= k:
                break
            if curr == self.head:
                break
        return result
```

### 约瑟夫问题

**LeetCode 1823**: 找到胜利者的编号。

```python
def find_the_winner(n, k):
    """
    模拟循环链表
    时间复杂度: O(n*k) 每次数 k 个，数 n 次
    """
    if n == 1:
        return 1

    # 创建循环链表
    class Node:
        def __init__(self, val):
            self.val = val
            self.next = None

    head = Node(1)
    curr = head
    for i in range(2, n + 1):
        curr.next = Node(i)
        curr = curr.next
    curr.next = head  # 形成环

    # 删除节点直到只剩一个
    count = 0
    while curr.next != curr:
        count += 1
        if count == k - 1:
            # 删除下一个节点
            curr.next = curr.next.next
            count = 0
        curr = curr.next

    return curr.val


def find_the_winner_math(n, k):
    """
    数学解法 - 约瑟夫问题的递推公式
    f(1) = 0
    f(i) = (f(i-1) + k) % i
    最终答案需要 +1 转为 1-indexed
    """
    result = 0
    for i in range(2, n + 1):
        result = (result + k) % i
    return result + 1
```

## LRU 缓存

Least Recently Used (LRU) 缓存是一种常用的缓存策略，最近使用的元素放在最前面，当缓存满时驱逐最久未使用的元素。

### 标准实现：哈希表 + 双向链表

```python
class LRUCache:
    """
    LRU 缓存实现

    数据结构：
    - 双向链表：维护访问顺序，头部是最近使用的，尾部是最久未使用的
    - 哈希表：O(1) 的 key 到节点的映射

    时间复杂度: O(1) 的 get 和 put
    空间复杂度: O(capacity)
    """

    class Node:
        def __init__(self, key=0, val=0):
            self.key = key
            self.val = val
            self.prev = None
            self.next = None

    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}  # key -> Node

        # 虚拟头尾节点简化边界处理
        self.head = self.Node()  # head.next 是最近使用的
        self.tail = self.Node()  # tail.prev 是最久未使用的
        self.head.next = self.tail
        self.tail.prev = self.head

    def _remove(self, node):
        """从链表中移除节点 O(1)"""
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_to_head(self, node):
        """添加到链表头部 O(1)"""
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node

    def _move_to_head(self, node):
        """将节点移到头部 O(1)"""
        self._remove(node)
        self._add_to_head(node)

    def get(self, key: int) -> int:
        """
        获取缓存值
        - 如果存在，将节点移到头部，返回值
        - 如果不存在，返回 -1
        """
        if key not in self.cache:
            return -1

        node = self.cache[key]
        self._move_to_head(node)
        return node.val

    def put(self, key: int, value: int) -> None:
        """
        设置缓存值
        - 如果存在，更新值并移到头部
        - 如果不存在：
          - 缓存满时，删除最久未使用的（tail.prev）
          - 添加新节点到头部
        """
        if key in self.cache:
            node = self.cache[key]
            node.val = value
            self._move_to_head(node)
        else:
            if len(self.cache) >= self.capacity:
                # 驱逐最久未使用的
                lru = self.tail.prev
                self._remove(lru)
                del self.cache[lru.key]

            new_node = self.Node(key, value)
            self.cache[key] = new_node
            self._add_to_head(new_node)
```

### Python 字典 + OrderedDict 实现

```python
from collections import OrderedDict


class LRUCacheOrderedDict:
    """
    使用 Python 内置的 OrderedDict 实现
    move_to_end() 是 O(1)，popitem(last=False) 是 O(1)

    时间复杂度: O(1) 的 get 和 put
    空间复杂度: O(capacity)
    """

    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)  # 标记为最近使用
        return self.cache[key]

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
            self.cache[key] = value
        else:
            if len(self.cache) >= self.capacity:
                self.cache.popitem(last=False)  # 删除最旧的（开头）
            self.cache[key] = value
```

### LRU 缓存变种

#### LFU（最不经常使用）

```python
class LFUCache:
    """
    LFU 缓存：删除使用频率最低的元素
    频率相同时删除最久未使用的

    数据结构：
    - 哈希表：key -> Node
    - 字典：freq -> 双向链表（虚拟头节点，头部是最新使用的）
    - min_freq 记录当前最小频率

    时间复杂度: O(1) 的 get 和 put
    空间复杂度: O(capacity)
    """

    class Node:
        def __init__(self, key=0, val=0):
            self.key = key
            self.val = val
            self.freq = 1
            self.prev = self.next = None

    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}  # key -> Node
        self.freq_map = {}  # freq -> 虚拟头节点的双向链表
        self.min_freq = 0

    def _add_to_freq_list(self, node):
        """将节点添加到对应频率链表的头部 O(1)"""
        freq = node.freq
        if freq not in self.freq_map:
            # 创建虚拟头节点（不存储数据）
            head = self.Node()
            head.next = head
            head.prev = head
            self.freq_map[freq] = head

        # 插入到 freq 链表头部
        head = self.freq_map[freq]
        node.next = head.next
        node.prev = head
        head.next.prev = node
        head.next = node

    def _remove_from_freq_list(self, node):
        """从当前频率链表移除节点 O(1)"""
        node.prev.next = node.next
        node.next.prev = node.prev

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1

        node = self.cache[key]
        # 从旧频率链表移除
        self._remove_from_freq_list(node)

        # 更新频率
        node.freq += 1
        self._add_to_freq_list(node)

        # 更新最小频率
        if self.min_freq not in self.freq_map or \
           self.freq_map[self.min_freq].next == self.freq_map[self.min_freq]:
            self.min_freq = node.freq

        return node.val

    def put(self, key: int, value: int) -> None:
        if self.capacity == 0:
            return

        if key in self.cache:
            node = self.cache[key]
            node.val = value
            # 更新频率逻辑
            self._remove_from_freq_list(node)
            node.freq += 1
            self._add_to_freq_list(node)

            if self.min_freq not in self.freq_map or \
               self.freq_map[self.min_freq].next == self.freq_map[self.min_freq]:
                self.min_freq = node.freq
        else:
            if len(self.cache) >= self.capacity:
                # 驱逐最小频率的最旧节点（链表尾部）
                min_head = self.freq_map[self.min_freq]
                lru = min_head.prev  # 尾部是最旧的
                self._remove_from_freq_list(lru)
                del self.cache[lru.key]

            node = self.Node(key, value)
            self.cache[key] = node
            self._add_to_freq_list(node)
            self.min_freq = 1
```

## 链表技巧总结

### 虚拟头节点

处理头节点边界情况时，使用虚拟头节点可以简化代码。

```python
def remove_elements_dummy(head, val):
    """使用虚拟头节点"""
    dummy = ListNode(0, head)
    curr = dummy

    while curr.next:
        if curr.next.val == val:
            curr.next = curr.next.next
        else:
            curr = curr.next

    return dummy.next
```

### 合并 K 个有序链表

```python
import heapq


def merge_k_lists(lists):
    """
    最小堆优化
    时间复杂度: O(n log k)，n 是总节点数，k 是链表数
    """
    heap = []
    for i, lst in enumerate(lists):
        if lst:
            heapq.heappush(heap, (lst.val, i, lst))

    dummy = ListNode()
    curr = dummy

    while heap:
        val, i, node = heapq.heappop(heap)
        curr.next = node
        curr = curr.next
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))

    return dummy.next
```

### 复杂链表复制

**LeetCode 138**: 复制带随机指针的链表。

```python
def copy_random_pointer(head):
    """
    三步法：
    1. 复制每个节点并插入原节点后面
    2. 设置随机指针
    3. 分离两个链表
    """
    if not head:
        return None

    # Step 1: 复制节点并插入
    curr = head
    while curr:
        new_node = Node(curr.val)
        new_node.next = curr.next
        curr.next = new_node
        curr = new_node.next

    # Step 2: 设置随机指针
    curr = head
    while curr:
        if curr.random:
            curr.next.random = curr.random.next
        curr = curr.next.next

    # Step 3: 分离两个链表
    dummy = head.next
    curr = head
    while curr:
        copy = curr.next
        curr.next = copy.next
        if copy.next:
            copy.next = copy.next.next
        curr = curr.next

    return dummy
```
