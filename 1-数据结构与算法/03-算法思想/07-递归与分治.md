# 递归与分治

### 解决什么问题
将大问题分解为同类子问题，通过解决子问题来解决原问题。适用于问题本身具有递归结构（树、图、分形）或可自然分解（排序、查找）的场景。

### 核心概念
- 递归三要素：终止条件、递归调用、返回值合并
- 分治：分解为独立子问题，分别求解后合并结果
- 与DP区别：分治的子问题独立，DP的子问题重叠
- 递归复杂度：可用Master定理分析

## 分治模式

分治是将问题分解为独立的子问题，分别求解后合并结果。

```
分治步骤：
1. 分解：将问题分成子问题
2. 解决：递归求解子问题
3. 合并：将子问题的结果合并
```

### 分治模板

```python
def divide_conquer(problem, params):
    # 1. 终止条件
    if problem is trivial:
        return solve_trivial(problem)

    # 2. 分解问题
    sub_problems = split(problem)

    # 3. 递归解决子问题
    sub_results = [divide_conquer(p, params) for p in sub_problems]

    # 4. 合并结果
    return merge(sub_results)
```

## 经典递归问题

### 1. 汉诺塔

```python
def hanoi(n, source='A', target='C', auxiliary='B'):
    """
    将 n 个盘子从 source 移动到 target，使用 auxiliary 作为辅助
    移动次数: 2^n - 1
    """
    if n == 1:
        print(f"Move disk 1 from {source} to {target}")
        return

    # 将 n-1 个盘子从 source 移动到 auxiliary
    hanoi(n - 1, source, auxiliary, target)

    # 移动第 n 个盘子
    print(f"Move disk {n} from {source} to {target}")

    # 将 n-1 个盘子从 auxiliary 移动到 target
    hanoi(n - 1, auxiliary, target, source)
```

### 2. 组合总和

```python
def combination_sum(candidates, target):
    """
    找出所有和为 target 的组合（可重复使用元素）
    LeetCode 39
    """
    result = []

    def backtrack(start, target, path):
        if target == 0:
            result.append(path[:])
            return
        if target < 0:
            return

        for i in range(start, len(candidates)):
            path.append(candidates[i])
            backtrack(i, target - candidates[i], path)  # i 可以重复
            path.pop()

    backtrack(0, target, [])
    return result
```

## 分治经典问题

### 1. 归并排序（分治）

```python
def merge_sort(arr):
    if len(arr) <= 1:
        return arr

    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])

    return merge(left, right)

def merge(left, right):
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result
```

### 2. 快速排序（分治）

```python
def quick_sort(arr):
    if len(arr) <= 1:
        return arr

    pivot = arr[len(arr) // 2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]

    return quick_sort(left) + middle + quick_sort(right)


def quick_sort_inplace(arr, low=0, high=None):
    if high is None:
        high = len(arr) - 1

    if low < high:
        p = partition(arr, low, high)
        quick_sort_inplace(arr, low, p - 1)
        quick_sort_inplace(arr, p + 1, high)

def partition(arr, low, high):
    pivot = arr[high]
    i = low - 1
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1
```

### 3. 最近点对（分治）

```python
import math

def closest_pair(points):
    """
    最近点对问题
    时间复杂度: O(n log n)
    """
    points = sorted(points, key=lambda p: p[0])  # 按 x 坐标排序

    def dist(p1, p2):
        return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

    def closest_in_strip(strip, d):
        """在 strip 中查找比 d 更近的点对"""
        min_d = d
        strip_sorted = sorted(strip, key=lambda p: p[1])

        for i in range(len(strip_sorted)):
            for j in range(i + 1, len(strip_sorted)):
                if strip_sorted[j][1] - strip_sorted[i][1] >= min_d:
                    break
                d = dist(strip_sorted[i], strip_sorted[j])
                min_d = d

        return min_d

    def closest_util(px, py):
        if len(px) <= 3:
            min_d = float('inf')
            for i in range(len(px)):
                for j in range(i + 1, len(px)):
                    d = dist(px[i], px[j])
                    if d < min_d:
                        min_d = d
            return min_d

        mid = len(px) // 2
        mid_x = px[mid][0]

        pyl = [p for p in py if p[0] <= mid_x]
        pyr = [p for p in py if p[0] > mid_x]

        dl = closest_util(px[:mid], pyl)
        dr = closest_util(px[mid:], pyr)
        d = min(dl, dr)

        strip = [p for p in py if abs(p[0] - mid_x) < d]
        return closest_in_strip(strip, d)

    return closest_util(points, sorted(points, key=lambda p: p[1]))
```

### 4. 大整数乘法（分治）

```python
def karatsuba(x, y):
    """
    Karatsuba 大整数乘法
    时间复杂度: O(n^log3) ≈ O(n^1.585)
    相比普通乘法的 O(n^2) 有显著提升
    """
    if x < 10 or y < 10:
        return x * y

    n = max(len(str(x)), len(str(y)))
    m = n // 2

    high1, low1 = divmod(x, 10**m)
    high2, low2 = divmod(y, 10**m)

    z0 = karatsuba(low1, low2)
    z1 = karatsuba((low1 + high1), (low2 + high2))
    z2 = karatsuba(high1, high2)

    return z2 * 10**(2*m) + (z1 - z2 - z0) * 10**m + z0
```

## 递归优化技巧

### 1. 记忆化（Memoization）

```python
def fibMemo(n, memo=None):
    """记忆化递归 - 避免重复计算"""
    if memo is None:
        memo = {}

    if n in memo:
        return memo[n]

    if n <= 1:
        return n

    memo[n] = fibMemo(n - 1, memo) + fibMemo(n - 2, memo)
    return memo[n]


# 或者用装饰器
from functools import lru_cache

@lru_cache(maxsize=None)
def fib_cache(n):
    if n < 2:
        return n
    return fib_cache(n - 1) + fib_cache(n - 2)
```

### 2. 尾递归优化

```python
# 尾递归版本（Python 不支持尾递归优化，但思路重要）
def factorial_tail(n, acc=1):
    if n == 0:
        return acc
    return factorial_tail(n - 1, n * acc)


# 对比普通递归
def factorial(n):
    if n == 0:
        return 1
    return n * factorial(n - 1)  # 不是尾递归，最后一步是乘法
```

### 3. 递归转迭代

```python
# 递归遍历树
def inorder_recursive(root):
    if not root:
        return []
    return inorder_recursive(root.left) + [root.val] + inorder_recursive(root.right)

# 迭代版本（使用栈）
def inorder_iterative(root):
    result = []
    stack = []
    current = root

    while current or stack:
        while current:
            stack.append(current)
            current = current.left
        current = stack.pop()
        result.append(current.val)
        current = current.right

    return result
```

## 递归思维训练

### 什么时候用递归

1. 问题可以分解为同类子问题
2. 子问题相互独立
3. 存在终止条件
4. 问题规模会逐渐缩小

### 递归常见错误

1. **忘记终止条件**：导致无限递归
2. **终止条件不正确**：结果错误
3. **子问题不独立**：导致重复计算
4. **栈溢出**：递归深度过大

### 递归调试技巧

```python
# 添加日志调试递归
def debug_recursion(n, depth=0):
    indent = "  " * depth
    print(f"{indent}Enter: n={n}")

    if n <= 0:
        print(f"{indent}Base case: n={n}")
        return 0

    result = n + debug_recursion(n - 1, depth + 1)

    print(f"{indent}Exit: n={n}, result={result}")
    return result
```

## 补充：分治代码实现

### 求众数 II（LeetCode 229）

```python
def majority_element(nums):
    """
    找出所有出现次数超过 n/3 的元素
    思路：分治
    出现超过 n/3 的元素最多有 2 个
    """
    if not nums:
        return []

    def divide_conquer(left, right):
        if left == right:
            return {nums[left]}

        mid = (left + right) // 2
        left_set = divide_conquer(left, mid)
        right_set = divide_conquer(mid + 1, right)

        result = set()
        for num in left_set | right_set:
            count = sum(1 for i in range(left, right + 1) if nums[i] == num)
            if count > (right - left + 1) // 3:
                result.add(num)
        return result

    return list(divide_conquer(0, len(nums) - 1))
```

### 重建二叉树（LeetCode 105）

```python
def build_tree(preorder, inorder):
    """
    根据前序和中序遍历结果重建二叉树
    前序：[根, 左子树, 右子树]
    中序：[左子树, 根, 右子树]
    """
    if not preorder or not inorder:
        return None

    root_val = preorder[0]
    root = TreeNode(root_val)

    # 在中序遍历中找根
    root_idx = inorder.index(root_val)

    # 递归重建
    left_size = root_idx
    root.left = build_tree(
        preorder[1:1 + left_size],
        inorder[:root_idx]
    )
    root.right = build_tree(
        preorder[1 + left_size:],
        inorder[root_idx + 1:]
    )

    return root
```

## 分治 vs 动态规划

| 特性 | 分治 | 动态规划 |
|------|------|----------|
| 子问题关系 | 独立 | 重叠 |
| 重复计算 | 有 | 无（记忆化） |
| 最优子结构 | 不一定 | 必须 |
| 典型应用 | 归并排序、最近点对 | 爬楼梯、背包 |
| 实现方式 | 递归 | 递归或迭代 |
