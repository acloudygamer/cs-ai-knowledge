# 位运算

## 核心要点

- 位运算是 O(1) 空间复杂度的利器
- 计算机以补码形式存储整数
- 位运算：与、或、异或、取反、左移、右移

## 基本操作

```python
def basic_bit_operations():
    """
    常用位运算操作
    """

    # 基本操作
    a = 5       # 0101
    b = 3       # 0011

    # 与 (AND)：都为 1 才为 1
    assert (a & b) == 1    # 0001

    # 或 (OR)：有一个为 1 就为 1
    assert (a | b) == 7    # 0111

    # 异或 (XOR)：不同为 1，相同为 0
    assert (a ^ b) == 6     # 0110

    # 取反 (NOT)：单目运算符
    assert (~5) == -6       # 补码表示

    # 左移：低位补 0
    assert (a << 1) == 10  # 1010

    # 右移：无符号右移，低位丢弃
    assert (a >> 1) == 2    # 0010


def swap(a, b):
    """不用临时变量交换两数"""
    a = a ^ b
    b = a ^ b  # b = (a ^ b) ^ b = a
    a = a ^ b  # a = (a ^ b) ^ a = b
    return a, b


def absolute_value(n):
    """求整数的绝对值（不使用 if）"""
    # 利用符号位
    mask = n >> 31  # 全 0 或 全 1
    return (n + mask) ^ mask


def is_power_of_two(n):
    """判断是否为 2 的幂"""
    # 2^k: 二进制只有一位为 1
    # n & (n-1) 会消除最低位的 1
    return n > 0 and (n & (n - 1)) == 0


def count_ones(n):
    """
    计算二进制中 1 的个数
    Brian & Kernighan 算法：O(k)，k 为 1 的个数
    """
    count = 0
    while n:
        n = n & (n - 1)  # 消除最低位的 1
        count += 1
    return count


def count_ones_fast(n):
    """
    查表法（适用于 32 位整数）：O(1)
    """
    TABLE = [0] * 256
    for i in range(1, 256):
        TABLE[i] = TABLE[i & (i - 1)] + 1

    count = 0
    while n:
        count += TABLE[n & 0xFF]
        n >>= 8
    return count
```

## 常用技巧

### 1. 掩码操作

```python
def get_bit(n, i):
    """获取第 i 位（从 0 开始）"""
    return (n >> i) & 1


def set_bit(n, i):
    """将第 i 位设为 1"""
    return n | (1 << i)


def clear_bit(n, i):
    """将第 i 位设为 0"""
    return n & ~(1 << i)


def update_bit(n, i, bit):
    """将第 i 位设为 bit"""
    return (n & ~(1 << i)) | (bit << i)


def is_odd(n):
    """判断奇偶（不用 %）"""
    return n & 1


def lowest_set_bit(n):
    """获取最低位的 1"""
    return n & (-n)


def clear_lowest_set_bit(n):
    """清除最低位的 1"""
    return n & (n - 1)
```

### 2. 字母数字转换

```python
def lowercase_to_uppercase(c):
    """小写转大写（利用 ASCII 差值）"""
    # 'a' = 97, 'A' = 65，差值为 32，即 1 << 5
    return chr(ord(c) ^ (1 << 5))


def uppercase_to_lowercase(c):
    return chr(ord(c) | (1 << 5))


def is_lowercase(c):
    """判断是否小写"""
    return 'a' <= c <= 'z'
```

### 3. 位运算实现加减乘除

```python
def add(a, b):
    """不用 +，实现加法"""
    while b:
        carry = a & b  # 进位
        a = a ^ b      # 非进位和
        b = carry << 1  # 进位左移
    return a


def negate(a):
    """取反"""
    return add(~a, 1)


def subtract(a, b):
    """减法：a - b = a + (-b)"""
    return add(a, negate(b))


def multiply(a, b):
    """乘法：二进制思想"""
    result = 0
    while b:
        if b & 1:
            result = add(result, a)
        a <<= 1
        b >>= 1
    return result


def divide(a, b):
    """除法：二进制思想"""
    if b == 0:
        raise ValueError("Division by zero")

    negative = (a < 0) ^ (b < 0)
    a, b = abs(a), abs(b)

    result = 0
    for i in range(31, -1, -1):
        if a >= (b << i):
            a = subtract(a, b << i)
            result = add(result, 1 << i)

    return negate(result) if negative else result
```

## 面试高频题型

### 1. 只出现一次的数字（LeetCode 136）

```python
def single_number(nums):
    """
    除一个元素外，其他元素都出现两次
    利用异或：a ^ a = 0, a ^ 0 = a
    """
    result = 0
    for num in nums:
        result ^= num
    return result


def single_number_ii(nums):
    """
    除一个元素外，其他元素都出现三次
    利用位运算统计每一位
    """
    result = 0
    for i in range(32):
        bit_sum = 0
        for num in nums:
            bit_sum += (num >> i) & 1
        if bit_sum % 3:
            result |= (1 << i)
    return result


def single_number_iii(nums):
    """
    两个元素只出现一次，其他出现两次
    分组异或
    """
    xor = 0
    for num in nums:
        xor ^= num

    # 找到最低位的 1（两数的差异位）
    diff = xor & (-xor)

    result = 0
    for num in nums:
        if num & diff:
            result ^= num

    return [result, xor ^ result]
```

### 2. 比特位计数（LeetCode 338）

```python
def count_bits(n):
    """
    统计 0 到 n 每个数二进制中 1 的个数
    动态规划：i & (i-1) 的结果 + 1
    """
    dp = [0] * (n + 1)
    for i in range(1, n + 1):
        dp[i] = dp[i & (i - 1)] + 1
    return dp


def count_bits_dp(n):
    """
    另一种 DP：利用 i >> 1 的结果
    dp[i] = dp[i >> 1] + (i & 1)
    """
    dp = [0] * (n + 1)
    for i in range(1, n + 1):
        dp[i] = dp[i >> 1] + (i & 1)
    return dp
```

### 3. 数字范围求和（LeetCode 338 变体）

```python
def range_bitwise_and(left, right):
    """
    [left, right] 范围内所有数字按位与
    技巧：找到公共前缀
    """
    shift = 0
    while left < right:
        left >>= 1
        right >>= 1
        shift += 1
    return left << shift


def bitwise_complement(n):
    """
    N 的比特位补码（无符号）
    """
    if n == 0:
        return 1
    mask = (1 << n.bit_length()) - 1
    return n ^ mask
```

### 4. 颠倒二进制位（LeetCode 190）

```python
def reverse_bits(n):
    """
    颠倒 32 位无符号整数的二进制位
    分治：两两交换，四四交换...
    """
    n = (n >> 16) | ((n & 0xFFFF) << 16)
    n = ((n & 0xFF00FF00) >> 8) | ((n & 0x00FF00FF) << 8)
    n = ((n & 0xF0F0F0F0) >> 4) | ((n & 0x0F0F0F0F) << 4)
    n = ((n & 0xCCCCCCCC) >> 2) | ((n & 0x33333333) << 2)
    n = ((n & 0xAAAAAAAA) >> 1) | ((n & 0x55555555) << 1)
    return n
```

### 5. O(1) 时间获取随机数

```python
import random


class RandomBit:
    """只用 rand7() 实现 rand10()"""

    def rand7(self):
        return random.randint(1, 7)

    def rand10(self):
        """
        拒绝采样：用 rand7 生成 rand10
        """
        while True:
            # 生成 0-48 的均匀分布
            num = (self.rand7() - 1) * 7 + (self.rand7() - 1)
            if num < 40:  # 40 是 10 的倍数
                return (num % 10) + 1


def rand5():
    """只用 rand7 实现 rand5"""
    while True:
        num = self.rand7()
        if num <= 5:
            return num
```

## 位运算优先级

```python
# 位运算优先级容易出错，加括号更安全
# 优先级（从高到低）：~ > >> << > & > ^ > |

# 错误写法
a & mask == 0    # 先比较，再按位与

# 正确写法
(a & mask) == 0
```

## 复杂度总结

| 操作 | 时间 | 空间 |
|------|------|------|
| 基本位运算 | O(1) | O(1) |
| 统计 1 的个数 | O(1) ~ O(k) | O(1) |
| 查找只出现一次的数 | O(n) | O(1) |
| 数字范围按位与 | O(1) | O(1) |
