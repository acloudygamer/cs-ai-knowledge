# 并查集 (Union-Find)

### 解决什么问题
处理不相交集合的合并与查询，适用于连通性判断、朋友圈问题、岛屿数量、集合合并等场景，比DFS/BFS更适合动态连通性问题。

### 核心概念
- Find：查找元素所属集合的根节点
- Union：合并两个不相交的集合
- 路径压缩：查询时将路径上节点直接指向根
- 按秩合并：小秩合并到大秩，保持树的扁平化
- 复杂度：近乎 O(1)，实际是 O(α(n))

### 怎么用

## 核心概念

并查集是一种处理不相交集合合并与查询的数据结构，主要支持两种操作：
- **Find**：查找元素所属集合的根节点
- **Union**：合并两个不相交的集合

常用于连通性判断、朋友圈问题、岛屿数量等。

## 基本实现

```python
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))  # 父节点
        self.rank = [0] * n            # 树的秩（用于优化）

    def find(self, x):
        """路径压缩"""
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # 递归压缩路径
        return self.parent[x]

    def union(self, x, y):
        """按秩合并"""
        root_x, root_y = self.find(x), self.find(y)
        if root_x == root_y:
            return False  # 已在同一集合

        # 小秩合并到大秩
        if self.rank[root_x] < self.rank[root_y]:
            root_x, root_y = root_y, root_x
        self.parent[root_y] = root_x
        if self.rank[root_x] == self.rank[root_y]:
            self.rank[root_x] += 1
        return True
```

## 复杂度分析

| 操作 | 朴素版 | 路径压缩 | 路径压缩 + 按秩合并 |
|------|--------|----------|-------------------|
| Find | O(n) | O(α(n)) | O(α(n)) |
| Union | O(n) | O(α(n)) | O(α(n)) |

其中 α(n) 是 Ackermann 函数的反函数，近似为常数。

## 经典问题

### 省份数量（连通分量）

```python
def find_circle_num(is_connected):
    n = len(is_connected)
    uf = UnionFind(n)
    for i in range(n):
        for j in range(i + 1, n):
            if is_connected[i][j]:
                uf.union(i, j)
    return sum(1 for i in range(n) if uf.find(i) == i)

# 更简洁的BFS版本
from collections import deque
def find_circle_num_bfs(is_connected):
    n = len(is_connected)
    visited = set()
    provinces = 0
    for i in range(n):
        if i not in visited:
            provinces += 1
            queue = deque([i])
            while queue:
                j = queue.popleft()
                if j in visited:
                    continue
                visited.add(j)
                for k in range(n):
                    if is_connected[j][k] and k not in visited:
                        queue.append(k)
    return provinces
```

### 岛屿数量（感染扩散）

```python
def num_islands(grid):
    if not grid:
        return 0
    m, n = len(grid), len(grid[0])
    uf = UnionFind(m * n)
    directions = [(1, 0), (0, 1)]

    # 只在横向和纵向合并
    for i in range(m):
        for j in range(n):
            if grid[i][j] == '1':
                idx = i * n + j
                for di, dj in directions:
                    ni, nj = i + di, j + dj
                    if 0 <= ni < m and 0 <= nj < n and grid[ni][nj] == '1':
                        uf.union(idx, ni * n + nj)

    # 统计根节点数量
    islands = set()
    for i in range(m):
        for j in range(n):
            if grid[i][j] == '1':
                islands.add(uf.find(i * n + j))
    return len(islands)
```

### 合并区间（区间集合）

```python
def merge_intervals(intervals):
    if not intervals:
        return []
    # 按起点排序，并查集连接重叠区间
    intervals.sort()
    uf = UnionFind(len(intervals))
    for i in range(len(intervals) - 1):
        if intervals[i][1] >= intervals[i + 1][0]:
            uf.union(i, i + 1)

    # 合并同一集合的区间
    merged = {}
    for i in range(len(intervals)):
        root = uf.find(i)
        if root not in merged:
            merged[root] = intervals[i][:]
        else:
            merged[root][1] = max(merged[root][1], intervals[i][1])
    return list(merged.values())
```

## 并查集 vs DFS

| 场景 | 并查集 | DFS/BFS |
|------|--------|---------|
| 动态连通性 | 更适合 | 需要重建访问状态 |
| 频繁查询两点是否连通 | O(1) | O(V+E) |
| 需要获知所有连通分量成员 | 需额外存储 | 直接可得 |
| 图结构固定不变 | 并查集略优 | 等效 |

### Kruskal 最小生成树

```python
def kruskal(n, edges):
    """edges: [(u, v, weight)]"""
    edges.sort(key=lambda x: x[2])
    uf = UnionFind(n)
    mst_weight = 0
    mst_edges = []

    for u, v, w in edges:
        if uf.union(u, v):
            mst_weight += w
            mst_edges.append((u, v, w))
            if len(mst_edges) == n - 1:
                break
    return mst_weight, mst_edges
```

### 路径压缩详解

#### 递归路径压缩（完全压缩）

```python
def find(self, x):
    if self.parent[x] != x:
        self.parent[x] = self.find(self.parent[x])  # 路径上所有节点直接指向根
    return self.parent[x]
```

#### 迭代路径压缩（ flatten ）

```python
def find(self, x):
    root = x
    # 找到根
    while self.parent[root] != root:
        root = self.parent[root]

    # 将路径上所有节点指向根
    while self.parent[x] != root:
        parent = self.parent[x]
        self.parent[x] = root
        x = parent

    return root
```

#### 路径压缩 + 按秩合并的复杂度

使用路径压缩和按秩合并后，每次操作的均摊复杂度为 **O(α(n))**，其中 α(n) 是 Ackermann 函数的反函数。

| n 值 | α(n) |
|------|------|
| 1 | 1 |
| 2 | 2 |
| 3 | 3 |
| 4 | 4 |
| 5 | 4 |
| ... | ... |
| 10^80 | ≤ 5 |

实际应用中，α(n) 可以认为是常数（< 5）。

### Tarjan 离线 LCA（最近公共祖先）

Tarjan 的离线 LCA 算法使用并查集，在一次 DFS 中回答所有 LCA 查询。

```python
def tarjan_lca(n, edges, queries):
    """
    Tarjan 离线 LCA
    n: 节点数
    edges: [(u, v)] 无向边
    queries: [(u, v)] 查询对

    返回: {query: lca}
    """
    from collections import defaultdict

    # 建图
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)

    # 查询分组
    query_dict = defaultdict(list)
    for u, v in queries:
        query_dict[u].append((v, (u, v)))
        query_dict[v].append((u, (u, v)))

    parent = list(range(n))
    ancestor = list(range(n))
    visited = [False] * n
    lca_result = {}

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(x, y):
        parent[find(y)] = find(x)

    def dfs(u, p):
        visited[u] = True
        ancestor[find(u)] = p  # u 的祖先设为 p

        for v in graph[u]:
            if v != p:
                dfs(v, u)
                union(u, v)  # 合并后 v 的祖先是 u
                ancestor[find(u)] = p

        # 处理以 u 为起点的查询
        for v, key in query_dict[u]:
            if visited[v]:
                lca_result[key] = ancestor[find(v)]

    dfs(0, 0)
    return lca_result
```

### 并查集求连通分量大小

```python
class UnionFindWithSize:
    """带连通分量大小的并查集"""

    def __init__(self, n):
        self.parent = list(range(n))
        self.size = [1] * n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        root_x, root_y = self.find(x), self.find(y)
        if root_x == root_y:
            return False

        # 小合并到大
        if self.size[root_x] < self.size[root_y]:
            root_x, root_y = root_y, root_x

        self.parent[root_y] = root_x
        self.size[root_x] += self.size[root_y]
        return True

    def get_size(self, x):
        return self.size[self.find(x)]

    def count_components(self):
        """统计连通分量数"""
        return sum(1 for i in range(len(self.parent)) if self.parent[i] == i)
```

### 实战例题

#### 1. 岛屿数量 II（动态添加岛屿）

```python
def num_islands2(m, n, operators):
    """
    动态添加岛屿，返回每次添加后的岛屿数量
    LeetCode 305
    """
    directions = [(1, 0), (0, 1), (-1, 0), (0, -1)]
    uf = UnionFindWithSize(m * n)
    land = [[False] * n for _ in range(m)]
    count = 0
    result = []

    def idx(i, j):
        return i * n + j

    for i, j in operators:
        if land[i][j]:
            result.append(count)
            continue

        land[i][j] = True
        uf.parent[idx(i, j)] = idx(i, j)
        uf.size[idx(i, j)] = 1
        count += 1

        for di, dj in directions:
            ni, nj = i + di, j + dj
            if 0 <= ni < m and 0 <= nj < n and land[ni][nj]:
                if uf.union(idx(i, j), idx(ni, nj)):
                    count -= 1

        result.append(count)

    return result
```

#### 2. 情侣牵手（分组问题）

```python
def min_swaps_couples(row):
    """
    最小交换次数使情侣牵手
    LeetCode 765
    方法：将每对情侣视为一个集合，统计连通分量
    """
    n = len(row) // 2
    uf = UnionFindWithSize(n)

    for i in range(0, len(row), 2):
        a = row[i] // 2  # 转换为情侣编号
        b = row[i + 1] // 2
        uf.union(a, b)

    # 每组连通分量需要 size-1 次交换
    return sum(uf.size[uf.find(i)] - 1 for i in range(n))
```

#### 3. 账户合并

```python
def accounts_merge(accounts):
    """
    合并同一人的多个账户
    LeetCode 721
    """
    from collections import defaultdict

    uf = UnionFindWithSize(len(accounts))
    email_to_account = {}

    # 建立邮箱到账户的映射
    for i, emails in enumerate(accounts):
        for email in emails[1:]:
            if email not in email_to_account:
                email_to_account[email] = i
            else:
                uf.union(i, email_to_account[email])

    # 按根节点分组
    merged = defaultdict(list)
    for i in range(len(accounts)):
        root = uf.find(i)
        merged[root].append(i)

    # 构建结果
    result = []
    for indices in merged.values():
        emails = set()
        name = accounts[indices[0]][0]
        for idx in indices:
            emails.update(accounts[idx][1:])
        result.append([name] + sorted(emails))

    return result
```

#### 4. 删除无效括号（并查集思维）

```python
def remove_invalid_parentheses(s):
    """
    删除最小数量的无效括号使字符串有效
    LeetCode 301
    方法：广度优先 + 并查集剪枝
    """
    def is_valid(t):
        count = 0
        for c in t:
            if c == '(':
                count += 1
            elif c == ')':
                count -= 1
                if count < 0:
                    return False
        return count == 0

    visited = {s}
    result = []
    found = False

    while visited and not found:
        next_level = set()
        for t in visited:
            if is_valid(t):
                result.append(t)
                found = True
            else:
                for i, c in enumerate(t):
                    if c in '()':
                        next_level.add(t[:i] + t[i+1:])
        visited = next_level

    return result
```

### 并查集 vs BFS/DFS 对比

| 场景 | 并查集 | BFS/DFS |
|------|--------|---------|
| 静态图查询 | O(α(n)) | O(V+E) |
| 动态加边 | O(α(n)) | 需要重建 |
| 路径相关信息 | 不直接支持 | 容易获取 |
| 内存占用 | 较小 | 较大（需要 visited） |
| 实现难度 | 中等 | 简单 |
