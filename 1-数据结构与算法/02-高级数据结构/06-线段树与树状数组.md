# 线段树与树状数组

## 核心要点

- 线段树：用于区间查询与更新，O(log n)
- 树状数组（Fenwick Tree）：前缀和查询与更新，O(log n)，实现更简洁
- 两者都是对数组区间操作的优化结构

## 树状数组（Fenwick Tree）

### 基本原理

利用二进制表示，任何数可以表示为 2 的幂之和。树状数组中每个节点管理特定范围的数组元素。

```python
class FenwickTree:
    """
    树状数组：支持 O(log n) 的单点更新和前缀和查询
    """

    def __init__(self, size_or_array):
        """
        初始化：支持传入大小或初始数组
        """
        if isinstance(size_or_array, int):
            self.n = size_or_array
            self.tree = [0] * (self.n + 1)
        else:
            arr = size_or_array
            self.n = len(arr)
            self.tree = [0] * (self.n + 1)
            for i, val in enumerate(arr, 1):
                self._add(i, val)

    def _add(self, idx, delta):
        """在 idx 位置增加 delta"""
        while idx <= self.n:
            self.tree[idx] += delta
            idx += idx & (-idx)  # lowbit 操作

    def _prefix_sum(self, idx):
        """返回 [1..idx] 的前缀和"""
        result = 0
        while idx > 0:
            result += self.tree[idx]
            idx -= idx & (-idx)
        return result

    def range_sum(self, left, right):
        """返回 [left..right] 的区间和（1-indexed）"""
        return self._prefix_sum(right) - self._prefix_sum(left - 1)

    def point_update(self, idx, value):
        """将 idx 位置设置为 value（增量更新）"""
        original = self.point_query(idx)
        self._add(idx, value - original)

    def point_query(self, idx):
        """查询 idx 位置的值"""
        return self.range_sum(idx, idx)

    def lower_bound(self, target):
        """
        找到最小的 idx 使得 prefix_sum(idx) >= target
        又称"前缀和二分查找"
        """
        idx = 0
        bit_mask = 1 << (self.n.bit_length() - 1)
        while bit_mask:
            next_idx = idx + bit_mask
            if next_idx <= self.n and self.tree[next_idx] < target:
                idx = next_idx
                target -= self.tree[next_idx]
            bit_mask >>= 1
        return idx + 1
```

### 树状数组实战

```python
# 动态数组中第 K 大的数（LeetCode 315 变体）
def kth_smallest(arr, k):
    """
    查找第 k 小的数（1-indexed）
    利用树状数组 + 二分查找
    """
    # 离散化
    sorted_vals = sorted(set(arr))
    ft = FenwickTree(len(sorted_vals))

    for val in arr:
        idx = bisect_left(sorted_vals, val) + 1
        ft._add(idx, 1)

    # 找到第 k 小的索引
    idx = ft.lower_bound(k)
    return sorted_vals[idx - 1]


# 翻转对数量（LeetCode 493 变体）
def reverse_pairs(nums):
    """
    统计翻转对数量：i < j 且 nums[i] > 2*nums[j]
    """
    if not nums:
        return 0

    # 离散化
    vals = sorted(set(nums + [x * 2 for x in nums]))
    ft = FenwickTree(len(vals))
    count = 0

    for j, num in enumerate(nums):
        # 查询有多少个 nums[i] > 2*nums[j]
        target_idx = bisect_right(vals, num * 2)
        if target_idx <= len(vals):
            count += ft.range_sum(target_idx, len(vals))
        # 将当前 num 加入树状数组
        idx = bisect_left(vals, num) + 1
        ft._add(idx, 1)

    return count
```

## 线段树

### 基本实现

```python
class SegTreeNode:
    def __init__(self, val=0):
        self.val = val
        self.left = None
        self.right = None


class SegTree:
    """
    线段树：支持区间查询（Sum/Min/Max）和单点/区间更新
    """

    def __init__(self, arr):
        self.n = len(arr)
        self.arr = arr
        self.root = self._build(0, self.n - 1)

    def _build(self, left, right):
        """构建线段树"""
        if left == right:
            return SegTreeNode(self.arr[left])

        mid = (left + right) // 2
        node = SegTreeNode()
        node.left = self._build(left, mid)
        node.right = self._build(mid + 1, right)
        node.val = node.left.val + node.right.val  # 聚合操作
        return node

    def query(self, ql, qr):
        """区间查询 [ql, qr]"""
        return self._query(self.root, 0, self.n - 1, ql, qr)

    def _query(self, node, left, right, ql, qr):
        if ql <= left and right <= qr:
            return node.val

        mid = (left + right) // 2
        result = 0
        if ql <= mid:
            result += self._query(node.left, left, mid, ql, qr)
        if qr > mid:
            result += self._query(node.right, mid + 1, right, ql, qr)
        return result

    def update(self, idx, value):
        """单点更新：将 idx 位置设为 value"""
        self._update(self.root, 0, self.n - 1, idx, value)

    def _update(self, node, left, right, idx, value):
        if left == right:
            node.val = value
            self.arr[idx] = value
            return

        mid = (left + right) // 2
        if idx <= mid:
            self._update(node.left, left, mid, idx, value)
        else:
            self._update(node.right, mid + 1, right, idx, value)
        node.val = node.left.val + node.right.val

    def range_update(self, ql, qr, delta):
        """区间更新：[ql, qr] 内的所有元素增加 delta"""
        self._range_update(self.root, 0, self.n - 1, ql, qr, delta)

    def _range_update(self, node, left, right, ql, qr, delta):
        if ql <= left and right <= qr:
            node.val += delta * (right - left + 1)
            # 懒标记传播（此处简化实现）
            return

        mid = (left + right) // 2
        if ql <= mid:
            self._range_update(node.left, left, mid, ql, qr, delta)
        if qr > mid:
            self._range_update(node.right, mid + 1, right, ql, qr, delta)
        node.val = node.left.val + node.right.val
```

### 数组实现的线段树（更常用）

```python
class SegTreeArray:
    """
    数组实现的线段树（堆式存储）
    支持：区间和查询、单点更新
    """

    def __init__(self, arr):
        self.n = len(arr)
        # 4n 大小足够容纳所有节点
        self.tree = [0] * (4 * self.n)
        if self.n > 0:
            self._build(1, 0, self.n - 1, arr)

    def _build(self, node, left, right, arr):
        if left == right:
            self.tree[node] = arr[left]
            return
        mid = (left + right) // 2
        self._build(node * 2, left, mid, arr)
        self._build(node * 2 + 1, mid + 1, right, arr)
        self.tree[node] = self.tree[node * 2] + self.tree[node * 2 + 1]

    def query(self, ql, qr):
        """区间和查询"""
        return self._query(1, 0, self.n - 1, ql, qr)

    def _query(self, node, left, right, ql, qr):
        if ql <= left and right <= qr:
            return self.tree[node]
        mid = (left + right) // 2
        result = 0
        if ql <= mid:
            result += self._query(node * 2, left, mid, ql, qr)
        if qr > mid:
            result += self._query(node * 2 + 1, mid + 1, right, ql, qr)
        return result

    def update(self, idx, value):
        """单点更新"""
        self._update(1, 0, self.n - 1, idx, value)

    def _update(self, node, left, right, idx, value):
        if left == right:
            self.tree[node] = value
            return
        mid = (left + right) // 2
        if idx <= mid:
            self._update(node * 2, left, mid, idx, value)
        else:
            self._update(node * 2 + 1, mid + 1, right, idx, value)
        self.tree[node] = self.tree[node * 2] + self.tree[node * 2 + 1]
```

```python
def reverse_pairs(nums):
    """
    统计翻转对：i < j 且 nums[i] > 2 * nums[j]
    使用归并排序思想（分治）
    """
    if not nums:
        return 0

    self.count = 0
    self._merge_sort(nums, 0, len(nums) - 1)
    return self.count


def reverse_pairs_better(nums):
    """
    优化版本：使用 BIT
    """
    if not nums:
        return 0

    # 离散化
    vals = sorted(set(nums + [x * 2 for x in nums]))
    ft = FenwickTree(len(vals))
    count = 0

    for j in range(len(nums)):
        # 统计有多少个 nums[i] > 2*nums[j]
        target_idx = bisect_right(vals, nums[j] * 2)
        if target_idx <= len(vals):
            count += ft.range_sum(target_idx, len(vals))
        # 插入 nums[j]
        idx = bisect_left(vals, nums[j]) + 1
        ft._add(idx, 1)

    return count
```

## Fenwick Tree vs Segment Tree

| 特性 | 树状数组 | 线段树 |
|------|---------|--------|
| 实现复杂度 | 低 | 中 |
| 空间复杂度 | O(n) | O(4n) |
| 区间查询 | O(log n) | O(log n) |
| 单点更新 | O(log n) | O(log n) |
| 区间更新 | 需要差分数组 | 原生支持 |
| 灵活性 | 低（只能前缀聚合） | 高（可自定义） |
| 难以实现的操作 | 区间最大值 | 区间多种聚合 |

## 复杂度总结

| 操作 | 树状数组 | 线段树 |
|------|---------|--------|
| 构建 | O(n log n) / O(n) | O(n) |
| 单点更新 | O(log n) | O(log n) |
| 区间查询 | O(log n) | O(log n) |
| 区间更新 | O(log n)（差分） | O(log n) |
| 空间 | O(n) | O(4n) |
