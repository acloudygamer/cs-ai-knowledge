# 图

### 解决什么问题
图用于表示实体间的复杂关系，适用于社交网络、路由算法、任务调度、依赖分析等场景，解决可达性、最短路径、最优分配等问题。

### 核心概念
- 存储结构：邻接表（稀疏图）、邻接矩阵（密集图）
- 图遍历：BFS（最短路径）、DFS（连通分量、拓扑排序）
- 最短路径：Dijkstra（正权）、Bellman-Ford（可负权）、Floyd（多源）
- 生成树：Prim（切分性质）、Kruskal（边排序）
- 匹配问题：匈牙利算法、Hopcroft-Karp

import heapq

INF = float('inf')

def dijkstra_matrix(graph, start):
    n = len(graph)
    dist = [INF] * n
    dist[start] = 0
    visited = [False] * n

    for _ in range(n):
        # 找最小未访问节点
        u = -1
        for i in range(n):
            if not visited[i] and (u == -1 or dist[i] < dist[u]):
                u = i
        if dist[u] == INF:
            break
        visited[u] = True

        # 更新邻居距离
        for v in range(n):
            if graph[u][v] != INF and not visited[v]:
                dist[v] = min(dist[v], dist[u] + graph[u][v])
    return dist

## BFS/DFS

```python
# 图的BFS
from collections import deque

def bfs(graph, start):
    visited = set()
    queue = deque([start])
    result = []
    while queue:
        node = queue.popleft()
        if node not in visited:
            visited.add(node)
            result.append(node)
            queue.extend(graph[node])
    return result

# 图的DFS（递归）
def dfs_recursive(graph, node, visited=None):
    if visited is None:
        visited = set()
    visited.add(node)
    result = [node]
    for neighbor in graph[node]:
        if neighbor not in visited:
            result.extend(dfs_recursive(graph, neighbor, visited))
    return result

# 图的DFS（迭代）
def dfs_iterative(graph, start):
    visited = set()
    stack = [start]
    result = []
    while stack:
        node = stack.pop()
        if node not in visited:
            visited.add(node)
            result.append(node)
            stack.extend(graph[node])
    return result
```

## 最短路径 - Dijkstra

```python
import heapq

def dijkstra(graph, start):
    dist = {start: 0}
    pq = [(0, start)]
    while pq:
        d, node = heapq.heappop(pq)
        if d > dist.get(node, float('inf')):
            continue
        for neighbor, weight in graph[node]:
            new_dist = d + weight
            if new_dist < dist.get(neighbor, float('inf')):
                dist[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    return dist

有向无环图（DAG）的顶点线性排序，使得每条有向边 (u, v) 都满足 u 在 v 之前。

### Kahn算法（BFS + 入度）

```python
from collections import deque, defaultdict

def topological_sort_kahn(num_courses, prerequisites):
    graph = defaultdict(list)
    in_degree = [0] * num_courses

    for u, v in prerequisites:
        graph[v].append(u)  # v -> u (u依赖v)
        in_degree[u] += 1

    queue = deque([i for i in range(num_courses) if in_degree[i] == 0])
    result = []

    while queue:
        node = queue.popleft()
        result.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    return result if len(result) == num_courses else []  # 空列表表示有环
```

### DFS方法

```python
def topological_sort_dfs(num_courses, prerequisites):
    graph = defaultdict(list)
    for u, v in prerequisites:
        graph[v].append(u)

    visited = [0] * num_courses  # 0=未访问, 1=访问中, 2=已完成
    result = []
    has_cycle = False

    def dfs(node):
        nonlocal has_cycle
        if visited[node] == 1:  # 发现环
            has_cycle = True
            return
        if visited[node] == 2:   # 已完成
            return

        visited[node] = 1
        for neighbor in graph[node]:
            dfs(neighbor)
            if has_cycle:
                return
        visited[node] = 2
        result.append(node)

    for i in range(num_courses):
        if visited[i] == 0:
            dfs(i)
            if has_cycle:
                return []

    return result[::-1]
```

## 最小生成树 (MST)

### Prim算法

```python
def prim(n, edges):
    """ edges: [(u, v, weight)], 返回MST总权重 """
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))

    visited = [False] * n
    min_edge = [(INF, -1)] * n  # (weight, from)
    min_edge[0] = (0, -1)
    total = 0

    for _ in range(n):
        u = -1
        for i in range(n):
            if not visited[i] and (u == -1 or min_edge[i][0] < min_edge[u][0]):
                u = i

        if u != -1 and min_edge[u][0] == INF:
            return -1  # 图不连通

        if u != -1:
            visited[u] = True
            total += min_edge[u][0]
            for v, w in graph[u]:
                if not visited[v] and w < min_edge[v][0]:
                    min_edge[v] = (w, u)

    return total
```

### Kruskal算法（使用并查集）

```python
def kruskal(n, edges):
    edges.sort(key=lambda x: x[2])
    parent = list(range(n))

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        parent[px] = py
        return True

    mst_weight = 0
    mst_edges = []
    for u, v, w in edges:
        if union(u, v):
            mst_weight += w
            mst_edges.append((u, v, w))
            if len(mst_edges) == n - 1:
                break
    return mst_weight, mst_edges
```

## 关键路径算法

```python
def find_critical_path(n, edges):
    """
    关键路径 = 项目最短完成时间
    edges: [(u, v, weight)]
    返回: 关键路径长度和路径
    """
    graph = defaultdict(list)
    in_degree = [0] * n

    for u, v, w in edges:
        graph[u].append((v, w))
        in_degree[v] += 1

    # 正向计算最早完成时间
    earliest = [0] * n
    queue = deque([i for i in range(n) if in_degree[i] == 0])

    while queue:
        u = queue.popleft()
        for v, w in graph[u]:
            earliest[v] = max(earliest[v], earliest[u] + w)
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)

    # 反向计算最晚完成时间
    latest = [earliest[n - 1]] * n
    queue = deque([n - 1])

    while queue:
        v = queue.popleft()
        for u in range(n):
            for to_, w in graph[u]:
                if to_ == v:
                    latest[u] = min(latest[u], latest[v] - w)
                    if latest[u] == latest[v] - w:
                        queue.append(u)

    # 简化的关键路径输出
    return earliest[n - 1]
```

## 最大流算法（网络流）

给定有向图和源点、汇点，求最大流量。

### Ford-Fulkerson 方法

```python
def ford_fulkerson(graph, source, sink):
    """
    Ford-Fulkerson 最大流算法（Edmonds-Karp 用 BFS 找增广路径）
    时间复杂度: O(V * E^2)
    graph: 邻接表，graph[u][v] 表示边(u,v)的容量
    """
    from collections import deque

    def bfs_find_path(graph, source, sink, parent):
        """BFS 找增广路径"""
        visited = set([source])
        queue = deque([source])
        parent_map = {source: None}

        while queue:
            u = queue.popleft()
            if u == sink:
                return True

            for v in graph.get(u, []):
                if v not in visited and graph[u][v] > 0:
                    visited.add(v)
                    queue.append(v)
                    parent_map[v] = u
                    if v == sink:
                        parent[:] = []
                        while v is not None:
                            parent.append(v)
                            v = parent_map[v]
                        parent.reverse()
                        return True
        return False

    def get_bottleneck(graph, parent):
        """找到路径上的最小剩余容量"""
        bottleneck = float('inf')
        for i in range(len(parent) - 1):
            u, v = parent[i], parent[i + 1]
            bottleneck = min(bottleneck, graph[u][v])
        return bottleneck

    def augment_path(graph, parent, flow):
        """沿路径增加流量"""
        for i in range(len(parent) - 1):
            u, v = parent[i], parent[i + 1]
            graph[u][v] -= flow
            graph[v][u] = graph.get(v, {}).get(u, 0) + flow

    parent = []
    max_flow = 0

    while bfs_find_path(graph, source, sink, parent):
        flow = get_bottleneck(graph, parent)
        augment_path(graph, parent, flow)
        max_flow += flow
        parent.clear()

    return max_flow
```

### Dinic 算法（更高效）

```python
def dinic_max_flow(n, graph_edges, source, sink):
    """
    Dinic 最大流算法
    时间复杂度: O(E * V^2)（实际更快）
    graph_edges: [(u, v, capacity), ...]
    """
    from collections import deque

    # 构建邻接表（包含残余网络）
    adj = [[] for _ in range(n)]
    for u, v, cap in graph_edges:
        adj[u].append([v, cap, len(adj[v])])      # 正向边
        adj[v].append([u, 0, len(adj[u]) - 1])    # 反向边

    def bfs(level):
        """构建层次图"""
        level[:] = [-1] * n
        level[source] = 0
        queue = deque([source])
        while queue:
            u = queue.popleft()
            for v, cap, rev in adj[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        return level[sink] >= 0

    def dfs(u, flow, level, it, sink):
        """在层次图中找增广路径"""
        if u == sink:
            return flow
        for i in range(it[u], len(adj[u])):
            it[u] = i
            v, cap, rev = adj[u][i]
            if cap > 0 and level[u] < level[v]:
                pushed = dfs(v, min(flow, cap), level, it, sink)
                if pushed:
                    # 更新残余网络
                    adj[u][i][1] -= pushed
                    adj[v][rev][1] += pushed
                    return pushed
        return 0

    level = [-1] * n
    max_flow = 0

    while bfs(level):
        it = [0] * n
        while True:
            pushed = dfs(source, float('inf'), level, it, sink)
            if not pushed:
                break
            max_flow += pushed

    return max_flow
```

## 图算法复杂度总结

| 算法 | 时间复杂度 | 空间复杂度 | 适用场景 |
|------|------------|------------|----------|
| BFS | O(V+E) | O(V) | 最短路径(无权) |
| DFS | O(V+E) | O(V) | 连通分量、拓扑排序 |
| Dijkstra | O((V+E)logV) | O(V) | 单源最短路径(正权) |
| Bellman-Ford | O(VE) | O(V) | 单源最短路径(可负权) |
| Floyd-Warshall | O(V^3) | O(V^2) | 全源最短路径 |
| Prim | O(V^2) 或 O(ElogV) | O(V) | MST |
| Kruskal | O(ElogE) | O(V) | MST |
| Kahn (拓扑) | O(V+E) | O(V) | DAG拓扑排序 |
| Tarjan (SCC) | O(V+E) | O(V) | 强连通分量 |
| Hungarian | O(V^3) | O(V^2) | 二分图最大匹配 |

## 强连通分量 (SCC)

### Tarjan 算法

基于 DFS 的强连通分量算法，使用栈和 lowlink 值。

```python
def tarjan_scc(n, edges):
    """
    Tarjan 算法求强连通分量
    n: 节点数 (0 ~ n-1)
    edges: 边列表 [(u, v), ...]
    返回: SCC列表，每个SCC是一个节点列表
    """
    from collections import defaultdict
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)

    disc = [-1] * n      # 发现时间
    low = [-1] * n       # Lowlink 值
    on_stack = [False] * n
    stack = []
    sccs = []
    time = [0]

    def dfs(u):
        disc[u] = low[u] = time[0]
        time[0] += 1
        stack.append(u)
        on_stack[u] = True

        for v in graph[u]:
            if disc[v] == -1:
                dfs(v)
                low[u] = min(low[u], low[v])
            elif on_stack[v]:
                low[u] = min(low[u], disc[v])

        # u 是 SCC 根
        if low[u] == disc[u]:
            scc = []
            while True:
                w = stack.pop()
                on_stack[w] = False
                scc.append(w)
                if w == u:
                    break
            sccs.append(scc)

    for i in range(n):
        if disc[i] == -1:
            dfs(i)

    return sccs


# Kosaraju 算法（两次 DFS）
def kosaraju_scc(n, edges):
    """
    Kosaraju 算法：两次 DFS 求强连通分量
    时间复杂度 O(V+E)
    """
    from collections import defaultdict
    graph = defaultdict(list)
    rev_graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        rev_graph[v].append(u)

    visited = [False] * n
    order = []

    def dfs1(u):
        visited[u] = True
        for v in graph[u]:
            if not visited[v]:
                dfs1(v)
        order.append(u)

    # 第一次 DFS：获取拓扑顺序
    for i in range(n):
        if not visited[i]:
            dfs1(i)

    # 第二次 DFS：在转置图上按逆序遍历
    sccs = []
    visited = [False] * n

    def dfs2(u, component):
        visited[u] = True
        component.append(u)
        for v in rev_graph[u]:
            if not visited[v]:
                dfs2(v, component)

    for u in reversed(order):
        if not visited[u]:
            component = []
            dfs2(u, component)
            sccs.append(component)

    return sccs
```

- 2-SAT 判定
- 依赖分析
- 强连通性判定

## 欧拉路径与哈密顿路径

### 欧拉路径

通过图中所有边恰好一次的路径。

```python
def has_euler_path(n, edges, is_directed=True):
    """
    判断是否存在欧拉路径/回路
    is_directed: 有向图还是无向图
    """
    if is_directed:
        in_deg = [0] * n
        out_deg = [0] * n
        for u, v in edges:
            out_deg[u] += 1
            in_deg[v] += 1
        # 欧拉路径：要么所有点入度=出度，要么恰好一个点出度=入度+1，一个点入度=出度+1
        start_plus = sum(1 for i in range(n) if out_deg[i] - in_deg[i] == 1)
        end_plus = sum(1 for i in range(n) if in_deg[i] - out_deg[i] == 1)
        return (start_plus == 0 and end_plus == 0) or \
               (start_plus == 1 and end_plus == 1)
    else:
        # 无向图：所有点度数都是偶数（欧拉回路），或恰好两个点度数为奇数（欧拉路径）
        deg = [0] * n
        for u, v in edges:
            deg[u] += 1
            deg[v] += 1
        odd = sum(1 for d in deg if d % 2 == 1)
        return odd == 0 or odd == 2


def fleury_path(n, edges, start, is_directed=True):
    """
    Fleury 算法：找欧拉路径（不适用于大规模图）
    """
    from collections import defaultdict

    if is_directed:
        graph = defaultdict(list)
        for u, v in edges:
            graph[u].append(v)
    else:
        graph = defaultdict(list)
        for u, v in edges:
            graph[u].append(v)
            graph[v].append(u)

    def edge_count(u):
        return len(graph[u])

    def can_use(u, v):
        """判断边 (u,v) 是否可以走（不走桥除非别无选择）"""
        if edge_count(u) == 1:
            return True
        # 简化：直接用 Hierholzer 算法

    # Hierholzer 算法（高效）
    if is_directed:
        in_deg = defaultdict(int)
        out_deg = defaultdict(int)
        for u, v in edges:
            out_deg[u] += 1
            in_deg[v] += 1

        # 找起点
        start_node = start
        for i in range(n):
            if out_deg[i] - in_deg[i] == 1:
                start_node = i
                break

        stack = [start_node]
        path = []
        while stack:
            u = stack[-1]
            if out_deg[u] > 0:
                v = next(v for v in graph[u] if out_deg[u] > 0)
                out_deg[u] -= 1
                stack.append(v)
            else:
                path.append(stack.pop())
        return path[::-1]
    else:
        deg = defaultdict(int)
        for u, v in edges:
            deg[u] += 1
            deg[v] += 1

        stack = [start]
        path = []
        while stack:
            u = stack[-1]
            if deg[u] > 0:
                v = graph[u][-1]
                deg[u] -= 1
                deg[v] -= 1
                graph[u].pop()
                graph[v].remove(u)
                stack.append(v)
            else:
                path.append(stack.pop())
        return path[::-1]
```

### 哈密顿路径

通过图中所有顶点恰好一次的路径（NP完全问题）。

```python
def hamiltonian_path_dp(n, graph):
    """
    哈密顿路径（DP，状态压缩）
    适用于小规模图 O(n^2 * 2^n)
    graph: 邻接矩阵
    返回: 是否存在哈密顿路径
    """
    # dp[mask][v] = 在 mask 表示的顶点集合中，以 v 结尾的路径是否存在
    dp = [[False] * n for _ in range(1 << n)]

    # 初始化：单个顶点的路径
    for i in range(n):
        dp[1 << i][i] = True

    # 状态转移
    for mask in range(1 << n):
        for v in range(n):
            if not dp[mask][v]:
                continue
            # 从 v 扩展到下一个顶点 u
            for u in range(n):
                if (mask & (1 << u)) == 0 and graph[v][u]:
                    dp[mask | (1 << u)][u] = True

    # 检查是否存在经过所有顶点的路径
    full_mask = (1 << n) - 1
    return any(dp[full_mask][v] for v in range(n))


def hamiltonian_path_backtrack(n, graph):
    """
    哈密顿路径（回溯）
    适用于更大规模但节点较少的情况
    """
    result = []

    def backtrack(path, visited):
        if len(path) == n:
            result.append(path[:])
            return True

        curr = path[-1]
        for u in range(n):
            if not visited[u] and graph[curr][u]:
                visited[u] = True
                path.append(u)
                if backtrack(path, visited):
                    return True
                path.pop()
                visited[u] = False
        return False

    for start in range(n):
        visited = [False] * n
        visited[start] = True
        if backtrack([start], visited):
            return result
    return []
```

### 欧拉 vs 哈密顿对比

| 特性 | 欧拉路径 | 哈密顿路径 |
|------|----------|------------|
| 定义 | 经过所有边恰好一次 | 经过所有顶点恰好一次 |
| 判定算法 | O(V+E) | NP完全（无多项式算法） |
| 存在条件 | 有明确的度数条件 | 无简单充分必要条件 |
| 求解 | Fleury/Hierholzer | DP/回溯（指数级） |

## 二分图匹配

### 匈牙利算法

求二分图最大匹配。

```python
def hungarian(max_left, max_right, edges):
    """
    Kuhn-Munkres算法（匈牙利算法）求二分图最大匹配
    注：这是Kuhn算法的实现，适合无权二分图最大匹配
    max_left: 左侧节点数 (0 ~ max_left-1)
    max_right: 右侧节点数 (0 ~ max_right-1)
    edges: 边列表 [(left_node, right_node), ...]
    返回: 最大匹配数
    """
    from collections import defaultdict, deque

    # 构建邻接表
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)

    match_to_left = [-1] * max_left   # 左侧节点匹配的右侧节点，-1表示未匹配
    match_to_right = [-1] * max_right  # 右侧节点匹配的左侧节点，-1表示未匹配
    result = 0

    def bfs():
        """BFS 找增广路径，构建层次图"""
        queue = deque()
        dist = [-1] * max_left

        # 从所有未匹配的左侧节点开始
        for u in range(max_left):
            if match_to_left[u] == -1:
                dist[u] = 0
                queue.append(u)

        found_augment = False
        while queue:
            u = queue.popleft()
            for v in graph[u]:
                matched_u = match_to_right[v]
                if matched_u == -1:
                    # 找到未匹配右侧节点，增广路径终点
                    found_augment = True
                elif dist[matched_u] == -1:
                    dist[matched_u] = dist[u] + 1
                    queue.append(matched_u)

        return found_augment

    def dfs(u, visited):
        """DFS 沿层次图找增广路径"""
        for v in graph[u]:
            if v in visited:
                continue
            visited.add(v)
            matched_u = match_to_right[v]
            if matched_u == -1 or dfs(matched_u, visited):
                match_to_left[u] = v
                match_to_right[v] = u
                return True
        return False

    # 尝试为每个左侧节点找匹配
    for u in range(max_left):
        if match_to_left[u] == -1:
            if dfs(u, set()):
                result += 1

    return result


def hopcroft_karp(max_left, max_right, edges):
    """
    Hopcroft-Karp 算法 O(sqrt(V) * E)
    适用于大规模二分图
    """
    from collections import defaultdict, deque

    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)

    match_u = [-1] * max_left
    match_v = [-1] * max_right
    dist = [-1] * max_left

    def bfs():
        """BFS 构建层次图，返回是否存在增广路径"""
        queue = deque()
        for u in range(max_left):
            if match_u[u] == -1:
                dist[u] = 0
                queue.append(u)
            else:
                dist[u] = -1

        found = False
        while queue:
            u = queue.popleft()
            for v in graph[u]:
                matched = match_v[v]
                if matched != -1 and dist[matched] == -1:
                    dist[matched] = dist[u] + 1
                    queue.append(matched)
                elif matched == -1:
                    # 找到未匹配右侧节点，存在增广路径
                    found = True
        return found

    def dfs(u):
        for v in graph[u]:
            matched = match_v[v]
            if matched == -1 or (dist[matched] == dist[u] + 1 and dfs(matched)):
                match_u[u] = v
                match_v[v] = u
                return True
        dist[u] = -1
        return False

    result = 0
    while bfs():
        for u in range(max_left):
            if match_u[u] == -1 and dfs(u):
                result += 1

    return result
