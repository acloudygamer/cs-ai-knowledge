import heapq

INF = float('inf')

def dijkstra_matrix(graph, start):
    n = len(graph)
    dist = [INF] * n
    dist[start] = 0
    visited = [False] * n

    for _ in range(n):
        # 找最小未访问节点
        u = -1
        for i in range(n):
            if not visited[i] and (u == -1 or dist[i] < dist[u]):
                u = i
        if dist[u] == INF:
            break
        visited[u] = True

        # 更新邻居距离
        for v in range(n):
            if graph[u][v] != INF and not visited[v]:
                dist[v] = min(dist[v], dist[u] + graph[u][v])
    return dist

## BFS/DFS

```python
# 图的BFS
from collections import deque

def bfs(graph, start):
    visited = set()
    queue = deque([start])
    result = []
    while queue:
        node = queue.popleft()
        if node not in visited:
            visited.add(node)
            result.append(node)
            queue.extend(graph[node])
    return result

# 图的DFS（递归）
def dfs_recursive(graph, node, visited=None):
    if visited is None:
        visited = set()
    visited.add(node)
    result = [node]
    for neighbor in graph[node]:
        if neighbor not in visited:
            result.extend(dfs_recursive(graph, neighbor, visited))
    return result

# 图的DFS（迭代）
def dfs_iterative(graph, start):
    visited = set()
    stack = [start]
    result = []
    while stack:
        node = stack.pop()
        if node not in visited:
            visited.add(node)
            result.append(node)
            stack.extend(graph[node])
    return result
```

## 最短路径 - Dijkstra

```python
import heapq

def dijkstra(graph, start):
    dist = {start: 0}
    pq = [(0, start)]
    while pq:
        d, node = heapq.heappop(pq)
        if d > dist.get(node, float('inf')):
            continue
        for neighbor, weight in graph[node]:
            new_dist = d + weight
            if new_dist < dist.get(neighbor, float('inf')):
                dist[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    return dist

## 拓扑排序

有向无环图（DAG）的顶点线性排序，使得每条有向边 (u, v) 都满足 u 在 v 之前。

### Kahn算法（BFS + 入度）

```python
from collections import deque, defaultdict

def topological_sort_kahn(num_courses, prerequisites):
    graph = defaultdict(list)
    in_degree = [0] * num_courses

    for u, v in prerequisites:
        graph[v].append(u)  # v -> u (u依赖v)
        in_degree[u] += 1

    queue = deque([i for i in range(num_courses) if in_degree[i] == 0])
    result = []

    while queue:
        node = queue.popleft()
        result.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    return result if len(result) == num_courses else []  # 空列表表示有环
```

### DFS方法

```python
def topological_sort_dfs(num_courses, prerequisites):
    graph = defaultdict(list)
    for u, v in prerequisites:
        graph[v].append(u)

    visited = [0] * num_courses  # 0=未访问, 1=访问中, 2=已完成
    result = []
    has_cycle = False

    def dfs(node):
        nonlocal has_cycle
        if visited[node] == 1:  # 发现环
            has_cycle = True
            return
        if visited[node] == 2:   # 已完成
            return

        visited[node] = 1
        for neighbor in graph[node]:
            dfs(neighbor)
            if has_cycle:
                return
        visited[node] = 2
        result.append(node)

    for i in range(num_courses):
        if visited[i] == 0:
            dfs(i)
            if has_cycle:
                return []

    return result[::-1]
```

## 最小生成树 (MST)

### Prim算法

```python
def prim(n, edges):
    """ edges: [(u, v, weight)], 返回MST总权重 """
    graph = defaultdict(list)
    for u, v, w in edges:
        graph[u].append((v, w))
        graph[v].append((u, w))

    visited = [False] * n
    min_edge = [(INF, -1)] * n  # (weight, from)
    min_edge[0] = (0, -1)
    total = 0

    for _ in range(n):
        u = -1
        for i in range(n):
            if not visited[i] and (u == -1 or min_edge[i][0] < min_edge[u][0]):
                u = i

        if u != -1 and min_edge[u][0] == INF:
            return -1  # 图不连通

        if u != -1:
            visited[u] = True
            total += min_edge[u][0]
            for v, w in graph[u]:
                if not visited[v] and w < min_edge[v][0]:
                    min_edge[v] = (w, u)

    return total
```

### Kruskal算法（使用并查集）

```python
def kruskal(n, edges):
    edges.sort(key=lambda x: x[2])
    parent = list(range(n))

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(x, y):
        px, py = find(x), find(y)
        if px == py:
            return False
        parent[px] = py
        return True

    mst_weight = 0
    mst_edges = []
    for u, v, w in edges:
        if union(u, v):
            mst_weight += w
            mst_edges.append((u, v, w))
            if len(mst_edges) == n - 1:
                break
    return mst_weight, mst_edges
```

## 关键路径算法

```python
def find_critical_path(n, edges):
    """
    关键路径 = 项目最短完成时间
    edges: [(u, v, weight)]
    返回: 关键路径长度和路径
    """
    graph = defaultdict(list)
    in_degree = [0] * n

    for u, v, w in edges:
        graph[u].append((v, w))
        in_degree[v] += 1

    # 正向计算最早完成时间
    earliest = [0] * n
    queue = deque([i for i in range(n) if in_degree[i] == 0])

    while queue:
        u = queue.popleft()
        for v, w in graph[u]:
            earliest[v] = max(earliest[v], earliest[u] + w)
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)

    # 反向计算最晚完成时间
    latest = [earliest[n - 1]] * n
    queue = deque([n - 1])

    while queue:
        v = queue.popleft()
        for u, w in graph.items():
            if (v, w) in [(to_, weight) for to_, weight in graph[u]]:
                latest[u] = min(latest[u], latest[v] - w)
                if latest[u] == latest[v] - w:
                    queue.append(u)

    # 简化的关键路径输出
    return earliest[n - 1]
```

## 最大流算法（网络流）

给定有向图和源点、汇点，求最大流量。

### Ford-Fulkerson 方法

```python
def ford_fulkerson(graph, source, sink):
    """
    Ford-Fulkerson 最大流算法（Edmonds-Karp 用 BFS 找增广路径）
    时间复杂度: O(V * E^2)
    graph: 邻接表，graph[u][v] 表示边(u,v)的容量
    """
    from collections import deque

    def bfs_find_path(graph, source, sink, parent):
        """BFS 找增广路径"""
        visited = set([source])
        queue = deque([source])
        parent_map = {source: None}

        while queue:
            u = queue.popleft()
            if u == sink:
                return True

            for v in graph.get(u, []):
                if v not in visited and graph[u][v] > 0:
                    visited.add(v)
                    queue.append(v)
                    parent_map[v] = u
                    if v == sink:
                        parent[:] = []
                        while v is not None:
                            parent.append(v)
                            v = parent_map[v]
                        parent.reverse()
                        return True
        return False

    def get_bottleneck(graph, parent):
        """找到路径上的最小剩余容量"""
        bottleneck = float('inf')
        for i in range(len(parent) - 1):
            u, v = parent[i], parent[i + 1]
            bottleneck = min(bottleneck, graph[u][v])
        return bottleneck

    def augment_path(graph, parent, flow):
        """沿路径增加流量"""
        for i in range(len(parent) - 1):
            u, v = parent[i], parent[i + 1]
            graph[u][v] -= flow
            graph[v][u] = graph.get(v, {}).get(u, 0) + flow

    parent = []
    max_flow = 0

    while bfs_find_path(graph, source, sink, parent):
        flow = get_bottleneck(graph, parent)
        augment_path(graph, parent, flow)
        max_flow += flow
        parent.clear()

    return max_flow
```

### Dinic 算法（更高效）

```python
def dinic_max_flow(n, graph_edges, source, sink):
    """
    Dinic 最大流算法
    时间复杂度: O(E * V^2)（实际更快）
    graph_edges: [(u, v, capacity), ...]
    """
    from collections import deque

    # 构建邻接表（包含残余网络）
    adj = [[] for _ in range(n)]
    for u, v, cap in graph_edges:
        adj[u].append([v, cap, len(adj[v])])      # 正向边
        adj[v].append([u, 0, len(adj[u]) - 1])    # 反向边

    def bfs(level):
        """构建层次图"""
        level[:] = [-1] * n
        level[source] = 0
        queue = deque([source])
        while queue:
            u = queue.popleft()
            for v, cap, rev in adj[u]:
                if cap > 0 and level[v] < 0:
                    level[v] = level[u] + 1
                    queue.append(v)
        return level[sink] >= 0

    def dfs(u, flow, level, it, sink):
        """在层次图中找增广路径"""
        if u == sink:
            return flow
        for i in range(it[u], len(adj[u])):
            it[u] = i
            v, cap, rev = adj[u][i]
            if cap > 0 and level[u] < level[v]:
                pushed = dfs(v, min(flow, cap), level, it, sink)
                if pushed:
                    # 更新残余网络
                    adj[u][i][1] -= pushed
                    adj[v][rev][1] += pushed
                    return pushed
        return 0

    level = [-1] * n
    max_flow = 0

    while bfs(level):
        it = [0] * n
        while True:
            pushed = dfs(source, float('inf'), level, it, sink)
            if not pushed:
                break
            max_flow += pushed

    return max_flow
```

### 最小割

最小割将图划分为两部分，使得从源到汇的边容量和最小。

```python
def min_cut(graph, source, sink):
    """找到最小割的顶点集合"""
    visited = set([source])

    def dfs(u, flow=float('inf')):
        for v in graph.get(u, []):
            if graph[u][v] > 0 and v not in visited:
                visited.add(v)
                dfs(v)

    dfs(source)
    return visited  # S 集合，与 T 集合构成最小割
```

## 图算法复杂度总结

| 算法 | 时间复杂度 | 空间复杂度 | 适用场景 |
|------|------------|------------|----------|
| BFS | O(V+E) | O(V) | 最短路径(无权) |
| DFS | O(V+E) | O(V) | 连通分量、拓扑排序 |
| Dijkstra | O((V+E)logV) | O(V) | 单源最短路径(正权) |
| Bellman-Ford | O(VE) | O(V) | 单源最短路径(可负权) |
| Floyd-Warshall | O(V^3) | O(V^2) | 全源最短路径 |
| Prim | O(V^2) 或 O(ElogV) | O(V) | MST |
| Kruskal | O(ElogE) | O(V) | MST |
| Kahn (拓扑) | O(V+E) | O(V) | DAG拓扑排序 |
